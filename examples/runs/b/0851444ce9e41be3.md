Written `support_bot.py` (Python, since no project language was set up here — swap for another SDK if you're using one).

How it works:
- `run_agent_loop` sends `messages` to `claude-opus-5` with two strict-schema tools, `look_up_order` and `issue_refund`.
- While `stop_reason == "tool_use"`, it executes each requested tool, wraps exceptions as `is_error: true` tool results instead of raising, and appends both the assistant turn and the tool results back into `messages` before calling again — that's the multi-turn loop.
- It exits on `end_turn` (returns the reply), guards `refusal` and `max_tokens` before touching `content`, and mirrors errors as data so Claude can recover (e.g. try a different order ID) instead of crashing.
- `issue_refund` is irreversible, so it gates on a human `y/N` prompt before touching the mock ledger — swap that `input()` for your real approval-queue call in production.

To run it for real: `pip install anthropic`, set `ANTHROPIC_API_KEY`, then `python support_bot.py`.

"""Support bot: looks up an order and issues a refund over a multi-turn tool-calling loop.

No project language was specified, so this defaults to Python with the Anthropic SDK
(pip install anthropic). Swap MOCK_ORDERS / the two tool implementations for real
order-service and payment-provider calls.
"""

import json

import anthropic

MODEL = "claude-opus-5"

# --- Mock backend -----------------------------------------------------------

MOCK_ORDERS = {
    "ORD-1001": {"order_id": "ORD-1001", "status": "delivered", "total": 49.99, "refunded": 0.0},
    "ORD-1002": {"order_id": "ORD-1002", "status": "cancelled", "total": 120.00, "refunded": 0.0},
}


def look_up_order(order_id: str) -> dict:
    order = MOCK_ORDERS.get(order_id)
    if order is None:
        return {"error": f"No order found with id '{order_id}'"}
    return order


def issue_refund(order_id: str, amount: float, reason: str) -> dict:
    order = MOCK_ORDERS.get(order_id)
    if order is None:
        return {"error": f"No order found with id '{order_id}'"}
    remaining = order["total"] - order["refunded"]
    if amount > remaining:
        return {"error": f"Refund of {amount} exceeds remaining refundable amount {remaining}"}

    # Refunds are irreversible, so a human approves each one before it is executed.
    approved = input(
        f"\n[APPROVAL REQUIRED] Refund ${amount:.2f} on {order_id} ({reason}). Approve? [y/N] "
    ).strip().lower() == "y"
    if not approved:
        return {"error": "Refund was not approved by a human reviewer"}

    order["refunded"] += amount
    return {"order_id": order_id, "refunded_amount": amount, "new_refunded_total": order["refunded"]}


TOOL_REGISTRY = {
    "look_up_order": look_up_order,
    "issue_refund": issue_refund,
}

TOOLS = [
    {
        "name": "look_up_order",
        "description": (
            "Look up an order by its ID and return its status, total, and amount already "
            "refunded. Use this before issuing a refund to confirm the order exists and "
            "check how much is still refundable."
        ),
        "strict": True,
        "input_schema": {
            "type": "object",
            "properties": {
                "order_id": {
                    "type": "string",
                    "description": "The order ID, e.g. 'ORD-1001'.",
                },
            },
            "required": ["order_id"],
            "additionalProperties": False,
        },
    },
    {
        "name": "issue_refund",
        "description": (
            "Issue a refund for an order. Only call this after looking up the order and "
            "confirming the requested amount does not exceed what's left to refund. This "
            "action requires human approval and cannot be undone."
        ),
        "strict": True,
        "input_schema": {
            "type": "object",
            "properties": {
                "order_id": {
                    "type": "string",
                    "description": "The order ID to refund, e.g. 'ORD-1001'.",
                },
                "amount": {
                    "type": "number",
                    "description": "Dollar amount to refund. Must not exceed the order's remaining refundable balance.",
                },
                "reason": {
                    "type": "string",
                    "description": "Short reason for the refund, shown to the human approver.",
                },
            },
            "required": ["order_id", "amount", "reason"],
            "additionalProperties": False,
        },
    },
]

SYSTEM_PROMPT = (
    "You are a support agent. Look up orders before refunding them, never guess an "
    "order's status or balance, and only refund what the customer is actually owed."
)


def run_agent_loop(client: anthropic.Anthropic, messages: list) -> str:
    while True:
        response = client.messages.create(
            model=MODEL,
            max_tokens=4096,
            system=SYSTEM_PROMPT,
            tools=TOOLS,
            messages=messages,
        )

        if response.stop_reason == "refusal":
            return "I can't help with that request."

        if response.stop_reason == "max_tokens":
            raise RuntimeError("Response was truncated at max_tokens")

        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason != "tool_use":
            return next(
                (b.text for b in response.content if b.type == "text"),
                "",
            )

        tool_results = []
        for block in response.content:
            if block.type != "tool_use":
                continue

            tool_fn = TOOL_REGISTRY.get(block.name)
            try:
                if tool_fn is None:
                    raise ValueError(f"Unknown tool: {block.name}")
                result = tool_fn(**block.input)
                is_error = "error" in result
            except Exception as e:
                result = {"error": str(e)}
                is_error = True

            tool_results.append({
                "type": "tool_result",
                "tool_use_id": block.id,
                "content": json.dumps(result),
                "is_error": is_error,
            })

        messages.append({"role": "user", "content": tool_results})
        # loop again so Claude can see the tool result and continue or respond


def main():
    client = anthropic.Anthropic()
    messages = []

    print("Support bot ready. Type 'quit' to exit.")
    while True:
        user_input = input("\nYou: ").strip()
        if user_input.lower() in ("quit", "exit"):
            break

        messages.append({"role": "user", "content": user_input})
        # run_agent_loop mutates `messages` in place, appending each assistant
        # turn and tool-result turn as it goes - don't append the reply again here.
        reply = run_agent_loop(client, messages)
        print(f"Bot: {reply}")


if __name__ == "__main__":
    main()
