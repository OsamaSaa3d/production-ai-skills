"""Read-only tool for querying the orders table."""

import sqlite3
from contextlib import closing
from typing import Literal, Optional

from langchain_core.tools import tool

DB_PATH = "orders.db"  # adjust to your real connection
MAX_ROWS = 200  # hard cap so one call can't dump the whole table

OrderStatus = Literal["pending", "paid", "shipped", "delivered", "cancelled", "refunded"]


@tool
def query_orders(
    order_id: Optional[int] = None,
    customer_id: Optional[int] = None,
    status: Optional[OrderStatus] = None,
    created_after: Optional[str] = None,
    created_before: Optional[str] = None,
    limit: int = 50,
) -> dict:
    """Look up rows from the orders table.

    Use this when the user asks about a specific order, a customer's order
    history, or orders matching a status/date range. Read-only -- it cannot
    modify data.

    Args:
        order_id: Exact order id to fetch, or null to not filter on it.
        customer_id: Exact customer id to fetch, or null to not filter on it.
        status: One of pending/paid/shipped/delivered/cancelled/refunded, or null.
        created_after: ISO-8601 date/time lower bound (inclusive), or null.
        created_before: ISO-8601 date/time upper bound (inclusive), or null.
        limit: Max rows to return (1-200); values outside that range are clamped.
    """
    clauses: list[str] = []
    params: list = []
    if order_id is not None:
        clauses.append("id = ?")
        params.append(order_id)
    if customer_id is not None:
        clauses.append("customer_id = ?")
        params.append(customer_id)
    if status is not None:
        clauses.append("status = ?")
        params.append(status)
    if created_after is not None:
        clauses.append("created_at >= ?")
        params.append(created_after)
    if created_before is not None:
        clauses.append("created_at <= ?")
        params.append(created_before)

    where = f"WHERE {' AND '.join(clauses)}" if clauses else ""
    safe_limit = max(1, min(limit or 50, MAX_ROWS))
    sql = (
        "SELECT id, customer_id, product, quantity, status, amount, created_at "
        f"FROM orders {where} ORDER BY created_at DESC LIMIT ?"
    )

    try:
        with closing(sqlite3.connect(DB_PATH)) as conn:
            conn.row_factory = sqlite3.Row
            rows = conn.execute(sql, [*params, safe_limit]).fetchall()
        return {"rows": [dict(row) for row in rows], "row_count": len(rows)}
    except sqlite3.Error as e:
        # Returned as tool content, not raised, so the model can react to it.
        return {"error": str(e)}

### `pipeline.py`

"""LangGraph pipeline step that lets the model query the orders table.

Drop the `query_orders` node/tool pair into an existing StateGraph by copying
the two `add_node` calls and the edges below into your graph builder --
everything here besides `build_pipeline` is reusable as-is.
"""

from typing import Annotated

from langchain_core.messages import SystemMessage
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.base import BaseCheckpointSaver
from langgraph.checkpoint.memory import MemorySaver
from langgraph.graph import END, START, StateGraph
from langgraph.graph.message import add_messages
from langgraph.prebuilt import ToolNode, tools_condition
from typing_extensions import TypedDict

from orders_tool import query_orders

TOOLS = [query_orders]

llm = ChatOpenAI(model="gpt-5.1", temperature=0).bind_tools(TOOLS, strict=True)

SYSTEM_PROMPT = SystemMessage(
    content=(
        "You can call query_orders to look up rows in the orders table. "
        "Only call it when answering the user actually requires order data."
    )
)


class PipelineState(TypedDict):
    messages: Annotated[list, add_messages]


def query_orders_node(state: PipelineState) -> dict:
    messages = state["messages"]
    if not any(isinstance(m, SystemMessage) for m in messages):
        messages = [SYSTEM_PROMPT, *messages]
    response = llm.invoke(messages)
    return {"messages": [response]}


def build_pipeline(checkpointer: BaseCheckpointSaver | None = None):
    graph = StateGraph(PipelineState)

    graph.add_node("query_orders", query_orders_node)
    graph.add_node("orders_tools", ToolNode(TOOLS))

    graph.add_edge(START, "query_orders")
    graph.add_conditional_edges(
        "query_orders",
        tools_condition,
        {"tools": "orders_tools", END: END},
    )
    graph.add_edge("orders_tools", "query_orders")

    return graph.compile(checkpointer=checkpointer or MemorySaver())


if __name__ == "__main__":
    app = build_pipeline()
    config = {"configurable": {"thread_id": "demo"}}
    result = app.invoke(
        {"messages": [{"role": "user", "content": "What's the status of order 42?"}]},
        config=config,
    )
    print(result["messages"][-1].content)
