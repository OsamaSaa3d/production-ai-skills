````py
import sqlite3

SCHEMA = """
CREATE TABLE IF NOT EXISTS orders (
    invoice_number TEXT PRIMARY KEY,
    issue_date     TEXT NOT NULL,
    total          REAL NOT NULL,
    customer_name  TEXT NOT NULL,
    po_number      TEXT,
    source_file    TEXT NOT NULL,
    extracted_at   TEXT NOT NULL
);
"""


def connect(db_path: str) -> sqlite3.Connection:
    conn = sqlite3.connect(db_path)
    conn.execute(SCHEMA)
    return conn


def upsert_order(conn: sqlite3.Connection, *, invoice_number: str, issue_date: str,
                  total: float, customer_name: str, po_number: str | None,
                  source_file: str, extracted_at: str) -> None:
    conn.execute(
        """
        INSERT INTO orders (invoice_number, issue_date, total, customer_name,
                             po_number, source_file, extracted_at)
        VALUES (?, ?, ?, ?, ?, ?, ?)
        ON CONFLICT(invoice_number) DO UPDATE SET
            issue_date = excluded.issue_date,
            total = excluded.total,
            customer_name = excluded.customer_name,
            po_number = excluded.po_number,
            source_file = excluded.source_file,
            extracted_at = excluded.extracted_at
        """,
        (invoice_number, issue_date, total, customer_name, po_number,
         source_file, extracted_at),
    )
    conn.commit()
````

### `extract_invoices.py`

````py
"""
Extract invoice_number, issue_date, total, customer_name, and po_number from
plain-text invoices and write them to the `orders` table in a SQLite database.

Usage:
    python extract_invoices.py <invoice.txt | directory> [--db orders.db]
"""
import argparse
import datetime as dt
import glob
import os
import sys

import anthropic

from db import connect, upsert_order
from schema import InvoiceExtraction

MODEL = "claude-opus-5"
SYSTEM_PROMPT = (
    "You extract structured billing data from plain-text invoices. "
    "Use only information present in the document; do not infer or invent values."
)


class Refused(Exception):
    def __init__(self, message: str, category: str | None = None):
        super().__init__(message)
        self.category = category


class Truncated(Exception):
    pass


def extract(client: anthropic.Anthropic, invoice_text: str, *,
            max_tokens: int = 2048, max_retries: int = 1) -> InvoiceExtraction:
    attempt = 0
    while True:
        attempt += 1
        response = client.messages.parse(
            model=MODEL,
            max_tokens=max_tokens,
            system=SYSTEM_PROMPT,
            messages=[{"role": "user", "content": invoice_text}],
            output_format=InvoiceExtraction,
        )

        if response.stop_reason == "refusal":
            category = getattr(response.stop_details, "category", None) if response.stop_details else None
            raise Refused("model refused to extract this invoice", category)

        if response.stop_reason == "max_tokens":
            if attempt <= max_retries:
                max_tokens *= 2
                continue
            raise Truncated(f"still truncated at max_tokens={max_tokens}")

        return response.parsed_output


def validate(invoice: InvoiceExtraction) -> list[str]:
    problems = []
    if invoice.total < 0:
        problems.append("negative total")
    try:
        dt.date.fromisoformat(invoice.issue_date)
    except ValueError:
        problems.append(f"issue_date not in ISO format: {invoice.issue_date!r}")
    return problems


def iter_invoice_files(path: str) -> list[str]:
    if os.path.isdir(path):
        return sorted(glob.glob(os.path.join(path, "*.txt")))
    return [path]


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("input", help="a .txt invoice file, or a directory of .txt invoices")
    parser.add_argument("--db", default="orders.db", help="SQLite database file (default: orders.db)")
    args = parser.parse_args()

    files = iter_invoice_files(args.input)
    if not files:
        print(f"No .txt files found at {args.input}", file=sys.stderr)
        return 1

    client = anthropic.Anthropic()
    conn = connect(args.db)

    ok, failed = 0, 0
    for path in files:
        with open(path, "r", encoding="utf-8") as f:
            text = f.read()

        try:
            invoice = extract(client, text)
        except Refused as e:
            print(f"[refused] {path}: {e} (category={e.category})", file=sys.stderr)
            failed += 1
            continue
        except Truncated as e:
            print(f"[truncated] {path}: {e}", file=sys.stderr)
            failed += 1
            continue
        except anthropic.APIStatusError as e:
            print(f"[api-error] {path}: {e.status_code} {e.message}", file=sys.stderr)
            failed += 1
            continue

        problems = validate(invoice)
        if problems:
            print(f"[invalid] {path}: {', '.join(problems)}", file=sys.stderr)
            failed += 1
            continue

        upsert_order(
            conn,
            invoice_number=invoice.invoice_number,
            issue_date=invoice.issue_date,
            total=invoice.total,
            customer_name=invoice.customer_name,
            po_number=invoice.po_number,
            source_file=os.path.abspath(path),
            extracted_at=dt.datetime.now(dt.timezone.utc).isoformat(),
        )
        print(f"[ok] {path}: invoice {invoice.invoice_number}")
        ok += 1

    conn.close()
    print(f"\n{ok} written, {failed} failed, out of {len(files)} file(s).")
    return 1 if failed and not ok else 0


if __name__ == "__main__":
    raise SystemExit(main())

````

````py
from pydantic import BaseModel, Field


class InvoiceExtraction(BaseModel):
    invoice_number: str = Field(
        description="The invoice number or ID exactly as printed on the document."
    )
    issue_date: str = Field(
        description="The invoice issue date, normalized to ISO 8601 (YYYY-MM-DD). "
                    "If the printed date is ambiguous (e.g. 03/04/2026), resolve it "
                    "using any locale cues in the document; if still ambiguous, use "
                    "the format as printed."
    )
    total: float = Field(
        description="The final total amount due on the invoice, as a plain number "
                    "with no currency symbol or thousands separators."
    )
    customer_name: str = Field(
        description="The name of the customer or company being billed (the "
                    "'bill to' party), not the vendor issuing the invoice."
    )
    po_number: str | None = Field(
        description="The purchase order (PO) number referenced on the invoice. "
                    "Null if the invoice does not reference a PO number."
    )
````
