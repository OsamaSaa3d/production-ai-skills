Wrote `orders_qa.py`. It uses Claude's native tool calling with `strict: true` on one tool, `query_orders`, rather than letting the model generate SQL:

- **Why not raw SQL:** an LLM-facing "run this SQL" tool is a standing SQL-injection/DROP-TABLE risk. Instead `query_orders` takes structured, schema-constrained args (`action`, filters, `aggregate_function`, `group_by`, `limit`); every value is bound as a `psycopg2` query parameter, and column/aggregate names come only from fixed allow-list dicts (`AGGREGATE_EXPR`, `GROUP_BY_EXPR`) — never from string interpolation.
- **Two modes:** `action="list"` returns matching rows (capped at 200); `action="aggregate"` computes count/sum/avg/min/max, optionally grouped by status, customer_email, day/week/month.
- **Defense in depth on the DB side:** each call opens a `readonly` session and sets a 5s `statement_timeout`. The real boundary should still be a Postgres role with SELECT-only grants — `DATABASE_URL` should point at that role, not an admin account.
- **Loop:** `ask()` sends the question, executes any tool calls, feeds `tool_result` blocks back, and repeats (capped at 5 rounds) until Claude returns a plain-text answer. It checks `stop_reason` before treating output as complete text.

To run: set `ANTHROPIC_API_KEY` and `DATABASE_URL` (pointed at a read-only role), `pip install anthropic psycopg2-binary`, then `python orders_qa.py`.

One thing I couldn't verify: I don't know your actual `status` values, so I left that filter as a free-form string (documented via description) rather than hardcoding an enum that might not match your data — tell me the real status set if you want it locked down as an enum.

"""
Plain-English Q&A over the `orders` table (id, customer_email, status, total_usd, placed_at).

Usage:
    set ANTHROPIC_API_KEY=...
    set DATABASE_URL=postgresql://readonly_user:pw@host:5432/dbname
    python orders_qa.py

Ask things like:
    "How many orders did jane@doe.com place?"
    "What's the average order total for shipped orders in August 2026?"
    "List the 10 most recent cancelled orders."

Design notes:
- The model never writes SQL. It calls `query_orders` with a structured,
  schema-constrained payload (filters + action). All values are bound as
  query parameters via psycopg2 — never string-concatenated — so there is
  no SQL-injection surface even though the input originates from an LLM.
- DATABASE_URL should point at a role with SELECT-only privileges on
  `orders`. The code also opens each transaction READ ONLY and sets a
  statement timeout as defense in depth, but the DB role is the real
  boundary.
"""

import decimal
import json
import os
import sys

import psycopg2
import psycopg2.sql as sql
from anthropic import Anthropic

MODEL = os.environ.get("ANTHROPIC_MODEL", "claude-sonnet-5")
DATABASE_URL = os.environ["DATABASE_URL"]
STATEMENT_TIMEOUT_MS = 5000
MAX_LIST_ROWS = 200
MAX_TOOL_ITERATIONS = 5

AGGREGATE_EXPR = {
    "count": sql.SQL("COUNT(*)"),
    "sum": sql.SQL("SUM(total_usd)"),
    "avg": sql.SQL("AVG(total_usd)"),
    "min": sql.SQL("MIN(total_usd)"),
    "max": sql.SQL("MAX(total_usd)"),
}

GROUP_BY_EXPR = {
    "status": sql.SQL("status"),
    "customer_email": sql.SQL("customer_email"),
    "day": sql.SQL("date_trunc('day', placed_at)"),
    "week": sql.SQL("date_trunc('week', placed_at)"),
    "month": sql.SQL("date_trunc('month', placed_at)"),
}

QUERY_ORDERS_TOOL = {
    "name": "query_orders",
    "description": (
        "Query the orders table. Use action='list' to fetch matching order rows, "
        "or action='aggregate' to compute a summary statistic (optionally grouped). "
        "All filters are optional (pass null to skip them) and are combined with AND."
    ),
    "strict": True,
    "input_schema": {
        "type": "object",
        "properties": {
            "action": {
                "type": "string",
                "enum": ["list", "aggregate"],
                "description": "'list' returns individual order rows; 'aggregate' returns a computed statistic.",
            },
            "status": {
                "anyOf": [{"type": "string"}, {"type": "null"}],
                "description": "Exact order status to filter on, e.g. 'pending', 'shipped', 'cancelled'. Null = any status.",
            },
            "customer_email": {
                "anyOf": [{"type": "string"}, {"type": "null"}],
                "description": "Exact customer email to filter on. Null = any customer.",
            },
            "placed_after": {
                "anyOf": [{"type": "string"}, {"type": "null"}],
                "description": "ISO 8601 date/timestamp; only include orders placed at or after this. Null = no lower bound.",
            },
            "placed_before": {
                "anyOf": [{"type": "string"}, {"type": "null"}],
                "description": "ISO 8601 date/timestamp; only include orders placed before this. Null = no upper bound.",
            },
            "min_total_usd": {
                "anyOf": [{"type": "number"}, {"type": "null"}],
                "description": "Only include orders with total_usd >= this value. Null = no lower bound.",
            },
            "max_total_usd": {
                "anyOf": [{"type": "number"}, {"type": "null"}],
                "description": "Only include orders with total_usd <= this value. Null = no upper bound.",
            },
            "aggregate_function": {
                "anyOf": [
                    {"type": "string", "enum": list(AGGREGATE_EXPR.keys())},
                    {"type": "null"},
                ],
                "description": "Required when action='aggregate'. 'count' counts orders; the others operate on total_usd. Ignored for action='list'.",
            },
            "group_by": {
                "anyOf": [
                    {"type": "string", "enum": list(GROUP_BY_EXPR.keys())},
                    {"type": "null"},
                ],
                "description": "Optional grouping for action='aggregate' (e.g. group by status, or by day/week/month). Null = one overall total. Ignored for action='list'.",
            },
            "limit": {
                "type": "integer",
                "description": f"Max rows to return for action='list' (capped at {MAX_LIST_ROWS} server-side). Ignored for action='aggregate'.",
            },
        },
        "required": [
            "action",
            "status",
            "customer_email",
            "placed_after",
            "placed_before",
            "min_total_usd",
            "max_total_usd",
            "aggregate_function",
            "group_by",
            "limit",
        ],
        "additionalProperties": False,
    },
}

SYSTEM_PROMPT = """You are a data assistant for a customer orders system.
The `orders` table has: id, customer_email, status, total_usd, placed_at.
Answer the user's question by calling `query_orders` to fetch the data you need,
then summarize the result in plain English. Never guess at numbers — always
query for them. If a question is ambiguous (e.g. "recent" with no timeframe),
make a reasonable assumption and say what you assumed."""


def _json_default(value):
    if isinstance(value, decimal.Decimal):
        return float(value)
    return str(value)


def _build_where(args):
    clauses = []
    params = []
    if args["status"] is not None:
        clauses.append(sql.SQL("status = %s"))
        params.append(args["status"])
    if args["customer_email"] is not None:
        clauses.append(sql.SQL("customer_email = %s"))
        params.append(args["customer_email"])
    if args["placed_after"] is not None:
        clauses.append(sql.SQL("placed_at >= %s"))
        params.append(args["placed_after"])
    if args["placed_before"] is not None:
        clauses.append(sql.SQL("placed_at < %s"))
        params.append(args["placed_before"])
    if args["min_total_usd"] is not None:
        clauses.append(sql.SQL("total_usd >= %s"))
        params.append(args["min_total_usd"])
    if args["max_total_usd"] is not None:
        clauses.append(sql.SQL("total_usd <= %s"))
        params.append(args["max_total_usd"])

    if not clauses:
        return sql.SQL(""), params
    return sql.SQL(" WHERE ") + sql.SQL(" AND ").join(clauses), params


def _run_list(cur, args):
    where_sql, params = _build_where(args)
    limit = max(1, min(int(args["limit"] or 50), MAX_LIST_ROWS))
    query = (
        sql.SQL("SELECT id, customer_email, status, total_usd, placed_at FROM orders")
        + where_sql
        + sql.SQL(" ORDER BY placed_at DESC LIMIT %s")
    )
    cur.execute(query, params + [limit])
    columns = [c.name for c in cur.description]
    rows = [dict(zip(columns, row)) for row in cur.fetchall()]
    return {"rows": rows, "row_count": len(rows), "truncated_at": limit}


def _run_aggregate(cur, args):
    func = args["aggregate_function"]
    if func not in AGGREGATE_EXPR:
        return {"error": f"aggregate_function must be one of {list(AGGREGATE_EXPR)}"}
    agg_expr = AGGREGATE_EXPR[func]
    where_sql, params = _build_where(args)

    group_by = args["group_by"]
    if group_by is None:
        query = (
            sql.SQL("SELECT ")
            + agg_expr
            + sql.SQL(" AS value, COUNT(*) AS order_count FROM orders")
            + where_sql
        )
        cur.execute(query, params)
        value, order_count = cur.fetchone()
        return {"value": value, "order_count": order_count}

    if group_by not in GROUP_BY_EXPR:
        return {"error": f"group_by must be one of {list(GROUP_BY_EXPR)}"}
    group_expr = GROUP_BY_EXPR[group_by]
    query = (
        sql.SQL("SELECT ")
        + group_expr
        + sql.SQL(" AS group_key, ")
        + agg_expr
        + sql.SQL(" AS value, COUNT(*) AS order_count FROM orders")
        + where_sql
        + sql.SQL(" GROUP BY ")
        + group_expr
        + sql.SQL(" ORDER BY ")
        + group_expr
    )
    cur.execute(query, params)
    columns = [c.name for c in cur.description]
    rows = [dict(zip(columns, row)) for row in cur.fetchall()]
    return {"groups": rows, "group_count": len(rows)}


def query_orders(**args):
    """Execute a validated, parameterized query against the orders table."""
    try:
        conn = psycopg2.connect(DATABASE_URL)
    except psycopg2.Error as e:
        return {"error": f"could not connect to database: {e}"}

    try:
        conn.set_session(readonly=True, autocommit=False)
        with conn.cursor() as cur:
            cur.execute(sql.SQL("SET LOCAL statement_timeout = %s"), [STATEMENT_TIMEOUT_MS])
            if args["action"] == "list":
                result = _run_list(cur, args)
            else:
                result = _run_aggregate(cur, args)
        conn.rollback()  # read-only session; nothing to commit
        return result
    except psycopg2.Error as e:
        conn.rollback()
        return {"error": f"query failed: {e}"}
    finally:
        conn.close()


TOOL_REGISTRY = {"query_orders": query_orders}


def ask(client, question, messages):
    messages.append({"role": "user", "content": question})

    for _ in range(MAX_TOOL_ITERATIONS):
        response = client.messages.create(
            model=MODEL,
            max_tokens=1024,
            system=SYSTEM_PROMPT,
            messages=messages,
            tools=[QUERY_ORDERS_TOOL],
        )
        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason == "max_tokens":
            return "[response truncated before completion]"
        if response.stop_reason != "tool_use":
            return "".join(block.text for block in response.content if block.type == "text")

        tool_results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            handler = TOOL_REGISTRY.get(block.name)
            if handler is None:
                result = {"error": f"unknown tool '{block.name}'"}
            else:
                result = handler(**block.input)
            tool_results.append({
                "type": "tool_result",
                "tool_use_id": block.id,
                "content": json.dumps(result, default=_json_default),
            })
        messages.append({"role": "user", "content": tool_results})

    return "[gave up after too many tool calls]"


def main():
    client = Anthropic()  # reads ANTHROPIC_API_KEY from the environment
    messages = []
    print("Ask a question about orders (Ctrl+C or 'exit' to quit).")
    while True:
        try:
            question = input("\n> ").strip()
        except (EOFError, KeyboardInterrupt):
            break
        if not question or question.lower() in ("exit", "quit"):
            break
        print(ask(client, question, messages))


if __name__ == "__main__":
    sys.exit(main())
