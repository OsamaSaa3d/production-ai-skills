"""Classify incoming support emails by urgency for routing."""

from __future__ import annotations

from enum import Enum
from typing import Literal

import anthropic
from pydantic import BaseModel

MODEL = "claude-opus-5"

# Routing queue for each urgency level.
ROUTING_QUEUE = {
    "critical": "oncall-page",
    "high": "priority-queue",
    "medium": "standard-queue",
    "low": "backlog",
}

SYSTEM_PROMPT = """\
You triage incoming customer support emails for a routing system.

Classify each email's urgency:
- critical: production outage, data loss, security incident, or the customer says they are blocked with no workaround
- high: significant functionality broken or a paying customer strongly impacted, but a workaround exists
- medium: a real issue with limited impact, or a time-sensitive question
- low: general question, feature request, or minor cosmetic issue

Base the classification only on what the email states or clearly implies. Do not assume urgency from tone alone."""


class Urgency(str, Enum):
    critical = "critical"
    high = "high"
    medium = "medium"
    low = "low"


class EmailClassification(BaseModel):
    urgency: Urgency
    reasoning: str


class Email(BaseModel):
    id: str
    subject: str
    body: str


class ClassifiedEmail(BaseModel):
    email: Email
    urgency: Urgency
    reasoning: str
    queue: str


def classify_email(client: anthropic.Anthropic, email: Email) -> ClassifiedEmail:
    response = client.messages.parse(
        model=MODEL,
        max_tokens=1024,
        system=SYSTEM_PROMPT,
        messages=[{
            "role": "user",
            "content": f"Subject: {email.subject}\n\n{email.body}",
        }],
        output_format=EmailClassification,
    )

    if response.stop_reason == "refusal":
        raise ValueError(f"Model refused to classify email {email.id!r}")
    if response.stop_reason == "max_tokens":
        raise ValueError(f"Classification for email {email.id!r} was truncated; raise max_tokens")

    result = response.parsed_output
    return ClassifiedEmail(
        email=email,
        urgency=result.urgency,
        reasoning=result.reasoning,
        queue=ROUTING_QUEUE[result.urgency.value],
    )


def classify_emails(client: anthropic.Anthropic, emails: list[Email]) -> list[ClassifiedEmail]:
    return [classify_email(client, email) for email in emails]


if __name__ == "__main__":
    sample_emails = [
        Email(
            id="1",
            subject="Production API returning 500s for all requests",
            body="Since 10 minutes ago every request to your API returns a 500 error. "
                 "Our checkout flow is completely down and customers can't pay. This is costing us money right now.",
        ),
        Email(
            id="2",
            subject="Question about upgrading my plan",
            body="Hi, I'm considering upgrading from the Starter to the Pro plan. "
                 "Could you send me a comparison of the features?",
        ),
        Email(
            id="3",
            subject="Export button missing on reports page",
            body="The CSV export button on the reports page seems to have disappeared. "
                 "Not urgent, but I use it occasionally and wanted to flag it.",
        ),
    ]

    client = anthropic.Anthropic()
    classified = classify_emails(client, sample_emails)

    for item in sorted(classified, key=lambda c: list(Urgency).index(c.urgency)):
        print(f"[{item.urgency.value.upper():8}] -> {item.queue:16} {item.email.subject!r}")
        print(f"           reason: {item.reasoning}")
