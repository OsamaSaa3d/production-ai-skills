## Summary

Built `triage_agent.py`, a **workflow** (fixed pipeline in code, not an autonomous agent loop — the steps are known in advance so an LLM shouldn't be deciding what to do next):

1. **Extract** — raw report text → structured fields (title, component, steps to reproduce, impact) via `client.messages.parse`
2. **Dedupe** — plain-code lexical shortlist against `issues.json` (`difflib`), then Claude judges which candidate, if any, is a true duplicate (not just superficially similar)
3. **Severity** — classified against a fixed rubric (critical/high/medium/low) baked into the prompt
4. **Route** — assigned to one of 8 fixed teams (frontend, backend, mobile, infra, security, data-ml, billing, support-docs), each with a description Claude routes against

All three LLM steps use Pydantic structured outputs on `claude-opus-5`, so results are always well-formed. Confirmed duplicates (confidence ≥0.7) get recorded as `duplicate_of:<id>` instead of opening a fresh issue.

**Verified locally:** CLI parsing, `--help`, the lexical dedupe shortlist (correctly surfaced the seeded checkout-500-error issue as the top candidate for a similar new report), issue-ID generation, and graceful failure when no API credentials are configured.

**Not verified:** the actual LLM calls (extraction/dedupe-judgment/severity/routing quality) — this sandbox has no `ANTHROPIC_API_KEY` / `ant auth login` session, so I couldn't exercise them live. Run `python triage_agent.py --text "..."` with credentials set to do that.

Files: `triage_agent.py` (pipeline + CLI), `issues_seed.json` (7 sample tracked issues across all severities/teams to seed the store), `requirements.txt`.

### `triage_agent.py`

````py
"""
Bug report triage workflow.

Fixed pipeline, not an autonomous agent: every run does the same four steps
in the same order, driven by this code rather than by model-chosen next actions.

    1. extract   - turn raw report text into structured fields (LLM, structured output)
    2. dedupe    - shortlist similar existing issues (plain code) and let the
                   model judge whether any is a true duplicate (LLM, structured output)
    3. severity  - classify against a fixed rubric (LLM, structured output)
    4. route     - pick a team from a fixed list (LLM, structured output)

Usage:
    python triage_agent.py --file report.txt
    python triage_agent.py --text "Checkout crashes when I click pay with a coupon applied"
    echo "some report text" | python triage_agent.py
"""

from __future__ import annotations

import argparse
import difflib
import json
import re
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Literal, Optional

import anthropic
from pydantic import BaseModel, Field

MODEL = "claude-opus-5"
DEFAULT_STORE_PATH = Path(__file__).parent / "issues.json"
SEED_STORE_PATH = Path(__file__).parent / "issues_seed.json"

TEAMS: dict[str, str] = {
    "frontend": "Web/UI client-side issues: rendering, layout, styling, browser-specific behavior.",
    "backend": "Server-side API and business logic issues not specific to infra or security.",
    "mobile": "iOS/Android native app issues.",
    "infra": "Deployment, CI/CD, hosting, scaling, outages, networking, performance under load.",
    "security": "Vulnerabilities, auth bypass, data exposure, injection, privilege escalation.",
    "data-ml": "Data pipelines, analytics, ML/AI model behavior.",
    "billing": "Payments, invoices, subscriptions, pricing and tax calculations.",
    "support-docs": "Documentation errors, unclear UX copy, non-bug user confusion.",
}

SEVERITY_RUBRIC = """
- critical: data loss/corruption, a security vulnerability, or a complete outage of a core
  flow, with no workaround.
- high: a major feature is broken for a large share of users, or there is a security-adjacent
  concern; a workaround exists but is painful.
- medium: a feature is partially broken or broken only in specific conditions; a reasonable
  workaround exists.
- low: cosmetic, edge-case, or low-traffic-path issue with negligible user impact.
""".strip()


class ExtractedReport(BaseModel):
    title: str = Field(description="Short, specific title for the bug (<=12 words).")
    summary: str = Field(description="1-3 sentence neutral summary of the bug.")
    component: str = Field(description="Best-guess component/area, e.g. checkout, auth, api, mobile-app, frontend, billing, docs.")
    steps_to_reproduce: list[str] = Field(default_factory=list)
    user_impact: str = Field(description="Who is affected and how, in one sentence.")
    environment: Optional[str] = Field(default=None, description="OS/browser/app version/etc. if mentioned, else null.")


class DuplicateMatch(BaseModel):
    is_duplicate: bool
    duplicate_of_id: Optional[str] = Field(default=None, description="ID of the matching existing issue, or null.")
    confidence: float = Field(ge=0.0, le=1.0)
    reasoning: str


class SeverityAssessment(BaseModel):
    severity: Literal["critical", "high", "medium", "low"]
    reasoning: str


class RoutingDecision(BaseModel):
    team: Literal["frontend", "backend", "mobile", "infra", "security", "data-ml", "billing", "support-docs"]
    reasoning: str


class TriageResult(BaseModel):
    extracted: ExtractedReport
    duplicate: DuplicateMatch
    severity: SeverityAssessment
    routing: RoutingDecision
    new_issue_id: Optional[str] = None


def load_store(path: Path) -> list[dict]:
    if path.exists():
        return json.loads(path.read_text(encoding="utf-8"))
    if SEED_STORE_PATH.exists():
        return json.loads(SEED_STORE_PATH.read_text(encoding="utf-8"))
    return []


def save_store(path: Path, issues: list[dict]) -> None:
    path.write_text(json.dumps(issues, indent=2), encoding="utf-8")


def next_issue_id(issues: list[dict]) -> str:
    numbers = []
    for issue in issues:
        match = re.search(r"(\d+)$", issue.get("id", ""))
        if match:
            numbers.append(int(match.group(1)))
    return f"ISSUE-{(max(numbers) + 1) if numbers else 1:03d}"


def find_candidates(extracted: ExtractedReport, issues: list[dict], top_k: int = 5, min_score: float = 0.3) -> list[dict]:
    """Plain-code lexical shortlist - ground truth from the store, no model guessing."""
    query = f"{extracted.title} {extracted.summary} {extracted.component}".lower()
    scored = []
    for issue in issues:
        candidate_text = f"{issue.get('title', '')} {issue.get('summary', '')} {issue.get('component', '')}".lower()
        score = difflib.SequenceMatcher(None, query, candidate_text).ratio()
        if issue.get("component", "").lower() == extracted.component.lower():
            score += 0.15
        if score >= min_score:
            scored.append((score, issue))
    scored.sort(key=lambda pair: pair[0], reverse=True)
    return [issue for _, issue in scored[:top_k]]


def extract_report(client: anthropic.Anthropic, raw_text: str) -> ExtractedReport:
    response = client.messages.parse(
        model=MODEL,
        max_tokens=2048,
        system=(
            "You extract structured fields from a raw, possibly messy bug report. "
            "Do not invent details that are not stated or clearly implied."
        ),
        messages=[{"role": "user", "content": raw_text}],
        output_format=ExtractedReport,
    )
    return response.parsed_output


def check_duplicate(client: anthropic.Anthropic, extracted: ExtractedReport, candidates: list[dict]) -> DuplicateMatch:
    if not candidates:
        return DuplicateMatch(
            is_duplicate=False,
            duplicate_of_id=None,
            confidence=1.0,
            reasoning="No similar existing issues were found in the store.",
        )

    candidate_lines = "\n".join(
        f"- id={c['id']} | title={c['title']} | summary={c['summary']} | component={c.get('component', 'unknown')}"
        for c in candidates
    )
    prompt = f"""New bug report:
title: {extracted.title}
summary: {extracted.summary}
component: {extracted.component}
user_impact: {extracted.user_impact}

Candidate existing issues (found by lexical similarity, may or may not be true duplicates):
{candidate_lines}

Judge whether the new report describes the same underlying bug as one of the candidates
(same root cause / same user-visible symptom), not just a superficially similar area.
If no candidate is a true duplicate, say so."""

    response = client.messages.parse(
        model=MODEL,
        max_tokens=1024,
        system="You are a meticulous bug triage engineer deciding whether a new report duplicates an existing tracked issue.",
        messages=[{"role": "user", "content": prompt}],
        output_format=DuplicateMatch,
    )
    return response.parsed_output


def classify_severity(client: anthropic.Anthropic, extracted: ExtractedReport) -> SeverityAssessment:
    prompt = f"""Bug report:
title: {extracted.title}
summary: {extracted.summary}
component: {extracted.component}
user_impact: {extracted.user_impact}
steps_to_reproduce: {extracted.steps_to_reproduce}
environment: {extracted.environment}

Severity rubric:
{SEVERITY_RUBRIC}

Assign the single best-fitting severity label."""

    response = client.messages.parse(
        model=MODEL,
        max_tokens=1024,
        system="You are a bug triage engineer applying a fixed severity rubric consistently.",
        messages=[{"role": "user", "content": prompt}],
        output_format=SeverityAssessment,
    )
    return response.parsed_output


def route_to_team(client: anthropic.Anthropic, extracted: ExtractedReport, severity: SeverityAssessment) -> RoutingDecision:
    team_lines = "\n".join(f"- {name}: {desc}" for name, desc in TEAMS.items())
    prompt = f"""Bug report:
title: {extracted.title}
summary: {extracted.summary}
component: {extracted.component}
user_impact: {extracted.user_impact}
severity: {severity.severity}

Teams:
{team_lines}

Pick the single team best positioned to own and fix this bug."""

    response = client.messages.parse(
        model=MODEL,
        max_tokens=1024,
        system="You route bug reports to the correct owning team from a fixed list.",
        messages=[{"role": "user", "content": prompt}],
        output_format=RoutingDecision,
    )
    return response.parsed_output


def triage(client: anthropic.Anthropic, raw_text: str, store_path: Path, persist: bool = True) -> TriageResult:
    issues = load_store(store_path)

    extracted = extract_report(client, raw_text)
    candidates = find_candidates(extracted, issues)
    duplicate = check_duplicate(client, extracted, candidates)
    severity = classify_severity(client, extracted)
    routing = route_to_team(client, extracted, severity)

    result = TriageResult(extracted=extracted, duplicate=duplicate, severity=severity, routing=routing)

    if persist:
        if duplicate.is_duplicate and duplicate.confidence >= 0.7:
            # Record as a duplicate reference rather than a new tracked issue.
            new_id = next_issue_id(issues)
            issues.append({
                "id": new_id,
                "title": extracted.title,
                "summary": extracted.summary,
                "component": extracted.component,
                "severity": severity.severity,
                "team": routing.team,
                "status": f"duplicate_of:{duplicate.duplicate_of_id}",
                "created_at": datetime.now(timezone.utc).isoformat(),
            })
            result.new_issue_id = new_id
        else:
            new_id = next_issue_id(issues)
            issues.append({
                "id": new_id,
                "title": extracted.title,
                "summary": extracted.summary,
                "component": extracted.component,
                "severity": severity.severity,
                "team": routing.team,
                "status": "open",
                "created_at": datetime.now(timezone.utc).isoformat(),
            })
            result.new_issue_id = new_id
        save_store(store_path, issues)

    return result


def format_result(result: TriageResult) -> str:
    lines = [
        f"Title:      {result.extracted.title}",
        f"Component:  {result.extracted.component}",
        f"Summary:    {result.extracted.summary}",
        f"Severity:   {result.severity.severity}  ({result.severity.reasoning})",
        f"Team:       {result.routing.team}  ({result.routing.reasoning})",
    ]
    if result.duplicate.is_duplicate:
        lines.append(
            f"Duplicate:  YES, likely duplicate of {result.duplicate.duplicate_of_id} "
            f"(confidence {result.duplicate.confidence:.2f}) - {result.duplicate.reasoning}"
        )
    else:
        lines.append(f"Duplicate:  no ({result.duplicate.reasoning})")
    if result.new_issue_id:
        lines.append(f"Issue ID:   {result.new_issue_id}")
    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(description="Triage an incoming bug report: dedupe, label severity, route to team.")
    parser.add_argument("--text", help="Raw bug report text.")
    parser.add_argument("--file", type=Path, help="Path to a file containing the raw bug report text.")
    parser.add_argument("--store", type=Path, default=DEFAULT_STORE_PATH, help="Path to the JSON issue store (created from issues_seed.json on first run).")
    parser.add_argument("--json", action="store_true", help="Print the full result as JSON instead of a human-readable summary.")
    parser.add_argument("--no-persist", action="store_true", help="Don't write the triaged report back to the store.")
    args = parser.parse_args()

    if args.file:
        raw_text = args.file.read_text(encoding="utf-8")
    elif args.text:
        raw_text = args.text
    elif not sys.stdin.isatty():
        raw_text = sys.stdin.read()
    else:
        parser.error("Provide a report via --text, --file, or stdin.")
        return

    client = anthropic.Anthropic()

    try:
        result = triage(client, raw_text, args.store, persist=not args.no_persist)
    except TypeError as e:
        if "authentication" in str(e).lower():
            print(
                "Error: no Anthropic credentials found. Set ANTHROPIC_API_KEY, "
                "ANTHROPIC_AUTH_TOKEN, or run `ant auth login`.",
                file=sys.stderr,
            )
            sys.exit(1)
        raise
    except anthropic.AuthenticationError:
        print("Error: invalid Anthropic API credentials.", file=sys.stderr)
        sys.exit(1)
    except anthropic.APIStatusError as e:
        print(f"Error: API request failed ({e.status_code}): {e.message}", file=sys.stderr)
        sys.exit(1)

    if args.json:
        print(result.model_dump_json(indent=2))
    else:
        print(format_result(result))


if __name__ == "__main__":
    main()
````
