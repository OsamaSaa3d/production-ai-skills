```
bug_triage/
  models.py       # Pydantic schemas: ExtractedReport, DuplicateJudgment, SeverityResult, RoutingResult, TriageResult
  issue_store.py  # JSON-backed issue store + TF-IDF-ish lexical retrieval for duplicate candidates
  triage.py       # the workflow: extract -> retrieve candidates -> LLM duplicate judgment -> severity -> routing -> file issue
  main.py         # CLI entry point
  data/issues.json # seed set of existing issues to check duplicates against
requirements.txt
```

**Why a workflow, not an autonomous agent:** the steps here are fixed and known in advance (always: read → dedupe → label severity → route), so I hardcoded that sequence in `triage_report()` rather than giving the model a tool loop to improvise through. The model still does the judgment-heavy parts — extraction, duplicate comparison, severity, routing — each as a `messages.parse()` call with a Pydantic schema, but your code owns the control flow, which makes it cheaper, deterministic, and easy to debug against your evals. Duplicate search uses a lexical (TF-IDF-like) retrieval step first to cut the corpus down to a handful of plausible candidates for the LLM to actually judge — it's a routing pattern for candidate selection, not a search agent.

Run it via:
```bash
python -m bug_triage.main --text "Checkout returns a 500 error when I try to apply a discount code..."
```

It needs `ANTHROPIC_API_KEY` set (or an `ant auth login` profile) — neither is available in this environment, so I verified the extraction schemas and the duplicate-candidate retrieval logic offline (both work correctly — a checkout/promo-code report correctly surfaces `BUG-101` as the top match), but I have not run the live Claude calls end-to-end. Try it with your own credentials and let me know if any of the prompts need tuning — severity/team rubrics in `triage.py` and `models.py` are the places to adjust for your actual team names and severity policy.

### `bug_triage/issue_store.py`

````py
import json
import math
import re
from collections import Counter
from pathlib import Path
from typing import List

DATA_PATH = Path(__file__).parent / "data" / "issues.json"

_STOPWORDS = {
    "a", "an", "the", "is", "are", "was", "were", "be", "been", "being", "to", "of", "in",
    "on", "for", "and", "or", "but", "with", "at", "by", "from", "up", "down", "this",
    "that", "it", "its", "as", "when", "after", "before", "some", "any", "user", "users",
    "report", "reports", "reported", "issue", "bug",
}


def _tokenize(text: str) -> List[str]:
    words = re.findall(r"[a-z0-9]+", text.lower())
    return [w for w in words if w not in _STOPWORDS and len(w) > 1]


def load_issues() -> list[dict]:
    return json.loads(DATA_PATH.read_text(encoding="utf-8"))


def save_issues(issues: list[dict]) -> None:
    DATA_PATH.write_text(json.dumps(issues, indent=2), encoding="utf-8")


def next_issue_id(issues: list[dict]) -> str:
    max_num = 0
    for issue in issues:
        match = re.match(r"BUG-(\d+)", issue["id"])
        if match:
            max_num = max(max_num, int(match.group(1)))
    return f"BUG-{max_num + 1}"


def search_candidates(query_text: str, issues: list[dict], top_k: int = 5) -> list[dict]:
    """Lexical (TF-IDF-ish) retrieval of the most similar existing issues.

    This is a cheap pre-filter, not the duplicate decision itself - it narrows
    an unbounded corpus down to a handful of plausible candidates for the LLM
    to actually judge, so the model never has to read every open issue.
    """
    corpus_tokens = [_tokenize(f"{i['title']} {i['description']}") for i in issues]
    doc_count = len(issues) or 1
    df = Counter()
    for tokens in corpus_tokens:
        df.update(set(tokens))

    def idf(term: str) -> float:
        return math.log((doc_count + 1) / (df.get(term, 0) + 1)) + 1.0

    query_tokens = Counter(_tokenize(query_text))
    query_vec = {t: c * idf(t) for t, c in query_tokens.items()}
    query_norm = math.sqrt(sum(v * v for v in query_vec.values())) or 1.0

    scored = []
    for issue, tokens in zip(issues, corpus_tokens):
        doc_tf = Counter(tokens)
        doc_vec = {t: c * idf(t) for t, c in doc_tf.items()}
        doc_norm = math.sqrt(sum(v * v for v in doc_vec.values())) or 1.0
        dot = sum(query_vec.get(t, 0.0) * v for t, v in doc_vec.items())
        score = dot / (query_norm * doc_norm)
        if score > 0:
            scored.append((score, issue))

    scored.sort(key=lambda pair: pair[0], reverse=True)
    return [issue for _, issue in scored[:top_k]]

````

### `bug_triage/main.py`

````py
import argparse
import sys

import anthropic

from .triage import triage_report


def _read_report(args: argparse.Namespace) -> str:
    if args.file:
        return open(args.file, encoding="utf-8").read()
    if args.text:
        return args.text
    if not sys.stdin.isatty():
        return sys.stdin.read()
    raise SystemExit("Provide a report via --file, --text, or stdin.")


def main() -> None:
    parser = argparse.ArgumentParser(description="Triage an incoming bug report.")
    parser.add_argument("--file", help="Path to a text file containing the bug report.")
    parser.add_argument("--text", help="The bug report text, inline.")
    parser.add_argument(
        "--no-file", action="store_true", help="Don't persist a new issue when it's not a duplicate."
    )
    args = parser.parse_args()

    raw_text = _read_report(args)
    client = anthropic.Anthropic()

    try:
        result = triage_report(client, raw_text, file_new_issue=not args.no_file)
    except anthropic.AuthenticationError:
        raise SystemExit("Invalid or missing API key. Set ANTHROPIC_API_KEY or run `ant auth login`.")
    except anthropic.RateLimitError as e:
        retry_after = e.response.headers.get("retry-after", "a bit")
        raise SystemExit(f"Rate limited - retry after {retry_after}s.")
    except anthropic.APIStatusError as e:
        raise SystemExit(f"API error ({e.status_code}): {e.message}")
    except anthropic.APIConnectionError:
        raise SystemExit("Network error reaching the Anthropic API.")

    print(f"Title:      {result.extracted.title}")
    print(f"Component:  {result.extracted.component}")
    print(f"Severity:   {result.severity.severity}  ({result.severity.reasoning})")
    print(f"Team:       {result.routing.team}  ({result.routing.reasoning})")
    if result.duplicate.is_duplicate:
        print(
            f"Duplicate:  yes, of {result.duplicate.duplicate_of_id} "
            f"(confidence {result.duplicate.confidence:.2f}) - {result.duplicate.reasoning}"
        )
        print("No new issue filed.")
    else:
        print(f"Duplicate:  no ({result.duplicate.reasoning})")
        if result.issue_id:
            print(f"Filed as:   {result.issue_id}")


if __name__ == "__main__":
    main()
````

### `bug_triage/models.py`

````py
from typing import List, Literal, Optional

from pydantic import BaseModel, Field

Severity = Literal["critical", "high", "medium", "low"]

TEAM_DESCRIPTIONS = {
    "frontend": "Web UI, client-side rendering, CSS/layout, browser-specific bugs.",
    "backend": "Server-side application logic, internal APIs, business logic, databases.",
    "mobile": "iOS/Android native apps and mobile-specific behavior.",
    "infra": "Deployment, CI/CD, hosting, networking, outages, performance/scaling, monitoring.",
    "security": "Authentication, authorization, data exposure, vulnerabilities, exploits.",
    "payments": "Billing, checkout, invoicing, payment processors, refunds.",
    "data-ml": "Analytics pipelines, data processing, ML models/predictions.",
    "support": "Unclear, needs-more-info, or general reports that don't fit another team.",
}

Team = Literal[
    "frontend", "backend", "mobile", "infra", "security", "payments", "data-ml", "support"
]


class ExtractedReport(BaseModel):
    title: str = Field(description="Short descriptive title for the bug, <= 12 words.")
    summary: str = Field(description="1-3 sentence summary of the problem.")
    component: str = Field(description="Best-guess product area/component, e.g. 'checkout', 'login', 'ios-app'.")
    steps_to_reproduce: List[str] = Field(default_factory=list)
    error_messages: List[str] = Field(default_factory=list, description="Verbatim error text/codes found in the report, if any.")
    affected_version: Optional[str] = None
    environment: Optional[str] = Field(default=None, description="OS/browser/device/environment if mentioned.")


class DuplicateJudgment(BaseModel):
    is_duplicate: bool
    duplicate_of_id: Optional[str] = Field(default=None, description="Issue ID this duplicates, if is_duplicate is true.")
    confidence: float = Field(ge=0.0, le=1.0)
    reasoning: str


class SeverityResult(BaseModel):
    severity: Severity
    reasoning: str


class RoutingResult(BaseModel):
    team: Team
    reasoning: str


class TriageResult(BaseModel):
    extracted: ExtractedReport
    duplicate: DuplicateJudgment
    severity: SeverityResult
    routing: RoutingResult
    issue_id: Optional[str] = Field(default=None, description="ID assigned if a new issue was filed.")
````

### `bug_triage/triage.py`

````py
import json

import anthropic

from . import issue_store
from .models import (
    TEAM_DESCRIPTIONS,
    DuplicateJudgment,
    ExtractedReport,
    RoutingResult,
    SeverityResult,
    TriageResult,
)

MODEL = "claude-opus-5"

SEVERITY_RUBRIC = """\
Classify bug severity using this rubric:
- critical: production outage, data loss/corruption, security exploit, or a core \
flow (login, checkout, payment) completely broken for all/most users with no workaround.
- high: a major feature is broken or badly degraded for a significant subset of \
users, and there is no reasonable workaround.
- medium: a feature is partially broken, affects a limited subset of users, or a \
workaround exists.
- low: cosmetic issue, minor inconvenience, edge case affecting very few users.
"""

TEAM_RUBRIC = "\n".join(f"- {name}: {desc}" for name, desc in TEAM_DESCRIPTIONS.items())


def extract_report(client: anthropic.Anthropic, raw_text: str) -> ExtractedReport:
    response = client.messages.parse(
        model=MODEL,
        max_tokens=2048,
        output_config={"effort": "medium"},
        system=(
            "You extract structured information from a raw incoming bug report. "
            "Only use information present in the report - never invent steps, "
            "versions, or error messages that aren't stated or clearly implied."
        ),
        messages=[{"role": "user", "content": raw_text}],
        output_format=ExtractedReport,
    )
    return response.parsed_output


def judge_duplicate(
    client: anthropic.Anthropic, extracted: ExtractedReport, candidates: list[dict]
) -> DuplicateJudgment:
    if not candidates:
        return DuplicateJudgment(
            is_duplicate=False,
            duplicate_of_id=None,
            confidence=1.0,
            reasoning="No existing issues are lexically similar enough to be plausible candidates.",
        )

    candidates_text = "\n\n".join(
        f"ID: {c['id']}\nTitle: {c['title']}\nDescription: {c['description']}\n"
        f"Component: {c['component']}\nStatus: {c['status']}"
        for c in candidates
    )
    prompt = (
        f"New bug report:\n"
        f"Title: {extracted.title}\n"
        f"Summary: {extracted.summary}\n"
        f"Component: {extracted.component}\n"
        f"Steps to reproduce: {extracted.steps_to_reproduce}\n"
        f"Error messages: {extracted.error_messages}\n\n"
        f"Candidate existing issues (already narrowed by keyword similarity):\n\n"
        f"{candidates_text}\n\n"
        "Does the new report describe the same underlying problem as one of these "
        "existing issues? Judge by root cause and user-visible symptom, not just "
        "wording overlap - e.g. two different error messages from the same broken "
        "flow are still a duplicate, while similar wording about a different "
        "component is not."
    )
    response = client.messages.parse(
        model=MODEL,
        max_tokens=1024,
        output_config={"effort": "medium"},
        system=(
            "You are a bug triage assistant deciding whether a new bug report is a "
            "duplicate of an already-tracked issue. Only mark is_duplicate true when "
            "you are reasonably confident they share the same root cause."
        ),
        messages=[{"role": "user", "content": prompt}],
        output_format=DuplicateJudgment,
    )
    return response.parsed_output


def classify_severity(client: anthropic.Anthropic, extracted: ExtractedReport) -> SeverityResult:
    prompt = (
        f"{SEVERITY_RUBRIC}\n\nBug report:\n"
        f"Title: {extracted.title}\n"
        f"Summary: {extracted.summary}\n"
        f"Component: {extracted.component}\n"
        f"Steps to reproduce: {extracted.steps_to_reproduce}\n"
        f"Error messages: {extracted.error_messages}\n"
        f"Affected version: {extracted.affected_version}\n"
    )
    response = client.messages.parse(
        model=MODEL,
        max_tokens=512,
        output_config={"effort": "low"},
        system="You are a bug triage assistant assigning a severity label to a bug report.",
        messages=[{"role": "user", "content": prompt}],
        output_format=SeverityResult,
    )
    return response.parsed_output


def route_team(client: anthropic.Anthropic, extracted: ExtractedReport) -> RoutingResult:
    prompt = (
        f"Teams and what they own:\n{TEAM_RUBRIC}\n\nBug report:\n"
        f"Title: {extracted.title}\n"
        f"Summary: {extracted.summary}\n"
        f"Component: {extracted.component}\n"
        f"Error messages: {extracted.error_messages}\n"
        f"Environment: {extracted.environment}\n\n"
        "Pick exactly one team to route this to. Use 'support' only when no other "
        "team clearly owns the problem area."
    )
    response = client.messages.parse(
        model=MODEL,
        max_tokens=512,
        output_config={"effort": "low"},
        system="You are a bug triage assistant routing a bug report to the right team.",
        messages=[{"role": "user", "content": prompt}],
        output_format=RoutingResult,
    )
    return response.parsed_output


def triage_report(client: anthropic.Anthropic, raw_text: str, file_new_issue: bool = True) -> TriageResult:
    extracted = extract_report(client, raw_text)

    issues = issue_store.load_issues()
    candidates = issue_store.search_candidates(f"{extracted.title} {extracted.summary}", issues)
    duplicate = judge_duplicate(client, extracted, candidates)

    severity = classify_severity(client, extracted)
    routing = route_team(client, extracted)

    issue_id = None
    if not duplicate.is_duplicate and file_new_issue:
        issue_id = issue_store.next_issue_id(issues)
        issues.append(
            {
                "id": issue_id,
                "title": extracted.title,
                "description": extracted.summary,
                "component": extracted.component,
                "severity": severity.severity,
                "team": routing.team,
                "status": "open",
            }
        )
        issue_store.save_issues(issues)

    return TriageResult(
        extracted=extracted,
        duplicate=duplicate,
        severity=severity,
        routing=routing,
        issue_id=issue_id,
    )
````
