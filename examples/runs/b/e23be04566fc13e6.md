- **`orders_db.py`** — connection helper (SQLite default; swap for your real driver, e.g. psycopg for Postgres).
- **`orders_tool.py`** — `QUERY_ORDERS_TOOL` schema (`strict: true`) and `query_orders()`. The model passes structured filters (order_id, customer_id, status, date range), never raw SQL, and the function builds a parameterized query — no injection surface. `status` and `limit` are re-validated in code since strict mode doesn't reliably enforce enums/ranges through `anyOf`-wrapped nullable fields.
- **`orders_step.py`** — `query_orders_step`, the LangGraph node. It's a plain function that calls the model directly with the tool (bounded loop, max 4 iterations), executes `query_orders` when called, and feeds the result back as a `role: "tool"` message. No LangChain agent executor — the loop is explicit and errors come back as data, not exceptions.

````py
"""Connection to the orders database.

This uses sqlite3 as a concrete, runnable default. Swap this module for your
real driver (psycopg, mysqlclient, etc.) — the parameterized-query shape in
orders_tool.py doesn't change.
"""

import os
import sqlite3
from contextlib import contextmanager

ORDERS_DB_PATH = os.environ.get("ORDERS_DB_PATH", "orders.db")


@contextmanager
def get_connection():
    conn = sqlite3.connect(ORDERS_DB_PATH)
    try:
        yield conn
    finally:
        conn.close()
````

### `orders_step.py`

````py
"""LangGraph step that lets the model query the orders table.

Wire it into an existing graph like:

    from orders_step import query_orders_step

    graph.add_node("query_orders", query_orders_step)
    graph.add_edge("previous_step", "query_orders")
    graph.add_edge("query_orders", "next_step")

The graph's checkpointer (whatever you pass to graph.compile(checkpointer=...))
already persists this node's state between steps, so no extra durability code
is needed here.

Tool calling is done directly against the OpenAI-compatible API with
strict mode, rather than through a LangChain agent executor, so the loop
stays inspectable and errors surface as data instead of being swallowed.
"""

import json
import os

from langgraph.graph import MessagesState
from openai import OpenAI

from orders_db import get_connection
from orders_tool import QUERY_ORDERS_TOOL, query_orders

MODEL = os.environ.get("ORDERS_STEP_MODEL", "gpt-4.1")
MAX_TOOL_ITERATIONS = 4

client = OpenAI()


def query_orders_step(state: MessagesState) -> MessagesState:
    messages = list(state["messages"])

    for _ in range(MAX_TOOL_ITERATIONS):
        response = client.chat.completions.create(
            model=MODEL,
            messages=messages,
            tools=[QUERY_ORDERS_TOOL],
            tool_choice="auto",
        )
        choice = response.choices[0]

        if choice.finish_reason == "length":
            raise RuntimeError("query_orders_step: response truncated mid-generation")

        message = choice.message
        messages.append(message)

        if not message.tool_calls:
            break

        with get_connection() as conn:
            for tool_call in message.tool_calls:
                if tool_call.function.name != "query_orders":
                    result = {"error": f"unknown tool: {tool_call.function.name}"}
                else:
                    try:
                        args = json.loads(tool_call.function.arguments)
                        result = query_orders(conn, **args)
                    except Exception as e:
                        result = {"error": str(e)}

                messages.append(
                    {
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "content": json.dumps(result),
                    }
                )
    else:
        raise RuntimeError("query_orders_step: exceeded max tool iterations")

    return {"messages": messages}

````

### `orders_tool.py`

````py
"""Tool definition and implementation for querying the orders table.

The model gets structured filters, never raw SQL — query_orders builds a
parameterized statement from them, so there's no injection surface.
"""

ALLOWED_STATUSES = {"pending", "paid", "shipped", "delivered", "cancelled", "refunded"}
MAX_LIMIT = 50
DEFAULT_LIMIT = 10

QUERY_ORDERS_TOOL = {
    "type": "function",
    "function": {
        "name": "query_orders",
        "description": (
            "Look up rows in the orders table. Filter by any combination of "
            "order_id, customer_id, and status, and optionally restrict to a "
            "created_at date range. Returns at most `limit` rows, most recent "
            "first. Use this whenever the user asks about a specific order, an "
            "order's status, or a customer's order history."
        ),
        "strict": True,
        "parameters": {
            "type": "object",
            "properties": {
                "order_id": {
                    "anyOf": [{"type": "integer"}, {"type": "null"}],
                    "description": "Exact order ID to look up, or null to not filter by it.",
                },
                "customer_id": {
                    "anyOf": [{"type": "integer"}, {"type": "null"}],
                    "description": "Exact customer ID to filter by, or null.",
                },
                "status": {
                    "anyOf": [
                        {
                            "type": "string",
                            "enum": sorted(ALLOWED_STATUSES),
                        },
                        {"type": "null"},
                    ],
                    "description": "Order status to filter by, or null for any status.",
                },
                "created_after": {
                    "anyOf": [{"type": "string"}, {"type": "null"}],
                    "description": "ISO-8601 date (YYYY-MM-DD). Only return orders created on or after this date, or null.",
                },
                "created_before": {
                    "anyOf": [{"type": "string"}, {"type": "null"}],
                    "description": "ISO-8601 date (YYYY-MM-DD). Only return orders created on or before this date, or null.",
                },
                "limit": {
                    "type": "integer",
                    "description": f"Max rows to return, 1-{MAX_LIMIT}.",
                },
            },
            "required": [
                "order_id",
                "customer_id",
                "status",
                "created_after",
                "created_before",
                "limit",
            ],
            "additionalProperties": False,
        },
    },
}


def query_orders(
    conn,
    *,
    order_id=None,
    customer_id=None,
    status=None,
    created_after=None,
    created_before=None,
    limit=DEFAULT_LIMIT,
):
    # Value constraints (enum, range) aren't reliably enforced by strict mode,
    # so re-check them here rather than trusting the schema.
    if status is not None and status not in ALLOWED_STATUSES:
        return {"error": f"invalid status: {status!r}"}

    try:
        limit = int(limit) if limit is not None else DEFAULT_LIMIT
    except (TypeError, ValueError):
        return {"error": f"invalid limit: {limit!r}"}
    limit = max(1, min(limit, MAX_LIMIT))

    clauses = []
    params = []
    if order_id is not None:
        clauses.append("id = ?")
        params.append(order_id)
    if customer_id is not None:
        clauses.append("customer_id = ?")
        params.append(customer_id)
    if status is not None:
        clauses.append("status = ?")
        params.append(status)
    if created_after is not None:
        clauses.append("created_at >= ?")
        params.append(created_after)
    if created_before is not None:
        clauses.append("created_at <= ?")
        params.append(created_before)

    where = f"WHERE {' AND '.join(clauses)}" if clauses else ""
    sql = (
        "SELECT id, customer_id, product, quantity, status, total, created_at "
        f"FROM orders {where} ORDER BY created_at DESC LIMIT ?"
    )
    params.append(limit)

    cursor = conn.execute(sql, params)
    columns = [c[0] for c in cursor.description]
    rows = [dict(zip(columns, row)) for row in cursor.fetchall()]
    return {"rows": rows, "count": len(rows)}
````
