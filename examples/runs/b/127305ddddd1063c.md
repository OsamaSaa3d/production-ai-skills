````py
import smtplib
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText

from summary import DailySummary


def render_html(summary: DailySummary, narrative: str | None) -> str:
    change_html = ""
    if summary.revenue_change_pct is not None:
        arrow = "&#9650;" if summary.revenue_change_pct >= 0 else "&#9660;"
        change_html = f"<p>{arrow} {summary.revenue_change_pct:+.1f}% vs. previous day</p>"

    top_products_rows = "".join(
        f"<tr><td>{name}</td><td style='text-align:right'>${rev:,.2f}</td></tr>"
        for name, rev in summary.top_products
    )

    category_rows = "".join(
        f"<tr><td>{cat}</td><td style='text-align:right'>${rev:,.2f}</td></tr>"
        for cat, rev in summary.category_breakdown.items()
    )
    category_section = (
        f"<h3>By Category</h3><table cellpadding='4'>{category_rows}</table>"
        if category_rows
        else ""
    )

    narrative_html = f"<p style='font-size:14px'>{narrative}</p>" if narrative else ""

    return f"""\
<html>
<body style="font-family: Arial, sans-serif; color:#222;">
  <h2>Sales Summary &mdash; {summary.date}</h2>
  {narrative_html}
  <table cellpadding="4">
    <tr><td><b>Total Revenue</b></td><td style="text-align:right">${summary.total_revenue:,.2f}</td></tr>
    <tr><td><b>Orders</b></td><td style="text-align:right">{summary.order_count}</td></tr>
    <tr><td><b>Units Sold</b></td><td style="text-align:right">{summary.total_units}</td></tr>
    <tr><td><b>Avg Order Value</b></td><td style="text-align:right">${summary.avg_order_value:,.2f}</td></tr>
  </table>
  {change_html}
  <h3>Top Products</h3>
  <table cellpadding="4">{top_products_rows}</table>
  {category_section}
</body>
</html>
"""


def send_email(subject: str, html_body: str, smtp_host: str, smtp_port: int,
                smtp_user: str, smtp_password: str, email_from: str, email_to: list[str]) -> None:
    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = email_from
    msg["To"] = ", ".join(email_to)
    msg.attach(MIMEText(html_body, "html"))

    with smtplib.SMTP(smtp_host, smtp_port) as server:
        server.starttls()
        server.login(smtp_user, smtp_password)
        server.sendmail(email_from, email_to, msg.as_string())
````

### `sales_summary_agent/main.py`

````py
import argparse
import logging
import sys
from datetime import date

from config import load_config
from emailer import render_html, send_email
from summary import compute_summary, load_sales_csv

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
logger = logging.getLogger(__name__)


def run(target_date: str, csv_override: str | None) -> None:
    cfg = load_config()
    csv_path = csv_override or cfg.sales_csv_path.format(date=target_date)

    logger.info("Reading sales CSV: %s", csv_path)
    df = load_sales_csv(csv_path)

    summary = compute_summary(df, target_date)
    logger.info(
        "Computed summary: revenue=$%.2f orders=%d units=%d",
        summary.total_revenue, summary.order_count, summary.total_units,
    )

    narrative = None
    if cfg.enable_ai_narrative:
        if not cfg.anthropic_api_key:
            logger.warning("ENABLE_AI_NARRATIVE=true but ANTHROPIC_API_KEY is unset; skipping narrative")
        else:
            from narrative import generate_narrative
            narrative = generate_narrative(summary, cfg.anthropic_api_key)

    html = render_html(summary, narrative)
    subject = f"{cfg.email_subject_prefix} {target_date}"

    send_email(
        subject=subject,
        html_body=html,
        smtp_host=cfg.smtp_host,
        smtp_port=cfg.smtp_port,
        smtp_user=cfg.smtp_user,
        smtp_password=cfg.smtp_password,
        email_from=cfg.email_from,
        email_to=cfg.email_to,
    )
    logger.info("Email sent to %s", ", ".join(cfg.email_to))


def main() -> None:
    parser = argparse.ArgumentParser(description="Email the team a daily sales summary.")
    parser.add_argument("--date", default=date.today().isoformat(), help="Target date, YYYY-MM-DD (default: today)")
    parser.add_argument("--csv", default=None, help="Explicit CSV path, overrides SALES_CSV_PATH")
    args = parser.parse_args()

    try:
        run(args.date, args.csv)
    except Exception:
        logger.exception("Daily sales summary run failed")
        sys.exit(1)


if __name__ == "__main__":
    main()

````

### `sales_summary_agent/narrative.py`

````py
"""Optional single LLM call that turns the computed stats into a short narrative blurb.

This is deliberately NOT an agent loop: one request, one response, no tools,
no retries-until-satisfied. If it fails or is disabled, the caller falls back
to the plain stats email.
"""
import logging

import anthropic

from summary import DailySummary

logger = logging.getLogger(__name__)

_MODEL = "claude-opus-5"


def generate_narrative(summary: DailySummary, api_key: str) -> str | None:
    client = anthropic.Anthropic(api_key=api_key)

    top_products_str = ", ".join(f"{name} (${rev:,.2f})" for name, rev in summary.top_products)
    change_str = (
        f"{summary.revenue_change_pct:+.1f}% vs. the previous recorded day"
        if summary.revenue_change_pct is not None
        else "no prior-day data to compare against"
    )

    prompt = (
        f"Write a 2-3 sentence plain-English summary of yesterday's sales for a team email. "
        f"Facts: total revenue ${summary.total_revenue:,.2f}, {summary.order_count} orders, "
        f"{summary.total_units} units sold, revenue change {change_str}. "
        f"Top products by revenue: {top_products_str}. "
        f"Be concise and factual, no headers, no bullet points, no greeting or sign-off."
    )

    try:
        response = client.messages.create(
            model=_MODEL,
            max_tokens=300,
            output_config={"effort": "low"},
            messages=[{"role": "user", "content": prompt}],
        )
    except anthropic.APIError as exc:
        logger.warning("Narrative generation failed, falling back to stats-only email: %s", exc)
        return None

    for block in response.content:
        if block.type == "text":
            return block.text.strip()
    return None

````

````md
# Daily Sales Summary

Reads the daily sales CSV, computes a summary (revenue, orders, units, top
products, category breakdown, day-over-day change), and emails it to the
team. Runs once per invocation and is triggered daily by the OS scheduler —
it is a fixed workflow, not an autonomous agent: there's no open-ended
decision loop, so a scheduled script is simpler and more predictable than
an agent would be.

## How it works

1. `summary.py` — loads the CSV (pandas) and computes deterministic stats.
   Day-over-day % change is derived from `data/history.json`, which the
   script appends to on every run — no need to hand it yesterday's file.
2. `narrative.py` — **optional** single call to Claude that turns the stats
   into a 2-3 sentence blurb for the top of the email. One request, no
   tools, no loop. If it fails or `ENABLE_AI_NARRATIVE=false`, the email
   still sends with just the stats table.
3. `emailer.py` — renders HTML and sends it via SMTP.
4. `main.py` — orchestrates the above, logs progress, and exits non-zero
   on failure (so Task Scheduler/cron marks the run as failed).
````

### `sales_summary_agent/summary.py`

````py
"""Deterministic sales-CSV aggregation. No LLM calls in this module."""
import json
from dataclasses import dataclass
from pathlib import Path

import pandas as pd

HISTORY_PATH = Path(__file__).parent / "data" / "history.json"


@dataclass(frozen=True)
class DailySummary:
    date: str
    total_revenue: float
    total_units: int
    order_count: int
    avg_order_value: float
    top_products: list[tuple[str, float]]
    category_breakdown: dict[str, float]
    revenue_change_pct: float | None  # vs. previous recorded day, if any


def load_sales_csv(path: str) -> pd.DataFrame:
    df = pd.read_csv(path)
    df.columns = [c.strip().lower() for c in df.columns]

    required = {"product", "quantity"}
    missing = required - set(df.columns)
    if missing:
        raise ValueError(f"CSV is missing required column(s): {sorted(missing)}")

    if "revenue" not in df.columns:
        if "unit_price" not in df.columns:
            raise ValueError("CSV needs either a 'revenue' column or a 'unit_price' column")
        df["revenue"] = df["quantity"] * df["unit_price"]

    return df


def _load_history() -> dict:
    if HISTORY_PATH.exists():
        return json.loads(HISTORY_PATH.read_text())
    return {}


def _save_history(history: dict) -> None:
    HISTORY_PATH.parent.mkdir(parents=True, exist_ok=True)
    HISTORY_PATH.write_text(json.dumps(history, indent=2, sort_keys=True))


def compute_summary(df: pd.DataFrame, date: str, top_n: int = 5) -> DailySummary:
    total_revenue = round(float(df["revenue"].sum()), 2)
    total_units = int(df["quantity"].sum())
    order_count = int(df["order_id"].nunique()) if "order_id" in df.columns else int(len(df))
    avg_order_value = round(total_revenue / order_count, 2) if order_count else 0.0

    top_products = [
        (name, round(float(rev), 2))
        for name, rev in df.groupby("product")["revenue"].sum().sort_values(ascending=False).head(top_n).items()
    ]

    category_breakdown = {}
    if "category" in df.columns:
        category_breakdown = {
            str(cat): round(float(rev), 2)
            for cat, rev in df.groupby("category")["revenue"].sum().sort_values(ascending=False).items()
        }

    history = _load_history()
    previous_dates = sorted(d for d in history if d < date)
    revenue_change_pct = None
    if previous_dates:
        prev_revenue = history[previous_dates[-1]]["total_revenue"]
        if prev_revenue:
            revenue_change_pct = round((total_revenue - prev_revenue) / prev_revenue * 100, 1)

    history[date] = {"total_revenue": total_revenue, "order_count": order_count}
    _save_history(history)

    return DailySummary(
        date=date,
        total_revenue=total_revenue,
        total_units=total_units,
        order_count=order_count,
        avg_order_value=avg_order_value,
        top_products=top_products,
        category_breakdown=category_breakdown,
        revenue_change_pct=revenue_change_pct,
    )
````
