import anthropic

MODEL = "claude-opus-5"

SYSTEM_PROMPT = (
    "You summarize meeting transcripts for a busy manager who was not present. "
    "Write a single short paragraph (3-5 sentences, no headers or bullet points) "
    "covering: what the meeting was about, the key decisions made, and any action "
    "items with owners if mentioned. Be concise and skip small talk or side "
    "tangents. If owners or deadlines aren't stated in the transcript, don't "
    "invent them."
)


def summarize_meeting(transcript: str, client: anthropic.Anthropic | None = None) -> str:
    """Summarize a meeting transcript into a short skimmable paragraph."""
    if not transcript.strip():
        raise ValueError("transcript is empty")

    client = client or anthropic.Anthropic()

    try:
        response = client.messages.create(
            model=MODEL,
            max_tokens=1024,
            output_config={"effort": "medium"},
            system=SYSTEM_PROMPT,
            messages=[{"role": "user", "content": transcript}],
        )
    except anthropic.APIStatusError as e:
        raise RuntimeError(f"Claude API error ({e.status_code}): {e.message}") from e
    except anthropic.APIConnectionError as e:
        raise RuntimeError("Could not reach the Claude API") from e

    if response.stop_reason == "refusal":
        raise RuntimeError("Summary request was refused by the model")

    return next(block.text for block in response.content if block.type == "text")


if __name__ == "__main__":
    example_transcript = """
    Alice: Let's kick off the Q3 roadmap review. First up, the mobile redesign.
    Bob: We're on track for the beta launch on Sept 30th. Need design sign-off by next Friday.
    Alice: I'll get design to review by then. Carol, what's the status on the API migration?
    Carol: About 80% done. Blocked on the auth team's rate-limit changes. I'll follow up with them today.
    Alice: Great, let's sync again next week. Thanks everyone.
    """
    print(summarize_meeting(example_transcript))
