Created `ticket_search_tool.py` — a `tickets_search` tool your agent can call.

Key design choice (per the tool-design skill): the agent never emits the DSL string. It passes typed fields — `status` (enum), `team`, `created_after`/`created_before` (validated `YYYY-MM-DD`), `text`, `limit` — and `build_query()` compiles those into `status:open AND team:platform AND created:>2026-01-01` in code. This closes off DSL injection, makes invalid states unrepresentable (bad status can't reach the endpoint), and keeps date-format mistakes out of the agent's hands.

Other things baked in:
- Results are trimmed to `id, title, status, team, created_at, url` — no raw internal fields.
- `limit` capped at 50; if the endpoint reports more matches than returned, the response includes a `note` telling the agent to narrow filters rather than crank the limit.
- Errors are actionable strings ("retry with limit=50", "check field values") instead of tracebacks.
- `input_examples` cover minimal/partial/full usage since date format and quoting conventions aren't obvious from the schema alone.

Assumptions I made without confirmation (flag if wrong): endpoint is `GET <TICKET_SEARCH_URL>?q=...&limit=...` with bearer-token auth via `TICKET_SEARCH_API_KEY`/`TICKET_SEARCH_URL` env vars, and `status` only takes the 5 values listed. """
Ticket search tool for an LLM agent.

Design choice: the agent never writes the ticket DSL string itself. It supplies
typed, validated fields; this module compiles them into the
`field:value AND field:>value` syntax the search endpoint expects. This keeps
injection and malformed-query risk out of model output entirely — the worst
a bad argument can do is produce a wrong-but-well-formed query over an
allowlisted set of fields.
"""

from __future__ import annotations

import os
import re
from dataclasses import dataclass
from datetime import date
from typing import Any, Optional

import requests

TICKET_SEARCH_URL = os.environ["TICKET_SEARCH_URL"]  # e.g. https://tickets.internal/api/search
TICKET_SEARCH_API_KEY = os.environ["TICKET_SEARCH_API_KEY"]

STATUS_VALUES = ("open", "closed", "pending", "in_progress", "resolved")

MAX_LIMIT = 50
DEFAULT_LIMIT = 20

_DATE_RE = re.compile(r"^\d{4}-\d{2}-\d{2}$")

# High-signal fields returned to the agent. The endpoint returns more
# (internal IDs, watcher lists, raw timestamps); everything else is dropped
# so the agent isn't scanning noise to find what it needs.
_RESULT_FIELDS = ("id", "title", "status", "team", "created_at", "url")


TOOL_DEFINITION: dict[str, Any] = {
    "name": "tickets_search",
    "description": (
        "Search tickets by status, team, creation date, and free text. "
        "Returns id, title, status, team, created_at, and url for each match, "
        "newest first. Combines all provided filters with AND — there is no "
        "way to express OR/NOT; if you need those, run separate searches."
    ),
    "input_schema": {
        "type": "object",
        "properties": {
            "text": {
                "type": "string",
                "description": "Free-text match against ticket title and body.",
            },
            "status": {
                "type": "string",
                "enum": list(STATUS_VALUES),
                "description": "Exact ticket status.",
            },
            "team": {
                "type": "string",
                "description": "Owning team slug, e.g. 'platform', 'billing'.",
            },
            "created_after": {
                "type": "string",
                "format": "date",
                "description": "Only tickets created on or after this date (YYYY-MM-DD).",
            },
            "created_before": {
                "type": "string",
                "format": "date",
                "description": "Only tickets created on or before this date (YYYY-MM-DD).",
            },
            "limit": {
                "type": "integer",
                "description": f"Max results to return (1-{MAX_LIMIT}). Defaults to {DEFAULT_LIMIT}.",
            },
        },
        "additionalProperties": False,
    },
    "input_examples": [
        {"team": "platform", "status": "open", "created_after": "2026-01-01"},
        {"text": "login page 500 error", "status": "open"},
        {"team": "billing", "created_after": "2026-06-01", "created_before": "2026-06-30", "limit": 10},
    ],
}


class TicketSearchError(Exception):
    """Raised with a message meant to be shown back to the agent verbatim."""


def _quote(value: str) -> str:
    return '"' + value.replace('"', '\\"') + '"'


def _validate_date(value: str, field: str) -> str:
    if not _DATE_RE.match(value):
        raise TicketSearchError(
            f"{field} must be in YYYY-MM-DD format, got {value!r}. "
            f"Retry with e.g. {field}='2026-01-01'."
        )
    return value


def build_query(
    text: Optional[str] = None,
    status: Optional[str] = None,
    team: Optional[str] = None,
    created_after: Optional[str] = None,
    created_before: Optional[str] = None,
) -> str:
    clauses: list[str] = []

    if status is not None:
        if status not in STATUS_VALUES:
            raise TicketSearchError(
                f"status must be one of {STATUS_VALUES}, got {status!r}."
            )
        clauses.append(f"status:{status}")

    if team is not None:
        clauses.append(f"team:{_quote(team)}")

    if created_after is not None:
        clauses.append(f"created:>{_validate_date(created_after, 'created_after')}")

    if created_before is not None:
        clauses.append(f"created:<{_validate_date(created_before, 'created_before')}")

    if text is not None:
        clauses.append(_quote(text))

    if not clauses:
        raise TicketSearchError(
            "At least one of text, status, team, created_after, or created_before is required."
        )

    return " AND ".join(clauses)


def tickets_search(
    text: Optional[str] = None,
    status: Optional[str] = None,
    team: Optional[str] = None,
    created_after: Optional[str] = None,
    created_before: Optional[str] = None,
    limit: Optional[int] = None,
) -> dict[str, Any]:
    """Tool implementation. Returns a JSON-serializable dict for the tool result."""
    resolved_limit = DEFAULT_LIMIT if limit is None else limit
    if not (1 <= resolved_limit <= MAX_LIMIT):
        return {
            "error": (
                f"limit must be between 1 and {MAX_LIMIT}, got {resolved_limit}. "
                f"Retry with limit={MAX_LIMIT} and page through results if you need more."
            )
        }

    try:
        query = build_query(text, status, team, created_after, created_before)
    except TicketSearchError as exc:
        return {"error": str(exc)}

    try:
        response = requests.get(
            TICKET_SEARCH_URL,
            params={"q": query, "limit": resolved_limit},
            headers={"Authorization": f"Bearer {TICKET_SEARCH_API_KEY}"},
            timeout=10,
        )
        response.raise_for_status()
    except requests.exceptions.Timeout:
        return {"error": "Ticket search timed out. Retry with a narrower query (add status or team)."}
    except requests.exceptions.HTTPError as exc:
        status_code = exc.response.status_code if exc.response is not None else None
        if status_code == 400:
            return {"error": f"Ticket search rejected the query {query!r} as invalid. Check field values and retry."}
        return {"error": f"Ticket search failed ({status_code}). Retry once; if it persists, report the outage."}
    except requests.exceptions.RequestException:
        return {"error": "Could not reach the ticket search service. Retry shortly."}

    payload = response.json()
    raw_results = payload.get("results", [])
    total = payload.get("total", len(raw_results))

    trimmed = [{k: item.get(k) for k in _RESULT_FIELDS} for item in raw_results]

    result: dict[str, Any] = {"query": query, "count": len(trimmed), "results": trimmed}
    if total > len(trimmed):
        result["note"] = (
            f"{total} tickets matched; showing {len(trimmed)}. "
            "Narrow the filters (status, team, or date range) rather than raising limit further."
        )
    return result
