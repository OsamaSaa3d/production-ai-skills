````md
# Daily sales summary

A scheduled script — not an autonomous agent. Every run does the same fixed
steps: read the CSV, aggregate numbers, optionally ask Claude for a one-line
narrative, build an email, send it. There's no loop and nothing decides its
own next step, so failures are easy to reproduce and debug.

## Setup

```powershell
pip install -r requirements.txt
copy .env.example .env
notepad .env   # fill in SMTP_* and TO_EMAILS at minimum
```

Point `CSV_PATH` in `.env` at wherever your POS/e-commerce system drops the
daily export (a sample file is at `data\daily_sales.csv` for testing).
Expected columns: `date, product, category, region, quantity, unit_price`
(optionally `total`). If your export uses different column names, edit the
`raw.get(...)` calls near the top of `load_csv` in `report.py`.

## Test it once

```powershell
python report.py
```

Check `logs\daily_sales_report.log` and your inbox. If SMTP fails, the script
tries to email the team an `[ALERT]` message instead of failing silently.
````

### `report.py`

````py
"""
Daily sales summary workflow.

Fixed pipeline (no autonomous decision-making):
  1. Load today's sales CSV
  2. Aggregate totals, top products/categories/regions, day-over-day change
  3. Optionally ask an LLM for a 2-3 sentence narrative of the numbers
  4. Build an HTML/text email
  5. Send it to the team via SMTP

Run daily via Windows Task Scheduler (see README.md).

Expected CSV columns: date, product, category, region, quantity, unit_price[, total]
`total` is computed as quantity * unit_price when the column is absent.
Adjust `load_csv`/`compute_summary` if your export uses different column names.
"""

from __future__ import annotations

import csv
import json
import logging
import os
import smtplib
import sys
from collections import defaultdict
from datetime import date, datetime
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from pathlib import Path

try:
    from dotenv import load_dotenv

    load_dotenv()
except ImportError:
    pass

BASE_DIR = Path(__file__).resolve().parent

CSV_PATH = Path(os.environ.get("CSV_PATH", BASE_DIR / "data" / "daily_sales.csv"))
HISTORY_PATH = Path(os.environ.get("HISTORY_PATH", BASE_DIR / "data" / "sales_history.json"))
LOG_PATH = Path(os.environ.get("LOG_PATH", BASE_DIR / "logs" / "daily_sales_report.log"))

SMTP_HOST = os.environ.get("SMTP_HOST", "")
SMTP_PORT = int(os.environ.get("SMTP_PORT", "587"))
SMTP_USER = os.environ.get("SMTP_USER", "")
SMTP_PASSWORD = os.environ.get("SMTP_PASSWORD", "")
USE_TLS = os.environ.get("SMTP_USE_TLS", "true").lower() == "true"
FROM_EMAIL = os.environ.get("FROM_EMAIL", SMTP_USER)
TO_EMAILS = [e.strip() for e in os.environ.get("TO_EMAILS", "").split(",") if e.strip()]

ENABLE_LLM_SUMMARY = os.environ.get("ENABLE_LLM_SUMMARY", "false").lower() == "true"
ANTHROPIC_API_KEY = os.environ.get("ANTHROPIC_API_KEY", "")
TOP_N = int(os.environ.get("TOP_N", "5"))


def setup_logging() -> None:
    LOG_PATH.parent.mkdir(parents=True, exist_ok=True)
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(message)s",
        handlers=[logging.FileHandler(LOG_PATH, encoding="utf-8"), logging.StreamHandler()],
    )


def _to_float(value: str, field: str, row_num: int) -> float:
    try:
        return float(value)
    except (TypeError, ValueError):
        raise ValueError(f"row {row_num}: could not parse '{field}'={value!r} as a number")


def load_csv(path: Path) -> list[dict]:
    if not path.exists():
        raise FileNotFoundError(f"sales CSV not found at {path}")

    rows = []
    skipped = 0
    with path.open(newline="", encoding="utf-8-sig") as f:
        reader = csv.DictReader(f)
        for i, raw in enumerate(reader, start=2):  # header is line 1
            try:
                quantity = _to_float(raw.get("quantity", "0"), "quantity", i)
                unit_price = _to_float(raw.get("unit_price", "0"), "unit_price", i)
                total = raw.get("total")
                total = _to_float(total, "total", i) if total not in (None, "") else quantity * unit_price
                rows.append(
                    {
                        "date": raw.get("date", ""),
                        "product": raw.get("product", "Unknown"),
                        "category": raw.get("category", "Uncategorized"),
                        "region": raw.get("region", "Unspecified"),
                        "quantity": quantity,
                        "unit_price": unit_price,
                        "total": total,
                    }
                )
            except ValueError as e:
                logging.warning("skipping bad row: %s", e)
                skipped += 1

    if not rows:
        raise ValueError(f"no usable rows parsed from {path} ({skipped} skipped)")

    logging.info("loaded %d rows from %s (%d skipped)", len(rows), path, skipped)
    return rows


def _top_n(totals: dict[str, float], n: int) -> list[tuple[str, float]]:
    return sorted(totals.items(), key=lambda kv: kv[1], reverse=True)[:n]


def compute_summary(rows: list[dict]) -> dict:
    revenue_by_product: dict[str, float] = defaultdict(float)
    revenue_by_category: dict[str, float] = defaultdict(float)
    revenue_by_region: dict[str, float] = defaultdict(float)

    total_revenue = 0.0
    total_units = 0.0

    for r in rows:
        revenue_by_product[r["product"]] += r["total"]
        revenue_by_category[r["category"]] += r["total"]
        revenue_by_region[r["region"]] += r["total"]
        total_revenue += r["total"]
        total_units += r["quantity"]

    total_orders = len(rows)
    avg_order_value = total_revenue / total_orders if total_orders else 0.0

    return {
        "report_date": date.today().isoformat(),
        "total_revenue": round(total_revenue, 2),
        "total_orders": total_orders,
        "total_units": total_units,
        "avg_order_value": round(avg_order_value, 2),
        "top_products": [(k, round(v, 2)) for k, v in _top_n(revenue_by_product, TOP_N)],
        "top_categories": [(k, round(v, 2)) for k, v in _top_n(revenue_by_category, TOP_N)],
        "top_regions": [(k, round(v, 2)) for k, v in _top_n(revenue_by_region, TOP_N)],
    }


def load_history(path: Path) -> dict:
    if not path.exists():
        return {}
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except (json.JSONDecodeError, OSError):
        logging.warning("could not read history file at %s, starting fresh", path)
        return {}


def save_history(path: Path, stats: dict) -> None:
    history = load_history(path)
    history[stats["report_date"]] = {
        "total_revenue": stats["total_revenue"],
        "total_orders": stats["total_orders"],
        "total_units": stats["total_units"],
    }
    # keep the last 90 days only
    for key in sorted(history.keys())[:-90]:
        del history[key]
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(history, indent=2), encoding="utf-8")


def compare_to_previous(stats: dict, history: dict) -> dict | None:
    past_days = sorted(d for d in history if d < stats["report_date"])
    if not past_days:
        return None
    prev = history[past_days[-1]]
    prev_revenue = prev["total_revenue"]
    if prev_revenue == 0:
        return None
    pct_change = round((stats["total_revenue"] - prev_revenue) / prev_revenue * 100, 1)
    return {"prev_date": past_days[-1], "prev_revenue": prev_revenue, "pct_change": pct_change}


def maybe_generate_narrative(stats: dict) -> str | None:
    if not ENABLE_LLM_SUMMARY:
        return None
    if not ANTHROPIC_API_KEY:
        logging.warning("ENABLE_LLM_SUMMARY is true but ANTHROPIC_API_KEY is not set; skipping narrative")
        return None

    try:
        import anthropic

        client = anthropic.Anthropic(api_key=ANTHROPIC_API_KEY)
        prompt = (
            "Here are today's computed sales metrics as JSON:\n"
            f"{json.dumps(stats, indent=2)}\n\n"
            "Write a 2-3 sentence plain-English highlight for a team email. "
            "Mention the standout number (best product/category/region or the "
            "day-over-day change if present). No preamble, no markdown."
        )
        response = client.messages.create(
            model="claude-sonnet-5",
            max_tokens=200,
            messages=[{"role": "user", "content": prompt}],
        )
        return response.content[0].text.strip()
    except Exception:
        logging.exception("LLM narrative generation failed; continuing without it")
        return None


def build_email(stats: dict, narrative: str | None) -> tuple[str, str]:
    def money(v: float) -> str:
        return f"${v:,.2f}"

    change_line = ""
    if stats.get("comparison"):
        c = stats["comparison"]
        arrow = "up" if c["pct_change"] >= 0 else "down"
        change_line = f"<p>That's <b>{arrow} {abs(c['pct_change'])}%</b> vs {c['prev_date']} ({money(c['prev_revenue'])}).</p>"

    def rows_html(pairs: list[tuple[str, float]]) -> str:
        return "".join(f"<tr><td>{name}</td><td style='text-align:right'>{money(val)}</td></tr>" for name, val in pairs)

    narrative_html = f"<p><i>{narrative}</i></p>" if narrative else ""

    html = f"""\
<html><body style="font-family: Arial, sans-serif; color: #222;">
  <h2>Daily Sales Summary — {stats['report_date']}</h2>
  {narrative_html}
  <p><b>Total revenue:</b> {money(stats['total_revenue'])}<br>
     <b>Orders:</b> {stats['total_orders']}<br>
     <b>Units sold:</b> {stats['total_units']:.0f}<br>
     <b>Avg order value:</b> {money(stats['avg_order_value'])}</p>
  {change_line}
  <h3>Top products</h3>
  <table>{rows_html(stats['top_products'])}</table>
  <h3>Top categories</h3>
  <table>{rows_html(stats['top_categories'])}</table>
  <h3>Top regions</h3>
  <table>{rows_html(stats['top_regions'])}</table>
</body></html>
"""

    text_lines = [
        f"Daily Sales Summary — {stats['report_date']}",
        narrative or "",
        f"Total revenue: {money(stats['total_revenue'])}",
        f"Orders: {stats['total_orders']}",
        f"Units sold: {stats['total_units']:.0f}",
        f"Avg order value: {money(stats['avg_order_value'])}",
    ]
    if stats.get("comparison"):
        c = stats["comparison"]
        text_lines.append(f"vs {c['prev_date']}: {c['pct_change']:+.1f}%")
    text_lines.append("Top products: " + ", ".join(f"{n} ({money(v)})" for n, v in stats["top_products"]))
    text = "\n".join(line for line in text_lines if line)

    return html, text


def send_email(subject: str, html: str, text: str) -> None:
    if not (SMTP_HOST and FROM_EMAIL and TO_EMAILS):
        raise RuntimeError("SMTP_HOST, FROM_EMAIL and TO_EMAILS must be set")

    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = FROM_EMAIL
    msg["To"] = ", ".join(TO_EMAILS)
    msg.attach(MIMEText(text, "plain"))
    msg.attach(MIMEText(html, "html"))

    with smtplib.SMTP(SMTP_HOST, SMTP_PORT) as server:
        if USE_TLS:
            server.starttls()
        if SMTP_USER:
            server.login(SMTP_USER, SMTP_PASSWORD)
        server.sendmail(FROM_EMAIL, TO_EMAILS, msg.as_string())

    logging.info("email sent to %s", TO_EMAILS)


def send_alert_email(error: Exception) -> None:
    try:
        subject = f"[ALERT] Daily sales report failed — {date.today().isoformat()}"
        text = f"The daily sales summary job failed at {datetime.now().isoformat()}:\n\n{error}"
        html = f"<p>{text.replace(chr(10), '<br>')}</p>"
        send_email(subject, html, text)
    except Exception:
        logging.exception("also failed to send the failure alert email")


def main() -> None:
    setup_logging()
    try:
        rows = load_csv(CSV_PATH)
        stats = compute_summary(rows)
        history = load_history(HISTORY_PATH)
        stats["comparison"] = compare_to_previous(stats, history)
        save_history(HISTORY_PATH, stats)
        narrative = maybe_generate_narrative(stats)
        html, text = build_email(stats, narrative)
        send_email(f"Daily Sales Summary — {stats['report_date']}", html, text)
        logging.info("daily sales summary sent successfully")
    except Exception as e:
        logging.exception("failed to generate/send daily sales summary")
        send_alert_email(e)
        sys.exit(1)


if __name__ == "__main__":
    main()
````
