I wrote `refund_bot.py` with the two tools and the loop. Key points, since a refund moves real money:

- **Two tools**: `look_up_order` (read-only, safe to call repeatedly) and `issue_refund` (irreversible — its description tells the model to look up the order first).
- **Value validation happens in code, not the schema**: strict mode enforces that `amount` is a number and all fields are present, but not that it's positive or within the order's refundable balance — `issue_refund` checks both server-side and returns a structured `{"error": ...}` the model can react to, per "Not reliably enforced (content)" in the skill.
- **Anthropic content-block shape**: assistant message appended verbatim before any tool runs; all `tool_result` blocks for a turn go in a single `user` message (splitting them across messages silently breaks parallel calling).
- **Stopping conditions**: `refusal` and `max_tokens` are checked before touching `content`; a repeated-identical-call counter breaks stuck loops by telling the model in-band rather than just killing the run; `max_iterations` is a return value, not an exception, so the caller/transcript is preserved.
- **`history` persists across calls** in the `__main__` loop so the multi-turn conversation (look up → discuss → refund) carries state naturally.

The `_ORDERS_DB` and tool bodies are stand-ins — swap them for your real order-lookup and payment-provider calls. One thing worth deciding explicitly: right now `issue_refund` executes immediately once the model calls it. For a real support bot you likely want a human-approval gate (or an amount threshold) before money actually moves — happy to wire that in if you want it.

"""Support bot: look up an order, then issue a refund against it, over multiple turns."""

import json
from dataclasses import dataclass, field

import anthropic

MODEL = "claude-sonnet-5"  # pin explicitly; see model-selection skill
MAX_ITERATIONS = 8

client = anthropic.Anthropic()

SYSTEM_PROMPT = (
    "You are a customer support agent. When a customer asks about an order or "
    "wants a refund, look the order up first so you have real data before acting. "
    "Never invent order details, amounts, or refund confirmations. If the order "
    "can't be found or the refund can't be issued, tell the customer plainly."
)

tools = [
    {
        "name": "look_up_order",
        "description": (
            "Look up an order by its order ID and return its status, line items, "
            "total charged, and refund history. Use this before discussing any "
            "order detail or before issuing a refund — never assume amounts. "
            "Does not modify anything; safe to call repeatedly."
        ),
        "strict": True,
        "input_schema": {
            "type": "object",
            "properties": {
                "order_id": {
                    "type": "string",
                    "description": "The order identifier, e.g. 'ORD-10234'.",
                },
            },
            "required": ["order_id"],
            "additionalProperties": False,
        },
    },
    {
        "name": "issue_refund",
        "description": (
            "Issue a refund against a specific order. Irreversible and moves real "
            "money — only call this after look_up_order has confirmed the order "
            "exists, is eligible, and the amount does not exceed what remains "
            "refundable. The amount is in the order's currency, not cents."
        ),
        "strict": True,
        "input_schema": {
            "type": "object",
            "properties": {
                "order_id": {
                    "type": "string",
                    "description": "The order identifier to refund, e.g. 'ORD-10234'.",
                },
                "amount": {
                    "type": "number",
                    "description": "Refund amount in the order's currency. Must not exceed the order's remaining refundable balance.",
                },
                "reason": {
                    "type": "string",
                    "description": "Short customer-facing reason for the refund, e.g. 'damaged on arrival'.",
                },
            },
            "required": ["order_id", "amount", "reason"],
            "additionalProperties": False,
        },
    },
]


# --- Tool implementations (replace with real order-system / payment calls) ---

_ORDERS_DB = {
    "ORD-10234": {"total": 84.50, "currency": "USD", "status": "delivered", "refunded": 0.0},
}


def look_up_order(order_id: str) -> dict:
    order = _ORDERS_DB.get(order_id)
    if order is None:
        return {"error": f"No order found with id {order_id!r}."}
    return {
        "order_id": order_id,
        "status": order["status"],
        "total": order["total"],
        "currency": order["currency"],
        "refunded_so_far": order["refunded"],
        "refundable_remaining": round(order["total"] - order["refunded"], 2),
    }


def issue_refund(order_id: str, amount: float, reason: str) -> dict:
    order = _ORDERS_DB.get(order_id)
    if order is None:
        return {"error": f"No order found with id {order_id!r}."}
    remaining = order["total"] - order["refunded"]
    # value constraints (ranges) aren't enforced by the schema — check them here
    if amount <= 0:
        return {"error": "Refund amount must be greater than zero."}
    if amount > remaining + 1e-9:
        return {
            "error": (
                f"Refund of {amount} exceeds remaining refundable balance "
                f"of {round(remaining, 2)} for {order_id}."
            )
        }
    order["refunded"] += amount  # in production: call the real payment API here
    return {
        "refund_id": f"RFND-{order_id}-{int(order['refunded'] * 100)}",
        "order_id": order_id,
        "amount_refunded": round(amount, 2),
        "currency": order["currency"],
        "reason": reason,
        "remaining_refundable": round(order["total"] - order["refunded"], 2),
    }


REGISTRY = {
    "look_up_order": look_up_order,
    "issue_refund": issue_refund,
}


# --- The loop ---

@dataclass
class RunResult:
    text: str | None
    stop: str  # "done" | "max_iterations" | "refusal" | "truncated"
    iterations: int
    tool_calls: list = field(default_factory=list)


def execute(block, registry: dict) -> tuple[dict, bool]:
    """Run one tool_use block. Returns (content, is_error)."""
    if block.name not in registry:
        return {"error": f"No tool named {block.name!r}."}, True
    try:
        return {"result": registry[block.name](**block.input)}, False
    except Exception as e:
        return {"error": f"{type(e).__name__}: {e}"}, True


def run(user_message: str, messages: list | None = None, max_iterations: int = MAX_ITERATIONS) -> RunResult:
    messages = messages if messages is not None else []
    messages.append({"role": "user", "content": user_message})
    calls: list[str] = []
    seen: dict[tuple, int] = {}

    for i in range(1, max_iterations + 1):
        response = client.messages.create(
            model=MODEL,
            max_tokens=1024,
            system=SYSTEM_PROMPT,
            tools=tools,
            messages=messages,
        )

        if response.stop_reason == "refusal":
            return RunResult(None, "refusal", i, calls)
        if response.stop_reason == "max_tokens":
            return RunResult(None, "truncated", i, calls)

        messages.append({"role": "assistant", "content": response.content})  # verbatim

        if response.stop_reason != "tool_use":
            text = "".join(b.text for b in response.content if b.type == "text")
            return RunResult(text, "done", i, calls)

        results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            calls.append(block.name)

            signature = (block.name, json.dumps(block.input, sort_keys=True))
            seen[signature] = seen.get(signature, 0) + 1
            if seen[signature] >= 3:
                results.append({
                    "type": "tool_result",
                    "tool_use_id": block.id,
                    "content": "You've already called this tool with these exact "
                                "arguments twice and gotten the same result. Try "
                                "something different or answer with what you have.",
                    "is_error": True,
                })
                continue

            content, is_error = execute(block, REGISTRY)
            results.append({
                "type": "tool_result",
                "tool_use_id": block.id,
                "content": json.dumps(content),
                **({"is_error": True} if is_error else {}),
            })

        messages.append({"role": "user", "content": results})  # all results, one message

    return RunResult(None, "max_iterations", max_iterations, calls)


if __name__ == "__main__":
    history: list = []
    print("Support bot ready. Type a message (Ctrl+C to quit).")
    while True:
        user_input = input("\nyou> ").strip()
        if not user_input:
            continue
        result = run(user_input, messages=history)
        if result.stop == "done":
            print(f"bot> {result.text}")
        else:
            print(f"[stopped: {result.stop} after {result.iterations} iterations, "
                  f"calls={result.tool_calls}]")
