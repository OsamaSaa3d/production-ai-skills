Created `agent.py` with a single `get_user` tool (native OpenAI tool calling, `strict: true`, `user_id` required string). Since the directory had no existing agent code, I scaffolded a minimal chat-completion loop backed by an in-memory `USERS` dict — swap that dict lookup for a real DB/API call when one exists. Run it with `python agent.py` (requires `OPENAI_API_KEY` set).

## Files written by the session

### `agent.py`

````py
import json

from openai import OpenAI

MODEL = "gpt-4.1"
client = OpenAI()

# In-memory stand-in for a real user store/database.
USERS = {
    "1": {"id": "1", "name": "Ada Lovelace", "email": "ada@example.com"},
    "2": {"id": "2", "name": "Alan Turing", "email": "alan@example.com"},
}


def get_user(user_id: str) -> dict:
    user = USERS.get(user_id)
    if user is None:
        return {"error": f"no user found with id {user_id!r}"}
    return user


TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "get_user",
            "description": "Fetch a user's profile (id, name, email) by their unique user id.",
            "strict": True,
            "parameters": {
                "type": "object",
                "properties": {
                    "user_id": {
                        "type": "string",
                        "description": "The unique id of the user to fetch, e.g. '1'.",
                    },
                },
                "required": ["user_id"],
                "additionalProperties": False,
            },
        },
    }
]

TOOL_REGISTRY = {"get_user": get_user}


def run(prompt: str) -> str:
    messages = [{"role": "user", "content": prompt}]

    response = client.chat.completions.create(
        model=MODEL, messages=messages, tools=TOOLS, tool_choice="auto"
    )
    message = response.choices[0].message
    messages.append(message)

    if message.tool_calls:
        for tool_call in message.tool_calls:
            args = json.loads(tool_call.function.arguments)
            try:
                result = TOOL_REGISTRY[tool_call.function.name](**args)
            except Exception as e:
                result = {"error": str(e)}
            messages.append(
                {
                    "role": "tool",
                    "tool_call_id": tool_call.id,
                    "content": json.dumps(result),
                }
            )

        response = client.chat.completions.create(
            model=MODEL, messages=messages, tools=TOOLS
        )
        message = response.choices[0].message

    return message.content


if __name__ == "__main__":
    print(run("Look up the user with id 1 and tell me their email."))

````
