"""Extract invoice fields from plain-text invoices into the `orders` table.

Reads every *.txt file in an invoices directory, asks Claude to extract
invoice_number, issue_date, total, customer_name, and po_number via
structured output, and upserts the result into a SQLite `orders` table
keyed on invoice_number.

Usage:
    ANTHROPIC_API_KEY=... python extract_invoices.py [--invoices-dir invoices] [--db orders.db]
"""

from __future__ import annotations

import argparse
import os
import sqlite3
import sys
from datetime import datetime, timezone
from pathlib import Path

from anthropic import Anthropic
from pydantic import BaseModel, Field

MODEL = os.environ.get("ANTHROPIC_MODEL", "claude-sonnet-5")

SYSTEM_PROMPT = (
    "Extract the requested fields from the plain-text invoice below. "
    "Use only information present in the text; do not invent values."
)


class InvoiceExtraction(BaseModel):
    invoice_number: str = Field(
        description="The invoice's own identifying number, exactly as printed."
    )
    issue_date: str = Field(
        description="ISO 8601 date (YYYY-MM-DD) the invoice was issued. "
        "Convert from whatever date format is printed on the invoice."
    )
    total: float = Field(
        description="The final total amount due on the invoice, as a plain "
        "number with no currency symbol or thousands separators."
    )
    customer_name: str = Field(
        description="The name of the customer or company being billed (the bill-to party)."
    )
    po_number: str | None = Field(
        description="The purchase order number referenced on the invoice, if any. "
        "Null if the invoice does not mention a PO number."
    )


def extract_invoice(client: Anthropic, text: str) -> InvoiceExtraction:
    response = client.messages.parse(
        model=MODEL,
        max_tokens=1024,
        system=SYSTEM_PROMPT,
        messages=[{"role": "user", "content": text}],
        output_format=InvoiceExtraction,
    )

    if response.stop_reason == "refusal":
        raise ValueError(f"model refused to extract: {response.stop_details}")
    if response.stop_reason == "max_tokens":
        raise ValueError("response truncated at max_tokens; raise the limit and retry")

    parsed = response.parsed_output
    if parsed is None:
        raise ValueError("no parsed output returned")
    return parsed


def init_db(db_path: Path) -> sqlite3.Connection:
    conn = sqlite3.connect(db_path)
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS orders (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            invoice_number TEXT NOT NULL UNIQUE,
            issue_date TEXT NOT NULL,
            total REAL NOT NULL,
            customer_name TEXT NOT NULL,
            po_number TEXT,
            source_file TEXT NOT NULL,
            extracted_at TEXT NOT NULL
        )
        """
    )
    conn.commit()
    return conn


def save_order(conn: sqlite3.Connection, invoice: InvoiceExtraction, source_file: str) -> None:
    conn.execute(
        """
        INSERT INTO orders
            (invoice_number, issue_date, total, customer_name, po_number, source_file, extracted_at)
        VALUES (?, ?, ?, ?, ?, ?, ?)
        ON CONFLICT(invoice_number) DO UPDATE SET
            issue_date = excluded.issue_date,
            total = excluded.total,
            customer_name = excluded.customer_name,
            po_number = excluded.po_number,
            source_file = excluded.source_file,
            extracted_at = excluded.extracted_at
        """,
        (
            invoice.invoice_number,
            invoice.issue_date,
            invoice.total,
            invoice.customer_name,
            invoice.po_number,
            source_file,
            datetime.now(timezone.utc).isoformat(),
        ),
    )
    conn.commit()


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--invoices-dir", default="invoices", type=Path)
    parser.add_argument("--db", default="orders.db", type=Path)
    args = parser.parse_args()

    if not os.environ.get("ANTHROPIC_API_KEY"):
        print("ANTHROPIC_API_KEY is not set.", file=sys.stderr)
        return 1

    files = sorted(args.invoices_dir.glob("*.txt"))
    if not files:
        print(f"No .txt invoices found in {args.invoices_dir}", file=sys.stderr)
        return 1

    client = Anthropic()
    conn = init_db(args.db)

    ok, failed = 0, 0
    for f in files:
        text = f.read_text(encoding="utf-8")
        try:
            invoice = extract_invoice(client, text)
            save_order(conn, invoice, f.name)
            print(
                f"OK   {f.name}: invoice {invoice.invoice_number}, "
                f"total {invoice.total}, PO {invoice.po_number}"
            )
            ok += 1
        except Exception as exc:
            print(f"FAIL {f.name}: {exc}", file=sys.stderr)
            failed += 1

    conn.close()
    print(f"\n{ok} extracted, {failed} failed. Written to {args.db}")
    return 0 if failed == 0 else 1


if __name__ == "__main__":
    raise SystemExit(main())
