**What I added** (3 files, `orders.db` cleaned up after testing — it regenerates on first run):

- `orders_db.py` — orders table (SQLite for this example) + `query_orders()`, which takes structured filters and builds parameterized SQL. The model never sees or writes raw SQL, so there's no injection surface.
- `orders_tool.py` — the `query_orders` tool schema with `strict: true`, explicit `enum` for status, `anyOf`-with-`null` for optional filters (required by strict mode), and a registry that catches errors and returns them as data instead of raising.
- `pipeline.py` — a `StateGraph` with a `query_orders` node. It calls the OpenAI-compatible API directly with the tool bound (not `create_react_agent`), loops up to 4 tool round-trips, and stops on truncation (`finish_reason == "length"`) rather than parsing a partial tool call. Compiled with `MemorySaver` — swap for `SqliteSaver`/`PostgresSaver` to get checkpoints that survive a restart.

**To integrate**: `query_orders_node` only reads/appends `state["messages"]`, so drop `graph.add_node("query_orders", query_orders_node)` into your real graph at whatever point the model should be able to look up orders, and rewire the edges. Set `ORDERS_AGENT_MODEL` (and standard `OPENAI_API_KEY`) before running.

One thing worth deciding on your end: I used structured filter parameters (`customer_id`, `status`, `date_from`, `date_to`) rather than letting the model write free-form SQL — safer by construction, but if you need arbitrary querying across many columns this will feel limited. Say the word if you'd rather I open that up with a more permissive (but still injection-safe) query DSL.

### `orders_db.py`

````py
"""SQLite-backed orders table used by the query_orders tool.

Swap the connection helper for your real warehouse/OLTP client — the
query_orders() signature (structured filters, parameterized SQL) is the
part worth keeping regardless of backend.
"""

import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).with_name("orders.db")

VALID_STATUSES = {"pending", "shipped", "delivered", "cancelled"}


def _connect() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db() -> None:
    conn = _connect()
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS orders (
            order_id    TEXT PRIMARY KEY,
            customer_id TEXT NOT NULL,
            product     TEXT NOT NULL,
            quantity    INTEGER NOT NULL,
            status      TEXT NOT NULL,
            order_date  TEXT NOT NULL,
            total       REAL NOT NULL
        )
        """
    )
    if conn.execute("SELECT COUNT(*) FROM orders").fetchone()[0] == 0:
        conn.executemany(
            "INSERT INTO orders VALUES (?, ?, ?, ?, ?, ?, ?)",
            [
                ("ORD-1001", "CUST-1", "Wireless Mouse", 1, "delivered", "2026-08-01", 24.99),
                ("ORD-1002", "CUST-1", "USB-C Hub", 2, "shipped", "2026-09-02", 59.98),
                ("ORD-1003", "CUST-2", "Mechanical Keyboard", 1, "pending", "2026-09-10", 89.99),
                ("ORD-1004", "CUST-3", "Monitor Stand", 1, "cancelled", "2026-07-20", 34.5),
                ("ORD-1005", "CUST-2", "Webcam", 1, "delivered", "2026-08-15", 45.0),
            ],
        )
    conn.commit()
    conn.close()


def query_orders(
    customer_id: str | None = None,
    status: str | None = None,
    date_from: str | None = None,
    date_to: str | None = None,
    limit: int = 20,
) -> list[dict]:
    """Look up orders with structured filters. Raises ValueError on bad input."""
    if status is not None and status not in VALID_STATUSES:
        raise ValueError(f"status must be one of {sorted(VALID_STATUSES)}, got {status!r}")
    if not 1 <= limit <= 100:
        raise ValueError("limit must be between 1 and 100")

    clauses = []
    params: list = []
    if customer_id is not None:
        clauses.append("customer_id = ?")
        params.append(customer_id)
    if status is not None:
        clauses.append("status = ?")
        params.append(status)
    if date_from is not None:
        clauses.append("order_date >= ?")
        params.append(date_from)
    if date_to is not None:
        clauses.append("order_date <= ?")
        params.append(date_to)

    where = f"WHERE {' AND '.join(clauses)}" if clauses else ""
    sql = f"SELECT * FROM orders {where} ORDER BY order_date DESC LIMIT ?"
    params.append(limit)

    conn = _connect()
    try:
        rows = conn.execute(sql, params).fetchall()
        return [dict(row) for row in rows]
    finally:
        conn.close()


if __name__ == "__main__":
    init_db()
    print(f"Seeded {DB_PATH}")

````

### `orders_tool.py`

````py
"""Tool definition + registry for letting the model query the orders table.

Strict mode is on, so the model can only pass a status from the fixed
enum and can't smuggle arbitrary SQL through this interface -- query_orders()
builds parameterized SQL from named filters, never from a model-supplied
string.
"""

from orders_db import VALID_STATUSES, query_orders

QUERY_ORDERS_TOOL = {
    "type": "function",
    "function": {
        "name": "query_orders",
        "description": (
            "Look up rows from the orders table by customer, status, and/or "
            "date range. Use this whenever the user asks about order status, "
            "history, or details for a specific customer. Returns up to "
            "`limit` orders, most recent first."
        ),
        "strict": True,
        "parameters": {
            "type": "object",
            "properties": {
                "customer_id": {
                    "anyOf": [{"type": "string"}, {"type": "null"}],
                    "description": "Exact customer id, e.g. 'CUST-1'. Omit (null) to search across customers.",
                },
                "status": {
                    "anyOf": [
                        {"type": "string", "enum": sorted(VALID_STATUSES)},
                        {"type": "null"},
                    ],
                    "description": "Order status to filter on, or null for any status.",
                },
                "date_from": {
                    "anyOf": [{"type": "string"}, {"type": "null"}],
                    "description": "ISO date (YYYY-MM-DD) lower bound on order_date, inclusive, or null.",
                },
                "date_to": {
                    "anyOf": [{"type": "string"}, {"type": "null"}],
                    "description": "ISO date (YYYY-MM-DD) upper bound on order_date, inclusive, or null.",
                },
                "limit": {
                    "type": "integer",
                    "description": "Max rows to return (1-100).",
                },
            },
            "required": ["customer_id", "status", "date_from", "date_to", "limit"],
            "additionalProperties": False,
        },
    },
}

TOOL_REGISTRY = {
    "query_orders": query_orders,
}


def call_tool(name: str, arguments: dict) -> dict:
    """Execute a tool call, returning errors as structured data instead of raising."""
    fn = TOOL_REGISTRY.get(name)
    if fn is None:
        return {"error": f"unknown tool: {name}"}
    try:
        result = fn(**arguments)
        return {"orders": result}
    except Exception as e:
        return {"error": str(e)}

````

### `pipeline.py`

````py
"""LangGraph pipeline with a step that lets the model query the orders table.

LangGraph is kept here for its durable checkpointing between steps (the
reason this pipeline uses it at all) -- the tool-calling step itself talks
to the OpenAI-compatible API directly with strict mode on, rather than
going through create_react_agent or another framework agent wrapper, so
the tool contract stays inspectable and the strict guarantee stays visible.

`MemorySaver` below only persists for the life of the process. Swap it for
`langgraph.checkpoint.sqlite.SqliteSaver` or
`langgraph.checkpoint.postgres.PostgresSaver` to get checkpoints that
survive a restart.
"""

import json
import os
from operator import add
from typing import Annotated, TypedDict

from langgraph.checkpoint.memory import MemorySaver
from langgraph.graph import END, START, StateGraph
from openai import OpenAI

from orders_db import init_db
from orders_tool import QUERY_ORDERS_TOOL, call_tool

MODEL_ID = os.environ["ORDERS_AGENT_MODEL"]  # pin explicitly, e.g. "gpt-4.1"
MAX_TOOL_ITERATIONS = 4

_client: OpenAI | None = None


def get_client() -> OpenAI:
    global _client
    if _client is None:
        _client = OpenAI()  # reads OPENAI_API_KEY / OPENAI_BASE_URL from env
    return _client


class PipelineState(TypedDict):
    messages: Annotated[list[dict], add]


def query_orders_node(state: PipelineState) -> dict:
    """Give the model access to the orders table, bounded to a few tool round-trips."""
    client = get_client()
    transcript = list(state["messages"])
    new_messages: list[dict] = []

    for _ in range(MAX_TOOL_ITERATIONS):
        response = client.chat.completions.create(
            model=MODEL_ID,
            messages=transcript + new_messages,
            tools=[QUERY_ORDERS_TOOL],
            tool_choice="auto",
        )
        choice = response.choices[0]
        message = choice.message

        assistant_msg: dict = {"role": "assistant", "content": message.content or ""}
        if message.tool_calls:
            assistant_msg["tool_calls"] = [tc.model_dump() for tc in message.tool_calls]
        new_messages.append(assistant_msg)

        if choice.finish_reason == "length":
            # truncated mid-generation -- do not treat a partial tool call as real
            break
        if not message.tool_calls:
            break

        for tool_call in message.tool_calls:
            args = json.loads(tool_call.function.arguments)
            result = call_tool(tool_call.function.name, args)
            new_messages.append(
                {
                    "role": "tool",
                    "tool_call_id": tool_call.id,
                    "content": json.dumps(result),
                }
            )
    else:
        raise RuntimeError("query_orders_node exceeded max tool iterations")

    return {"messages": new_messages}


def build_pipeline():
    graph = StateGraph(PipelineState)
    # Splice query_orders_node into your existing graph wherever the model
    # needs order data -- it only reads state["messages"] and appends to it.
    graph.add_node("query_orders", query_orders_node)
    graph.add_edge(START, "query_orders")
    graph.add_edge("query_orders", END)
    return graph.compile(checkpointer=MemorySaver())


if __name__ == "__main__":
    init_db()
    app = build_pipeline()
    config = {"configurable": {"thread_id": "demo-thread-1"}}

    result = app.invoke(
        {"messages": [{"role": "user", "content": "What orders has CUST-2 placed?"}]},
        config=config,
    )
    for msg in result["messages"]:
        print(msg)

````
