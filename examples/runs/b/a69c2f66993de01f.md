Wrote `triage.py` — a support-email urgency classifier using Anthropic's native structured outputs (`client.messages.parse()` with a Pydantic schema), following the structured-output skill's classification pattern:

- **`Urgency` enum** (critical/high/medium/low) with the routing criteria spelled out in the field description, since enum tokens alone are ambiguous.
- **`reasoning` before `urgency`** in the schema so the model reasons before committing to a label.
- **`needs_human_review`** as an abstention flag — genuinely ambiguous or sensitive emails get flagged instead of forced into a bucket.
- **Refusal and truncation handled explicitly** as distinct exceptions (`Refused`, `Truncated`) per `stop_reason`, with one bounded retry (doubled token budget) only for truncation — refusals aren't retried since a repeat would just refuse again.
- **CLI entry point** reads a JSONL file of `{id, body}` emails and prints one classification per line, routing refusals to `needs_human_review` rather than crashing or silently dropping them.

Note: requires `ANTHROPIC_API_KEY` in the environment and the `anthropic`/`pydantic` packages installed. I verified the API shape (`output_format=`, `response.parsed_output`, `response.stop_reason`) against current Anthropic docs rather than assuming it — one part of that fetch listed model names I couldn't verify (`claude-mythos-*`), so I ignored those and used `claude-sonnet-5`, which is confirmed valid elsewhere in this session.

Since there's no support-email dataset here to run it against, I compiled it for syntax but haven't executed a live classification — let me know if you have sample emails and I'll test it end-to-end.

"""Classify incoming support emails by urgency for routing.

Usage:
    python triage.py emails.jsonl

Input: a JSON Lines file where each line is {"id": "...", "body": "..."}
Output: one JSON object per line on stdout with the triage result.
"""

from __future__ import annotations

import json
import logging
import sys
from dataclasses import dataclass
from enum import Enum
from typing import Generic, TypeVar

from anthropic import Anthropic
from pydantic import BaseModel, Field

logger = logging.getLogger("email_triage")

MODEL = "claude-sonnet-5"

client = Anthropic()

SYSTEM_PROMPT = (
    "You triage incoming customer support emails for urgency-based routing. "
    "Base the urgency only on what the email actually says — do not infer "
    "severity from tone, punctuation, or politeness alone."
)


class Urgency(str, Enum):
    CRITICAL = "critical"
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"


class EmailTriage(BaseModel):
    reasoning: str = Field(
        description="One sentence citing the specific phrase in the email that decided the urgency."
    )
    urgency: Urgency = Field(
        description=(
            "critical: full outage, data loss, security/privacy breach, or a payment/legal "
            "deadline within hours — the customer cannot work at all and needs a response now. "
            "high: a core feature is broken or blocking the customer's work with no workaround, "
            "but it is not affecting everyone or everything. "
            "medium: a real problem that has a workaround, or a time-sensitive but non-blocking "
            "question (billing, account access, order status). "
            "low: general questions, how-to requests, feature requests, or feedback with no "
            "deadline pressure."
        )
    )
    needs_human_review: bool = Field(
        description=(
            "True if the urgency is genuinely ambiguous, the email is threatening or legally "
            "sensitive, or the routing decision should not be made automatically."
        )
    )


class Refused(Exception):
    """The model declined to classify this email; do not retry, route to a human."""


class Truncated(Exception):
    """The response hit the token limit before completing; caller should retry with more tokens."""


T = TypeVar("T", bound=BaseModel)


@dataclass
class Parsed(Generic[T]):
    value: T
    attempts: int


def classify_email(body: str, *, max_tokens: int = 512, max_retries: int = 1) -> Parsed[EmailTriage]:
    attempt = 0
    while True:
        attempt += 1
        response = client.messages.parse(
            model=MODEL,
            max_tokens=max_tokens,
            system=SYSTEM_PROMPT,
            messages=[{"role": "user", "content": body}],
            output_format=EmailTriage,
        )

        if response.stop_reason == "refusal":
            logger.warning("triage_refused", extra={"body_preview": body[:200]})
            raise Refused("model declined to classify this email")

        if response.stop_reason == "max_tokens":
            logger.warning("triage_truncated", extra={"max_tokens": max_tokens})
            if attempt <= max_retries:
                max_tokens *= 2
                continue
            raise Truncated(f"still truncated at max_tokens={max_tokens}")

        if response.parsed_output is None:
            raise ValueError("no parsed output and no refusal — inspect the raw response")

        return Parsed(response.parsed_output, attempt)


def main() -> None:
    logging.basicConfig(level=logging.INFO)

    if len(sys.argv) != 2:
        print("usage: python triage.py emails.jsonl", file=sys.stderr)
        raise SystemExit(1)

    with open(sys.argv[1], encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            email = json.loads(line)

            try:
                result = classify_email(email["body"])
            except Refused:
                print(json.dumps({"id": email["id"], "status": "needs_human_review", "reason": "refused"}))
                continue
            except Truncated as e:
                print(json.dumps({"id": email["id"], "status": "error", "reason": str(e)}))
                continue

            triage = result.value
            print(
                json.dumps(
                    {
                        "id": email["id"],
                        "status": "classified",
                        "urgency": triage.urgency.value,
                        "reasoning": triage.reasoning,
                        "needs_human_review": triage.needs_human_review,
                    }
                )
            )


if __name__ == "__main__":
    main()
