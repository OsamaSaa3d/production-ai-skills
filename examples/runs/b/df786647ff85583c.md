Created `agent.py` with a single tool, `get_user_by_id`, defined with `strict: true` so the provider enforces the schema, plus `users.json` as a mock data store it reads from. The tool takes one required `user_id` string, returns the user record or a structured `{"error": ...}` if not found (never raises). Uses native OpenAI-compatible tool calling per the `llm-tool-calling` skill — no framework, no extra tools.

Assumption: no backend was specified, so I used a local JSON file as the user store — swap `get_user_by_id`'s body for a real DB/API call when one exists; the tool schema and calling loop don't need to change.

## Files written by the session

### `agent.py`

````py
import json
import os
from pathlib import Path

from openai import OpenAI

MODEL = "gpt-4.1"
USERS_PATH = Path(__file__).parent / "users.json"

client = OpenAI(base_url=os.environ.get("OPENAI_BASE_URL"), api_key=os.environ["OPENAI_API_KEY"])

tools = [
    {
        "type": "function",
        "function": {
            "name": "get_user_by_id",
            "description": "Fetch a single user's profile (id, name, email) by their unique user id.",
            "strict": True,
            "parameters": {
                "type": "object",
                "properties": {
                    "user_id": {
                        "type": "string",
                        "description": "The unique id of the user to fetch, e.g. '1'.",
                    },
                },
                "required": ["user_id"],
                "additionalProperties": False,
            },
        },
    }
]


def get_user_by_id(user_id: str) -> dict:
    users = json.loads(USERS_PATH.read_text())
    user = users.get(user_id)
    if user is None:
        return {"error": f"No user found with id '{user_id}'"}
    return user


TOOL_REGISTRY = {
    "get_user_by_id": get_user_by_id,
}


def run(user_message: str) -> str:
    messages = [{"role": "user", "content": user_message}]

    response = client.chat.completions.create(
        model=MODEL,
        messages=messages,
        tools=tools,
        tool_choice="auto",
    )
    message = response.choices[0].message
    messages.append(message)

    if message.tool_calls:
        for tool_call in message.tool_calls:
            name = tool_call.function.name
            args = json.loads(tool_call.function.arguments)
            try:
                result = TOOL_REGISTRY[name](**args)
            except Exception as e:
                result = {"error": str(e)}
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call.id,
                "content": json.dumps(result),
            })

        response = client.chat.completions.create(
            model=MODEL,
            messages=messages,
            tools=tools,
        )
        message = response.choices[0].message

    return message.content


if __name__ == "__main__":
    print(run("Who is user 2?"))

````
