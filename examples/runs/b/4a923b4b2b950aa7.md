## What I built

Per the tool-design skill's core rule — never let the model emit a query-DSL string — I split this into a typed tool boundary plus a query builder that owns the grammar:

- **`ticket-search/query-builder.ts`** — `buildTicketQuery()` takes typed fields (`status`, `team`, `assignee`, `priority`, `createdAfter`/`createdBefore`, `text`) and constructs the `field:value AND field:>value` DSL string. Validates enums and dates, safely quotes free text, and rejects anything that doesn't fit the allowlisted grammar — so the agent can never inject arbitrary query syntax.
- **`ticket-search/tool.ts`** — the Anthropic tool definition (`search_tickets`) with enums for `status`/`priority`, `input_examples` showing common field combinations, and a handler that calls the search endpoint, caps results at 20, returns only high-signal fields (id/title/status/team/priority/assignee/created_at), and returns actionable error strings (e.g. "narrow the filters") instead of tracebacks.
- **`ticket-search/agent.ts`** — a minimal Claude tool-use loop wiring it together end to end.

Assumptions I made since there's no existing codebase or API spec here (flag if wrong):
- Statuses: `open`, `in_progress`, `pending`, `closed`; priorities: `low`/`medium`/`high`/`critical` — adjust the enums in `tool.ts`/`query-builder.ts` to your real values.
- Search endpoint is `GET {TICKET_API_BASE_URL}/tickets/search?q=<dsl>`, bearer-token auth via `TICKET_API_TOKEN` env var — swap in `tool.ts` for your actual endpoint/auth.
- Node/TypeScript with `@anthropic-ai/sdk`, since nothing else was in the directory.

`npx tsc --noEmit` passes cleanly. Next step if you want confidence beyond type-checking: point `TICKET_API_BASE_URL` at a real (or mocked) endpoint and run `npm run agent -- "find open platform tickets from this year"`.

````ts
/**
 * Minimal agent loop demonstrating search_tickets wired into Claude tool use.
 * Run: ANTHROPIC_API_KEY=... TICKET_API_BASE_URL=... npx tsx ticket-search/agent.ts "find open platform tickets from this year"
 */
import Anthropic from "@anthropic-ai/sdk";
import { searchTickets, searchTicketsToolDefinition } from "./tool.js";
import type { TicketSearchParams } from "./query-builder.js";

const anthropic = new Anthropic();

async function runAgent(userRequest: string): Promise<string> {
  const messages: Anthropic.MessageParam[] = [{ role: "user", content: userRequest }];

  for (let turn = 0; turn < 5; turn++) {
    const response = await anthropic.messages.create({
      model: "claude-sonnet-5",
      max_tokens: 1024,
      tools: [searchTicketsToolDefinition as Anthropic.Tool],
      messages,
    });

    const toolUses = response.content.filter(
      (block): block is Anthropic.ToolUseBlock => block.type === "tool_use",
    );

    if (toolUses.length === 0) {
      const text = response.content.find((b): b is Anthropic.TextBlock => b.type === "text");
      return text?.text ?? "";
    }

    messages.push({ role: "assistant", content: response.content });

    const toolResults: Anthropic.ToolResultBlockParam[] = [];
    for (const use of toolUses) {
      if (use.name !== "search_tickets") continue;
      const result = await searchTickets(use.input as TicketSearchParams);
      toolResults.push({
        type: "tool_result",
        tool_use_id: use.id,
        content: JSON.stringify(result),
      });
    }
    messages.push({ role: "user", content: toolResults });
  }

  return "Gave up after 5 turns without a final answer.";
}

runAgent(request).then((answer) => console.log(answer));
````

### `ticket-search/query-builder.ts`

````ts
/**
 * Builds `status:open AND team:platform AND created:>2026-01-01`-style query
 * strings for the ticket search endpoint from typed, validated fields.
 *
 * The model never sees or emits this grammar. It only fills in the typed
 * `TicketSearchParams` object; this module is the single place that knows
 * the DSL's syntax, quoting rules, and operators.
 */

export type TicketStatus = "open" | "in_progress" | "pending" | "closed";
export type TicketPriority = "low" | "medium" | "high" | "critical";

export interface TicketSearchParams {
  /** Free-text match against ticket title/body. */
  text?: string;
  status?: TicketStatus;
  /** Team slug, e.g. "platform", "billing", "growth". */
  team?: string;
  assignee?: string;
  priority?: TicketPriority;
  /** Inclusive lower bound, e.g. "2026-01-01". */
  createdAfter?: string;
  /** Inclusive upper bound, e.g. "2026-06-30". */
  createdBefore?: string;
}

const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
const STATUSES: readonly TicketStatus[] = ["open", "in_progress", "pending", "closed"];
const PRIORITIES: readonly TicketPriority[] = ["low", "medium", "high", "critical"];
// Conservative allowlist for unquoted DSL tokens (team slugs, assignee handles).
const SAFE_TOKEN_RE = /^[A-Za-z0-9._@-]+$/;

export class InvalidTicketSearchParamsError extends Error {}

function quoteIfNeeded(value: string): string {
  if (SAFE_TOKEN_RE.test(value)) return value;
  // Escape embedded quotes/backslashes, then wrap so the DSL parser treats
  // this as a single literal rather than additional query syntax.
  return `"${value.replace(/\\/g, "\\\\").replace(/"/g, '\\"')}"`;
}

function requireDate(label: string, value: string): string {
  if (!DATE_RE.test(value)) {
    throw new InvalidTicketSearchParamsError(
      `${label} must be an ISO date (YYYY-MM-DD), got "${value}".`,
    );
  }
  return value;
}

/**
 * Converts typed search params into the ticket system's DSL query string.
 * Throws InvalidTicketSearchParamsError on anything that doesn't fit the
 * allowlisted grammar, rather than passing untrusted text through.
 */
export function buildTicketQuery(params: TicketSearchParams): string {
  const clauses: string[] = [];

  if (params.status !== undefined) {
    if (!STATUSES.includes(params.status)) {
      throw new InvalidTicketSearchParamsError(
        `status must be one of ${STATUSES.join(", ")}, got "${params.status}".`,
      );
    }
    clauses.push(`status:${params.status}`);
  }

  if (params.priority !== undefined) {
    if (!PRIORITIES.includes(params.priority)) {
      throw new InvalidTicketSearchParamsError(
        `priority must be one of ${PRIORITIES.join(", ")}, got "${params.priority}".`,
      );
    }
    clauses.push(`priority:${params.priority}`);
  }

  if (params.team !== undefined) {
    clauses.push(`team:${quoteIfNeeded(params.team)}`);
  }

  if (params.assignee !== undefined) {
    clauses.push(`assignee:${quoteIfNeeded(params.assignee)}`);
  }

  if (params.createdAfter !== undefined) {
    clauses.push(`created:>${requireDate("createdAfter", params.createdAfter)}`);
  }

  if (params.createdBefore !== undefined) {
    clauses.push(`created:<${requireDate("createdBefore", params.createdBefore)}`);
  }

  if (params.text !== undefined && params.text.trim() !== "") {
    clauses.push(`text:${quoteIfNeeded(params.text.trim())}`);
  }

  if (clauses.length === 0) {
    throw new InvalidTicketSearchParamsError(
      "At least one search field is required (status, team, assignee, priority, createdAfter, createdBefore, or text).",
    );
  }

  return clauses.join(" AND ");
}

````

### `ticket-search/tool.ts`

````ts
import {
  buildTicketQuery,
  InvalidTicketSearchParamsError,
  type TicketSearchParams,
} from "./query-builder.js";

const TICKET_API_BASE_URL = process.env.TICKET_API_BASE_URL ?? "https://tickets.example.internal/api";
const TICKET_API_TOKEN = process.env.TICKET_API_TOKEN;

// Anthropic Messages API tool_use block. Note: no query-string parameter —
// the DSL is built in buildTicketQuery, never emitted by the model.
export const searchTicketsToolDefinition = {
  name: "search_tickets",
  description:
    "Search support/engineering tickets by status, team, assignee, priority, " +
    "and creation date. Returns up to 20 matching tickets with id, title, " +
    "status, team, priority, assignee, and created_at. At least one field " +
    "must be set. Use 'text' for free-text matches on title/body.",
  input_schema: {
    type: "object",
    properties: {
      text: {
        type: "string",
        description: "Free-text match against ticket title and body.",
      },
      status: {
        type: "string",
        enum: ["open", "in_progress", "pending", "closed"],
      },
      team: {
        type: "string",
        description: "Team slug that owns the ticket, e.g. 'platform', 'billing', 'growth'.",
      },
      assignee: {
        type: "string",
        description: "Assignee username or email.",
      },
      priority: {
        type: "string",
        enum: ["low", "medium", "high", "critical"],
      },
      createdAfter: {
        type: "string",
        format: "date",
        description: "Inclusive lower bound on ticket creation date, ISO 8601 (YYYY-MM-DD).",
      },
      createdBefore: {
        type: "string",
        format: "date",
        description: "Inclusive upper bound on ticket creation date, ISO 8601 (YYYY-MM-DD).",
      },
    },
    additionalProperties: false,
  },
  // Anthropic-specific: teaches format/co-occurrence conventions a JSON
  // Schema can't express (date shape, which fields travel together).
  input_examples: [
    { status: "open", team: "platform", createdAfter: "2026-01-01" },
    { text: "login page 500 error", status: "open" },
    { team: "billing", assignee: "jane@acme.com", priority: "critical" },
  ],
} as const;

export interface TicketSummary {
  id: string;
  title: string;
  status: string;
  team: string;
  priority: string;
  assignee: string | null;
  created_at: string;
}

interface RawTicketApiResponse {
  results: Array<{
    id: string;
    title: string;
    status: string;
    team: string;
    priority: string;
    assignee: string | null;
    created_at: string;
    // ...other fields the API returns that we deliberately drop
  }>;
  total: number;
}

const MAX_RESULTS = 20;

/**
 * Tool handler: validates+converts typed params to the DSL query, calls the
 * search endpoint, and returns a capped, high-signal ticket list. Errors are
 * returned as actionable strings (per tool-design guidance) rather than
 * thrown tracebacks, since this return value goes straight into a tool_result.
 */
export async function searchTickets(
  params: TicketSearchParams,
): Promise<{ tickets: TicketSummary[]; total: number } | { error: string }> {
  let query: string;
  try {
    query = buildTicketQuery(params);
  } catch (err) {
    if (err instanceof InvalidTicketSearchParamsError) {
      return { error: err.message };
    }
    throw err;
  }

  const url = new URL("/tickets/search", TICKET_API_BASE_URL);
  url.searchParams.set("q", query);
  url.searchParams.set("limit", String(MAX_RESULTS));

  const response = await fetch(url, {
    headers: TICKET_API_TOKEN ? { Authorization: `Bearer ${TICKET_API_TOKEN}` } : {},
  });

  if (!response.ok) {
    if (response.status === 400) {
      return {
        error:
          `The ticket search endpoint rejected the query (${query}). ` +
          "Double-check field values against the allowed statuses/priorities and retry with narrower or corrected filters.",
      };
    }
    if (response.status === 429) {
      return { error: "Ticket search is rate-limited. Wait a moment and retry with fewer calls." };
    }
    return { error: `Ticket search failed with HTTP ${response.status}. Retry once; if it persists, report the failure.` };
  }

  const data = (await response.json()) as RawTicketApiResponse;

  const tickets: TicketSummary[] = data.results.slice(0, MAX_RESULTS).map((t) => ({
    id: t.id,
    title: t.title,
    status: t.status,
    team: t.team,
    priority: t.priority,
    assignee: t.assignee,
    created_at: t.created_at,
  }));

  const result: { tickets: TicketSummary[]; total: number; note?: string } = {
    tickets,
    total: data.total,
  };
  if (data.total > MAX_RESULTS) {
    result.note = `Showing ${MAX_RESULTS} of ${data.total} matches. Narrow the filters (e.g. add createdAfter, assignee, or a more specific status) to see the rest.`;
  }

  return result;
}
````
