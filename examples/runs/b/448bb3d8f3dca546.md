- **Strict-mode schemas**: both tools have `additionalProperties: false` and all params in `required`, so the model can't send malformed args.
- **`qty > 0` is checked in code**, not the schema — strict mode enforces `type: integer` but not positivity, so that's validated before the tool runs.
- **Errors from the real functions** (unknown SKU, insufficient stock) are caught and returned as `{"error": ...}` tool results instead of raising, so the model can react (e.g. tell the user stock is insufficient) instead of crashing the loop.
- **`finish_reason == "length"`** is checked before trusting `tool_calls`, per the skill's guidance on truncation.
- Model is read from `OPENAI_MODEL` env var, defaulting to `gpt-4.1` — pin to whatever your org actually uses.

"""
Lets the model call get_inventory(sku) and reserve_stock(sku, qty) as tools.

Uses native tool calling with strict mode (OpenAI-compatible Chat Completions API):
the provider constrains tool names/argument shape during generation, so we don't
parse JSON out of free text or hand-roll a ReAct loop.
"""

import json
import os

from openai import OpenAI

MODEL = os.environ.get("OPENAI_MODEL", "gpt-4.1")  # pin explicitly; check current model list before relying on this
client = OpenAI()


# --- Existing business logic -------------------------------------------------
# Replace these two stubs with the actual functions from your codebase.
# Signatures/semantics assumed: get_inventory(sku) -> int on-hand qty (raises
# KeyError if unknown); reserve_stock(sku, qty) -> reservation id (raises
# ValueError if qty exceeds on-hand). Adjust the wrappers below if the real
# functions differ.

_FAKE_WAREHOUSE = {"SKU-100": 42, "SKU-200": 0, "SKU-300": 7}


def get_inventory(sku: str) -> int:
    if sku not in _FAKE_WAREHOUSE:
        raise KeyError(f"unknown sku: {sku}")
    return _FAKE_WAREHOUSE[sku]


def reserve_stock(sku: str, qty: int) -> str:
    on_hand = get_inventory(sku)
    if qty > on_hand:
        raise ValueError(f"cannot reserve {qty} of {sku}; only {on_hand} on hand")
    _FAKE_WAREHOUSE[sku] -= qty
    return f"RES-{sku}-{qty}"


# --- Tool declarations ---------------------------------------------------

TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "get_inventory",
            "description": (
                "Look up the current on-hand quantity for a single SKU. "
                "Use this before reserving stock, or whenever the user asks "
                "how much of an item is in stock."
            ),
            "strict": True,
            "parameters": {
                "type": "object",
                "properties": {
                    "sku": {
                        "type": "string",
                        "description": "The exact SKU code to look up, e.g. 'SKU-100'.",
                    },
                },
                "required": ["sku"],
                "additionalProperties": False,
            },
        },
    },
    {
        "type": "function",
        "function": {
            "name": "reserve_stock",
            "description": (
                "Reserve a quantity of a SKU against on-hand inventory. "
                "Only call this after confirming enough stock is available "
                "(e.g. via get_inventory) or when the user has explicitly "
                "asked to place a reservation. Fails if qty exceeds on-hand."
            ),
            "strict": True,
            "parameters": {
                "type": "object",
                "properties": {
                    "sku": {
                        "type": "string",
                        "description": "The exact SKU code to reserve, e.g. 'SKU-100'.",
                    },
                    "qty": {
                        "type": "integer",
                        "description": "Number of units to reserve. Must be a positive integer.",
                    },
                },
                "required": ["sku", "qty"],
                "additionalProperties": False,
            },
        },
    },
]

TOOL_REGISTRY = {
    "get_inventory": get_inventory,
    "reserve_stock": reserve_stock,
}


def _call_tool(name: str, args: dict):
    fn = TOOL_REGISTRY[name]
    try:
        result = fn(**args)
        return {"result": result}
    except Exception as e:  # noqa: BLE001 - deliberately broad: return any failure as tool data
        return {"error": str(e)}


def run(user_message: str, *, max_turns: int = 8) -> str:
    messages = [{"role": "user", "content": user_message}]

    for _ in range(max_turns):
        response = client.chat.completions.create(
            model=MODEL,
            messages=messages,
            tools=TOOLS,
            tool_choice="auto",
        )
        choice = response.choices[0]
        message = choice.message
        messages.append(message)

        if choice.finish_reason == "length":
            raise RuntimeError("response truncated mid-generation; increase max_tokens")

        if not message.tool_calls:
            return message.content or ""

        for tool_call in message.tool_calls:
            name = tool_call.function.name
            args = json.loads(tool_call.function.arguments)
            # Structural correctness (types, required fields) is guaranteed by
            # strict mode. qty > 0 is a value constraint strict mode does not
            # reliably enforce, so it's checked here rather than trusted to the schema.
            if name == "reserve_stock" and args.get("qty", 0) <= 0:
                result = {"error": "qty must be a positive integer"}
            else:
                result = _call_tool(name, args)

            messages.append({
                "role": "tool",
                "tool_call_id": tool_call.id,
                "content": json.dumps(result),
            })

    raise RuntimeError(f"exceeded max_turns ({max_turns}) without a final answer")


if __name__ == "__main__":
    print(run("How many units of SKU-100 are in stock? Reserve 5 if there's enough."))
