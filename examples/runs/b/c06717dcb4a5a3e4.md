I built `extract_invoices.py`, which:

1. Reads each `.txt` invoice from an `invoices/` directory (created a `sample-001.txt` for reference/testing).
2. Extracts `invoice_number`, `issue_date`, `total`, `customer_name`, `po_number` via Claude's native structured output (`output_format=` with a Pydantic schema, strict-mode conformance — no regex/JSON-parsing).
3. Made `po_number` nullable since not every invoice references a PO — required fields would otherwise force the model to invent one.
4. Handles the two failure modes strict mode doesn't cover: refusal (skipped, logged) and truncation (retried once with doubled token budget, then skipped).
5. Writes into a SQLite `orders` table (`orders.db`, auto-created) keyed on `invoice_number`, upserting so reruns don't duplicate rows.

Assumptions made (flag if wrong): plain-text invoices live as `.txt` files in a folder; SQLite is the target ("orders table" with no DB specified); model is `claude-sonnet-5`, easy to swap to Haiku 4.5 in the `MODEL` constant if cost matters more than accuracy here.

To run: `pip install -r requirements.txt`, set `ANTHROPIC_API_KEY`, then `python extract_invoices.py`. Point it at your real invoice directory if it's not `invoices/`.

"""Extract invoice number, issue date, total, customer name, and PO number
from plain-text invoices and write the results into the `orders` table
of a SQLite database.

Usage:
    python extract_invoices.py [invoices_dir] [--db orders.db]

Requires ANTHROPIC_API_KEY to be set in the environment.
"""
import argparse
import sqlite3
import sys
from datetime import datetime, timezone
from pathlib import Path

from anthropic import Anthropic
from pydantic import BaseModel, Field

MODEL = "claude-sonnet-5"


class InvoiceExtraction(BaseModel):
    invoice_number: str = Field(
        description="The invoice's unique identifier, exactly as printed on the document."
    )
    issue_date: str = Field(
        description="ISO 8601 date (YYYY-MM-DD) the invoice was issued. "
        "Convert from whatever date format is printed on the document."
    )
    total: float = Field(
        description="The final total amount due on the invoice (numeric value only, "
        "no currency symbol)."
    )
    customer_name: str = Field(
        description="The name of the customer or company being billed "
        "(the bill-to party, not the vendor issuing the invoice)."
    )
    po_number: str | None = Field(
        description="The purchase order number referenced on the invoice. "
        "Null if the invoice does not reference a PO."
    )


class Refused(Exception):
    pass


class Truncated(Exception):
    pass


def extract_invoice(
    client: Anthropic, text: str, *, max_tokens: int = 1024, max_retries: int = 1
) -> InvoiceExtraction:
    attempt = 0
    while True:
        attempt += 1
        response = client.messages.parse(
            model=MODEL,
            max_tokens=max_tokens,
            messages=[
                {
                    "role": "user",
                    "content": (
                        "Extract the invoice fields from the following invoice text. "
                        "Use null for any field that is not present in the document.\n\n"
                        f"{text}"
                    ),
                }
            ],
            output_format=InvoiceExtraction,
        )

        if response.stop_reason == "refusal":
            raise Refused(f"model refused to extract: {response.stop_reason}")

        if response.stop_reason == "max_tokens":
            if attempt <= max_retries:
                max_tokens *= 2
                continue
            raise Truncated(f"still truncated at max_tokens={max_tokens}")

        return response.parsed_output


def ensure_table(conn: sqlite3.Connection) -> None:
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS orders (
            invoice_number TEXT PRIMARY KEY,
            issue_date     TEXT NOT NULL,
            total          REAL NOT NULL,
            customer_name  TEXT NOT NULL,
            po_number      TEXT,
            source_file    TEXT NOT NULL,
            extracted_at   TEXT NOT NULL
        )
        """
    )
    conn.commit()


def write_order(conn: sqlite3.Connection, invoice: InvoiceExtraction, source_file: str) -> None:
    conn.execute(
        """
        INSERT INTO orders (invoice_number, issue_date, total, customer_name, po_number, source_file, extracted_at)
        VALUES (?, ?, ?, ?, ?, ?, ?)
        ON CONFLICT(invoice_number) DO UPDATE SET
            issue_date=excluded.issue_date,
            total=excluded.total,
            customer_name=excluded.customer_name,
            po_number=excluded.po_number,
            source_file=excluded.source_file,
            extracted_at=excluded.extracted_at
        """,
        (
            invoice.invoice_number,
            invoice.issue_date,
            invoice.total,
            invoice.customer_name,
            invoice.po_number,
            source_file,
            datetime.now(timezone.utc).isoformat(),
        ),
    )
    conn.commit()


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Extract invoice fields from plain-text invoices into the orders table."
    )
    parser.add_argument(
        "invoices_dir",
        nargs="?",
        default="invoices",
        help="Directory of plain-text invoice files (default: invoices)",
    )
    parser.add_argument("--db", default="orders.db", help="SQLite database path (default: orders.db)")
    args = parser.parse_args()

    invoices_dir = Path(args.invoices_dir)
    files = sorted(invoices_dir.glob("*.txt"))
    if not files:
        print(f"No .txt invoices found in {invoices_dir}", file=sys.stderr)
        sys.exit(1)

    client = Anthropic()
    conn = sqlite3.connect(args.db)
    ensure_table(conn)

    ok, failed = 0, 0
    for path in files:
        text = path.read_text(encoding="utf-8")
        try:
            invoice = extract_invoice(client, text)
        except (Refused, Truncated) as exc:
            print(f"SKIP {path.name}: {exc}", file=sys.stderr)
            failed += 1
            continue
        write_order(conn, invoice, path.name)
        print(
            f"OK   {path.name}: invoice {invoice.invoice_number}, "
            f"total {invoice.total}, PO {invoice.po_number}"
        )
        ok += 1

    conn.close()
    print(f"\n{ok} written, {failed} failed, out of {len(files)} invoice(s).")


if __name__ == "__main__":
    main()
