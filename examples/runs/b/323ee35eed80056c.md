All 14 endpoints are wrapped and verified end-to-end with mocked HTTP calls.

## What's built

`C:\tmp\skills-eval\arm-b\t10-r1\crm_agent\`:

- **`crm_client.py`** — thin HTTP layer over your 14 assumed REST endpoints (Contacts/Companies/Deals CRUD + search, plus notes). **This is the file to edit** if your real paths, auth, or field names differ.
- **`tools.py`** — 14 agent-facing tools, task-named rather than endpoint-mirrored: `get_contact`/`get_company` each fold in a related deals lookup (2 HTTP calls → 1 tool call); `delete_contact` stays its own tool rather than a flag on `update_contact` since destructive-vs-not is exactly the kind of semantic an agent shouldn't be able to invert silently; `create_contact`/`create_deal`/`add_note` carry `input_examples` for the convention-heavy nested fields (id formats, which optional fields co-occur).
- **`dispatch.py`** — executes each tool call, returns only high-signal fields (no raw pass-through of every API field), paginates with a "narrow your query" hint when results are truncated, and turns 404/422/429/5xx into instructions the agent can act on instead of tracebacks.
- **`agent.py`** — the actual tool-use loop against `claude-sonnet-5`.

Tested via mocks (no live API to hit): schema validity, pagination shaping, error-message mapping, and the update/delete paths all check out.

**Before this runs for real:** fill in `.env.example` (`CRM_API_BASE_URL`, `CRM_API_KEY`, `ANTHROPIC_API_KEY`), `pip install -r requirements.txt`, and correct any endpoint paths/field names in `crm_client.py` that don't match your actual API. Once it's live, the tool-design skill's eval loop (`evals-before-shipping`, `ToolCorrectnessMetric`/`ArgumentCorrectnessMetric`) is the way to check whether the consolidation choices above actually hold up — I designed this against the skill's defaults, not against measured failures on your real traffic.

### `crm_agent/agent.py`

````py
"""
Agent loop: Claude + the 14-endpoint CRM exposed as 14 task-oriented tools.

Usage:
    python -m crm_agent.agent "Find Acme Corp's open deals and summarize them"
"""

import sys

import anthropic

from . import config
from .dispatch import execute_tool
from .tools import TOOLS

MODEL = "claude-sonnet-5"

SYSTEM_PROMPT = (
    "You are a CRM assistant. Use the provided tools to look up and modify "
    "contacts, companies, and deals. Prefer search_* tools to find ids before "
    "calling get_*/update_*/delete_* with them. Always confirm with the user "
    "before calling delete_contact — it is irreversible."
)


def run_agent(user_message: str, max_turns: int = 8) -> str:
    client = anthropic.Anthropic(api_key=config.ANTHROPIC_API_KEY or None)
    messages = [{"role": "user", "content": user_message}]

    for _ in range(max_turns):
        response = client.messages.create(
            model=MODEL,
            max_tokens=2048,
            system=SYSTEM_PROMPT,
            tools=TOOLS,
            messages=messages,
        )
        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason != "tool_use":
            return "".join(block.text for block in response.content if block.type == "text")

        tool_results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            result_json = execute_tool(block.name, block.input)
            tool_results.append({
                "type": "tool_result",
                "tool_use_id": block.id,
                "content": result_json,
            })
        messages.append({"role": "user", "content": tool_results})

    return "Stopped after max_turns without a final answer."


if __name__ == "__main__":
    prompt = " ".join(sys.argv[1:]) or "List the 5 most recent contacts."
    print(run_agent(prompt))

````

"""
Thin wrapper around the 14 CRM REST endpoints.

ASSUMED shape (adjust paths/fields to match your real API):
  Contacts:  GET /contacts, GET /contacts/{id}, POST /contacts,
             PATCH /contacts/{id}, DELETE /contacts/{id}
  Companies: GET /companies, GET /companies/{id}, POST /companies,
             PATCH /companies/{id}
  Deals:     GET /deals, GET /deals/{id}, POST /deals, PATCH /deals/{id}
  Notes:     POST /notes

This module knows nothing about agents or tool schemas — it's the
deterministic HTTP layer. `tools.py` / `dispatch.py` build the agent-facing
surface on top of it.
"""

import requests

from . import config


class CRMAPIError(Exception):
    def __init__(self, status_code: int, message: str):
        self.status_code = status_code
        self.message = message
        super().__init__(message)


class CRMClient:
    def __init__(self, base_url: str = None, api_key: str = None):
        self.base_url = (base_url or config.CRM_API_BASE_URL).rstrip("/")
        self.api_key = api_key or config.CRM_API_KEY

    def _headers(self):
        return {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
        }

    def _request(self, method: str, path: str, params: dict = None, json: dict = None):
        resp = requests.request(
            method,
            f"{self.base_url}{path}",
            headers=self._headers(),
            params={k: v for k, v in (params or {}).items() if v is not None},
            json=json,
            timeout=15,
        )
        if not resp.ok:
            try:
                detail = resp.json().get("message", resp.text)
            except ValueError:
                detail = resp.text
            raise CRMAPIError(resp.status_code, detail)
        if resp.status_code == 204 or not resp.content:
            return {}
        return resp.json()

    # ---- Contacts (5 endpoints) ----
    def list_contacts(self, q=None, email=None, company_id=None, page=1, page_size=20):
        return self._request(
            "GET", "/contacts",
            params={"q": q, "email": email, "company_id": company_id, "page": page, "page_size": page_size},
        )

    def get_contact(self, contact_id: str):
        return self._request("GET", f"/contacts/{contact_id}")

    def create_contact(self, payload: dict):
        return self._request("POST", "/contacts", json=payload)

    def update_contact(self, contact_id: str, payload: dict):
        return self._request("PATCH", f"/contacts/{contact_id}", json=payload)

    def delete_contact(self, contact_id: str):
        return self._request("DELETE", f"/contacts/{contact_id}")

    # ---- Companies (4 endpoints) ----
    def list_companies(self, q=None, industry=None, page=1, page_size=20):
        return self._request(
            "GET", "/companies",
            params={"q": q, "industry": industry, "page": page, "page_size": page_size},
        )

    def get_company(self, company_id: str):
        return self._request("GET", f"/companies/{company_id}")

    def create_company(self, payload: dict):
        return self._request("POST", "/companies", json=payload)

    def update_company(self, company_id: str, payload: dict):
        return self._request("PATCH", f"/companies/{company_id}", json=payload)

    # ---- Deals (4 endpoints) ----
    def list_deals(self, q=None, stage=None, company_id=None, contact_id=None,
                   min_amount=None, max_amount=None, page=1, page_size=20):
        return self._request(
            "GET", "/deals",
            params={
                "q": q, "stage": stage, "company_id": company_id, "contact_id": contact_id,
                "min_amount": min_amount, "max_amount": max_amount,
                "page": page, "page_size": page_size,
            },
        )

    def get_deal(self, deal_id: str):
        return self._request("GET", f"/deals/{deal_id}")

    def create_deal(self, payload: dict):
        return self._request("POST", "/deals", json=payload)

    def update_deal(self, deal_id: str, payload: dict):
        return self._request("PATCH", f"/deals/{deal_id}", json=payload)

    # ---- Notes (1 endpoint) ----
    def create_note(self, payload: dict):
        return self._request("POST", "/notes", json=payload)

### `crm_agent/dispatch.py`

````py
"""
Executes agent tool calls against CRMClient and shapes the results:
high-signal fields only, pagination guidance, and errors that tell the
agent what to do next instead of raw tracebacks.
"""

import json

from . import config
from .crm_client import CRMAPIError, CRMClient

client = CRMClient()


def _clamp_page_size(page_size):
    return max(1, min(int(page_size or config.DEFAULT_PAGE_SIZE), config.MAX_PAGE_SIZE))


def _pick(obj: dict, fields: list) -> dict:
    return {f: obj[f] for f in fields if f in obj}


def _paginated(resource_label: str, resp: dict, fields: list, narrow_hint: str) -> dict:
    items = resp.get("items", resp.get("results", []))
    total = resp.get("total", len(items))
    page = resp.get("page", 1)
    page_size = resp.get("page_size", len(items))
    shaped = [_pick(item, fields) for item in items]
    result = {"total": total, "page": page, "results": shaped}
    if total > page * page_size:
        result["note"] = (
            f"Showing {len(shaped)} of {total} {resource_label}. "
            f"Call again with page={page + 1} for more, or narrow the search with {narrow_hint}."
        )
    return result


_SEARCH_TOOL_BY_RESOURCE = {"contact": "search_contacts", "company": "search_companies", "deal": "search_deals"}


def _error_payload(e: CRMAPIError, resource: str) -> dict:
    if e.status_code == 404:
        search_tool = _SEARCH_TOOL_BY_RESOURCE.get(resource, f"search_{resource}s")
        msg = f"No {resource} found with that id. Double-check the id, or use {search_tool} first."
    elif e.status_code in (400, 422):
        msg = f"The CRM API rejected the request: {e.message}. Fix the flagged field(s) and retry."
    elif e.status_code == 429:
        msg = "The CRM API is rate-limiting requests. Wait a moment, then retry with a smaller page_size."
    elif e.status_code >= 500:
        msg = f"The CRM API returned a server error ({e.status_code}). Retry once; if it persists, tell the user."
    else:
        msg = f"CRM API error ({e.status_code}): {e.message}"
    return {"error": msg}


CONTACT_FIELDS = ["id", "name", "email", "phone", "company_id", "company_name"]
COMPANY_FIELDS = ["id", "name", "industry", "website"]
DEAL_FIELDS = ["id", "title", "amount", "stage", "company_id", "company_name", "contact_id", "contact_name"]


def search_contacts(query=None, email=None, company_id=None, page=1, page_size=20):
    resp = client.list_contacts(q=query, email=email, company_id=company_id,
                                 page=page, page_size=_clamp_page_size(page_size))
    return _paginated("contacts", resp, CONTACT_FIELDS, "email or company_id")


def get_contact(contact_id, include_deals=True):
    contact = _pick(client.get_contact(contact_id), CONTACT_FIELDS)
    if include_deals:
        deals_resp = client.list_deals(contact_id=contact_id, page=1, page_size=config.MAX_PAGE_SIZE)
        contact["deals"] = [_pick(d, DEAL_FIELDS) for d in deals_resp.get("items", deals_resp.get("results", []))]
    return contact


def create_contact(name, email=None, phone=None, company_id=None):
    payload = {"name": name, "email": email, "phone": phone, "company_id": company_id}
    created = client.create_contact({k: v for k, v in payload.items() if v is not None})
    return _pick(created, CONTACT_FIELDS)


def update_contact(contact_id, **fields):
    updated = client.update_contact(contact_id, {k: v for k, v in fields.items() if v is not None})
    return _pick(updated, CONTACT_FIELDS)


def delete_contact(contact_id):
    client.delete_contact(contact_id)
    return {"deleted": True, "contact_id": contact_id}


def search_companies(query=None, industry=None, page=1, page_size=20):
    resp = client.list_companies(q=query, industry=industry, page=page, page_size=_clamp_page_size(page_size))
    return _paginated("companies", resp, COMPANY_FIELDS, "industry")


def get_company(company_id):
    company = _pick(client.get_company(company_id), COMPANY_FIELDS)
    contacts_resp = client.list_contacts(company_id=company_id, page=1, page_size=config.MAX_PAGE_SIZE)
    deals_resp = client.list_deals(company_id=company_id, page=1, page_size=config.MAX_PAGE_SIZE)
    company["contacts"] = [_pick(c, CONTACT_FIELDS) for c in contacts_resp.get("items", contacts_resp.get("results", []))]
    company["deals"] = [_pick(d, DEAL_FIELDS) for d in deals_resp.get("items", deals_resp.get("results", []))]
    return company


def create_company(name, industry=None, website=None):
    payload = {"name": name, "industry": industry, "website": website}
    created = client.create_company({k: v for k, v in payload.items() if v is not None})
    return _pick(created, COMPANY_FIELDS)


def update_company(company_id, **fields):
    updated = client.update_company(company_id, {k: v for k, v in fields.items() if v is not None})
    return _pick(updated, COMPANY_FIELDS)


def search_deals(query=None, stage=None, company_id=None, contact_id=None,
                  min_amount=None, max_amount=None, page=1, page_size=20):
    resp = client.list_deals(q=query, stage=stage, company_id=company_id, contact_id=contact_id,
                              min_amount=min_amount, max_amount=max_amount,
                              page=page, page_size=_clamp_page_size(page_size))
    return _paginated("deals", resp, DEAL_FIELDS, "stage, company_id, or an amount range")


def get_deal(deal_id):
    return _pick(client.get_deal(deal_id), DEAL_FIELDS)


def create_deal(title, amount, stage="prospecting", company_id=None, contact_id=None):
    payload = {"title": title, "amount": amount, "stage": stage, "company_id": company_id, "contact_id": contact_id}
    created = client.create_deal({k: v for k, v in payload.items() if v is not None})
    return _pick(created, DEAL_FIELDS)


def update_deal(deal_id, **fields):
    updated = client.update_deal(deal_id, {k: v for k, v in fields.items() if v is not None})
    return _pick(updated, DEAL_FIELDS)


def add_note(entity_type, entity_id, body):
    created = client.create_note({"entity_type": entity_type, "entity_id": entity_id, "body": body})
    return {"note_id": created.get("id"), "entity_type": entity_type, "entity_id": entity_id}


TOOL_HANDLERS = {
    "search_contacts": lambda i: search_contacts(**i),
    "get_contact": lambda i: get_contact(**i),
    "create_contact": lambda i: create_contact(**i),
    "update_contact": lambda i: update_contact(i.pop("contact_id"), **i),
    "delete_contact": lambda i: delete_contact(**i),
    "search_companies": lambda i: search_companies(**i),
    "get_company": lambda i: get_company(**i),
    "create_company": lambda i: create_company(**i),
    "update_company": lambda i: update_company(i.pop("company_id"), **i),
    "search_deals": lambda i: search_deals(**i),
    "get_deal": lambda i: get_deal(**i),
    "create_deal": lambda i: create_deal(**i),
    "update_deal": lambda i: update_deal(i.pop("deal_id"), **i),
    "add_note": lambda i: add_note(**i),
}

# resource label used to pick the right 404/validation message per tool
_RESOURCE_BY_TOOL = {
    "search_contacts": "contact", "get_contact": "contact", "create_contact": "contact",
    "update_contact": "contact", "delete_contact": "contact",
    "search_companies": "company", "get_company": "company", "create_company": "company", "update_company": "company",
    "search_deals": "deal", "get_deal": "deal", "create_deal": "deal", "update_deal": "deal",
    "add_note": "note",
}


def execute_tool(name: str, tool_input: dict) -> str:
    """Run one tool call and return a JSON string suitable for a tool_result block."""
    handler = TOOL_HANDLERS.get(name)
    if handler is None:
        return json.dumps({"error": f"Unknown tool '{name}'."})
    try:
        result = handler(dict(tool_input))
        return json.dumps(result, default=str)
    except CRMAPIError as e:
        return json.dumps(_error_payload(e, _RESOURCE_BY_TOOL.get(name, "record")))
    except TypeError as e:
        return json.dumps({"error": f"Invalid arguments for {name}: {e}"})

````
