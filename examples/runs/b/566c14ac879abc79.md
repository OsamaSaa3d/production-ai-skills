````py
"""
Existing inventory functions. Replace this module's contents with your
real implementation — the integration in tool_agent.py only depends on
these two signatures and their return/raise behavior.
"""

_STOCK = {
    "SKU-001": 42,
    "SKU-002": 0,
    "SKU-003": 17,
}


def get_inventory(sku: str) -> dict:
    """Look up current stock for a SKU.

    Raises KeyError if the SKU does not exist.
    """
    if sku not in _STOCK:
        raise KeyError(f"Unknown SKU: {sku}")
    return {"sku": sku, "quantity_available": _STOCK[sku]}


def reserve_stock(sku: str, qty: int) -> dict:
    """Reserve `qty` units of `sku`.

    Raises KeyError if the SKU does not exist, and ValueError if there
    is not enough stock available to satisfy the reservation.
    """
    if sku not in _STOCK:
        raise KeyError(f"Unknown SKU: {sku}")
    if qty <= 0:
        raise ValueError("qty must be positive")
    if _STOCK[sku] < qty:
        raise ValueError(
            f"Insufficient stock for {sku}: requested {qty}, "
            f"available {_STOCK[sku]}"
        )
    _STOCK[sku] -= qty
    return {"sku": sku, "reserved": qty, "remaining": _STOCK[sku]}
````

````py
"""
Wires the existing get_inventory/reserve_stock functions up as tools the
model can call natively (Anthropic Messages API, strict mode). See the
llm-tool-calling skill for the pattern this follows.
"""

import json

import anthropic

from inventory import get_inventory, reserve_stock

MODEL = "claude-sonnet-5"  # pin explicitly; see model-selection

client = anthropic.Anthropic()

TOOLS = [
    {
        "name": "get_inventory",
        "description": (
            "Look up the current available quantity for a single SKU. "
            "Use this before reserving stock, or whenever the user asks "
            "how much of an item is in stock. Returns the SKU and the "
            "quantity currently available; it does not reserve anything. "
            "If the SKU does not exist, returns an error the model can "
            "surface to the user."
        ),
        "strict": True,
        "input_schema": {
            "type": "object",
            "properties": {
                "sku": {
                    "type": "string",
                    "description": "The exact SKU identifier to look up, e.g. 'SKU-001'.",
                },
            },
            "required": ["sku"],
            "additionalProperties": False,
        },
    },
    {
        "name": "reserve_stock",
        "description": (
            "Reserve a quantity of a SKU, decrementing available inventory. "
            "Only call this after confirming with the user (or with "
            "get_inventory) that enough stock exists — this call is not "
            "idempotent and each call reserves additional units. Returns "
            "the quantity reserved and the remaining stock, or a "
            "structured error if the SKU is unknown or stock is insufficient."
        ),
        "strict": True,
        "input_schema": {
            "type": "object",
            "properties": {
                "sku": {
                    "type": "string",
                    "description": "The exact SKU identifier to reserve, e.g. 'SKU-001'.",
                },
                "qty": {
                    "type": "integer",
                    "description": "Number of units to reserve. Must be a positive integer.",
                },
            },
            "required": ["sku", "qty"],
            "additionalProperties": False,
        },
    },
]

TOOL_REGISTRY = {
    "get_inventory": get_inventory,
    "reserve_stock": reserve_stock,
}

MAX_ITERATIONS = 10


def execute(block) -> dict:
    """Run one tool_use block, returning errors as data instead of raising."""
    name = block.name
    if name not in TOOL_REGISTRY:
        return {"error": f"No tool named {name!r}. Available: {sorted(TOOL_REGISTRY)}"}

    args = block.input  # already-parsed, no json.loads needed
    if name == "reserve_stock" and args.get("qty", 0) <= 0:
        # value constraint strict mode doesn't enforce — check it ourselves
        return {"error": "qty must be a positive integer"}

    try:
        return {"result": TOOL_REGISTRY[name](**args)}
    except Exception as e:
        return {"error": f"{type(e).__name__}: {e}"}


def run(user_message: str, max_iterations: int = MAX_ITERATIONS) -> str:
    messages = [{"role": "user", "content": user_message}]

    for _ in range(max_iterations):
        response = client.messages.create(
            model=MODEL,
            max_tokens=1024,
            tools=TOOLS,
            messages=messages,
        )

        if response.stop_reason == "refusal":
            return "The model declined to respond to this request."

        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason != "tool_use":
            return "".join(
                block.text for block in response.content if block.type == "text"
            )

        results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            outcome = execute(block)
            results.append({
                "type": "tool_result",
                "tool_use_id": block.id,
                "content": json.dumps(outcome),
                **({"is_error": True} if "error" in outcome else {}),
            })
        messages.append({"role": "user", "content": results})

    return "Stopped after max iterations without a final answer."


if __name__ == "__main__":
    print(run("How much stock do we have for SKU-001, and can you reserve 5 units?"))

````
