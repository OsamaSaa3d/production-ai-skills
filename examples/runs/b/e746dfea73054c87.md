````py
"""
Tool-calling loop that gives an Anthropic model access to the 14 CRM tools
in tools.py. Direct Messages API use, no framework — see the
llm-tool-calling skill for why.

Usage:
    export ANTHROPIC_API_KEY=...
    export CRM_API_BASE_URL=https://api.yourcrm.com/v1
    export CRM_API_KEY=...
    python agent.py "Find the contact jane@acme.com and add a note to her
                      open deal saying we followed up today"
"""

from __future__ import annotations

import json
import sys
from collections import Counter
from dataclasses import dataclass, field

import anthropic

from tools import TOOL_REGISTRY, TOOLS

MODEL = "claude-sonnet-5"
MAX_ITERATIONS = 10
MAX_TOKENS = 4096

SYSTEM_PROMPT = (
    "You are a CRM assistant. Use the provided tools to look up and modify "
    "contacts, companies, and deals. Prefer search tools to find an id before "
    "calling a tool that requires one — never guess an id. Ask the user before "
    "calling delete_contact with confirm=true; it is irreversible."
)

client = anthropic.Anthropic()


@dataclass
class RunResult:
    text: str | None
    stop: str  # "done" | "max_iterations" | "refusal" | "stuck"
    iterations: int
    tool_calls: list[str] = field(default_factory=list)


def execute_tool(name: str, tool_input: dict) -> dict:
    if name not in TOOL_REGISTRY:
        return {"error": f"No tool named {name!r}. Available: {sorted(TOOL_REGISTRY)}"}
    try:
        return TOOL_REGISTRY[name](**tool_input)
    except Exception as e:
        return {"error": f"{type(e).__name__}: {e}"}


def run(user_message: str, max_iterations: int = MAX_ITERATIONS) -> RunResult:
    messages = [{"role": "user", "content": user_message}]
    tool_calls: list[str] = []
    seen: Counter[tuple[str, str]] = Counter()

    for i in range(1, max_iterations + 1):
        response = client.messages.create(
            model=MODEL,
            max_tokens=MAX_TOKENS,
            system=SYSTEM_PROMPT,
            tools=TOOLS,
            messages=messages,
        )

        if response.stop_reason == "refusal":
            return RunResult(None, "refusal", i, tool_calls)

        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason != "tool_use":
            text = "".join(b.text for b in response.content if b.type == "text")
            return RunResult(text, "done", i, tool_calls)

        results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            tool_calls.append(block.name)

            signature = (block.name, json.dumps(block.input, sort_keys=True))
            seen[signature] += 1
            if seen[signature] >= 3:
                results.append({
                    "type": "tool_result",
                    "tool_use_id": block.id,
                    "content": (
                        "You've called this tool with these exact arguments 3 times "
                        "and gotten the same result. Stop retrying — either use "
                        "different arguments or report what you have to the user."
                    ),
                    "is_error": True,
                })
                continue

            output = execute_tool(block.name, block.input)
            results.append({
                "type": "tool_result",
                "tool_use_id": block.id,
                "content": json.dumps(output),
                "is_error": bool(isinstance(output, dict) and output.get("error")),
            })

        messages.append({"role": "user", "content": results})

    return RunResult(None, "max_iterations", max_iterations, tool_calls)


if __name__ == "__main__":
    prompt = " ".join(sys.argv[1:]) or "List the 5 most recently updated deals."
    result = run(prompt)
    print(f"[stop={result.stop} iterations={result.iterations} tool_calls={result.tool_calls}]\n")
    print(result.text or "(no final text — see stop reason above)")

````

````py
"""
Thin REST client for the CRM API.

ASSUMPTION: no endpoint spec was provided, so this targets a conventional
14-endpoint CRUD surface over contacts / companies / deals:

    GET    /contacts                list/search contacts
    GET    /contacts/{id}           get one contact
    POST   /contacts                create contact
    PATCH  /contacts/{id}           update contact
    DELETE /contacts/{id}           delete contact
    GET    /companies               list/search companies
    GET    /companies/{id}          get one company
    POST   /companies               create company
    PATCH  /companies/{id}          update company
    GET    /deals                   list/search deals
    GET    /deals/{id}              get one deal
    POST   /deals                   create deal
    PATCH  /deals/{id}              update deal
    POST   /deals/{id}/notes        add a note to a deal

If your real API differs, this is the only file that needs to change —
method signatures below are what tools.py depends on, not the URL shapes.
"""

from __future__ import annotations

import os
from typing import Any

import requests

BASE_URL = os.environ.get("CRM_API_BASE_URL", "https://api.example-crm.com/v1")
API_KEY = os.environ.get("CRM_API_KEY")
TIMEOUT_S = 15


class CRMAPIError(Exception):
    def __init__(self, status_code: int, body: str):
        self.status_code = status_code
        self.body = body
        super().__init__(f"CRM API returned {status_code}: {body[:500]}")


class CRMClient:
    def __init__(self, base_url: str = BASE_URL, api_key: str | None = API_KEY):
        if not api_key:
            raise RuntimeError("CRM_API_KEY environment variable is not set")
        self.base_url = base_url.rstrip("/")
        self.session = requests.Session()
        self.session.headers.update(
            {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
        )

    def _request(self, method: str, path: str, **kwargs) -> Any:
        resp = self.session.request(
            method, f"{self.base_url}{path}", timeout=TIMEOUT_S, **kwargs
        )
        if not resp.ok:
            raise CRMAPIError(resp.status_code, resp.text)
        if resp.status_code == 204 or not resp.content:
            return {}
        return resp.json()

    # -- contacts ----------------------------------------------------------

    def list_contacts(
        self,
        query: str | None = None,
        email: str | None = None,
        company_id: str | None = None,
        limit: int = 25,
        cursor: str | None = None,
    ) -> Any:
        params = {
            k: v
            for k, v in {
                "q": query,
                "email": email,
                "company_id": company_id,
                "limit": limit,
                "cursor": cursor,
            }.items()
            if v is not None
        }
        return self._request("GET", "/contacts", params=params)

    def get_contact(self, contact_id: str) -> Any:
        return self._request("GET", f"/contacts/{contact_id}")

    def create_contact(self, payload: dict) -> Any:
        return self._request("POST", "/contacts", json=payload)

    def update_contact(self, contact_id: str, payload: dict) -> Any:
        return self._request("PATCH", f"/contacts/{contact_id}", json=payload)

    def delete_contact(self, contact_id: str) -> Any:
        return self._request("DELETE", f"/contacts/{contact_id}")

    # -- companies -----------------------------------------------------------

    def list_companies(
        self, query: str | None = None, limit: int = 25, cursor: str | None = None
    ) -> Any:
        params = {
            k: v
            for k, v in {"q": query, "limit": limit, "cursor": cursor}.items()
            if v is not None
        }
        return self._request("GET", "/companies", params=params)

    def get_company(self, company_id: str) -> Any:
        return self._request("GET", f"/companies/{company_id}")

    def create_company(self, payload: dict) -> Any:
        return self._request("POST", "/companies", json=payload)

    def update_company(self, company_id: str, payload: dict) -> Any:
        return self._request("PATCH", f"/companies/{company_id}", json=payload)

    # -- deals ---------------------------------------------------------------

    def list_deals(
        self,
        query: str | None = None,
        stage: str | None = None,
        company_id: str | None = None,
        contact_id: str | None = None,
        limit: int = 25,
        cursor: str | None = None,
    ) -> Any:
        params = {
            k: v
            for k, v in {
                "q": query,
                "stage": stage,
                "company_id": company_id,
                "contact_id": contact_id,
                "limit": limit,
                "cursor": cursor,
            }.items()
            if v is not None
        }
        return self._request("GET", "/deals", params=params)

    def get_deal(self, deal_id: str) -> Any:
        return self._request("GET", f"/deals/{deal_id}")

    def create_deal(self, payload: dict) -> Any:
        return self._request("POST", "/deals", json=payload)

    def update_deal(self, deal_id: str, payload: dict) -> Any:
        return self._request("PATCH", f"/deals/{deal_id}", json=payload)

    def add_deal_note(self, deal_id: str, text: str) -> Any:
        return self._request("POST", f"/deals/{deal_id}/notes", json={"text": text})
````

````md
# CRM agent tools

Exposes the CRM REST API to an Anthropic-model agent as 14 tools.

## Files

- `crm_client.py` — thin REST client. **Has the assumed endpoint list and
  payload shapes at the top of the file** — swap these for your real paths,
  everything else is unaffected as long as the method signatures stay the same.
- `tools.py` — the 14 agent-facing tools: schemas, descriptions, and the
  handlers that call `crm_client` and shape responses.
- `agent.py` — the tool-calling loop (Anthropic Messages API, no framework).

## Assumption made

No endpoint spec was given, so this targets a conventional CRUD surface —
contacts / companies / deals, list+get+create+update+delete, 14 endpoints
total (see the docstring in `crm_client.py`). **Point `crm_client.py` at your
actual paths and response fields before running this for real** — field
names like `company_name` on a contact, or the shape of a paginated list
response (`{"items": [...], "next_cursor": ...}`), are guesses about a
typical CRM API, not confirmed facts about yours.

## Why 14 tools isn't a 1:1 endpoint wrap

Per the tool-design skill, wrapping each endpoint as a tool 1:1 is the
default anti-pattern. Here that's avoided two ways even though the tool
count still happens to equal the endpoint count:

- `get_contact`, `get_company`, and `get_deal` take an `include` parameter
  that folds in commonly-chained related data (a contact's open deals, a
  company's deals *and* contacts, a deal's notes) in one call, instead of
  making the agent chain a get + a filtered list itself.
- Create/update/delete stay as separate tools rather than a single
  `mode`-flag tool, because create/update/delete are semantics an agent must
  not invert — see "wrong arguments on a mode flag" in the tool-design
  skill. `delete_contact` additionally requires `confirm: true`, since it's
  the one irreversible call in the set.

14 tools is comfortably inside the ~10-20 safe zone before tool-selection
accuracy degrades, so no dispatcher or deferred-loading was needed here —
revisit if this tool set grows past ~20.
````

````py
"""
Agent-facing tools over the CRM API.

Deliberately NOT a 1:1 wrap of the 14 REST endpoints (see the tool-design
skill: "build tools for tasks, not for API endpoints"). Two consolidations
are applied on top of the raw client:

1. `get_contact` / `get_company` / `get_deal` take an `include` field that
   pulls in commonly-chained related data (e.g. a contact's open deals) in
   one call, instead of making the agent chain get -> list -> merge itself.
2. Every list/search tool returns a curated, high-signal field set (names
   instead of raw foreign-key ids where practical) and caps page size, so a
   broad search doesn't flood the transcript.

Reads and writes are named as separate tools (search/get vs create/update/
delete) rather than collapsed into one dispatcher, because create/update/
delete are semantics an agent must not invert. `delete_contact` additionally
requires an explicit `confirm: true` — the one irreversible operation in
this set.

14 tools total, one per endpoint-level task, with the destructive one
explicitly guarded and the read ones enriched via `include`.
"""

from __future__ import annotations

from crm_client import CRMAPIError, CRMClient

_client: CRMClient | None = None


def client() -> CRMClient:
    """Lazily construct the CRM client so importing this module (e.g. to
    inspect TOOLS) doesn't require CRM_API_KEY to already be set."""
    global _client
    if _client is None:
        _client = CRMClient()
    return _client


MAX_PAGE_SIZE = 25
DEFAULT_PAGE_SIZE = 10


def _nullable(schema: dict) -> dict:
    return {"anyOf": [schema, {"type": "null"}]}


def _err(message: str) -> dict:
    return {"error": message}


def _clamp_limit(limit: int | None) -> int:
    if limit is None:
        return DEFAULT_PAGE_SIZE
    return max(1, min(limit, MAX_PAGE_SIZE))

def search_contacts(
    query: str | None, email: str | None, company_id: str | None,
    limit: int | None, cursor: str | None,
) -> dict:
    try:
        page = client().list_contacts(
            query=query, email=email, company_id=company_id,
            limit=_clamp_limit(limit), cursor=cursor,
        )
    except CRMAPIError as e:
        return _err(f"Contact search failed ({e.status_code}): {e.body[:200]}")
    items = page.get("items", page if isinstance(page, list) else [])
    return {
        "contacts": [
            {
                "id": c["id"],
                "name": c.get("name"),
                "email": c.get("email"),
                "company_name": c.get("company_name"),
            }
            for c in items
        ],
        "next_cursor": page.get("next_cursor") if isinstance(page, dict) else None,
    }


def get_contact(contact_id: str, include: str) -> dict:
    try:
        contact = client().get_contact(contact_id)
    except CRMAPIError as e:
        if e.status_code == 404:
            return _err(f"No contact with id {contact_id!r}. Use search_contacts to find the right id.")
        return _err(f"get_contact failed ({e.status_code}): {e.body[:200]}")

    result = {
        "id": contact["id"],
        "name": contact.get("name"),
        "email": contact.get("email"),
        "phone": contact.get("phone"),
        "title": contact.get("title"),
        "company_id": contact.get("company_id"),
        "company_name": contact.get("company_name"),
    }
    if include == "with_deals":
        try:
            deals_page = client().list_deals(contact_id=contact_id, limit=5)
            deal_items = deals_page.get("items", [])
            result["open_deals"] = [
                {"id": d["id"], "title": d.get("title"), "stage": d.get("stage"), "value": d.get("value")}
                for d in deal_items
            ]
            if deals_page.get("next_cursor"):
                result["open_deals_truncated"] = True
        except CRMAPIError:

    return {"id": created["id"], "name": created.get("name"), "email": created.get("email")}

def update_deal(
    deal_id: str, title: str | None, value: float | None, stage: str | None,
    company_id: str | None, contact_id: str | None,
) -> dict:
    if stage is not None and stage not in DEAL_STAGES:
        return _err(f"Invalid stage {stage!r}. Must be one of: {', '.join(DEAL_STAGES)}.")
    payload = {k: v for k, v in {
        "title": title, "value": value, "stage": stage,
        "company_id": company_id, "contact_id": contact_id,
    }.items() if v is not None}
    if not payload:
        return _err("No fields provided to update. Set at least one of title/value/stage/company_id/contact_id.")
    try:
        updated = client().update_deal(deal_id, payload)
    except CRMAPIError as e:
        if e.status_code == 404:
            return _err(f"No deal with id {deal_id!r}. Use search_deals to find the right id.")
        return _err(f"update_deal failed ({e.status_code}): {e.body[:200]}")
    return {"id": updated["id"], "updated_fields": list(payload.keys())}


def add_deal_note(deal_id: str, text: str) -> dict:
    if not text.strip():
        return _err("Note text cannot be empty.")
    try:
        note = client().add_deal_note(deal_id, text)
    except CRMAPIError as e:
        if e.status_code == 404:
            return _err(f"No deal with id {deal_id!r}. Use search_deals to find the right id.")
        return _err(f"add_deal_note failed ({e.status_code}): {e.body[:200]}")
    return {"deal_id": deal_id, "note_id": note.get("id"), "added": True}
````
