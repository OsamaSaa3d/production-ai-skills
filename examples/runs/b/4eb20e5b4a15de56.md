Syntax checks out. Here's the design:

**`order_qa.py`** — a REPL that turns English questions into answers over the `orders` table, using Claude (Sonnet 5) with one strict tool: `query_orders`.

Key decision: the model **never writes SQL**. It can only pick from fixed enums — column names, comparison operators, aggregate functions, sort direction. The Python code assembles the actual SQL from a whitelist dict and binds every value as a parameterized query argument. That's what makes it safe: even a maliciously-crafted question can't produce anything outside the combinations already enumerated in the schema, so there's no SQL-injection surface regardless of what the model outputs.

Flow:
1. User types a question → sent to Claude with the `query_orders` tool.
2. Claude emits structured args (filters, optional aggregate/group_by/order_by, limit).
3. `_build_sql` validates and assembles parameterized SQL (e.g. `group_by` without `aggregate` is rejected as a query error and handed back to the model to retry).
4. Results go back as a `tool_result`; row count is capped at 200 server-side no matter what the model requests.
5. Claude turns the JSON rows/aggregate into a plain-English answer.

To run:
```
pip install -r requirements.txt
export ANTHROPIC_API_KEY=...
export DATABASE_URL=postgresql://user:pass@host:5432/dbname
python order_qa.py
```

Assumptions I made (flag if wrong): `status` values aren't known ahead of time so I didn't enum-constrain them (users can still filter/`like` on status text); grouping is limited to `status`/`customer_email` since grouping by `id`/`total_usd`/`placed_at` isn't meaningful without bucketing. I haven't run this against a live database — worth testing against real data before relying on it.

"""
Plain-English Q&A over the Postgres `orders` table.

Usage:
    export ANTHROPIC_API_KEY=...
    export DATABASE_URL=postgresql://user:pass@host:5432/dbname
    python order_qa.py

The model is given ONE strict tool, `query_orders`. It never writes raw SQL —
it can only pick columns/operators/aggregates from fixed enums, so the actual
SQL is assembled server-side from a whitelist and all values are bound as
query parameters. That's what makes this safe against prompt/SQL injection:
even a fully adversarial "question" can only produce combinations this code
already enumerates.
"""

import json
import os
import sys
from decimal import Decimal

import psycopg2
import psycopg2.extras
import anthropic

MODEL = "claude-sonnet-5"
MAX_ROWS = 200  # hard server-side cap, regardless of what the model asks for

COLUMNS = ["id", "customer_email", "status", "total_usd", "placed_at"]
COLUMN_CAST = {
    "id": "integer",
    "customer_email": "text",
    "status": "text",
    "total_usd": "numeric",
    "placed_at": "timestamp",
}
OPERATORS = {"=": "=", "!=": "!=", ">": ">", ">=": ">=", "<": "<", "<=": "<=", "like": "LIKE"}
AGGREGATES = {"count": "COUNT", "sum": "SUM", "avg": "AVG", "min": "MIN", "max": "MAX"}
GROUPABLE_COLUMNS = ["status", "customer_email"]

QUERY_ORDERS_TOOL = {
    "name": "query_orders",
    "description": (
        "Query the orders table (columns: id, customer_email, status, total_usd, placed_at). "
        "Use this for any question about orders: looking up individual orders, filtering by "
        "customer/status/date/amount, or computing aggregates like counts, sums, or averages, "
        "optionally grouped by status or customer_email. Returns rows or a single aggregate "
        "value as JSON. Does not modify data — read-only."
    ),
    "strict": True,
    "input_schema": {
        "type": "object",
        "properties": {
            "filters": {
                "type": "array",
                "description": (
                    "Conditions to filter orders by, combined with AND. Empty array = no filter."
                ),
                "items": {
                    "type": "object",
                    "properties": {
                        "column": {"type": "string", "enum": COLUMNS},
                        "operator": {
                            "type": "string",
                            "enum": list(OPERATORS.keys()),
                            "description": "Comparison operator. Use 'like' only with customer_email or status, with SQL wildcards, e.g. '%@example.com'.",
                        },
                        "value": {
                            "type": "string",
                            "description": (
                                "Comparison value as a string. For total_usd use a plain number "
                                "like '100.00'. For placed_at use an ISO date or timestamp like "
                                "'2026-01-01' or '2026-01-01T00:00:00'."
                            ),
                        },
                    },
                    "required": ["column", "operator", "value"],
                    "additionalProperties": False,
                },
            },
            "aggregate": {
                "anyOf": [
                    {
                        "type": "object",
                        "properties": {
                            "function": {"type": "string", "enum": list(AGGREGATES.keys())},
                            "column": {"type": "string", "enum": COLUMNS},
                        },
                        "required": ["function", "column"],
                        "additionalProperties": False,
                    },
                    {"type": "null"},
                ],
                "description": (
                    "Set to compute a single aggregate (e.g. sum of total_usd) instead of "
                    "returning individual rows. Use column 'id' with function 'count' to count rows. "
                    "Null means return raw rows."
                ),
            },
            "group_by": {
                "anyOf": [{"type": "string", "enum": GROUPABLE_COLUMNS}, {"type": "null"}],
                "description": "Group aggregate results by this column (e.g. revenue per status). Only used with aggregate. Null means no grouping.",
            },
            "order_by": {
                "anyOf": [
                    {
                        "type": "object",
                        "properties": {
                            "column": {"type": "string", "enum": COLUMNS},
                            "direction": {"type": "string", "enum": ["asc", "desc"]},
                        },
                        "required": ["column", "direction"],
                        "additionalProperties": False,
                    },
                    {"type": "null"},
                ],
                "description": "How to sort raw-row results. Null defaults to placed_at desc.",
            },
            "limit": {
                "type": "integer",
                "description": f"Max rows to return. Use a small number unless the user asks for more; server caps this at {MAX_ROWS} regardless.",
            },
        },
        "required": ["filters", "aggregate", "group_by", "order_by", "limit"],
        "additionalProperties": False,
    },
}


class QueryError(Exception):
    pass


def _build_sql(filters, aggregate, group_by, order_by, limit):
    where_clauses = []
    params = []
    for f in filters:
        col, op, val = f["column"], f["operator"], f["value"]
        if op == "like" and col not in ("customer_email", "status"):
            raise QueryError(f"'like' is only valid on customer_email or status, not {col}")
        cast = COLUMN_CAST[col]
        if op == "like":
            where_clauses.append(f"{col} LIKE %s")
            params.append(val)
        else:
            where_clauses.append(f"{col} {OPERATORS[op]} %s::{cast}")
            params.append(val)
    where_sql = f"WHERE {' AND '.join(where_clauses)}" if where_clauses else ""

    if group_by and not aggregate:
        raise QueryError("group_by requires aggregate to be set")

    if aggregate:
        func = AGGREGATES[aggregate["function"]]
        agg_col = aggregate["column"]
        select_sql = f"{func}({agg_col}) AS result"
        group_sql = ""
        if group_by:
            select_sql = f"{group_by}, {select_sql}"
            group_sql = f"GROUP BY {group_by}"
        order_sql = f"ORDER BY {group_by}" if group_by else ""
        sql = f"SELECT {select_sql} FROM orders {where_sql} {group_sql} {order_sql}".strip()
        return sql, params

    select_sql = "id, customer_email, status, total_usd, placed_at"
    if order_by:
        order_sql = f"ORDER BY {order_by['column']} {order_by['direction'].upper()}"
    else:
        order_sql = "ORDER BY placed_at DESC"
    safe_limit = max(1, min(int(limit), MAX_ROWS))
    sql = f"SELECT {select_sql} FROM orders {where_sql} {order_sql} LIMIT %s".strip()
    params.append(safe_limit)
    return sql, params


def _row_to_json(row):
    out = {}
    for k, v in row.items():
        if isinstance(v, Decimal):
            out[k] = float(v)
        elif hasattr(v, "isoformat"):
            out[k] = v.isoformat()
        else:
            out[k] = v
    return out


def query_orders(conn, filters, aggregate, group_by, order_by, limit):
    sql, params = _build_sql(filters, aggregate, group_by, order_by, limit)
    with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
        cur.execute(sql, params)
        rows = cur.fetchall()
    return {"row_count": len(rows), "rows": [_row_to_json(r) for r in rows]}


def ask(client, conn, question, max_turns=4):
    messages = [{"role": "user", "content": question}]

    for _ in range(max_turns):
        response = client.messages.create(
            model=MODEL,
            max_tokens=1024,
            tools=[QUERY_ORDERS_TOOL],
            messages=messages,
        )

        if response.stop_reason == "refusal":
            return "I can't help with that question."
        if response.stop_reason == "max_tokens":
            return "The answer was too long and got cut off — try a narrower question."

        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason != "tool_use":
            return "".join(b.text for b in response.content if b.type == "text")

        tool_results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            try:
                result = query_orders(conn, **block.input)
                content = json.dumps(result, default=str)
                is_error = False
            except (QueryError, psycopg2.Error) as e:
                conn.rollback()
                content = f"Query error: {e}. Adjust the filters/columns and try again."
                is_error = True
            tool_results.append(
                {
                    "type": "tool_result",
                    "tool_use_id": block.id,
                    "content": content,
                    "is_error": is_error,
                }
            )
        messages.append({"role": "user", "content": tool_results})

    return "I couldn't resolve that question in time — try rephrasing it."


def main():
    conn = psycopg2.connect(os.environ.get("DATABASE_URL", ""))
    client = anthropic.Anthropic()

    print("Ask a question about orders (Ctrl+C to quit).")
    try:
        while True:
            question = input("\n> ").strip()
            if not question:
                continue
            print(ask(client, conn, question))
    except (KeyboardInterrupt, EOFError):
        print()
    finally:
        conn.close()


if __name__ == "__main__":
    main()
