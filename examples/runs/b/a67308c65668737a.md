Built `ticket_search_tool.py` — a `search_tickets` tool the agent calls with typed fields (`status`, `team`, `priority`, `assignee`, date ranges, free `text`), never the raw DSL string. Key points, per the tool-design skill's guidance on query languages:

- **`build_query()`** is the only place that knows the `field:value AND field:>date` grammar — it assembles the query string from validated, enum-constrained inputs. The model can't produce a malformed or injected query because it never emits query syntax.
- **Enums for `status`/`team`/`priority`** make invalid values structurally impossible; I seeded plausible values (`open/in_progress/pending/resolved/closed`, `platform/billing/growth/infra/support`) — swap these for your real ones before shipping, since I couldn't verify them against a live system.
- **`input_examples`** show minimal, filtered, and detailed-follow-up call shapes so date format and field co-occurrence are unambiguous.
- **`response_format` enum** (concise/detailed) keeps normal results cheap and only pulls assignee/description when needed.
- **Capped, trimmed results** (`MAX_RESULTS = 25`) with a `note` steering the agent to narrow filters instead of raising the limit.
- **Errors name the next action** (timeout → narrow filter; 429 → wait and retry) instead of surfacing raw exceptions.
- A minimal agent loop at the bottom wires it into the Anthropic Messages API tool-use cycle.

Needs `TICKET_SEARCH_URL` and `TICKET_SEARCH_API_TOKEN` env vars, plus `pip install requests anthropic`. I couldn't test against a real endpoint since none was provided — you'll want to point it at your actual search API and confirm the status/team/priority value lists match.

"""
Ticket search tool for an LLM agent.

The ticket system's search endpoint takes a query-DSL string, e.g.:

    status:open AND team:platform AND created:>2026-01-01

Per the tool-design guidance on query languages: never hand the agent
the raw DSL and ask it to emit query strings. Give it typed, enum-
constrained fields instead, and build the DSL string here in code.
That keeps invalid states unrepresentable (an enum of five statuses
can't produce a sixth) and keeps the grammar itself out of the system
prompt, where it would compete for attention on every single turn.

Adjust STATUSES / TEAMS / PRIORITIES below to match your real ticket
system's values before shipping this -- the ones here are reasonable
guesses, not confirmed against any live API.
"""

from __future__ import annotations

import os
import re
from datetime import date
from typing import Any

import requests

# --------------------------------------------------------------------------
# Config -- confirm these against your actual ticket system.
# --------------------------------------------------------------------------

SEARCH_URL = os.environ["TICKET_SEARCH_URL"]  # e.g. https://tickets.internal/api/v1/search
API_TOKEN = os.environ["TICKET_SEARCH_API_TOKEN"]

STATUSES = ["open", "in_progress", "pending", "resolved", "closed"]
TEAMS = ["platform", "billing", "growth", "infra", "support"]
PRIORITIES = ["low", "medium", "high", "critical"]

MAX_RESULTS = 25  # hard cap on rows returned to the agent per call
DATE_RE = re.compile(r"^\d{4}-\d{2}-\d{2}$")


# --------------------------------------------------------------------------
# Tool definition (Anthropic Messages API tool-use format)
# --------------------------------------------------------------------------

SEARCH_TICKETS_TOOL = {
    "name": "search_tickets",
    "description": (
        "Search tickets by status, team, priority, assignee, free text, and "
        "creation/update date range. Returns id, title, status, team, priority, "
        "and created_at for each match. Use response_format='detailed' only when "
        "you need a ticket's assignee or full description for a follow-up action."
    ),
    "input_schema": {
        "type": "object",
        "properties": {
            "text": {
                "type": "string",
                "description": "Free-text match against ticket title and body.",
            },
            "status": {
                "type": "string",
                "enum": STATUSES,
                "description": "Exact ticket status.",
            },
            "team": {
                "type": "string",
                "enum": TEAMS,
                "description": "Owning team.",
            },
            "priority": {
                "type": "string",
                "enum": PRIORITIES,
            },
            "assignee": {
                "type": "string",
                "description": "Assignee username or email, exact match.",
            },
            "created_after": {
                "type": "string",
                "format": "date",
                "description": "ISO date (YYYY-MM-DD). Matches tickets created on or after this date.",
            },
            "created_before": {
                "type": "string",
                "format": "date",
                "description": "ISO date (YYYY-MM-DD). Matches tickets created on or before this date.",
            },
            "updated_after": {
                "type": "string",
                "format": "date",
                "description": "ISO date (YYYY-MM-DD). Matches tickets updated on or after this date.",
            },
            "limit": {
                "type": "integer",
                "description": f"Max rows to return (default 10, max {MAX_RESULTS}).",
            },
            "response_format": {
                "type": "string",
                "enum": ["concise", "detailed"],
                "description": "Use 'concise' unless you need assignee/description for a follow-up call.",
            },
        },
        "additionalProperties": False,
    },
    "input_examples": [
        # minimal
        {"text": "login page 500 error"},
        # common filtered search
        {"status": "open", "team": "platform", "created_after": "2026-01-01"},
        # narrow follow-up needing full detail
        {
            "status": "open",
            "team": "billing",
            "priority": "critical",
            "response_format": "detailed",
            "limit": 5,
        },
    ],
}


# --------------------------------------------------------------------------
# Query construction -- the only place that knows the DSL grammar.
# --------------------------------------------------------------------------


def _validate_date(value: str, field: str) -> None:
    if not DATE_RE.match(value):
        raise ValueError(f"{field} must be an ISO date (YYYY-MM-DD), got {value!r}")


def build_query(
    *,
    text: str | None = None,
    status: str | None = None,
    team: str | None = None,
    priority: str | None = None,
    assignee: str | None = None,
    created_after: str | None = None,
    created_before: str | None = None,
    updated_after: str | None = None,
) -> str:
    """Build the ticket system's query-DSL string from typed fields.

    This is the single place that touches the DSL grammar. The agent
    never sees or emits this string -- it only ever provides the typed
    arguments above.
    """
    clauses: list[str] = []

    if status is not None:
        if status not in STATUSES:
            raise ValueError(f"status must be one of {STATUSES}, got {status!r}")
        clauses.append(f"status:{status}")

    if team is not None:
        if team not in TEAMS:
            raise ValueError(f"team must be one of {TEAMS}, got {team!r}")
        clauses.append(f"team:{team}")

    if priority is not None:
        if priority not in PRIORITIES:
            raise ValueError(f"priority must be one of {PRIORITIES}, got {priority!r}")
        clauses.append(f"priority:{priority}")

    if assignee is not None:
        clauses.append(f'assignee:"{assignee}"')

    if created_after is not None:
        _validate_date(created_after, "created_after")
        clauses.append(f"created:>{created_after}")

    if created_before is not None:
        _validate_date(created_before, "created_before")
        clauses.append(f"created:<{created_before}")

    if updated_after is not None:
        _validate_date(updated_after, "updated_after")
        clauses.append(f"updated:>{updated_after}")

    if text is not None:
        escaped = text.replace('"', '\\"')
        clauses.append(f'text:"{escaped}"')

    if not clauses:
        raise ValueError("search_tickets requires at least one filter field")

    return " AND ".join(clauses)


# --------------------------------------------------------------------------
# Tool implementation
# --------------------------------------------------------------------------

CONCISE_FIELDS = ("id", "title", "status", "team", "priority", "created_at")
DETAILED_FIELDS = CONCISE_FIELDS + ("assignee", "description", "updated_at")


def search_tickets(
    *,
    text: str | None = None,
    status: str | None = None,
    team: str | None = None,
    priority: str | None = None,
    assignee: str | None = None,
    created_after: str | None = None,
    created_before: str | None = None,
    updated_after: str | None = None,
    limit: int = 10,
    response_format: str = "concise",
) -> dict[str, Any]:
    """Execute the search and shape the response for the agent.

    Returns a dict with either "results" (+ "total") or "error". On
    error the message names the next action, per the tool-design
    guidance on steering rather than surfacing tracebacks.
    """
    try:
        query = build_query(
            text=text,
            status=status,
            team=team,
            priority=priority,
            assignee=assignee,
            created_after=created_after,
            created_before=created_before,
            updated_after=updated_after,
        )
    except ValueError as exc:
        return {"error": str(exc)}

    capped_limit = max(1, min(limit, MAX_RESULTS))

    try:
        resp = requests.get(
            SEARCH_URL,
            params={"q": query, "limit": capped_limit},
            headers={"Authorization": f"Bearer {API_TOKEN}"},
            timeout=10,
        )
        resp.raise_for_status()
    except requests.exceptions.Timeout:
        return {"error": "Ticket search timed out. Retry with a narrower filter (e.g. add a date range or team)."}
    except requests.exceptions.HTTPError as exc:
        status_code = exc.response.status_code if exc.response is not None else None
        if status_code == 429:
            return {"error": "Ticket search is rate-limited. Wait a moment before retrying."}
        return {"error": f"Ticket search failed ({status_code}). Check filter values and retry."}
    except requests.exceptions.RequestException:
        return {"error": "Could not reach the ticket search service. Retry once; if it persists, report the outage."}

    payload = resp.json()
    rows = payload.get("results", payload if isinstance(payload, list) else [])
    total = payload.get("total", len(rows))

    fields = DETAILED_FIELDS if response_format == "detailed" else CONCISE_FIELDS
    trimmed = [{k: row.get(k) for k in fields if k in row} for row in rows[:capped_limit]]

    result: dict[str, Any] = {"query": query, "total": total, "results": trimmed}
    if total > len(trimmed):
        result["note"] = (
            f"Showing {len(trimmed)} of {total} matches. Narrow the search "
            "(add status/team/date filters) rather than raising limit further."
        )
    return result


# --------------------------------------------------------------------------
# Minimal agent loop wiring the tool in (Anthropic Messages API)
# --------------------------------------------------------------------------


def run_agent(user_message: str, model: str = "claude-sonnet-5") -> str:
    import anthropic

    client = anthropic.Anthropic()
    messages: list[dict[str, Any]] = [{"role": "user", "content": user_message}]

    while True:
        response = client.messages.create(
            model=model,
            max_tokens=1024,
            tools=[SEARCH_TICKETS_TOOL],
            messages=messages,
        )
        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason != "tool_use":
            return "".join(block.text for block in response.content if block.type == "text")

        tool_results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            if block.name == "search_tickets":
                output = search_tickets(**block.input)
            else:
                output = {"error": f"Unknown tool {block.name}"}
            tool_results.append(
                {
                    "type": "tool_result",
                    "tool_use_id": block.id,
                    "content": str(output),
                }
            )
        messages.append({"role": "user", "content": tool_results})


if __name__ == "__main__":
    import sys

    question = " ".join(sys.argv[1:]) or "Find open platform team tickets created after Jan 1 2026"
    print(run_agent(question))
