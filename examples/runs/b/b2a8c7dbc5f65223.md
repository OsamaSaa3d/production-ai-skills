import anthropic

SYSTEM_PROMPT = (
    "You summarize meeting transcripts for a busy manager who has 30 seconds to skim. "
    "Write exactly one paragraph (3-6 sentences, no headings, no bullet points) covering: "
    "the meeting's purpose, the key decisions made, any action items with owners if named, "
    "and open questions or blockers. Be concrete and skip small talk."
)


def summarize_meeting_transcript(
    transcript: str,
    client: anthropic.Anthropic | None = None,
) -> str:
    """Summarize a meeting transcript into a short, skimmable paragraph."""
    if not transcript.strip():
        raise ValueError("transcript is empty")

    client = client or anthropic.Anthropic()

    try:
        response = client.messages.create(
            model="claude-opus-5",
            max_tokens=1024,
            output_config={"effort": "low"},
            system=SYSTEM_PROMPT,
            messages=[{"role": "user", "content": transcript}],
        )
    except anthropic.RateLimitError as e:
        retry_after = e.response.headers.get("retry-after", "60")
        raise RuntimeError(f"rate limited, retry after {retry_after}s") from e
    except anthropic.APIStatusError as e:
        raise RuntimeError(f"Claude API error ({e.status_code}): {e.message}") from e
    except anthropic.APIConnectionError as e:
        raise RuntimeError("network error calling Claude API") from e

    if response.stop_reason == "refusal":
        raise RuntimeError(f"summary refused: {response.stop_details}")

    return next(block.text for block in response.content if block.type == "text")


if __name__ == "__main__":
    example_transcript = """
    Alice: Let's kick off the Q3 roadmap sync. First up, the mobile redesign.
    Bob: Design is done, engineering starts Monday. Should ship by end of August.
    Alice: Great. Second, the API rate limiting issue from last week.
    Carol: I root-caused it, fix is in review. Bob, can you review my PR today?
    Bob: Yes, will do by EOD.
    Alice: Last thing, we still don't have a decision on the pricing page copy.
    Carol: Marketing owes us that by Friday, we're blocked until then.
    Alice: Okay, let's reconvene next week. Thanks everyone.
    """
    print(summarize_meeting_transcript(example_transcript))
