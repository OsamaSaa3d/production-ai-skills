Wrote `triage_emails.py` — compiles cleanly (not run live, since it needs an `OPENAI_API_KEY` and network access).

How it works:
- **Schema**: `Urgency` enum (`critical`/`high`/`low`/`medium`) with the level definitions in the field description, `reasoning` generated *before* the label (per skill guidance — thinking, not rationalization), and each `EmailTriage` echoes the input `id`.
- **Batching**: emails classified in chunks of 15 via `client.chat.completions.parse(response_format=BatchTriage)` (strict structured output — no JSON parsing/regex needed). Results are matched back by id, not position; any missing ids are re-run individually rather than silently dropped.
- **Failure handling**: refusals are logged and raised distinctly (not retried); truncation is retried once with doubled `max_tokens` — matching the two failure modes strict mode doesn't eliminate.
- **Routing**: `ROUTING` maps each urgency level to a queue + SLA (oncall pager for critical down to a backlog queue for low), and `route()` produces the dicts you'd hand to a router.

To use it for real: set `OPENAI_API_KEY`, swap `MODEL` for whatever you're standardized on, and replace the `sample_emails` block with your actual email source.

"""Classify incoming support emails by urgency for routing.

Uses OpenAI structured outputs (strict JSON schema) so the response is
guaranteed to match the schema below -- no fence-stripping or retry-on-
malformed-JSON needed. See classify_batch() for the two failure modes
(refusal, truncation) that strict mode does *not* eliminate.
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from enum import Enum

from openai import OpenAI
from pydantic import BaseModel, Field

logging.basicConfig(level=logging.INFO)
log = logging.getLogger("triage")

MODEL = "gpt-4.1"
BATCH_SIZE = 15  # keep truncation cheap: one bad batch, not one bad hundred

client = OpenAI()


class Urgency(str, Enum):
    CRITICAL = "critical"
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"


# Where each urgency level routes to, and the SLA that implies.
ROUTING = {
    Urgency.CRITICAL: ("oncall_pager", "15 minutes"),
    Urgency.HIGH: ("priority_queue", "4 hours"),
    Urgency.MEDIUM: ("standard_queue", "1 business day"),
    Urgency.LOW: ("backlog_queue", "3 business days"),
}


class EmailTriage(BaseModel):
    id: str = Field(description="Echo the input email id exactly.")
    reasoning: str = Field(
        description="One sentence citing the phrase(s) that decided the urgency."
    )
    urgency: Urgency = Field(
        description=(
            "critical: production down, data loss, security breach, customer "
            "actively blocked with no workaround, explicit threat to cancel/escalate. "
            "high: major feature broken, no workaround, affects many users, "
            "billing failure blocking access. "
            "medium: partial functionality issue, workaround exists, single user "
            "affected, general how-to question tied to a real blocker. "
            "low: feature request, praise/complaint with no blocker, question with "
            "no urgency indicated, documentation request."
        )
    )


class BatchTriage(BaseModel):
    results: list[EmailTriage]


class Refused(Exception):
    pass


class Truncated(Exception):
    pass


@dataclass
class Email:
    id: str
    subject: str
    body: str


def _format_batch(emails: list[Email]) -> str:
    parts = []
    for e in emails:
        parts.append(f"id: {e.id}\nsubject: {e.subject}\nbody: {e.body}")
    return "\n\n---\n\n".join(parts)


def _classify_chunk(emails: list[Email], max_tokens: int, max_retries: int) -> list[EmailTriage]:
    messages = [
        {
            "role": "system",
            "content": (
                "You triage incoming customer support emails by urgency so they "
                "can be routed to the right queue. Classify every email in the "
                "batch and echo its id."
            ),
        },
        {"role": "user", "content": _format_batch(emails)},
    ]

    attempt = 0
    while True:
        attempt += 1
        completion = client.chat.completions.parse(
            model=MODEL,
            messages=messages,
            response_format=BatchTriage,
            max_completion_tokens=max_tokens,
        )
        choice = completion.choices[0]

        if choice.message.refusal:
            log.warning("triage_refused: %s", choice.message.refusal)
            raise Refused(choice.message.refusal)

        if choice.finish_reason == "length":
            log.warning("triage_truncated at max_tokens=%d", max_tokens)
            if attempt <= max_retries:
                max_tokens *= 2
                continue
            raise Truncated(f"still truncated at max_tokens={max_tokens}")

        return choice.message.parsed.results


def classify_batch(
    emails: list[Email], max_tokens: int = 2048, max_retries: int = 1
) -> list[EmailTriage]:
    """Classify emails in chunks, verifying every id comes back before trusting it."""
    results: list[EmailTriage] = []

    for start in range(0, len(emails), BATCH_SIZE):
        chunk = emails[start : start + BATCH_SIZE]
        chunk_results = _classify_chunk(chunk, max_tokens, max_retries)

        returned_ids = {r.id for r in chunk_results}
        expected_ids = {e.id for e in chunk}
        missing = expected_ids - returned_ids
        if missing:
            log.warning("triage_missing_ids, re-running individually: %s", missing)
            missing_emails = [e for e in chunk if e.id in missing]
            for email in missing_emails:
                chunk_results.extend(_classify_chunk([email], max_tokens, max_retries))

        results.extend(chunk_results)

    return results


def route(triaged: list[EmailTriage]) -> list[dict]:
    routed = []
    for t in triaged:
        queue, sla = ROUTING[t.urgency]
        routed.append(
            {
                "id": t.id,
                "urgency": t.urgency.value,
                "queue": queue,
                "sla": sla,
                "reasoning": t.reasoning,
            }
        )
    return routed


if __name__ == "__main__":
    sample_emails = [
        Email(
            id="1",
            subject="Production API returning 500s",
            body="All our checkout requests have been failing for 20 minutes. "
            "We are losing sales right now, please help immediately.",
        ),
        Email(
            id="2",
            subject="Can't export reports to CSV",
            body="Export button does nothing when I click it. Not urgent, "
            "I can pull the data manually for now, but wanted to flag it.",
        ),
        Email(
            id="3",
            subject="Feature idea: dark mode",
            body="Would be nice to have a dark mode in the dashboard someday.",
        ),
        Email(
            id="4",
            subject="Billing charged twice",
            body="I was charged twice for my subscription this month and now "
            "my card is maxed out, I can't make any other payments.",
        ),
    ]

    triaged = classify_batch(sample_emails)
    for row in route(triaged):
        print(row)
