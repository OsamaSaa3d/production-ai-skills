````py
"""Minimal Claude agent with a single tool: fetch a user by id.

Requires: pip install anthropic
Credentials: set ANTHROPIC_API_KEY, or `ant auth login`.
"""

import sys

import anthropic
from anthropic import beta_tool

# Placeholder user store - replace with a real database/API call.
_USERS = {
    "1": {"id": "1", "name": "Ada Lovelace", "email": "ada@example.com"},
    "2": {"id": "2", "name": "Grace Hopper", "email": "grace@example.com"},
    "3": {"id": "3", "name": "Alan Turing", "email": "alan@example.com"},
}


@beta_tool
def get_user(user_id: str) -> str:
    """Fetch a user by their id.

    Args:
        user_id: The unique identifier of the user to look up.
    """
    user = _USERS.get(user_id)
    if user is None:
        return f"No user found with id '{user_id}'."
    return f"id={user['id']} name={user['name']} email={user['email']}"


def run(prompt: str) -> str:
    client = anthropic.Anthropic()

    runner = client.beta.messages.tool_runner(
        model="claude-opus-5",
        max_tokens=16000,
        tools=[get_user],
        messages=[{"role": "user", "content": prompt}],
    )

    final = None
    for message in runner:
        final = message

    if final is None:
        return ""
    return "".join(block.text for block in final.content if block.type == "text")


if __name__ == "__main__":
    query = " ".join(sys.argv[1:]) or "Look up the user with id 2."
    print(run(query))

````
