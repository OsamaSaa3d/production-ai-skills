**Built and verified:** `fix_failing_test.py` — a tool that takes any failing-test shell command and drives it to green.

How it works:
1. Runs your test command; if it passes, exits immediately.
2. If it fails, invokes headless `claude -p` with the failure output, scoped via `--allowedTools` to only `Read`/`Edit`/`Glob`/`Grep` and a `Bash` permission narrowed to the test command's own executable (e.g. it could run `pytest ...` but nothing else) — so a "fix" can't wander into unrelated shell commands. Capped by `--max-budget-usd` per attempt.
3. Reruns the test command itself (not trusting Claude's self-report) to verify, and loops up to `--max-attempts` (default 3).

Usage:
```
python fix_failing_test.py "pytest path/to/test.py -q"
python fix_failing_test.py "npm test" --max-attempts 5 --max-budget-usd 1.5
```

### `fix_failing_test.py`

````py
#!/usr/bin/env python3
"""Take a failing test command and get it passing, autonomously.

Runs TEST_CMD; if it fails, hands the failure output to Claude Code
(headless `claude -p`) scoped to Read/Edit/Glob/Grep and to running
TEST_CMD itself via Bash, then reruns the tests. Repeats until the
tests pass or --max-attempts is exhausted.

Usage:
    python fix_failing_test.py "pytest test_calculator.py -q"
    python fix_failing_test.py "npm test" --max-attempts 5 --max-budget-usd 1.5
"""

import argparse
import shlex
import shutil
import subprocess
import sys


def run_tests(test_cmd: str) -> tuple[bool, str]:
    result = subprocess.run(test_cmd, shell=True, capture_output=True, text=True)
    return result.returncode == 0, (result.stdout + result.stderr)


def resolve_claude() -> str:
    # On Windows, "claude" alone resolves to claude.ps1, which subprocess
    # (shell=False) can't exec directly. claude.cmd is the runnable shim.
    for name in ("claude.cmd", "claude"):
        path = shutil.which(name)
        if path:
            return path
    sys.exit("error: 'claude' CLI not found on PATH")


def invoke_claude(claude_path: str, test_cmd: str, failure_output: str,
                   allow_bash_prefix: str, max_budget_usd: float) -> subprocess.CompletedProcess:
    prompt = f"""A test is failing in this repository. Make it pass by fixing the
underlying source code. Do not edit, weaken, skip, or delete the test itself
unless the test is factually incorrect about the intended behavior.

Command to make pass:
    {test_cmd}

Most recent failure output:
---
{failure_output[-8000:]}
---

Investigate the relevant source, apply the minimal correct fix, and rerun the
command yourself to confirm it passes before finishing."""

    cmd = [
        claude_path, "-p", prompt,
        "--permission-mode", "acceptEdits",
        "--permission-prompts", "none",
        "--allowedTools", "Read", "Edit", "Glob", "Grep", f"Bash({allow_bash_prefix} *)",
        "--max-budget-usd", str(max_budget_usd),
        "--no-session-persistence",
    ]
    return subprocess.run(cmd, shell=False, capture_output=True, text=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("test_cmd", help="Shell command that runs the failing test, e.g. 'pytest test_calculator.py -q'")
    parser.add_argument("--max-attempts", type=int, default=3, help="Max fix attempts before giving up (default: 3)")
    parser.add_argument("--max-budget-usd", type=float, default=2.0, help="Per-attempt USD budget cap passed to claude -p (default: 2.0)")
    args = parser.parse_args()

    claude_path = resolve_claude()
    # First word of the test command is the only Bash prefix Claude is allowed to run,
    # so a "fix" attempt can't wander off into unrelated shell commands.
    allow_bash_prefix = shlex.split(args.test_cmd)[0]

    for attempt in range(1, args.max_attempts + 1):
        print(f"[{attempt}/{args.max_attempts}] running: {args.test_cmd}")
        passed, output = run_tests(args.test_cmd)
        if passed:
            print("PASSING.")
            return 0

        print(f"[{attempt}/{args.max_attempts}] failing, invoking claude to fix it...")
        result = invoke_claude(claude_path, args.test_cmd, output, allow_bash_prefix, args.max_budget_usd)
        if result.stdout.strip():
            print(result.stdout.strip())
        if result.returncode != 0 and result.stderr.strip():
            print(result.stderr.strip(), file=sys.stderr)

    print(f"[final check] running: {args.test_cmd}")
    passed, output = run_tests(args.test_cmd)
    if passed:
        print("PASSING.")
        return 0

    print(f"STILL FAILING after {args.max_attempts} attempts:\n{output}")
    return 1


if __name__ == "__main__":
    sys.exit(main())

````
