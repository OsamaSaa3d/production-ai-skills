### `README.md`

````md
# crm-api MCP server

Exposes your CRM's REST API to an AI agent (e.g. Claude Code) as MCP tools —
one tool per endpoint, generated from [`endpoints.json`](./endpoints.json).

**⚠️ The endpoints currently in `endpoints.json` are placeholders.** No API
spec was available when this was built, so it ships with a typical 14-route
CRM surface (Contacts / Companies / Deals / Tasks — list/get/create/update/
delete). Edit that file to match your real API before relying on it — see
"Pointing it at your real API" below.

## Setup

```bash
npm install
cp .env.example .env   # then fill in CRM_API_BASE_URL and CRM_API_TOKEN
```

## Try it without a real API

Print the tool definitions that would be generated, without starting the
server or needing credentials:

```bash
npm run list-tools
```

## Register with Claude Code

A project-scoped `.mcp.json` is already included. From this directory:

```bash
claude mcp list          # confirm it's picked up
```

Or register manually:

```bash
claude mcp add crm-api -- node src/index.js
```

Set `CRM_API_BASE_URL` and `CRM_API_TOKEN` in your environment (or `.env`,
loaded via your shell) before launching Claude Code so the server can read
them at startup — it fails fast with a clear error if either is missing.

## Pointing it at your real API

Edit `endpoints.json`. Each entry looks like:

```json
{
  "name": "get_contact",
  "method": "GET",
  "path": "/contacts/{id}",
  "description": "Fetch a single contact by ID.",
  "pathParams": [
    { "name": "id", "type": "string", "required": true, "description": "Contact ID" }
  ]
}
```

Supported fields per endpoint:

- `name` — tool name the agent will call (must be unique)
- `method` — `GET` / `POST` / `PUT` / `PATCH` / `DELETE`
- `path` — URL path, `{param}` segments become required path params
- `description` — shown to the agent; make it specific
- `pathParams` — array of `{ name, type, required, description }`
- `queryParams` — same shape, appended as `?key=value`
- `bodyRequired` — for POST/PUT/PATCH, whether a JSON `body` arg is mandatory
- `bodyDescription` — free text describing expected body shape (there's no
  strict schema — the agent sends whatever JSON object it constructs)

Top-level config:

- `baseUrl` — supports `${ENV_VAR}` interpolation
- `auth` — `{ "type": "bearer", "envVar": "CRM_API_TOKEN" }` by default; set
  `header`/`prefix` to use a custom header (e.g. an `X-API-Key` scheme) or
  change `type` away from `"bearer"` to send the token with no prefix
- `defaultHeaders` — sent on every request (defaults to JSON content type)

No restart-worthy code changes are needed for endpoint changes — just edit
the JSON and restart the MCP server.

## How it works

- `src/config.js` loads and validates `endpoints.json`, expands env vars,
  and resolves the auth header.
- `src/tools.js` turns each endpoint into an MCP tool (JSON Schema input)
  and performs the actual HTTP call, substituting path params, appending
  query params, and forwarding the `body` arg for writes.
- `src/index.js` wires those into an MCP `Server` over stdio.

Errors from the CRM API (non-2xx) are returned to the agent as tool errors
with the response body included, rather than crashing the server.
````

````js
// Dev helper: prints the MCP tool definitions that would be generated from
// endpoints.json, without starting the server or requiring live credentials.
import { readFileSync } from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { endpointToToolDef } from "../src/tools.js";

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const raw = readFileSync(path.join(__dirname, "..", "endpoints.json"), "utf-8");
const config = JSON.parse(raw);

console.log(`${config.endpoints.length} endpoint(s) defined:\n`);
````

````js
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import path from "node:path";

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const DEFAULT_CONFIG_PATH = path.join(__dirname, "..", "endpoints.json");

/** Expand ${ENV_VAR} references in a string using process.env. */
function expandEnv(str) {
  if (typeof str !== "string") return str;
  return str.replace(/\$\{([A-Z0-9_]+)\}/gi, (_, name) => {
    const val = process.env[name];
    if (val === undefined) {
      throw new Error(
        `Missing required environment variable "${name}" referenced in endpoints.json`
      );
    }
    return val;
  });
}

export function loadConfig(configPath = process.env.CRM_ENDPOINTS_FILE || DEFAULT_CONFIG_PATH) {
  const raw = readFileSync(configPath, "utf-8");
  const config = JSON.parse(raw);

  if (!config.baseUrl) {
    throw new Error('endpoints.json is missing "baseUrl"');
  }
  if (!Array.isArray(config.endpoints) || config.endpoints.length === 0) {
    throw new Error('endpoints.json must define a non-empty "endpoints" array');
  }

  config.baseUrl = expandEnv(config.baseUrl).replace(/\/+$/, "");

  // Resolve auth token now so we fail fast at startup, not on first tool call.
  if (config.auth) {
    if (config.auth.envVar) {
      const token = process.env[config.auth.envVar];
      if (!token) {
        throw new Error(
          `Missing required environment variable "${config.auth.envVar}" for CRM API auth`
        );
      }
      config.auth.resolvedToken = token;
    }
  }

  const seen = new Set();
  for (const ep of config.endpoints) {
    if (!ep.name || !ep.method || !ep.path) {
      throw new Error(`Invalid endpoint entry: ${JSON.stringify(ep)}`);
    }
    if (seen.has(ep.name)) {
      throw new Error(`Duplicate endpoint name "${ep.name}" in endpoints.json`);
    }
    seen.add(ep.name);
  }

  return config;
}

export function buildAuthHeaders(config) {
  const headers = { ...(config.defaultHeaders || {}) };
  if (config.auth && config.auth.resolvedToken) {
    const headerName = config.auth.header || "Authorization";
    const prefix = config.auth.type === "bearer" ? (config.auth.prefix ?? "Bearer ") : (config.auth.prefix ?? "");
    headers[headerName] = `${prefix}${config.auth.resolvedToken}`;
  }
  return headers;
}
````

````js
#!/usr/bin/env node
import { Server } from "@modelcontextprotocol/sdk/server/index.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { ListToolsRequestSchema, CallToolRequestSchema } from "@modelcontextprotocol/sdk/types.js";
import { loadConfig, buildAuthHeaders } from "./config.js";
import { endpointToToolDef, callEndpoint } from "./tools.js";

const config = loadConfig();
const endpointsByName = new Map(config.endpoints.map((ep) => [ep.name, ep]));
const toolDefs = config.endpoints.map(endpointToToolDef);
const authHeaders = buildAuthHeaders(config);

const server = new Server(
  { name: "crm-api", version: "1.0.0" },
  { capabilities: { tools: {} } }
);

server.setRequestHandler(ListToolsRequestSchema, async () => ({
  tools: toolDefs,
}));

server.setRequestHandler(CallToolRequestSchema, async (request) => {
  const { name, arguments: args } = request.params;
  const ep = endpointsByName.get(name);
  if (!ep) {
    return {
      isError: true,
      content: [{ type: "text", text: `Unknown tool "${name}"` }],
    };
  }

  try {
    const result = await callEndpoint(config, ep, args, authHeaders);
    return {
      content: [
        {
          type: "text",
          text: JSON.stringify(result.data, null, 2),
        },
      ],
    };
  } catch (err) {
    return {
      isError: true,
      content: [
        {
          type: "text",
          text: `Error calling ${name}: ${err.message}${
            err.payload ? `\nResponse body: ${JSON.stringify(err.payload)}` : ""
          }`,
        },
      ],
    };
  }
});

async function main() {
  const transport = new StdioServerTransport();
  await server.connect(transport);
  console.error(
    `crm-api MCP server running — ${toolDefs.length} tools exposed from ${config.baseUrl}`
  );
}

main().catch((err) => {
  console.error("Fatal error starting crm-api MCP server:", err);
  process.exit(1);
});
````

````js
/**
 * Converts endpoints.json entries into MCP tool definitions (name + JSON Schema input)
 * and into HTTP request executors.
 */

function jsonSchemaType(type) {
  switch (type) {
    case "number":
    case "integer":
      return "number";
    case "boolean":
      return "boolean";
    default:
      return "string";
  }
}

export function endpointToToolDef(ep) {
  const properties = {};
  const required = [];

  for (const p of ep.pathParams || []) {
    properties[p.name] = { type: jsonSchemaType(p.type), description: p.description || `Path parameter: ${p.name}` };
    if (p.required !== false) required.push(p.name);
  }

  for (const q of ep.queryParams || []) {
    properties[q.name] = { type: jsonSchemaType(q.type), description: q.description || `Query parameter: ${q.name}` };
    if (q.required) required.push(q.name);
  }

  if (["POST", "PUT", "PATCH"].includes(ep.method.toUpperCase())) {
    properties.body = {
      type: "object",
      description: ep.bodyDescription || "JSON request body",
      additionalProperties: true,
    };
    if (ep.bodyRequired) required.push("body");
  }

  return {
    name: ep.name,
    description: `[${ep.method.toUpperCase()} ${ep.path}] ${ep.description || ""}`.trim(),
    inputSchema: {
      type: "object",
      properties,
      required: required.length ? required : undefined,
    },
  };
}

function resolvePath(template, pathParams, args) {
  return template.replace(/\{([a-zA-Z0-9_]+)\}/g, (_, name) => {
    if (!(name in args)) {
      throw new Error(`Missing required path parameter "${name}"`);
    }
    return encodeURIComponent(String(args[name]));
  });
}

/**
 * Executes the HTTP call for a given endpoint definition + tool call arguments.
 */
export async function callEndpoint(config, ep, args, headers) {
  args = args || {};
  const pathParamNames = new Set((ep.pathParams || []).map((p) => p.name));
  const resolvedPath = resolvePath(ep.path, pathParamNames, args);

  const url = new URL(config.baseUrl + resolvedPath);
  for (const q of ep.queryParams || []) {
    if (args[q.name] !== undefined && args[q.name] !== null) {
      url.searchParams.set(q.name, String(args[q.name]));
    }
  }

  const init = {
    method: ep.method.toUpperCase(),
    headers,
  };

  if (["POST", "PUT", "PATCH"].includes(init.method) && args.body !== undefined) {
    init.body = JSON.stringify(args.body);
  }

  const res = await fetch(url, init);
  const contentType = res.headers.get("content-type") || "";
  const payload = contentType.includes("application/json")
    ? await res.json().catch(() => null)
    : await res.text();

  if (!res.ok) {
    const err = new Error(`CRM API request failed: ${res.status} ${res.statusText}`);
    err.status = res.status;
    err.payload = payload;
    throw err;
  }

  return { status: res.status, data: payload };
}
````
