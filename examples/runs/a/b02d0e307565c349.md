````py
"""Entry point: an agent that can search tickets via the ticket system's
search endpoint.

Usage:
    python -m ticket_agent.agent "How many open platform tickets were created this year?"

Requires:
    ANTHROPIC_API_KEY      (or another credential source picked up by the SDK)
    TICKET_API_BASE_URL
    TICKET_API_TOKEN
"""

from __future__ import annotations

import sys

import anthropic

from .tools import search_tickets

SYSTEM_PROMPT = (
    "You are a support-ops assistant with access to the ticket system's "
    "search tool. Translate the user's request into the ticket search DSL "
    "(fields combined with AND / OR, e.g. "
    "\"status:open AND team:platform AND created:>2026-01-01\") and use the "
    "search_tickets tool to answer. If a search is ambiguous (e.g. which "
    "team, what date range), ask a clarifying question instead of guessing."
)


def run(user_message: str) -> str:
    client = anthropic.Anthropic()

    runner = client.beta.messages.tool_runner(
        model="claude-opus-5",
        max_tokens=16000,
        system=SYSTEM_PROMPT,
        thinking={"type": "adaptive"},
        tools=[search_tickets],
        messages=[{"role": "user", "content": user_message}],
    )

    final_text = ""
    for message in runner:
        for block in message.content:
            if block.type == "text":
                final_text = block.text
    return final_text


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print('Usage: python -m ticket_agent.agent "<question about tickets>"')
        sys.exit(1)

    print(run(" ".join(sys.argv[1:])))

````

### `ticket_agent/ticket_client.py`

````py
"""Thin HTTP client for the ticket system's search endpoint.

The endpoint takes a query string in a small DSL, e.g.:

    status:open AND team:platform AND created:>2026-01-01

This client just forwards that query string as-is and parses the JSON
response - it does not attempt to validate or build the DSL itself.

Configuration (environment variables):
    TICKET_API_BASE_URL   Required. e.g. "https://tickets.example.com"
    TICKET_API_TOKEN      Required. Sent as "Authorization: Bearer <token>".
    TICKET_API_SEARCH_PATH  Optional. Defaults to "/api/v1/tickets/search".

Adjust `_headers()` / the request shape in `search()` if your ticket
system uses a different auth scheme (API key header, Basic auth, ...)
or a POST-with-body search endpoint instead of GET+query-param.
"""

from __future__ import annotations

import os
from dataclasses import dataclass
from typing import Any

import requests

DEFAULT_SEARCH_PATH = "/api/v1/tickets/search"
DEFAULT_TIMEOUT_SECONDS = 15


class TicketSearchError(Exception):
    """Raised when the ticket search request fails."""


@dataclass
class TicketSearchClient:
    base_url: str
    token: str
    search_path: str = DEFAULT_SEARCH_PATH
    timeout: float = DEFAULT_TIMEOUT_SECONDS

    @classmethod
    def from_env(cls) -> "TicketSearchClient":
        base_url = os.environ.get("TICKET_API_BASE_URL")
        token = os.environ.get("TICKET_API_TOKEN")
        if not base_url:
            raise TicketSearchError(
                "TICKET_API_BASE_URL is not set - point it at your ticket "
                "system's base URL, e.g. https://tickets.example.com"
            )
        if not token:
            raise TicketSearchError(
                "TICKET_API_TOKEN is not set - a bearer token for the "
                "ticket search API is required"
            )
        return cls(
            base_url=base_url.rstrip("/"),
            token=token,
            search_path=os.environ.get("TICKET_API_SEARCH_PATH", DEFAULT_SEARCH_PATH),
        )

    def _headers(self) -> dict[str, str]:
        return {
            "Authorization": f"Bearer {self.token}",
            "Accept": "application/json",
        }

    def search(
        self,
        query: str,
        limit: int = 25,
        cursor: str | None = None,
    ) -> dict[str, Any]:
        """Run a search query against the ticket system.

        Args:
            query: The DSL query string, e.g.
                "status:open AND team:platform AND created:>2026-01-01".
            limit: Max number of tickets to return (API may cap this).
            cursor: Opaque pagination cursor from a previous response's
                "next_cursor" field, if the result set was truncated.

        Returns:
            The parsed JSON response body.

        Raises:
            TicketSearchError: On network failure, timeout, or a non-2xx
                response.
        """
        url = f"{self.base_url}{self.search_path}"
        params: dict[str, Any] = {"q": query, "limit": limit}
        if cursor:
            params["cursor"] = cursor

        try:
            response = requests.get(
                url, headers=self._headers(), params=params, timeout=self.timeout
            )
        except requests.Timeout as e:
            raise TicketSearchError(f"Ticket search timed out after {self.timeout}s") from e
        except requests.RequestException as e:
            raise TicketSearchError(f"Ticket search request failed: {e}") from e

        if response.status_code == 401:
            raise TicketSearchError("Ticket search auth failed (401) - check TICKET_API_TOKEN")
        if response.status_code == 422 or response.status_code == 400:
            raise TicketSearchError(
                f"Ticket system rejected the query (HTTP {response.status_code}): "
                f"{response.text[:500]}"
            )
        if not response.ok:
            raise TicketSearchError(
                f"Ticket search failed (HTTP {response.status_code}): {response.text[:500]}"
            )

        try:
            return response.json()
        except ValueError as e:
            raise TicketSearchError("Ticket search returned non-JSON response") from e

````

### `ticket_agent/tools.py`

````py
"""Claude tool definitions for the ticket agent."""

from __future__ import annotations

import json

from anthropic import beta_tool

from .ticket_client import TicketSearchClient, TicketSearchError

_client: TicketSearchClient | None = None


def _get_client() -> TicketSearchClient:
    global _client
    if _client is None:
        _client = TicketSearchClient.from_env()
    return _client


@beta_tool
def search_tickets(query: str, limit: int = 25, cursor: str = "") -> str:
    """Search tickets in the ticket system.

    Call this when the user asks to find, count, or list tickets, or
    asks a question that requires looking up ticket state (e.g. "how
    many open platform tickets are there", "show me tickets created
    this month").

    The query is a small boolean DSL over ticket fields, combined with
    AND / OR and parentheses. Examples:
        status:open AND team:platform AND created:>2026-01-01
        (status:open OR status:in_progress) AND assignee:me
        priority:high AND team:infra AND updated:<2026-06-01

    Common fields: status (open, in_progress, closed, ...), team,
    assignee, priority (low, medium, high, urgent), created, updated
    (both take >, <, >=, <= followed by an ISO date), and label.

    Args:
        query: The DSL query string, e.g.
            "status:open AND team:platform AND created:>2026-01-01".
        limit: Max number of tickets to return. Defaults to 25; raise
            it for broader searches, but prefer a narrower query over a
            very large limit.
        cursor: Pagination cursor from a previous result's
            "next_cursor" field. Leave empty for the first page.
    """
    try:
        result = _get_client().search(query, limit=limit, cursor=cursor or None)
    except TicketSearchError as e:
        return json.dumps({"error": str(e)})

    return json.dumps(result)

````
