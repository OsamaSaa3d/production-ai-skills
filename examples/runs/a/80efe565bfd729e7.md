````py
"""Minimal agent with a single tool: fetch a user by id.

Usage:
    pip install anthropic requests
    export ANTHROPIC_API_KEY=...   # or use `ant auth login`
    python agent.py "Look up user 3 and tell me their name and email."
"""

import sys

import anthropic
import requests
from anthropic import beta_tool

# Swap this out for your real backend (DB lookup, internal API, etc.) -
# the tool interface below stays the same.
USERS_API_URL = "https://jsonplaceholder.typicode.com/users/{user_id}"


@beta_tool
def get_user(user_id: int) -> str:
    """Fetch a user by their id.

    Args:
        user_id: The numeric id of the user to look up.
    """
    response = requests.get(USERS_API_URL.format(user_id=user_id), timeout=10)
    if response.status_code == 404:
        return f"Error: no user found with id {user_id}."
    response.raise_for_status()
    return response.text  # JSON string; Claude can read it directly


def main() -> None:
    prompt = " ".join(sys.argv[1:]) or "Look up user 1 and summarize who they are."

    client = anthropic.Anthropic()
    runner = client.beta.messages.tool_runner(
        model="claude-opus-5",
        max_tokens=16000,
        tools=[get_user],
        messages=[{"role": "user", "content": prompt}],
    )

    final = None
    for message in runner:
        final = message

    if final is not None:
        for block in final.content:
            if block.type == "text":
                print(block.text)


if __name__ == "__main__":
    main()

````
