- **Error handling:** a most-specific-first exception chain (rate limit → 5xx → 4xx → connection error) rather than one broad `except`.
- **API:** `classify_email(subject, body, sender)` for one email, `classify_batch(list_of_dicts)` for many, plus a CLI entry point reading a JSON array and printing annotated results.

"""
Classify support emails by urgency using the Claude API, for routing to the
right queue (e.g. page on-call vs. drop in the weekly digest).

Uses structured outputs (Pydantic + `client.messages.parse`) so every
response is guaranteed-valid JSON - no manual parsing or retry-on-bad-JSON
logic needed.

Setup:
    pip install anthropic pydantic
    # Auth resolves from ANTHROPIC_API_KEY / ANTHROPIC_AUTH_TOKEN / `ant auth login`

Usage:
    python classify_emails.py emails.json > results.json

`emails.json` is a JSON array of {"subject": ..., "body": ..., "sender": ...}
objects ("sender" is optional).
"""

from __future__ import annotations

import json
import sys
from enum import Enum
from pathlib import Path

import anthropic
from pydantic import BaseModel, Field

# Model choice is deliberately not tuned down for cost here - classification
# quality directly determines routing correctness (a missed "critical" is
# expensive). Swap to "claude-sonnet-5" or "claude-haiku-4-5" yourself if you
# want cheaper high-volume triage and have measured that it holds up.
MODEL = "claude-opus-5"

SYSTEM_PROMPT = """You triage incoming customer support emails for a SaaS company.
Classify each email's urgency so it can be routed to the right queue.

Urgency levels:
- critical: production down, data loss, security breach, payment/billing outage \
affecting many users, or a churn/legal/PR risk that needs a response within the hour
- high: a paying customer is blocked from core functionality with no workaround, \
or the request is time-sensitive (e.g. a contract or renewal deadline)
- medium: a real bug or question that degrades the experience but has a workaround, \
or affects a single non-blocking feature
- low: general question, feature request, feedback, or anything with no working impact

Base the classification on business impact and time sensitivity, not on tone alone -
a politely worded email about a full outage is still critical, and an angry email
about a cosmetic typo is still low.
"""


class Urgency(str, Enum):
    CRITICAL = "critical"
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"


class EmailClassification(BaseModel):
    urgency: Urgency
    category: str = Field(
        description="Short label for the issue type, e.g. 'billing', 'outage', 'how-to', 'feature-request'"
    )
    reasoning: str = Field(description="One-sentence justification for the urgency level")


client = anthropic.Anthropic()  # reads ANTHROPIC_API_KEY / ANTHROPIC_AUTH_TOKEN from env


def classify_email(subject: str, body: str, sender: str | None = None) -> EmailClassification:
    """Classify a single support email's urgency via Claude structured outputs."""
    email_text = f"From: {sender or 'unknown'}\nSubject: {subject}\n\n{body}"

    try:
        response = client.messages.parse(
            model=MODEL,
            max_tokens=1024,
            system=SYSTEM_PROMPT,
            messages=[{"role": "user", "content": email_text}],
            output_format=EmailClassification,
        )
    except anthropic.RateLimitError as e:
        retry_after = int(e.response.headers.get("retry-after", "60"))
        raise RuntimeError(f"Rate limited by Claude API; retry after {retry_after}s") from e
    except anthropic.APIStatusError as e:
        if e.status_code >= 500:
            raise RuntimeError(f"Claude API server error ({e.status_code}); retry later") from e
        raise RuntimeError(f"Bad request to Claude API: {e.message}") from e
    except anthropic.APIConnectionError as e:
        raise RuntimeError("Network error contacting Claude API") from e

    return response.parsed_output


def classify_batch(emails: list[dict]) -> list[dict]:
    """Classify a list of {subject, body, sender?} dicts; returns them annotated."""
    results = []
    for email in emails:
        classification = classify_email(
            subject=email["subject"],
            body=email["body"],
            sender=email.get("sender"),
        )
        results.append({**email, "classification": classification.model_dump()})
    return results


def main() -> None:
    if len(sys.argv) != 2:
        print("Usage: python classify_emails.py <emails.json>", file=sys.stderr)
        sys.exit(1)

    emails = json.loads(Path(sys.argv[1]).read_text(encoding="utf-8"))
    results = classify_batch(emails)
    print(json.dumps(results, indent=2))


if __name__ == "__main__":
    main()
