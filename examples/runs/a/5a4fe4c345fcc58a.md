Built at `C:\tmp\skills-eval\arm-a\t14-r1`. Summary of the design and the assumptions I made:

**Assumptions** (no existing repo/tracker was specified, so I made it pluggable rather than guessing wrong): plain Python, `IssueStore` backed by a local JSON file standing in for a real tracker, and a configurable team registry rather than a hardcoded org chart.

**Pipeline** (`bug_triage/agent.py`):
1. **Parse** the report into title/body.
2. **Retrieve candidates** (`retrieval.py`) — a dependency-free TF-IDF + cosine similarity search over the existing issue store. This is cheap lexical pre-filtering, not the duplicate verdict itself.
3. **Classify** (`llm_client.py`) — one Claude call, forced tool-use (`submit_triage`), given the report, the top-K candidates, the severity rubric, and the team list from `config/teams.json`. Returns duplicate verdict + reason, severity + reason, team + reason, and labels.
4. **Apply guardrails + write back** (`agent.py`): a duplicate verdict below a confidence threshold (default 0.6) is downgraded to "not a duplicate" rather than silently merging two different bugs; a team name the model invents that isn't in the config falls back to "Triage / Unassigned"; a duplicate target that doesn't exist in the store is filed fresh instead of dropped.

**Why two-stage duplicate detection**: lexical retrieval narrows a large store to a handful of plausible matches cheaply (no embeddings API needed), then the model — which can reason about paraphrases and root cause vs. symptom — makes the actual duplicate call and explains it.

**Testing**: 18 pytest cases, all passing, all offline — a `FakeLLMClient` stands in for the real Anthropic client so retrieval, routing, and the agent's guardrails (confidence downgrade, unknown-team fallback, dangling-duplicate fallback, dry-run) are verified without an API key.

**Usage**: `python -m bug_triage.cli triage sample_reports/report_login_500.txt` (needs `pip install -r requirements.txt` and `ANTHROPIC_API_KEY` for a real run — I didn't run this against the live API, only verified the CLI parsing/plumbing and the offline test suite). Full usage and extension notes (swapping in a real tracker, editing the rubric/team config) are in `README.md`.

````py
"""Orchestrates the triage pipeline: retrieve -> classify -> apply."""

from __future__ import annotations

from .llm_client import LLMClient
from .models import BugReport, TriageResult
from .retrieval import IssueStore, retrieve_candidates
from .routing import RoutingConfig


class TriageAgent:
    def __init__(
        self,
        llm_client: LLMClient,
        store: IssueStore,
        routing: RoutingConfig,
        top_k_candidates: int = 5,
        duplicate_confidence_threshold: float = 0.6,
    ):
        self.llm_client = llm_client
        self.store = store
        self.routing = routing
        self.top_k_candidates = top_k_candidates
        self.duplicate_confidence_threshold = duplicate_confidence_threshold

    def triage(self, report: BugReport, apply: bool = True) -> TriageResult:
        candidates = retrieve_candidates(
            report.as_text(), self.store.list_issues(), top_k=self.top_k_candidates
        )

        raw = self.llm_client.classify(
            report,
            candidates,
            team_block=self.routing.as_prompt_block(),
            fallback_team=self.routing.fallback_team(),
        )

        result = self._parse(raw)

        if apply:
            self._apply(report, result)

        return result

    def _parse(self, raw: dict) -> TriageResult:
        is_duplicate = bool(raw["is_duplicate"])
        confidence = float(raw.get("duplicate_confidence", 0.0))

        # Guard rail: don't let a low-confidence call silently collapse two
        # different bugs into one. If the model says "duplicate" but isn't
        # confident, downgrade to "not a duplicate" and let a human confirm.
        if is_duplicate and confidence < self.duplicate_confidence_threshold:
            is_duplicate = False

        team = raw.get("team") or self.routing.fallback_team()
        if team not in self.routing.names():
            team = self.routing.fallback_team()

        return TriageResult(
            is_duplicate=is_duplicate,
            duplicate_of=raw.get("duplicate_of") if is_duplicate else None,
            duplicate_confidence=confidence,
            duplicate_reason=raw.get("duplicate_reason", ""),
            severity=raw["severity"],
            severity_reason=raw.get("severity_reason", ""),
            team=team,
            team_reason=raw.get("team_reason", ""),
            labels=list(raw.get("labels", [])),
        )

    def _apply(self, report: BugReport, result: TriageResult) -> None:
        if result.is_duplicate and result.duplicate_of and self.store.get(result.duplicate_of):
            issue_id = self.store.add_issue(
                title=report.title,
                body=report.body,
                severity=result.severity,
                team=result.team,
                labels=result.labels,
                status="duplicate",
            )
            self.store.link_duplicate(issue_id, result.duplicate_of)
            result.new_issue_id = issue_id
        else:
            # Either not a duplicate, or the model named a duplicate target
            # that doesn't actually exist in the store — file it fresh rather
            # than silently dropping the report.
            if result.is_duplicate and not (result.duplicate_of and self.store.get(result.duplicate_of)):
                result.is_duplicate = False
                result.duplicate_of = None
            issue_id = self.store.add_issue(
                title=report.title,
                body=report.body,
                severity=result.severity,
                team=result.team,
                labels=result.labels,
                status="open",
            )
            result.new_issue_id = issue_id

        self.store.save()
````

````py
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

from .agent import TriageAgent
from .llm_client import AnthropicLLMClient
from .models import BugReport
from .retrieval import IssueStore
from .routing import DEFAULT_TEAMS_PATH, RoutingConfig

DEFAULT_STORE_PATH = Path(__file__).resolve().parent.parent / "data" / "issues_store.json"


def parse_report_file(path: Path) -> BugReport:
    text = path.read_text(encoding="utf-8").strip()
    lines = text.splitlines()
    if lines and lines[0].lower().startswith("title:"):
        title = lines[0].split(":", 1)[1].strip()
        body = "\n".join(lines[1:]).strip()
    else:
        title = lines[0].strip() if lines else "(untitled)"
        body = "\n".join(lines[1:]).strip() or text
    return BugReport(title=title, body=body, source=str(path))


def build_arg_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="bug_triage", description="Triage a bug report.")
    sub = parser.add_subparsers(dest="command", required=True)

    triage_p = sub.add_parser("triage", help="Triage a single bug report file.")
    triage_p.add_argument("report_file", type=Path, help="Path to the bug report text file.")
    triage_p.add_argument("--store", type=Path, default=DEFAULT_STORE_PATH, help="Path to the issue store JSON file.")
    triage_p.add_argument("--teams", type=Path, default=DEFAULT_TEAMS_PATH, help="Path to the teams routing config.")
    triage_p.add_argument("--dry-run", action="store_true", help="Don't write the result back to the store.")
    triage_p.add_argument("--model", default=None, help="Override the Claude model id.")

    return parser


def main(argv: list[str] | None = None) -> int:
    parser = build_arg_parser()
    args = parser.parse_args(argv)

    if args.command == "triage":
        if not args.report_file.exists():
            print(f"error: report file not found: {args.report_file}", file=sys.stderr)
            return 1

        report = parse_report_file(args.report_file)
        store = IssueStore(args.store)
        routing = RoutingConfig.load(args.teams)
        client_kwargs = {"model": args.model} if args.model else {}
        llm_client = AnthropicLLMClient(**client_kwargs)
        agent = TriageAgent(llm_client, store, routing)

        result = agent.triage(report, apply=not args.dry_run)
        print(json.dumps(result.to_dict(), indent=2))
        return 0

    return 1


if __name__ == "__main__":
    raise SystemExit(main())
````

````py
Kept separate from agent.py so the prompt/schema can be unit tested and so

from __future__ import annotations

import json
import os
from typing import Protocol

from .models import BugReport, Candidate

DEFAULT_MODEL = os.environ.get("ANTHROPIC_MODEL", "claude-sonnet-5")

SEVERITY_RUBRIC = """\
Severity rubric — pick exactly one:

CRITICAL: Production is down, data loss/corruption, a security vulnerability,
  or a core flow (login, checkout, etc.) is broken for all/most users with no
  workaround.
HIGH: A major feature is broken or a core flow fails for a meaningful subset
  of users, and there is no reasonable workaround.
MEDIUM: A feature is broken or behaves incorrectly, but a workaround exists,
  or the impact is limited to an edge case / small subset of users.
LOW: Cosmetic issues, minor inconvenience, typos, or requests that don't
  block any user from completing their task.
"""

SYSTEM_PROMPT_TEMPLATE = """\
You are a bug triage assistant for an engineering organization. Given a new
bug report and a short list of possibly-related existing issues, you decide:

1. Whether the new report is a duplicate of one of the listed existing
   issues. Judge this on the underlying bug (same root cause / same user
   impact), not surface wording — a paraphrase or a report with a different
   stack trace but the same root cause still counts as a duplicate. When
   unsure, prefer NOT marking it a duplicate (false "not a duplicate" is
   cheaper than incorrectly merging two different bugs).
2. Its severity, using this rubric:

{rubric}

3. Which team should own it, from this list:

{teams}

If nothing fits well, use "{fallback_team}".

Always call the submit_triage tool exactly once with your full decision.
Keep every "_reason" field to one or two sentences grounded in specifics from
the report (and the matching candidate, if a duplicate).
"""

TRIAGE_TOOL = {
    "name": "submit_triage",
    "description": "Submit the triage decision for the bug report.",
    "input_schema": {
        "type": "object",
        "properties": {
            "is_duplicate": {"type": "boolean"},
            "duplicate_of": {
                "type": ["string", "null"],
                "description": "issue_id of the existing issue this duplicates, or null.",
            },
            "duplicate_confidence": {
                "type": "number",
                "description": "0.0-1.0 confidence in the duplicate verdict.",
            },
            "duplicate_reason": {"type": "string"},
            "severity": {
                "type": "string",
                "enum": ["CRITICAL", "HIGH", "MEDIUM", "LOW"],
            },
            "severity_reason": {"type": "string"},
            "team": {"type": "string"},
            "team_reason": {"type": "string"},
            "labels": {
                "type": "array",
                "items": {"type": "string"},
                "description": "Short freeform labels, e.g. component or symptom tags.",
            },
        },
        "required": [
            "is_duplicate",
            "duplicate_of",
            "duplicate_confidence",
            "duplicate_reason",
            "severity",
            "severity_reason",
            "team",
            "team_reason",
            "labels",
        ],
    },
}


def build_system_prompt(team_block: str, fallback_team: str) -> str:
    return SYSTEM_PROMPT_TEMPLATE.format(
        rubric=SEVERITY_RUBRIC, teams=team_block, fallback_team=fallback_team
    )


def build_user_message(report: BugReport, candidates: list[Candidate]) -> str:
    parts = [
        "New bug report:",
        "---",
        report.as_text(),
        "---",
    ]
    if candidates:
        parts.append("\nPossibly related existing issues (candidates, may be empty of true matches):")
        for c in candidates:
            parts.append(
                f"\n[{c.issue_id}] (lexical similarity {c.score}) {c.title}\n{c.body}"
            )
    else:
        parts.append("\nNo similar existing issues were found by the retrieval step.")
    return "\n".join(parts)


class LLMClient(Protocol):
    """Interface the agent depends on. Implement/fake this, not Anthropic directly."""

    def classify(
        self, report: BugReport, candidates: list[Candidate], team_block: str, fallback_team: str
    ) -> dict: ...


class AnthropicLLMClient:
    """Real implementation backed by the Anthropic Messages API."""

    def __init__(self, api_key: str | None = None, model: str = DEFAULT_MODEL, max_tokens: int = 1024):
        import anthropic  # imported lazily so tests don't need it installed

        self._client = anthropic.Anthropic(api_key=api_key)
        self._model = model
        self._max_tokens = max_tokens

    def classify(
        self, report: BugReport, candidates: list[Candidate], team_block: str, fallback_team: str
    ) -> dict:
        system = build_system_prompt(team_block, fallback_team)
        user = build_user_message(report, candidates)

        response = self._client.messages.create(
            model=self._model,
            max_tokens=self._max_tokens,
            system=system,
            tools=[TRIAGE_TOOL],
            tool_choice={"type": "tool", "name": "submit_triage"},
            messages=[{"role": "user", "content": user}],
        )

        for block in response.content:
            if block.type == "tool_use" and block.name == "submit_triage":
                return dict(block.input)

        raise RuntimeError("Model did not return a submit_triage tool call")
````

### `bug_triage/retrieval.py`

````py
"""Cheap, dependency-free candidate retrieval for duplicate detection.

This is deliberately *not* the final duplicate judgment. It's a fast lexical
pre-filter (TF-IDF + cosine similarity, stdlib only) that narrows a large
issue store down to a handful of plausible candidates so the LLM only has to
reason about, say, 5 issues instead of 5,000. The actual "is this really the
same bug" call is made by the model in llm_client.py / agent.py, which can
account for paraphrasing, stack-trace equivalence, etc. that pure lexical
overlap misses.
"""

from __future__ import annotations

import json
import math
import re
from collections import Counter
from pathlib import Path
from typing import Iterable

from .models import Candidate

_TOKEN_RE = re.compile(r"[a-z0-9]+")

_STOPWORDS = {
    "the", "a", "an", "and", "or", "but", "is", "are", "was", "were", "be",
    "to", "of", "in", "on", "for", "with", "this", "that", "it", "as", "at",
    "when", "then", "than", "i", "we", "you", "our", "my", "not", "no",
    "do", "does", "did", "has", "have", "had", "can", "could", "would",
    "should", "will", "if", "from", "by", "into", "up", "so",
}


def tokenize(text: str) -> list:
    return [t for t in _TOKEN_RE.findall(text.lower()) if t not in _STOPWORDS]


class IssueStore:
    """A JSON-file-backed store of existing issues.

    Kept intentionally minimal (list_issues / add_issue / link_duplicate) so
    it can be swapped for a real GitHub/Jira-backed implementation later
    without touching the rest of the pipeline.
    """

    def __init__(self, path: str | Path):
        self.path = Path(path)
        if self.path.exists():
            with open(self.path, "r", encoding="utf-8") as f:
                data = json.load(f)
        else:
            data = {"issues": [], "next_id": 1}
        self._issues: list[dict] = data.get("issues", [])
        self._next_id: int = data.get("next_id", len(self._issues) + 1)

    def list_issues(self) -> list[dict]:
        return list(self._issues)

    def get(self, issue_id: str) -> dict | None:
        for issue in self._issues:
            if issue["id"] == issue_id:
                return issue
        return None

    def add_issue(
        self,
        title: str,
        body: str,
        severity: str,
        team: str,
        labels: Iterable[str] = (),
        status: str = "open",
    ) -> str:
        issue_id = f"ISSUE-{self._next_id}"
        self._next_id += 1
        self._issues.append(
            {
                "id": issue_id,
                "title": title,
                "body": body,
                "severity": severity,
                "team": team,
                "labels": list(labels),
                "status": status,
                "duplicate_of": None,
            }
        )
        return issue_id

    def link_duplicate(self, issue_id: str, duplicate_of: str) -> None:
        issue = self.get(issue_id)
        if issue is not None:
            issue["duplicate_of"] = duplicate_of
            issue["status"] = "duplicate"

    def save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with open(self.path, "w", encoding="utf-8") as f:
            json.dump({"issues": self._issues, "next_id": self._next_id}, f, indent=2)


def _tfidf_vectors(docs: list[list[str]]) -> list[Counter]:
    df = Counter()
    for tokens in docs:
        for term in set(tokens):
            df[term] += 1
    n_docs = len(docs) or 1

    vectors = []
    for tokens in docs:
        tf = Counter(tokens)
        vec = Counter()
        for term, count in tf.items():
            idf = math.log((n_docs + 1) / (df[term] + 1)) + 1.0
            vec[term] = count * idf
        vectors.append(vec)
    return vectors


def _cosine(a: Counter, b: Counter) -> float:
    if not a or not b:
        return 0.0
    common = set(a) & set(b)
    dot = sum(a[t] * b[t] for t in common)
    norm_a = math.sqrt(sum(v * v for v in a.values()))
    norm_b = math.sqrt(sum(v * v for v in b.values()))
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return dot / (norm_a * norm_b)


def retrieve_candidates(
    report_text: str, issues: list[dict], top_k: int = 5, min_score: float = 0.05
) -> list[Candidate]:
    """Return up to `top_k` existing issues most lexically similar to the report."""
    open_issues = [i for i in issues if i.get("status") != "duplicate"]
    if not open_issues:
        return []

    corpus_texts = [report_text] + [f"{i['title']} {i['body']}" for i in open_issues]
    tokenized = [tokenize(t) for t in corpus_texts]
    vectors = _tfidf_vectors(tokenized)
    report_vec, issue_vecs = vectors[0], vectors[1:]

    scored = []
    for issue, vec in zip(open_issues, issue_vecs):
        score = _cosine(report_vec, vec)
        if score >= min_score:
            scored.append((score, issue))

    scored.sort(key=lambda pair: pair[0], reverse=True)
    return [
        Candidate(
            issue_id=issue["id"],
            title=issue["title"],
            body=issue["body"],
            status=issue.get("status", "open"),
            score=round(score, 4),
        )
        for score, issue in scored[:top_k]
    ]

````

````py
"""Loads the team routing table used to steer the model's routing decision.

The routing table is data, not code, so ops can add/rename/redescribe teams
by editing config/teams.json without touching the agent.
"""

from __future__ import annotations

import json
from pathlib import Path


DEFAULT_TEAMS_PATH = Path(__file__).resolve().parent.parent / "config" / "teams.json"


class RoutingConfig:
    def __init__(self, teams: list[dict]):
        if not teams:
            raise ValueError("routing config must define at least one team")
        self.teams = teams

    @classmethod
    def load(cls, path: str | Path = DEFAULT_TEAMS_PATH) -> "RoutingConfig":
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        return cls(data["teams"])

    def names(self) -> list[str]:
        return [t["name"] for t in self.teams]

    def as_prompt_block(self) -> str:
        """Render the team list as text for inclusion in the system prompt."""
        lines = []
        for t in self.teams:
            keywords = ", ".join(t.get("keywords", []))
            lines.append(f"- {t['name']}: {t['description']} (keywords: {keywords})")
        return "\n".join(lines)

    def fallback_team(self) -> str:
        for t in self.teams:
            if t.get("is_fallback"):
                return t["name"]
        return self.teams[-1]["name"]

````

### `README.md`

````md
# Bug Triage Agent

An agent that takes an incoming bug report and decides three things:

1. **Is this a duplicate?** — retrieves similar existing issues and asks the
   model to confirm whether the new report is really the same underlying bug.
2. **How severe is it?** — labels the report `CRITICAL` / `HIGH` / `MEDIUM` / `LOW`
   against an explicit rubric.
3. **Who should own it?** — routes it to one of a configurable set of teams.

It's built as a small, pluggable Python library + CLI rather than a single
prompt, so each stage can be tested, tuned, and swapped out independently
(e.g. point `IssueStore` at a real GitHub/Jira backend later).

## How it works

```
bug report (text)
      │
      ▼
 parse_report()               → title + body
      │
      ▼
 retrieve_candidates()        → cheap local TF-IDF/cosine search over the
      │                          existing issue store (no API call, no network)
      ▼
 TriageAgent.triage()         → one Claude call, forced tool-use, given:
      │                          - the report
      │                          - the rubric
      │                          - the top-K candidate issues
      │                          - the allowed team list
      │                        returns structured JSON: duplicate verdict,
      │                        severity + reason, team + reason, labels
      ▼
 IssueStore.apply()           → either links the report to the duplicate,
                                  or files a new labeled/routed issue
```

Duplicate detection is deliberately two-stage: a cheap lexical retrieval step
(stdlib only, no embeddings API needed) narrows thousands of issues down to a
handful of plausible candidates, then the model — which can reason about
paraphrases, stack traces, and symptoms vs. root cause — makes the actual
duplicate/not-duplicate call and explains why. This keeps token/dollar cost
low while giving the judgment call to the model instead of a similarity
threshold.

## Layout

```
bug_triage/
  models.py      dataclasses: BugReport, Candidate, TriageResult
  retrieval.py   stdlib TF-IDF + cosine similarity candidate search
  routing.py     team registry, loaded from config/teams.json
  llm_client.py  Anthropic wrapper: builds the tool-use schema + system prompt
  agent.py       orchestrates retrieval → LLM call → store update
  cli.py         `python -m bug_triage.cli` entry point
config/teams.json     editable routing table (team name, description, keywords)
data/issues_store.json  sample "existing issues" backing store
sample_reports/        example incoming bug reports to try
tests/                 unit tests; LLM calls are mocked, no network/API key needed
```

## Usage

```bash
pip install -r requirements.txt
export ANTHROPIC_API_KEY=sk-ant-...        # only needed for real runs, not tests

# Triage a single report, print the decision, and update the store in place
python -m bug_triage.cli triage sample_reports/report_login_500.txt

# Dry run: show what would happen without writing to the store
python -m bug_triage.cli triage sample_reports/report_login_500.txt --dry-run

# Point at a different store/routing config
python -m bug_triage.cli triage my_report.txt --store data/issues_store.json --teams config/teams.json
```

Output looks like:

  "team": "Identity & Access",
  "team_reason": "Report involves the login/2FA endpoint, owned by Identity & Access per routing config.",

## Extending

- **Real issue tracker**: replace `IssueStore` (in `bug_triage/store` module,
  see `retrieval.py`'s `IssueStore` class) with one backed by the GitHub or
  Jira API — keep the same `list_issues()` / `add_issue()` / `link_duplicate()`
  interface and nothing else needs to change.
- **Routing rules**: edit `config/teams.json`. No code changes needed to add,
  rename, or re-describe a team — the model reads the config at call time.
- **Severity rubric**: edit the `SEVERITY_RUBRIC` string in `bug_triage/llm_client.py`.
- **Model**: set `ANTHROPIC_MODEL` (defaults to `claude-sonnet-5`).
````

````py
"""Test doubles shared across the test suite."""

from __future__ import annotations

from bug_triage.models import BugReport, Candidate


class FakeLLMClient:
    """Returns a pre-baked response (or one produced by a callback) instead
    of calling the real Anthropic API. Records the last call's arguments so
    tests can assert on what the agent sent it.
    """

    def __init__(self, response: dict | None = None, response_fn=None):
        self.response = response
        self.response_fn = response_fn
        self.last_call: dict | None = None

    def classify(
        self, report: BugReport, candidates: list[Candidate], team_block: str, fallback_team: str
    ) -> dict:
        self.last_call = {
            "report": report,
            "candidates": candidates,
            "team_block": team_block,
            "fallback_team": fallback_team,
        }
        if self.response_fn is not None:
            return self.response_fn(report, candidates)
        return dict(self.response)


def make_default_triage_response(**overrides) -> dict:
    base = {
        "is_duplicate": False,
        "duplicate_of": None,
        "duplicate_confidence": 0.0,
        "duplicate_reason": "",
        "severity": "MEDIUM",
        "severity_reason": "Some impact but a workaround exists.",
        "team": "Backend API",
        "team_reason": "Matches backend keywords.",
        "labels": ["bug"],
    }
    base.update(overrides)
    return base
````
