````py
"""
Classify incoming support emails by urgency so they can be routed to the
right queue.

Uses Claude API structured outputs (client.messages.parse + a Pydantic
model) so every classification comes back as a validated object instead
of free-form text that needs its own parsing/error handling.

Usage:
    python classify_support_emails.py emails.json

`emails.json` is a list of objects: [{"id", "sender", "subject", "body"}, ...]
"""

from __future__ import annotations

import json
import logging
import sys
from dataclasses import dataclass
from enum import Enum
from pathlib import Path
from typing import Iterable, Optional

import anthropic
from pydantic import BaseModel, Field

logger = logging.getLogger(__name__)

# Opus is the default per house policy; classification is cheap/fast work,
# so cost is controlled via low reasoning effort rather than a smaller model.
# Swap to "claude-sonnet-5" or "claude-haiku-4-5" if you want a cheaper model
# for this high-volume, latency-sensitive route.
MODEL = "claude-opus-5"


class Urgency(str, Enum):
    CRITICAL = "critical"  # outage, security/data-loss incident, payments down, enterprise churn threat
    HIGH = "high"  # core feature broken with no workaround, paying customer fully blocked
    MEDIUM = "medium"  # degraded but has a workaround, or a real but non-urgent deadline
    LOW = "low"  # general questions, feature requests, no time pressure


class EmailClassification(BaseModel):
    urgency: Urgency
    routing_team: str = Field(
        description="Team that should own this ticket, e.g. 'on-call-sre', 'billing', 'support-tier1'"
    )
    reason: str = Field(description="One-sentence justification for the urgency level")
    customer_sentiment: str = Field(description="'angry', 'frustrated', 'neutral', or 'positive'")


SYSTEM_PROMPT = """You are a support ticket triage assistant for a SaaS company.
Classify each incoming support email into an urgency tier so it can be routed correctly.

Urgency tiers:
- critical: Service outage, security/data-loss incident, payment system down, or an
  enterprise customer threatening to churn immediately.
- high: A core feature is broken with no workaround, or a paying customer is fully blocked.
- medium: Something is degraded but the customer has a workaround, or there's a real
  but non-urgent deadline.
- low: General questions, feature requests, or anything with no time pressure.

Base the urgency strictly on business impact and time sensitivity, not on tone alone -
a politely worded outage report is still critical, and an angry email about a minor
cosmetic bug is still low.
"""


@dataclass
class SupportEmail:
    id: str
    sender: str
    subject: str
    body: str


def classify_email(client: anthropic.Anthropic, email: SupportEmail) -> EmailClassification:
    """Classify a single support email's urgency via Claude structured outputs."""
    response = client.messages.parse(
        model=MODEL,
        max_tokens=1024,
        system=SYSTEM_PROMPT,
        output_config={"effort": "low"},  # routine classification; raise if quality needs it
        messages=[
            {
                "role": "user",
                "content": f"From: {email.sender}\nSubject: {email.subject}\n\n{email.body}",
            }
        ],
        output_format=EmailClassification,
    )
    return response.parsed_output


def classify_batch(
    client: anthropic.Anthropic, emails: Iterable[SupportEmail]
) -> list[tuple[SupportEmail, Optional[EmailClassification]]]:
    """Classify a sequence of emails, continuing past individual failures."""
    results: list[tuple[SupportEmail, Optional[EmailClassification]]] = []
    for email in emails:
        try:
            classification: Optional[EmailClassification] = classify_email(client, email)
        except anthropic.APIStatusError as e:
            logger.error("Classification failed for email %s: %s", email.id, e)
            classification = None
        results.append((email, classification))
    return results


def load_emails(path: Path) -> list[SupportEmail]:
    """Load emails from a JSON file: a list of {id, sender, subject, body} objects."""
    data = json.loads(path.read_text(encoding="utf-8"))
    return [SupportEmail(**item) for item in data]


def main() -> None:
    logging.basicConfig(level=logging.INFO, format="%(levelname)s: %(message)s")

    if len(sys.argv) != 2:
        print(f"Usage: python {Path(__file__).name} <emails.json>", file=sys.stderr)
        sys.exit(1)

    client = anthropic.Anthropic()  # reads ANTHROPIC_API_KEY from the environment
    emails = load_emails(Path(sys.argv[1]))
    results = classify_batch(client, emails)

    # Highest urgency first so the most time-sensitive tickets are visible immediately
    order = {u: i for i, u in enumerate(Urgency)}
    results.sort(
        key=lambda pair: order[pair[1].urgency] if pair[1] is not None else len(order)
    )

    for email, classification in results:
        if classification is None:
            print(f"[ERROR   ] {email.id}: classification failed")
            continue
        print(
            f"[{classification.urgency.value.upper():8}] {email.id} -> {classification.routing_team} "
            f"({classification.customer_sentiment}): {classification.reason}"
        )


if __name__ == "__main__":
    main()

````
