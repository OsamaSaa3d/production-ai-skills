**How it works:**
1. Claude (`claude-opus-5`, via the beta Tool Runner) gets the `orders` schema in its system prompt and one tool, `query_orders(sql)`.
2. Claude writes a `SELECT`, the tool validates it (single statement, `SELECT`/`WITH` only, blocklist of DDL/DML keywords, auto-appends `LIMIT 200` if missing) and runs it in a read-only, timeout-capped transaction via `psycopg2`.
3. Results come back as JSON; Claude turns them into a plain-English answer.

**Before running:**
- `pip install -r requirements.txt`
- `export ANTHROPIC_API_KEY=...` (or `ant auth login`)
- `export DATABASE_URL=postgresql://orders_readonly:...@host:5432/dbname` — **use a dedicated Postgres role that only has `SELECT` on `orders`** (example `CREATE ROLE`/`GRANT` statements are in the file's docstring). The in-app SQL validation is defense in depth, not a substitute for real DB-level privilege restriction — treat model-generated SQL as untrusted input.

**Usage:**
```
python nl_orders_query.py "How much revenue came from shipped orders last week?"
python nl_orders_query.py     # interactive mode
```

"""
Ask plain-English questions about the `orders` table and get plain-English answers.

How it works
------------
1. Claude is given the table schema and one tool: `query_orders(sql)`.
2. Claude writes a SELECT statement, the tool runs it read-only against
   Postgres, and Claude turns the rows into a natural-language answer.
3. The tool layer enforces SELECT-only, single-statement, row-capped,
   time-limited queries before anything touches the database.

Setup
-----
    pip install -r requirements.txt
    export ANTHROPIC_API_KEY=...        # or `ant auth login`
    export DATABASE_URL=postgresql://orders_readonly:...@host:5432/dbname

Strongly recommended: point DATABASE_URL at a dedicated Postgres role that
only has SELECT on the `orders` table, e.g.:

    CREATE ROLE orders_readonly LOGIN PASSWORD '...';
    GRANT CONNECT ON DATABASE mydb TO orders_readonly;
    GRANT USAGE ON SCHEMA public TO orders_readonly;
    GRANT SELECT ON orders TO orders_readonly;

The validation in this file (SELECT-only, no semicolons, keyword blocklist)
is defense in depth, not a substitute for that role -- a determined prompt
injection (e.g. text hidden inside order data Claude later reads) should
still not be able to do anything the DB grants don't allow.

Run
---
    python nl_orders_query.py "How much revenue came from shipped orders last week?"
    python nl_orders_query.py            # interactive mode
"""

from __future__ import annotations

import decimal
import datetime
import json
import os
import re
import sys

import psycopg2
import psycopg2.extras
from anthropic import Anthropic, beta_tool

MODEL = "claude-opus-5"
DATABASE_URL = os.environ["DATABASE_URL"]

MAX_ROWS = 200
STATEMENT_TIMEOUT_MS = 5000

SCHEMA_DESCRIPTION = """\
Table: orders
  id             bigint        primary key
  customer_email text          customer's email address
  status         text          one of: 'pending', 'paid', 'shipped', 'delivered', 'cancelled', 'refunded'
  total_usd      numeric       order total in US dollars
  placed_at      timestamptz   when the order was placed
"""

SYSTEM_PROMPT = f"""\
You answer plain-English questions about a Postgres `orders` table by writing
SQL and calling the `query_orders` tool to run it.

Schema:
{SCHEMA_DESCRIPTION}

Rules:
- Only ever write a single SELECT statement (CTEs with WITH are fine). The
  tool will reject anything else.
- Reference only the `orders` table and the columns listed above -- don't
  invent columns or tables.
- Prefer aggregates (COUNT, SUM, AVG) over dumping raw rows when the question
  asks for a number or a trend.
- If a question is ambiguous (e.g. "recent" orders), state the assumption you
  used (e.g. "last 7 days") in your final answer.
- If the question can't be answered from this table, say so plainly instead
  of guessing.
- After you get results back, answer in plain English. Don't dump raw JSON --
  summarize it, and mention the actual numbers.
"""

# Single-statement SELECT/WITH only, no stacked queries.
_FORBIDDEN_KEYWORDS = re.compile(
    r"\b(insert|update|delete|drop|alter|create|truncate|grant|revoke|copy|"
    r"vacuum|call|execute|merge|attach|detach|reindex|comment|listen|notify|"
    r"do|into)\b",
    re.IGNORECASE,
)
_STARTS_WITH_SELECT = re.compile(r"^\s*(select|with)\b", re.IGNORECASE)
_HAS_LIMIT = re.compile(r"\blimit\s+\d+\b", re.IGNORECASE)


class QueryRejected(Exception):
    """Raised when the model-generated SQL fails the safety checks."""


def _validate_and_cap(sql: str) -> str:
    stripped = sql.strip().rstrip(";").strip()
    if not stripped:
        raise QueryRejected("Empty query.")
    if ";" in stripped:
        raise QueryRejected("Only a single statement is allowed (no ';' inside the query).")
    if not _STARTS_WITH_SELECT.match(stripped):
        raise QueryRejected("Only SELECT (or WITH ... SELECT) statements are allowed.")
    if _FORBIDDEN_KEYWORDS.search(stripped):
        raise QueryRejected("Query contains a disallowed keyword. Only read-only SELECTs are allowed.")
    if not _HAS_LIMIT.search(stripped):
        stripped = f"{stripped} LIMIT {MAX_ROWS}"
    return stripped


def _json_default(value):
    if isinstance(value, decimal.Decimal):
        return float(value)
    if isinstance(value, (datetime.datetime, datetime.date)):
        return value.isoformat()
    raise TypeError(f"Object of type {type(value).__name__} is not JSON serializable")


def _run_readonly_query(sql: str) -> list[dict]:
    conn = psycopg2.connect(DATABASE_URL)
    try:
        conn.set_session(readonly=True, autocommit=True)
        with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute("SET LOCAL statement_timeout = %s", (STATEMENT_TIMEOUT_MS,))
            cur.execute(sql)
            return [dict(row) for row in cur.fetchall()]
    finally:
        conn.close()


@beta_tool
def query_orders(sql: str) -> str:
    """Run a single read-only SQL SELECT against the `orders` table and return the results as JSON.

    Args:
        sql: A single SELECT (or WITH ... SELECT) statement. Only the
            `orders` table and its documented columns may be referenced.
            No semicolons, no DDL/DML, no other statements.
    """
    try:
        capped_sql = _validate_and_cap(sql)
    except QueryRejected as exc:
        return f"Error: {exc}"

    try:
        rows = _run_readonly_query(capped_sql)
    except psycopg2.Error as exc:
        # str(exc) includes Postgres's own error message (e.g. bad column name),
        # which is useful for Claude to self-correct the next query attempt.
        return f"Error: query failed: {exc}"

    return json.dumps(
        {"row_count": len(rows), "rows": rows, "sql_used": capped_sql},
        default=_json_default,
    )


def ask(question: str) -> str:
    client = Anthropic()

    runner = client.beta.messages.tool_runner(
        model=MODEL,
        max_tokens=16000,
        system=SYSTEM_PROMPT,
        tools=[query_orders],
        messages=[{"role": "user", "content": question}],
    )

    final_message = None
    for message in runner:
        final_message = message

    if final_message is None:
        return "Sorry, I didn't get a response."

    if final_message.stop_reason == "refusal":
        return "I can't help with that request."

    text = "".join(block.text for block in final_message.content if block.type == "text")
    return text or "Sorry, I couldn't come up with an answer."


def main() -> None:
    if len(sys.argv) > 1:
        print(ask(" ".join(sys.argv[1:])))
        return

    print("Ask questions about the orders table (Ctrl+C to quit).")
    while True:
        try:
            question = input("\n> ").strip()
        except (KeyboardInterrupt, EOFError):
            print()
            break
        if not question:
            continue
        print(ask(question))


if __name__ == "__main__":
    main()
