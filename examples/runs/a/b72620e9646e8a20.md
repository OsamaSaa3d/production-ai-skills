````py
#!/usr/bin/env python
"""
extract_invoices.py

Parse plain-text invoices and load the key fields (invoice number, issue
date, total, customer name, PO number) into the `orders` SQLite table.

Usage:
    python extract_invoices.py --input invoices --db orders.db
    python extract_invoices.py --input invoices/single_invoice.txt --db orders.db
    python extract_invoices.py --input invoices --db orders.db --csv review.csv

Notes:
    - Invoice text formats vary, so each field is matched against a list of
      regex patterns (most specific first) and the first hit wins.
    - Any invoice missing a field is still written to the table, but flagged
      with needs_review=1 and printed to stderr so it can be checked by hand.
    - Re-running on the same invoice (same invoice_number) updates the
      existing row rather than duplicating it.
"""

import argparse
import csv
import re
import sqlite3
import sys
from datetime import datetime
from pathlib import Path

SCHEMA_PATH = Path(__file__).parent / "schema.sql"

# ---------------------------------------------------------------------------
# Field extraction patterns
# ---------------------------------------------------------------------------

INVOICE_NUMBER_PATTERNS = [
    r"Invoice\s*(?:Number|No\.?)\s*[:\-]\s*([A-Za-z0-9][A-Za-z0-9\-\/_]*)",
    r"Invoice\s*#\s*[:\-]?\s*([A-Za-z0-9][A-Za-z0-9\-\/_]*)",
    r"Invoice\s*[:\-]\s*([A-Za-z0-9][A-Za-z0-9\-\/_]*)",
]

ISSUE_DATE_PATTERNS = [
    r"(?:Issue\s*Date|Invoice\s*Date)\s*[:\-]\s*([A-Za-z0-9,\/\-\s]+?)\s*(?:\n|$)",
    r"(?<!Due\s)(?<!Due)\bDate\s*[:\-]\s*([A-Za-z0-9,\/\-\s]+?)\s*(?:\n|$)",
]

# "Total"/"Amount Due" etc, but not "Subtotal". Order matters: prefer the
# most explicit "final amount" label first.
TOTAL_PATTERNS = [
    r"(?:Total\s*Due|Amount\s*Due|Balance\s*Due|Grand\s*Total)\s*[:\-]\s*\$?\s*([\d,]+\.\d{2})",
    r"(?<!Sub)(?<!Sub )Total\s*[:\-]\s*\$?\s*([\d,]+\.\d{2})",
]

CUSTOMER_NAME_PATTERNS = [
    r"(?:Bill\s*To|Sold\s*To|Customer\s*Name|Customer|Client)\s*[:\-]\s*([^\n]+)",
]

PO_NUMBER_PATTERNS = [
    r"(?:P\.?O\.?\s*(?:Number|No\.?|#)|Purchase\s*Order\s*(?:Number|No\.?|#)?)\s*[:\-]\s*([A-Za-z0-9][A-Za-z0-9\-\/_]*)",
]

DATE_INPUT_FORMATS = [
    "%Y-%m-%d",
    "%m/%d/%Y",
    "%d/%m/%Y",
    "%B %d, %Y",
    "%b %d, %Y",
    "%d %B %Y",
    "%d %b %Y",
    "%m-%d-%Y",
]


def first_match(patterns, text):
    for pattern in patterns:
        m = re.search(pattern, text, re.IGNORECASE)
        if m:
            value = m.group(1).strip()
            if value:
                return value
    return None


def normalize_date(raw):
    if not raw:
        return None
    raw = raw.strip().rstrip(".,")
    for fmt in DATE_INPUT_FORMATS:
        try:
            return datetime.strptime(raw, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    # Couldn't confidently parse; keep the raw text rather than dropping data.
    return raw


def normalize_total(raw):
    if not raw:
        return None
    try:
        return float(raw.replace(",", ""))
    except ValueError:
        return None


def extract_fields(text, source_name):
    invoice_number = first_match(INVOICE_NUMBER_PATTERNS, text)
    issue_date_raw = first_match(ISSUE_DATE_PATTERNS, text)
    total_raw = first_match(TOTAL_PATTERNS, text)
    customer_name = first_match(CUSTOMER_NAME_PATTERNS, text)
    po_number = first_match(PO_NUMBER_PATTERNS, text)

    record = {
        "invoice_number": invoice_number,
        "issue_date": normalize_date(issue_date_raw),
        "total": normalize_total(total_raw),
        "customer_name": customer_name,
        "po_number": po_number,
        "source_file": source_name,
    }
    missing = [k for k in ("invoice_number", "issue_date", "total", "customer_name", "po_number")
               if not record[k]]
    record["needs_review"] = 1 if missing else 0
    return record, missing


def ensure_schema(conn):
    conn.executescript(SCHEMA_PATH.read_text(encoding="utf-8"))


def upsert_order(conn, record):
    if not record["invoice_number"]:
        # Without an invoice number we have no natural key to upsert on;
        # fall back to the source filename so the row isn't silently dropped.
        record["invoice_number"] = f"UNKNOWN-{record['source_file']}"

    conn.execute(
        """
        INSERT INTO orders (invoice_number, issue_date, total, customer_name,
                             po_number, source_file, needs_review)
        VALUES (:invoice_number, :issue_date, :total, :customer_name,
                :po_number, :source_file, :needs_review)
        ON CONFLICT(invoice_number) DO UPDATE SET
            issue_date    = excluded.issue_date,
            total         = excluded.total,
            customer_name = excluded.customer_name,
            po_number     = excluded.po_number,
            source_file   = excluded.source_file,
            needs_review  = excluded.needs_review,
            imported_at   = datetime('now')
        """,
        record,
    )


def iter_invoice_files(input_path: Path):
    if input_path.is_file():
        yield input_path
    elif input_path.is_dir():
        for path in sorted(input_path.glob("*.txt")):
            yield path
    else:
        raise FileNotFoundError(f"Input path not found: {input_path}")


def main():
    parser = argparse.ArgumentParser(description="Extract invoice fields into the orders table.")
    parser.add_argument("--input", required=True, help="Invoice .txt file or directory of .txt files")
    parser.add_argument("--db", default="orders.db", help="SQLite database file (default: orders.db)")
    parser.add_argument("--csv", help="Optional path to also write a CSV review export")
    args = parser.parse_args()

    input_path = Path(args.input)
    files = list(iter_invoice_files(input_path))
    if not files:
        print(f"No .txt invoices found under {input_path}", file=sys.stderr)
        return 1

    conn = sqlite3.connect(args.db)
    ensure_schema(conn)

    results = []
    any_missing = False
    for path in files:
        text = path.read_text(encoding="utf-8")
        record, missing = extract_fields(text, path.name)
        upsert_order(conn, record)
        results.append(record)
        if missing:
            any_missing = True
            print(f"[review] {path.name}: could not extract {', '.join(missing)}", file=sys.stderr)

    conn.commit()
    conn.close()

    print(f"Processed {len(files)} invoice(s) into {args.db} (table: orders).")

    if args.csv:
        with open(args.csv, "w", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(
                f,
                fieldnames=["invoice_number", "issue_date", "total", "customer_name",
                            "po_number", "source_file", "needs_review"],
            )
            writer.writeheader()
            writer.writerows(results)
        print(f"Wrote CSV export to {args.csv}")

    return 1 if any_missing else 0


if __name__ == "__main__":
    sys.exit(main())

````

# Invoice extraction → orders table

Parses plain-text invoices and writes `invoice_number`, `issue_date`,
`total`, `customer_name`, and `po_number` into an `orders` table.

## Assumptions made

This directory had no existing invoices, database, or schema to plug into,
so:

- **Storage**: a local SQLite database (`orders.db`), created from
  `schema.sql`. It's a real, queryable table — not a CSV dump — and needs
  no server setup. If your "orders table" actually lives in Postgres/MySQL/a
  spreadsheet system, point that out and this can be adapted (the extraction
  logic is decoupled from the write step in `extract_fields()` /
  `upsert_order()`).
- **Input**: invoices as `.txt` files in a folder (defaults to `invoices/`,
  see samples included there). Point `--input` at wherever your real
  invoice text lands.
- **Key**: `invoice_number` is treated as unique. Re-processing the same
  invoice updates its row instead of creating a duplicate.
- **Label formats**: invoice text isn't standardized, so each field is
  matched against several regex variants (e.g. "Invoice Number:",
  "Invoice #", "Invoice:"; "Total Due"/"Amount Due"/"Grand Total" before a
  plain "Total" so "Subtotal" is never mistaken for the total). Extend the
  pattern lists in `extract_invoices.py` as you see new invoice formats.
- **Missing fields**: an invoice is still written even if a field can't be
  found, but the row is flagged `needs_review = 1` and a warning is printed
  to stderr, so nothing is silently lost or silently wrong.

## Usage

```
python extract_invoices.py --input invoices --db orders.db
python extract_invoices.py --input invoices/one_invoice.txt --db orders.db
python extract_invoices.py --input invoices --db orders.db --csv review.csv
```

Exit code is `1` if any invoice needs manual review, `0` otherwise (useful
for wiring into a scheduled job).

## Files

- `extract_invoices.py` — parses invoices and upserts into SQLite.
- `schema.sql` — `orders` table definition.
- `invoices/` — three sample invoices (different formats, including one
  missing a PO number) used to verify the extraction.
- `orders.db` — generated by running the script against the samples.

-- Orders table populated from plain-text invoice extraction.
-- invoice_number is the natural key: re-importing the same invoice
-- updates the existing row instead of creating a duplicate.
CREATE TABLE IF NOT EXISTS orders (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    invoice_number  TEXT NOT NULL UNIQUE,
    issue_date      TEXT,               -- normalized to YYYY-MM-DD when parseable
    total           REAL,
    customer_name   TEXT,
    po_number       TEXT,
    source_file     TEXT,               -- which invoice .txt this row came from
    imported_at     TEXT DEFAULT (datetime('now')),
    needs_review    INTEGER NOT NULL DEFAULT 0  -- 1 if any field couldn't be extracted
);
