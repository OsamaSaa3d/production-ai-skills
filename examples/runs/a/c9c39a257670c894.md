"""Summarize a meeting transcript into a short, manager-skimmable paragraph.

Requires: pip install anthropic
Auth: set ANTHROPIC_API_KEY, or run `ant auth login` first.
"""

from __future__ import annotations

import anthropic

SYSTEM_PROMPT = (
    "You summarize meeting transcripts for a busy manager who was not in the "
    "meeting. Write exactly one short paragraph (3-6 sentences, no headers, "
    "no bullet points) that a manager can skim in under 15 seconds. Cover, "
    "in order of importance: the key decisions made, any action items with "
    "their owners, and open questions or blockers that need the manager's "
    "attention. Skip small talk and restating who attended. If the "
    "transcript contains no substantive content, say so in one sentence "
    "instead of inventing detail."
)


def summarize_meeting_transcript(
    transcript: str,
    client: anthropic.Anthropic | None = None,
    model: str = "claude-opus-5",
) -> str:
    """Summarize a meeting transcript into a short paragraph.

    Args:
        transcript: The raw meeting transcript text (speaker labels, notes,
            or a verbatim caption dump all work).
        client: An existing anthropic.Anthropic client to reuse (e.g. for
            connection pooling). A new one is created if omitted.
        model: The Claude model to use.

    Returns:
        A short paragraph summarizing the meeting.

    Raises:
        ValueError: If the transcript is empty.
        anthropic.APIError: On unrecoverable API failures (the SDK already
            retries transient network/429/5xx errors internally).
    """
    if not transcript or not transcript.strip():
        raise ValueError("transcript must be non-empty")

    client = client or anthropic.Anthropic()

    try:
        response = client.messages.create(
            model=model,
            max_tokens=1024,
            system=SYSTEM_PROMPT,
            # Summarization is a low-reasoning workload; low effort keeps
            # cost/latency down without hurting quality here.
            output_config={"effort": "low"},
            messages=[
                {
                    "role": "user",
                    "content": f"Transcript:\n\n{transcript}",
                }
            ],
        )
    except anthropic.AuthenticationError:
        raise RuntimeError("Invalid or missing Anthropic API key.") from None
    except anthropic.RateLimitError as e:
        retry_after = e.response.headers.get("retry-after", "unknown")
        raise RuntimeError(f"Rate limited; retry after {retry_after}s.") from e
    except anthropic.APIStatusError as e:
        raise RuntimeError(f"Anthropic API error ({e.status_code}): {e.message}") from e
    except anthropic.APIConnectionError as e:
        raise RuntimeError("Network error contacting the Anthropic API.") from e

    if response.stop_reason == "refusal":
        category = response.stop_details.category if response.stop_details else None
        raise RuntimeError(f"Summary request was refused (category: {category}).")

    for block in response.content:
        if block.type == "text":
            return block.text.strip()

    raise RuntimeError("No text content returned by the model.")


if __name__ == "__main__":
    example_transcript = """
    [10:00] Alice: Let's kick off. Sprint review first - we shipped the new
    onboarding flow, conversion is up 4%.
    [10:03] Bob: Nice. Payments migration is still blocked on the vendor's
    sandbox creds, I need someone from IT to escalate.
    [10:07] Alice: I'll ping IT today. Carol, where are we on the Q3 roadmap doc?
    [10:08] Carol: Draft is done, sharing by EOD for review. Need decisions on
    whether we cut the mobile redesign or push it to Q4.
    [10:15] Alice: Let's push mobile redesign to Q4, focus Q3 on payments and
    onboarding polish.
    [10:20] Bob: Agreed. I'll update the roadmap once Carol's draft is out.
    """

    print(summarize_meeting_transcript(example_transcript))
