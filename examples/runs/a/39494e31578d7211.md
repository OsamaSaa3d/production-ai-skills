"""Classify incoming support emails by urgency, for routing.

Requires: pip install anthropic pydantic
Auth: resolves ANTHROPIC_API_KEY / ANTHROPIC_AUTH_TOKEN / an `ant auth login`
profile automatically - no key needs to be hardcoded.
"""

from __future__ import annotations

import json
import random
import time
from enum import Enum
from typing import List, Optional

import anthropic
from pydantic import BaseModel, Field

MODEL = "claude-opus-5"

client = anthropic.Anthropic()


class Urgency(str, Enum):
    CRITICAL = "critical"  # outage / data loss / security / total blockage
    HIGH = "high"          # major feature broken, no workaround, losing money
    MEDIUM = "medium"      # partial impact, workaround exists
    LOW = "low"            # question, cosmetic issue, feature request


class EmailClassification(BaseModel):
    urgency: Urgency
    category: str = Field(description="Short label, e.g. 'billing', 'bug', 'account_access', 'feature_request'")
    reasoning: str = Field(description="One sentence justifying the urgency level")
    customer_sentiment: str = Field(description="One of: angry, frustrated, neutral, positive")


# Routing table: urgency -> (queue, target response time)
ROUTING = {
    Urgency.CRITICAL: ("on_call_engineer", "15 minutes"),
    Urgency.HIGH: ("senior_support", "2 hours"),
    Urgency.MEDIUM: ("general_support", "1 business day"),
    Urgency.LOW: ("general_support_backlog", "3 business days"),
}

SYSTEM_PROMPT = """You triage incoming customer support emails for a B2B SaaS product.

Classify each email's urgency using these rules:
- critical: production outage, data loss, security incident, or the customer is completely blocked with no workaround
- high: a major feature is broken, there's no workaround, or the customer is losing money/time at scale
- medium: partial impact with a workaround, or a time-sensitive but non-blocking request
- low: general question, minor/cosmetic issue, or a feature request

Base the classification only on the email's actual content and business impact -
not on politeness, length, or how urgently the customer claims it is."""


def classify_email(subject: str, body: str) -> EmailClassification:
    """Classify a single email. Retries on rate limits / transient server errors."""
    max_retries = 5
    for attempt in range(max_retries):
        try:
            response = client.messages.parse(
                model=MODEL,
                max_tokens=1024,
                system=SYSTEM_PROMPT,
                # Low effort is enough for classification and keeps cost/latency down.
                output_config={"effort": "low"},
                messages=[{
                    "role": "user",
                    "content": f"Subject: {subject}\n\n{body}",
                }],
                output_format=EmailClassification,
            )
            return response.parsed_output

        except anthropic.RateLimitError as e:
            retry_after = int(e.response.headers.get("retry-after", "0")) or None
            delay = retry_after or min(2 ** attempt + random.uniform(0, 1), 30)
            time.sleep(delay)
        except anthropic.APIStatusError as e:
            if e.status_code >= 500 and attempt < max_retries - 1:
                time.sleep(min(2 ** attempt + random.uniform(0, 1), 30))
            else:
                raise
        except anthropic.APIConnectionError:
            if attempt < max_retries - 1:
                time.sleep(min(2 ** attempt + random.uniform(0, 1), 30))
            else:
                raise

    raise RuntimeError(f"Failed to classify email after {max_retries} attempts")


def route_email(subject: str, body: str) -> dict:
    """Classify an email and attach routing info."""
    classification = classify_email(subject, body)
    queue, sla = ROUTING[classification.urgency]
    return {
        "subject": subject,
        "urgency": classification.urgency.value,
        "category": classification.category,
        "sentiment": classification.customer_sentiment,
        "reasoning": classification.reasoning,
        "queue": queue,
        "target_response_time": sla,
    }


def classify_emails_batch(emails: List[dict]) -> List[dict]:
    """Classify many emails via the Message Batches API (50% cheaper, async).

    `emails` is a list of {"id": ..., "subject": ..., "body": ...} dicts.
    Use this instead of `route_email` in a loop when processing a large
    backlog where sub-second latency per email doesn't matter.
    """
    requests = [
        anthropic.types.message_create_params.Request(
            custom_id=email["id"],
            params={
                "model": MODEL,
                "max_tokens": 1024,
                "system": SYSTEM_PROMPT,
                "output_config": {
                    "effort": "low",
                    "format": {
                        "type": "json_schema",
                        "schema": {
                            "type": "object",
                            "properties": {
                                "urgency": {"type": "string", "enum": [u.value for u in Urgency]},
                                "category": {"type": "string"},
                                "reasoning": {"type": "string"},
                                "customer_sentiment": {"type": "string"},
                            },
                            "required": ["urgency", "category", "reasoning", "customer_sentiment"],
                            "additionalProperties": False,
                        },
                    },
                },
                "messages": [{
                    "role": "user",
                    "content": f"Subject: {email['subject']}\n\n{email['body']}",
                }],
            },
        )
        for email in emails
    ]

    batch = client.messages.batches.create(requests=requests)
    while True:
        batch = client.messages.batches.retrieve(batch.id)
        if batch.processing_status == "ended":
            break
        time.sleep(10)

    results = {}
    for result in client.messages.batches.results(batch.id):
        if result.result.type != "succeeded":
            results[result.custom_id] = {"error": result.result.type}
            continue
        text = next(b.text for b in result.result.message.content if b.type == "text")
        data = json.loads(text)
        urgency = Urgency(data["urgency"])
        queue, sla = ROUTING[urgency]
        results[result.custom_id] = {
            "urgency": urgency.value,
            "category": data["category"],
            "sentiment": data["customer_sentiment"],
            "reasoning": data["reasoning"],
            "queue": queue,
            "target_response_time": sla,
        }

    # Preserve input order; batch results arrive in arbitrary order.
    return [{"id": email["id"], **results[email["id"]]} for email in emails]


if __name__ == "__main__":
    sample_emails = [
        {
            "subject": "URGENT: production down for all users",
            "body": "Our entire team is locked out and customers are seeing 500 errors. "
                    "This started 10 minutes ago and we're losing sales every minute.",
        },
        {
            "subject": "Question about invoice",
            "body": "Hi, could you clarify what the 'platform fee' line item on our latest "
                    "invoice covers? No rush, just curious.",
        },
        {
            "subject": "Export feature broken for CSV files",
            "body": "CSV export has been failing all morning with a generic error. "
                    "We can still export to Excel as a workaround, but our nightly "
                    "pipeline expects CSV.",
        },
    ]

    for email in sample_emails:
        result = route_email(email["subject"], email["body"])
        print(json.dumps(result, indent=2))
