````py
#!/usr/bin/env python3
"""
extract_invoices.py

Pulls (invoice number, issue date, total, customer name, PO number) out of
plain-text invoices and writes/upserts them into an `orders` table.

Usage:
    python extract_invoices.py <file-or-directory> [more files/dirs ...]
    python extract_invoices.py invoices/ --db orders.db --csv orders.csv

Behavior:
    - Accepts one or more .txt files, or directories (scanned non-recursively
      for *.txt).
    - Writes rows into a SQLite table `orders` (file: orders.db by default).
    - Re-running on an invoice with the same invoice_number UPDATES that row
      instead of inserting a duplicate.
    - Also mirrors the full table out to a CSV file (orders.csv by default)
      after every run, so it's easy to eyeball / open in Excel.
    - Any invoice where a required field can't be found is skipped and
      reported at the end instead of silently writing a blank row.

Field matching assumptions (see README.md for details / how to extend):
    - Invoice number:  "Invoice Number/No/#: ..."
    - Issue date:       "Issue Date: ..." or "Date: ..." or "Invoice Date: ..."
    - Total:            "Total / Total Due / Amount Due / Grand Total: $..."
                         (deliberately does NOT match "Subtotal")
    - Customer name:    "Bill To / Customer / Client: ..."
    - PO number:        "PO Number / PO# / P.O. / Purchase Order: ..."
"""

import argparse
import csv
import re
import sqlite3
import sys
from datetime import datetime
from pathlib import Path

FIELD_PATTERNS = {
    "invoice_number": re.compile(
        r"^\s*Invoice\s*(?:Number|No\.?|#)\s*[:\-]\s*(\S.*?)\s*$", re.IGNORECASE
    ),
    "issue_date": re.compile(
        r"^\s*(?:Issue\s*Date|Invoice\s*Date|Date)\s*[:\-]\s*(\S.*?)\s*$",
        re.IGNORECASE,
    ),
    "total": re.compile(
        r"^\s*(?:Total(?:\s*Due|\s*Amount)?|Grand\s*Total|Amount\s*Due)\s*[:\-]\s*"
        r"\$?\s*([\d,]+(?:\.\d{1,2})?)\s*$",
        re.IGNORECASE,
    ),
    "customer_name": re.compile(
        r"^\s*(?:Bill\s*To|Customer(?:\s*Name)?|Client)\s*[:\-]\s*(\S.*?)\s*$",
        re.IGNORECASE,
    ),
    "po_number": re.compile(
        r"^\s*(?:P\.?O\.?\s*(?:Number|No\.?|#)?|Purchase\s*Order\s*(?:Number|No\.?|#)?)"
        r"\s*[:\-]\s*(\S.*?)\s*$",
        re.IGNORECASE,
    ),
}

REQUIRED_FIELDS = ["invoice_number", "issue_date", "total", "customer_name"]
# PO number is treated as optional since not every order has one.
OPTIONAL_FIELDS = ["po_number"]
ALL_FIELDS = REQUIRED_FIELDS + OPTIONAL_FIELDS

DATE_INPUT_FORMATS = [
    "%Y-%m-%d",
    "%m/%d/%Y",
    "%m/%d/%y",
    "%d/%m/%Y",
    "%B %d, %Y",
    "%b %d, %Y",
    "%d %B %Y",
    "%d %b %Y",
]


def normalize_date(raw: str) -> str:
    """Try to normalize a date string to ISO YYYY-MM-DD; fall back to raw text."""
    for fmt in DATE_INPUT_FORMATS:
        try:
            return datetime.strptime(raw, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    return raw


def normalize_total(raw: str) -> float:
    return float(raw.replace(",", ""))


def parse_invoice_text(text: str) -> dict:
    """Extract fields from invoice text. Returns dict with whatever was found
    (missing fields are simply absent from the dict)."""
    found = {}
    for line in text.splitlines():
        for field, pattern in FIELD_PATTERNS.items():
            if field in found:
                continue
            m = pattern.match(line)
            if m:
                found[field] = m.group(1).strip()
    return found


def parse_invoice_file(path: Path):
    """Returns (record_dict, list_of_missing_required_fields)."""
    text = path.read_text(encoding="utf-8", errors="replace")
    found = parse_invoice_text(text)

    missing = [f for f in REQUIRED_FIELDS if f not in found]

    record = {f: found.get(f) for f in ALL_FIELDS}
    if not missing:
        record["issue_date"] = normalize_date(record["issue_date"])
        try:
            record["total"] = normalize_total(record["total"])
        except ValueError:
            missing.append("total")
    return record, missing


def gather_txt_files(paths):
    files = []
    for p in paths:
        p = Path(p)
        if p.is_dir():
            files.extend(sorted(p.glob("*.txt")))
        elif p.is_file():
            files.append(p)
        else:
            print(f"WARNING: path not found, skipping: {p}", file=sys.stderr)
    return files


def ensure_table(conn: sqlite3.Connection):
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS orders (
            invoice_number TEXT PRIMARY KEY,
            issue_date     TEXT NOT NULL,
            total          REAL NOT NULL,
            customer_name  TEXT NOT NULL,
            po_number      TEXT,
            source_file    TEXT,
            updated_at     TEXT NOT NULL
        )
        """
    )
    conn.commit()


def upsert_order(conn: sqlite3.Connection, record: dict, source_file: str):
    conn.execute(
        """
        INSERT INTO orders (invoice_number, issue_date, total, customer_name,
                             po_number, source_file, updated_at)
        VALUES (:invoice_number, :issue_date, :total, :customer_name,
                :po_number, :source_file, :updated_at)
        ON CONFLICT(invoice_number) DO UPDATE SET
            issue_date=excluded.issue_date,
            total=excluded.total,
            customer_name=excluded.customer_name,
            po_number=excluded.po_number,
            source_file=excluded.source_file,
            updated_at=excluded.updated_at
        """,
        {**record, "source_file": source_file, "updated_at": datetime.now().isoformat(timespec="seconds")},
    )


def export_csv(conn: sqlite3.Connection, csv_path: Path):
    cur = conn.execute(
        "SELECT invoice_number, issue_date, total, customer_name, po_number, "
        "source_file, updated_at FROM orders ORDER BY issue_date, invoice_number"
    )
    rows = cur.fetchall()
    with csv_path.open("w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(
            ["invoice_number", "issue_date", "total", "customer_name",
             "po_number", "source_file", "updated_at"]
        )
        writer.writerows(rows)


def main():
    ap = argparse.ArgumentParser(description="Extract invoice fields into an orders table.")
    ap.add_argument("paths", nargs="+", help="Invoice .txt file(s) and/or directories of them.")
    ap.add_argument("--db", default="orders.db", help="SQLite DB file (default: orders.db)")
    ap.add_argument("--csv", default="orders.csv", help="CSV mirror file (default: orders.csv)")
    args = ap.parse_args()

    files = gather_txt_files(args.paths)
    if not files:
        print("No .txt invoice files found.", file=sys.stderr)
        sys.exit(1)

    conn = sqlite3.connect(args.db)
    ensure_table(conn)

    written, skipped = [], []
    for path in files:
        record, missing = parse_invoice_file(path)
        if missing:
            skipped.append((path, missing))
            continue
        upsert_order(conn, record, str(path))
        written.append((path, record))

    conn.commit()
    export_csv(conn, Path(args.csv))
    conn.close()

    print(f"Wrote {len(written)} order(s) to {args.db} (table 'orders') and mirrored to {args.csv}.")
    for path, record in written:
        print(
            f"  OK   {path.name}: invoice={record['invoice_number']} "
            f"date={record['issue_date']} total={record['total']} "
            f"customer={record['customer_name']!r} po={record['po_number']}"
        )
    for path, missing in skipped:
        print(f"  SKIP {path.name}: missing field(s) {missing}", file=sys.stderr)

    if skipped:
        sys.exit(2)


if __name__ == "__main__":
    main()

````

### `README.md`

````md
# Invoice Extractor

Pulls **invoice number, issue date, total, customer name, and PO number** out
of plain-text invoices and writes them into an `orders` table.

## Usage

```
python extract_invoices.py <file-or-directory> [more paths...] [--db orders.db] [--csv orders.csv]
```

Examples:

```
python extract_invoices.py invoices/                 # scan a folder of *.txt invoices
python extract_invoices.py invoices/inv_2044.txt      # a single invoice
python extract_invoices.py inbox_invoices/ archived/  # multiple folders at once
```

## Output

- **`orders.db`** (SQLite) — table `orders`, columns:
  `invoice_number` (primary key), `issue_date` (ISO `YYYY-MM-DD`), `total`
  (numeric), `customer_name`, `po_number`, `source_file`, `updated_at`.
- **`orders.csv`** — full mirror of the table, regenerated on every run, for
  opening in Excel/Sheets.

Re-running on an invoice with the same invoice number **updates** that row
(no duplicates) — safe to re-run over a folder repeatedly, e.g. as new
invoices arrive.

## What it looks for

Matched case-insensitively, one label per line:

| Field           | Recognized labels |
|-----------------|--------------------|
| Invoice number  | `Invoice Number:`, `Invoice No:`, `Invoice #:` |
| Issue date      | `Issue Date:`, `Invoice Date:`, `Date:` |
| Total           | `Total:`, `Total Due:`, `Total Amount:`, `Amount Due:`, `Grand Total:` (deliberately does **not** match `Subtotal:`) |
| Customer name   | `Bill To:`, `Customer:`, `Customer Name:`, `Client:` |
| PO number       | `PO Number:`, `PO No:`, `PO#:`, `P.O.:`, `Purchase Order:`, `Purchase Order Number:` |

Dates are normalized to `YYYY-MM-DD` when they match a common format
(`MM/DD/YYYY`, `DD/MM/YYYY`, `Month D, YYYY`, etc.); otherwise the raw text is
kept as-is.

Invoice number, issue date, total, and customer name are **required** — if
any is missing, the invoice is skipped (not written with blank fields) and
reported at the end, e.g.:

```
SKIP invoice_1003_incomplete.txt: missing field(s) ['issue_date']
```

The script exits with code `2` if anything was skipped, `0` if everything
was written, `1` if no `.txt` files were found. PO number is treated as
**optional** (some orders legitimately have none) and stored as `NULL` if
absent.

## Extending it

All matching lives in `FIELD_PATTERNS` at the top of `extract_invoices.py` —
one regex per field, anchored to the start of a line. If your real invoices
use a label not listed above (e.g. `Bill No:` or `Ship To:` for the customer),
add an alternative into the relevant pattern.

script — two well-formed (different label styles, to prove the aliasing
works) and one intentionally missing a field (to prove skip+report works).
Safe to delete once you point the script at real invoices.
````
