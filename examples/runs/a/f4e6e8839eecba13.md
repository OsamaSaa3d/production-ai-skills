### `issue_store.py`

````py
"""Mock issue backlog used for duplicate-detection retrieval.

In production, swap `search_similar_issues` for a call to your real issue
tracker's search API (e.g. GitHub "search issues", Jira JQL full-text search,
Linear's `issueSearch`) - keep the same return shape (list of dicts with at
least `id`, `title`, `status`) and the rest of the agent needs no changes.

The scoring here is a cheap lexical heuristic (word overlap + fuzzy ratio).
It's deliberately dumb: its only job is to hand the LLM a short list of
*candidates* worth reading. The actual "is this really the same bug"
judgment is made by Claude in triage_agent.py, which can read nuance
(different root cause, different affected platform, etc.) that keyword
overlap can't.
"""

from __future__ import annotations

import json
import re
from difflib import SequenceMatcher
from pathlib import Path

DATA_PATH = Path(__file__).parent / "data" / "issues.json"

_STOPWORDS = {
    "the", "a", "an", "is", "are", "was", "were", "on", "in", "at", "to",
    "of", "and", "or", "for", "with", "this", "that", "it", "my", "i",
    "me", "when", "after", "before", "up", "down", "just", "can", "not",
}


def load_issues() -> list[dict]:
    with open(DATA_PATH, encoding="utf-8") as f:
        return json.load(f)


def _words(text: str) -> set[str]:
    tokens = re.findall(r"[a-z0-9]+", text.lower())
    return {t for t in tokens if t not in _STOPWORDS and len(t) > 2}


def _score(query: str, issue: dict) -> float:
    haystack = f"{issue['title']} {issue['description']}"
    ratio = SequenceMatcher(None, query.lower(), haystack.lower()).ratio()

    q_words = _words(query)
    h_words = _words(haystack)
    overlap = len(q_words & h_words) / max(1, len(q_words))

    return 0.5 * ratio + 0.5 * overlap


def search_similar_issues(query: str, max_results: int = 5) -> list[dict]:
    """Return the existing issues most lexically similar to `query`.

    Args:
        query: Free text to search with - typically the new report's title
            plus a snippet of its description.
        max_results: Maximum number of candidates to return.
    """
    issues = load_issues()
    scored = [(_score(query, issue), issue) for issue in issues]
    scored.sort(key=lambda pair: pair[0], reverse=True)

    results = []
    for score, issue in scored[:max_results]:
        if score <= 0.05:
            continue
        results.append(
            {
                "id": issue["id"],
                "title": issue["title"],
                "description": issue["description"],
                "status": issue["status"],
                "team": issue["team"],
                "severity": issue["severity"],
                "similarity_score": round(score, 3),
            }
        )
    return results

````

### `README.md`

````md
# Bug Triage Agent

Reads an incoming bug report and produces one structured triage decision:

- **Duplicate check** - searches the existing issue backlog and judges whether
  the report is the same bug as something already filed.
- **Severity** - `critical` / `high` / `medium` / `low`, per the rubric in
  `triage_agent.py`.
- **Routing** - which team owns it: `security`, `backend`, `frontend`,
  `mobile`, `infra`, `data`, or `support`.

## How it works

`triage_agent.py` runs a small agent loop against the Claude API
(`claude-opus-5`): Claude reads the report, can call a `search_similar_issues`
tool (as many times as it wants) to pull candidate duplicates from the issue
backlog, then returns one JSON object validated against a fixed schema
(`output_config.format`) - no free-form prose to parse.

```
report.json ──▶ triage_agent.triage() ──▶ [search_similar_issues tool]──▶ issue_store.py (data/issues.json)
                        │
                        ▼
              { duplicate, severity, routing, summary }
```

`issue_store.py` is a **mock** backlog (`data/issues.json`, 10 sample issues)
searched with cheap lexical scoring - just enough to hand Claude a short list
of plausible candidates. The actual duplicate judgment (same root cause vs.
superficially similar wording) is made by the model, which can reason about
things keyword matching can't.

**To point this at a real tracker:** replace `search_similar_issues` in
`issue_store.py` with a call to your tracker's search API (GitHub "search
issues", Jira JQL, Linear `issueSearch`, ...) and keep the same return shape
(`id`, `title`, `description`, `status`, ...). Nothing else in
`triage_agent.py` needs to change.

  care how candidates were found, only what's in them.
````

````py
"""Bug report triage agent.

Given one incoming bug report, the agent:
  1. Searches the existing issue backlog for likely duplicates (via a tool
     call it can invoke itself, as many times as it needs).
  2. Assigns a severity label (critical / high / medium / low).
  3. Routes the report to the owning team.

It returns one structured JSON decision, suitable for a bot to post as a
comment/label/assignment on the real ticket.

Usage:
    python triage_agent.py path/to/report.json
    python triage_agent.py < report.json
    python triage_agent.py --sample          # run every file in data/sample_reports/

Auth: resolves credentials the standard way (ANTHROPIC_API_KEY,
ANTHROPIC_AUTH_TOKEN, or an `ant auth login` profile) - see README.md.
"""

from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

import anthropic

from issue_store import search_similar_issues

MODEL = "claude-opus-5"
SAMPLE_DIR = Path(__file__).parent / "data" / "sample_reports"

SYSTEM_PROMPT = """\
You are a bug triage agent for a software company's engineering org. You are \
given one incoming bug report and must produce a single triage decision.

Do this in order:

1. DUPLICATE CHECK
   Call `search_similar_issues` with a short query built from the report's \
title and key symptoms. You may call it more than once (e.g. retry with \
different phrasing) if the first results look irrelevant. Read the \
candidates it returns and judge duplication on substance, not just wording: \
same underlying bug/root cause and same affected surface counts as a \
duplicate even if worded differently; superficially similar text about a \
*different* root cause, platform, or endpoint does not. Only mark \
is_duplicate true when you are reasonably confident, and cite the specific \
existing issue id.

2. SEVERITY
   Assign exactly one of: critical, high, medium, low.
   - critical: data loss/corruption, security vulnerability, full outage, \
payment/billing failure, or any bug with no workaround affecting all/most users.
   - high: a major feature is broken for a significant share of users, or a \
security-adjacent issue with limited exposure; a workaround may exist but is \
painful.
   - medium: a feature is partially broken or degraded, impact is limited to \
a subset of users, and a reasonable workaround exists.
   - low: cosmetic, minor UX annoyance, or an edge case affecting very few \
users with no functional impact.

3. ROUTING
   Assign exactly one team:
   - security: authN/authZ, data exposure/leakage, injection, XSS/CSRF, \
permission bypass, credential handling.
   - backend: API/server errors, database, business logic, third-party \
integrations, background jobs.
   - frontend: web UI rendering, layout/CSS, client-side JS errors, \
browser-specific behavior.
   - mobile: iOS/Android app crashes or UI issues.
   - infra: deployment, CI/CD, performance/scaling, infrastructure outages \
and alerts.
   - data: analytics, reporting, data pipelines/ETL, data correctness in \
exports/dashboards.
   - support: documentation, account/billing questions, and other non-bug \
requests.

   A report can plausibly span two areas (e.g. a security bug in a frontend \
form) - pick the team that owns the root cause and fix, not the team that \
merely surfaces the symptom.

Respond with only the final JSON object described by the schema - no prose \
outside it. Keep every "reasoning" field to 1-2 sentences.
"""

TRIAGE_SCHEMA = {
    "type": "object",
    "properties": {
        "title": {
            "type": "string",
            "description": "A clean, concise restatement of the bug title.",
        },
        "duplicate": {
            "type": "object",
            "properties": {
                "is_duplicate": {"type": "boolean"},
                "issue_id": {
                    "type": ["string", "null"],
                    "description": "Existing issue id if is_duplicate is true, else null.",
                },
                "confidence": {"type": "string", "enum": ["low", "medium", "high"]},
                "reasoning": {"type": "string"},
            },
            "required": ["is_duplicate", "issue_id", "confidence", "reasoning"],
            "additionalProperties": False,
        },
        "severity": {
            "type": "object",
            "properties": {
                "level": {
                    "type": "string",
                    "enum": ["critical", "high", "medium", "low"],
                },
                "reasoning": {"type": "string"},
            },
            "required": ["level", "reasoning"],
            "additionalProperties": False,
        },
        "routing": {
            "type": "object",
            "properties": {
                "team": {
                    "type": "string",
                    "enum": [
                        "security",
                        "backend",
                        "frontend",
                        "mobile",
                        "infra",
                        "data",
                        "support",
                    ],
                },
                "reasoning": {"type": "string"},
            },
            "required": ["team", "reasoning"],
            "additionalProperties": False,
        },
        "summary": {
            "type": "string",
            "description": "One or two sentences summarizing the report for a triage queue.",
        },
    },
    "required": ["title", "duplicate", "severity", "routing", "summary"],
    "additionalProperties": False,
}

SEARCH_TOOL = {
    "name": "search_similar_issues",
    "description": (
        "Search the existing issue backlog for issues that might be the same "
        "bug as the one being triaged. Returns up to `max_results` candidates "
        "ranked by lexical similarity, each with id, title, description, "
        "status, team, severity, and similarity_score. A high score does not "
        "guarantee a true duplicate - read the candidate's description."
    ),
    "input_schema": {
        "type": "object",
        "properties": {
            "query": {
                "type": "string",
                "description": "Search text - e.g. the report title plus key symptoms.",
            },
            "max_results": {
                "type": "integer",
                "description": "Max candidates to return (default 5).",
            },
        },
        "required": ["query"],
        "additionalProperties": False,
    },
}


def format_report(report: dict) -> str:
    lines = [f"Title: {report.get('title', '(no title provided)')}"]
    if report.get("description"):
        lines.append(f"Description: {report['description']}")
    if report.get("steps_to_reproduce"):
        lines.append(f"Steps to reproduce: {report['steps_to_reproduce']}")
    if report.get("environment"):
        lines.append(f"Environment: {report['environment']}")
    if report.get("reporter"):
        lines.append(f"Reporter: {report['reporter']}")
    return "\n".join(lines)


def run_tool(name: str, tool_input: dict) -> list[dict]:
    if name == "search_similar_issues":
        return search_similar_issues(
            tool_input["query"], tool_input.get("max_results", 5)
        )
    raise ValueError(f"Unknown tool: {name}")


def triage(client: anthropic.Anthropic, report: dict, *, max_tool_turns: int = 5) -> dict:
    """Run the triage loop for one report and return the parsed decision."""
    messages = [{"role": "user", "content": format_report(report)}]

    for _ in range(max_tool_turns):
        try:
            response = client.messages.create(
                model=MODEL,
                max_tokens=4096,
                system=SYSTEM_PROMPT,
                thinking={"type": "adaptive"},
                output_config={
                    "effort": "medium",
                    "format": {"type": "json_schema", "schema": TRIAGE_SCHEMA},
                },
                tools=[SEARCH_TOOL],
                messages=messages,
            )
        except anthropic.RateLimitError as e:
            retry_after = e.response.headers.get("retry-after", "unknown")
            raise RuntimeError(f"Rate limited; retry after {retry_after}s") from e
        except anthropic.AuthenticationError as e:
            raise RuntimeError(
                "Authentication failed - set ANTHROPIC_API_KEY or run `ant auth login`."
            ) from e
        except anthropic.APIStatusError as e:
            raise RuntimeError(f"API error ({e.status_code}): {e.message}") from e
        except anthropic.APIConnectionError as e:
            raise RuntimeError("Network error reaching the Anthropic API.") from e

        if response.stop_reason == "refusal":
            category = getattr(response.stop_details, "category", None)
            raise RuntimeError(f"Model declined to triage this report (category={category}).")

        tool_uses = [b for b in response.content if b.type == "tool_use"]

        if not tool_uses:
            text = next((b.text for b in response.content if b.type == "text"), None)
            if text is None:
                raise RuntimeError(
                    f"No text in final response (stop_reason={response.stop_reason})."
                )
            return json.loads(text)

        messages.append({"role": "assistant", "content": response.content})

        tool_results = []
        for call in tool_uses:
            try:
                result = run_tool(call.name, call.input)
                tool_results.append(
                    {
                        "type": "tool_result",
                        "tool_use_id": call.id,
                        "content": json.dumps(result),
                    }
                )
            except Exception as exc:  # tool failures go back to the model, not raised
                tool_results.append(
                    {
                        "type": "tool_result",
                        "tool_use_id": call.id,
                        "content": f"Error: {exc}",
                        "is_error": True,
                    }
                )
        messages.append({"role": "user", "content": tool_results})

    raise RuntimeError(f"Gave up after {max_tool_turns} tool-calling turns without a final answer.")


def print_decision(report: dict, decision: dict) -> None:
    dup = decision["duplicate"]
    sev = decision["severity"]
    route = decision["routing"]

    print(f"\n=== {decision['title']} ===")
    print(f"Summary:   {decision['summary']}")
    print(f"Severity:  {sev['level'].upper()}  -  {sev['reasoning']}")
    print(f"Team:      {route['team']}  -  {route['reasoning']}")
    if dup["is_duplicate"]:
        print(f"Duplicate: YES of {dup['issue_id']} (confidence: {dup['confidence']})  -  {dup['reasoning']}")
    else:
        print(f"Duplicate: no (confidence: {dup['confidence']})  -  {dup['reasoning']}")


def load_report(path: str | None) -> dict:
    if path:
        with open(path, encoding="utf-8") as f:
            return json.load(f)
    return json.load(sys.stdin)


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("report", nargs="?", help="Path to a bug report JSON file (title, description, ...). Reads stdin if omitted.")
    parser.add_argument("--sample", action="store_true", help="Run every sample report in data/sample_reports/.")
    parser.add_argument("--json", action="store_true", help="Print raw JSON decisions instead of the human-readable summary.")
    args = parser.parse_args()

    client = anthropic.Anthropic()

    if args.sample:
        report_paths = sorted(SAMPLE_DIR.glob("*.json"))
        if not report_paths:
            print(f"No sample reports found in {SAMPLE_DIR}", file=sys.stderr)
            return 1
        exit_code = 0
        for path in report_paths:
            report = json.loads(path.read_text(encoding="utf-8"))
            try:
                decision = triage(client, report)
            except RuntimeError as e:
                print(f"\n=== {path.name}: FAILED ===\n{e}", file=sys.stderr)
                exit_code = 1
                continue
            if args.json:
                print(json.dumps(decision, indent=2))
            else:
                print_decision(report, decision)
        return exit_code

    try:
        report = load_report(args.report)
    except (json.JSONDecodeError, FileNotFoundError) as e:
        print(f"Could not read report: {e}", file=sys.stderr)
        return 1

    try:
        decision = triage(client, report)
    except RuntimeError as e:
        print(f"Triage failed: {e}", file=sys.stderr)
        return 1

    if args.json:
        print(json.dumps(decision, indent=2))
    else:
        print_decision(report, decision)
    return 0


if __name__ == "__main__":
    sys.exit(main())
````
