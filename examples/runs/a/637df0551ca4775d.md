"""Summarize a meeting transcript into a short, manager-skimmable paragraph."""

import anthropic

MODEL = "claude-opus-5"

SYSTEM_PROMPT = (
    "You summarize meeting transcripts for a busy manager who has 15 seconds to "
    "skim. Write ONE short paragraph (3-5 sentences, no bullet points, no headers) "
    "covering: what the meeting was about, the key decisions made, any action items "
    "with their owners, and open questions or blockers. Be concrete (names, dates, "
    "numbers) and skip small talk, pleasantries, and filler. If the transcript is "
    "too short or unclear to summarize, say so plainly in one sentence instead of "
    "guessing."
)


def summarize_meeting_transcript(
    transcript: str,
    *,
    meeting_title: str | None = None,
    client: anthropic.Anthropic | None = None,
) -> str:
    """Summarize a meeting transcript into a short paragraph a manager can skim.

    Args:
        transcript: The raw meeting transcript text (speaker labels, notes, etc.
            are fine as-is - no special formatting required).
        meeting_title: Optional title/context (e.g. "Q3 Roadmap Sync") to help
            Claude frame the summary.
        client: Optional pre-built anthropic.Anthropic client. If omitted, one
            is created from the environment (ANTHROPIC_API_KEY / ANTHROPIC_AUTH_TOKEN
            / an `ant auth login` profile).

    Returns:
        A short paragraph (plain text) summarizing the meeting.

    Raises:
        ValueError: If the transcript is empty.
        anthropic.APIError: On unrecoverable API failures (see error handling below).
    """
    if not transcript or not transcript.strip():
        raise ValueError("transcript must be non-empty")

    client = client or anthropic.Anthropic()

    user_content = transcript.strip()
    if meeting_title:
        user_content = f"Meeting: {meeting_title}\n\n{user_content}"

    try:
        response = client.messages.create(
            model=MODEL,
            max_tokens=400,
            system=SYSTEM_PROMPT,
            output_config={"effort": "low"},  # summarization doesn't need deep reasoning
            messages=[{"role": "user", "content": user_content}],
        )
    except anthropic.RateLimitError as e:
        retry_after = int(e.response.headers.get("retry-after", "60"))
        raise RuntimeError(f"Rate limited by the API; retry after {retry_after}s") from e
    except anthropic.APIStatusError as e:
        if e.status_code >= 500:
            raise RuntimeError(f"Anthropic API server error ({e.status_code}); try again later") from e
        raise RuntimeError(f"Anthropic API rejected the request: {e.message}") from e
    except anthropic.APIConnectionError as e:
        raise RuntimeError("Network error while contacting the Anthropic API") from e

    if response.stop_reason == "refusal":
        raise RuntimeError("Claude declined to summarize this transcript")

    for block in response.content:
        if block.type == "text":
            return block.text.strip()

    raise RuntimeError("No text content returned by the API")


if __name__ == "__main__":
    example_transcript = """
    [10:00] Alice: Let's kick off the Q3 roadmap sync. First up, mobile redesign.
    [10:02] Bob: Design mocks are done. Dev estimates 3 weeks starting next Monday.
    [10:05] Alice: Great, let's lock that in. Bob, can you own the kickoff?
    [10:05] Bob: Yep, I'll send the plan by Friday.
    [10:07] Carol: Quick flag - we're blocked on the payments API until legal signs
    off on the new vendor contract. No ETA yet.
    [10:10] Alice: Ok, I'll chase legal this week. Let's revisit payments next sync.
    [10:12] Alice: Anything else? ... Ok, that's a wrap.
    """
    print(summarize_meeting_transcript(example_transcript, meeting_title="Q3 Roadmap Sync"))
