````py
"""Minimal Anthropic tool-use agent with a single tool: get_user_by_id.

Usage:
    set ANTHROPIC_API_KEY=...   (PowerShell: $env:ANTHROPIC_API_KEY = "...")
    python agent.py "Look up user 2 and tell me their email"

Requires: pip install anthropic
"""

import os
import sys
import json

import anthropic

from users_db import get_user_by_id

MODEL = "claude-sonnet-5"

TOOLS = [
    {
        "name": "get_user_by_id",
        "description": "Fetch a user's record (id, name, email) given their user id.",
        "input_schema": {
            "type": "object",
            "properties": {
                "user_id": {
                    "type": "string",
                    "description": "The id of the user to fetch.",
                }
            },
            "required": ["user_id"],
        },
    }
]


def call_tool(name: str, tool_input: dict) -> dict:
    if name == "get_user_by_id":
        return get_user_by_id(tool_input["user_id"])
    raise ValueError(f"Unknown tool: {name}")


def run(prompt: str) -> str:
    client = anthropic.Anthropic()
    messages = [{"role": "user", "content": prompt}]

    while True:
        response = client.messages.create(
            model=MODEL,
            max_tokens=1024,
            tools=TOOLS,
            messages=messages,
        )
        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason != "tool_use":
            return "".join(
                block.text for block in response.content if block.type == "text"
            )

        tool_results = []
        for block in response.content:
            if block.type == "tool_use":
                result = call_tool(block.name, block.input)
                tool_results.append(
                    {
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(result),
                    }
                )
        messages.append({"role": "user", "content": tool_results})


if __name__ == "__main__":
    prompt = " ".join(sys.argv[1:]) or "Who is user 1?"
    print(run(prompt))

````

### `users_db.py`

````py
"""Tiny in-memory user store, standing in for a real database/service.

Swap `get_user_by_id` for a real DB/API call when wiring this up for real.
"""

_USERS = {
    "1": {"id": "1", "name": "Ada Lovelace", "email": "ada@example.com"},
    "2": {"id": "2", "name": "Grace Hopper", "email": "grace@example.com"},
    "3": {"id": "3", "name": "Alan Turing", "email": "alan@example.com"},
}


def get_user_by_id(user_id: str) -> dict:
    """Look up a user by id.

    Returns the user record, or an error dict if no user has that id.
    """
    user = _USERS.get(str(user_id))
    if user is None:
        return {"error": f"No user found with id={user_id!r}"}
    return user
````
