### `graph.py`

````py
"""LangGraph pipeline with a durable checkpointer and an orders-query step.

Wire your existing nodes in where `intake`/`respond` are — those two are
placeholders standing in for the rest of "our pipeline". The new piece is
the orders_agent -> orders_tools loop: the model decides whether it needs
to query the orders table, calls the `query_orders` tool if so, and the
graph loops back until the model has enough to answer.
"""
from __future__ import annotations

import sqlite3
from typing import Annotated, TypedDict

from langchain_anthropic import ChatAnthropic
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.graph import END, START, StateGraph
from langgraph.graph.message import add_messages
from langgraph.prebuilt import ToolNode, tools_condition

from tools import query_orders

MODEL = "claude-opus-5"
ORDERS_TOOLS = [query_orders]


class PipelineState(TypedDict):
    messages: Annotated[list, add_messages]


def intake(state: PipelineState) -> PipelineState:
    """Placeholder for whatever step(s) precede this in the real pipeline."""
    return state


def orders_agent(state: PipelineState) -> PipelineState:
    """Model turn: answer directly, or call query_orders if it needs data."""
    model = ChatAnthropic(model=MODEL).bind_tools(ORDERS_TOOLS)
    response = model.invoke(state["messages"])
    return {"messages": [response]}


def build_graph(checkpointer: SqliteSaver):
    builder = StateGraph(PipelineState)
    builder.add_node("intake", intake)
    builder.add_node("orders_agent", orders_agent)
    builder.add_node("orders_tools", ToolNode(ORDERS_TOOLS))

    builder.add_edge(START, "intake")
    builder.add_edge("intake", "orders_agent")
    # tools_condition returns "tools" if the model emitted a tool call, else END.
    builder.add_conditional_edges(
        "orders_agent", tools_condition, {"tools": "orders_tools", END: END}
    )
    builder.add_edge("orders_tools", "orders_agent")  # loop back until the model stops calling tools

    return builder.compile(checkpointer=checkpointer)


def get_checkpointer(db_path: str = "checkpoints.sqlite") -> SqliteSaver:
    """SQLite-backed checkpointer: state survives process restarts, unlike MemorySaver."""
    conn = sqlite3.connect(db_path, check_same_thread=False)
    return SqliteSaver(conn)

````

````md
# LangGraph + orders-query step

Minimal scaffold showing a durable-checkpointed LangGraph pipeline with a
step that lets Claude query an `orders` table via tool use.

## Files

- `db_setup.py` — creates a demo SQLite `orders` table. Replace with your
  real database connection in `tools.py` when integrating.
- `tools.py` — `query_orders`, the tool exposed to the model. Enforces
  SELECT-only, single-statement queries and opens the connection read-only.
  Treat this as defense-in-depth, not a replacement for a real read-only
  DB role/user.
- `graph.py` — the graph: `intake -> orders_agent -> (orders_tools loop) -> END`,
  compiled with a `SqliteSaver` checkpointer so state persists across
  process restarts (unlike `MemorySaver`, which is in-memory only).
- `main.py` — example invocation with a `thread_id`.

## How the new step works

`orders_agent` binds `query_orders` to `ChatAnthropic(model="claude-opus-5")`
and calls the model with the running message state. `tools_condition` routes
to `orders_tools` (a `ToolNode`) whenever the model emits a tool call, which
executes the query and appends the result as a `ToolMessage`; the graph loops
back to `orders_agent` so the model can keep querying or produce a final
answer. Because the graph is compiled with a checkpointer, each superstep
(model turn, tool turn) is persisted — if the process crashes mid-loop, a
fresh `graph.invoke(..., config={"configurable": {"thread_id": "..."}})`
with the same `thread_id` resumes from the last completed step instead of
restarting the whole pipeline.

## Wiring into your real pipeline

Replace `intake`/`END` in `graph.py` with your existing upstream/downstream
nodes and drop `orders_agent`/`orders_tools` in wherever the model should be
able to look up order data. The `PipelineState` TypedDict should be extended
to match whatever state your other steps already carry — `messages` is the
minimum needed for the tool-calling loop.

```

export ANTHROPIC_API_KEY=...   # or use `ant auth login`
python main.py
````

### `tools.py`

````py
"""Tool the model uses to query the orders table."""
import re
import sqlite3

from langchain_core.tools import tool

from db_setup import DB_PATH

MAX_ROWS = 200

_SELECT_ONLY = re.compile(r"^\s*SELECT\b", re.IGNORECASE)
_FORBIDDEN_KEYWORDS = re.compile(
    r"\b(INSERT|UPDATE|DELETE|DROP|ALTER|CREATE|ATTACH|DETACH|PRAGMA|REPLACE|TRUNCATE|VACUUM)\b",
    re.IGNORECASE,
)


@tool
def query_orders(sql: str) -> str:
    """Run a read-only SQL SELECT query against the `orders` table and return the results.

    Only SELECT statements are permitted. The connection is opened read-only
    and a single statement is enforced; results are capped at 200 rows, so
    include your own LIMIT/aggregation for large queries.

    Args:
        sql: A single SQL SELECT statement, e.g.
             "SELECT id, customer_id, status, total FROM orders WHERE status = 'pending' LIMIT 20"
    """
    statement = sql.strip().rstrip(";")

    if ";" in statement:
        return "Error: only a single SQL statement is allowed."
    if not _SELECT_ONLY.match(statement):
        return "Error: only SELECT statements are allowed."
    if _FORBIDDEN_KEYWORDS.search(statement):
        return "Error: statement contains a disallowed keyword."
    if "orders" not in statement.lower():
        return "Error: query must reference the orders table."

    conn = None
    try:
        # mode=ro makes SQLite reject any write at the driver level, as a
        # second line of defense behind the keyword check above.
        conn = sqlite3.connect(f"file:{DB_PATH}?mode=ro", uri=True, timeout=5)
        conn.row_factory = sqlite3.Row
        rows = conn.execute(statement).fetchmany(MAX_ROWS)
    except sqlite3.Error as exc:
        return f"Error: {exc}"
    finally:
        if conn is not None:
            conn.close()

    if not rows:
        return "No rows returned."

    columns = rows[0].keys()
    lines = [", ".join(columns)]
    lines += [", ".join(str(row[c]) for c in columns) for row in rows]
    if len(rows) == MAX_ROWS:
        lines.append(f"(truncated to first {MAX_ROWS} rows)")
    return "\n".join(lines)
````
