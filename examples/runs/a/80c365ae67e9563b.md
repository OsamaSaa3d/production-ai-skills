"""
Example: an agent that can search tickets via search_tickets().

Run:
    pip install anthropic requests
    export ANTHROPIC_API_KEY=...      # or `ant auth login`
    export TICKET_BASE_URL=https://tickets.example.com
    export TICKET_API_KEY=...
    python agent_example.py "How many open platform tickets were created this year?"
"""

import sys
import anthropic
from ticket_search_tool import search_tickets

SYSTEM_PROMPT = """You are a support triage assistant with access to the
ticket system's search tool. Its query syntax supports fields (status,
team, assignee, priority, created, updated, title, reporter) combined with
AND / OR / NOT and parentheses, ':' for exact match and >, <, >=, <= for
dates/numbers, e.g. status:open AND team:platform AND created:>2026-01-01.
Build a query from the user's request, call search_tickets, and summarize
the results - don't guess ticket details that weren't returned."""


def main() -> None:
    user_input = sys.argv[1] if len(sys.argv) > 1 else "Show me open platform tickets from this year"

    client = anthropic.Anthropic()
    runner = client.beta.messages.tool_runner(
        model="claude-opus-5",
        max_tokens=16000,
        system=SYSTEM_PROMPT,
        tools=[search_tickets],
        messages=[{"role": "user", "content": user_input}],
    )

    for message in runner:
        for block in message.content:
            if block.type == "text":
                print(block.text)


if __name__ == "__main__":
    main()

### `ticket_search_tool.py`

"""
Ticket-search tool for a Claude agent.

Wraps our ticket system's search endpoint, which accepts Lucene-ish query
strings like:

    status:open AND team:platform AND created:>2026-01-01

The tool forwards whatever query string Claude builds straight to the
endpoint - it does not try to parse or validate the DSL client-side, since
the server is the authority on that grammar and returns a useful 400 on
malformed queries anyway.

Env vars:
    TICKET_BASE_URL  - e.g. "https://tickets.example.com"  (required)
    TICKET_API_KEY   - bearer token for the ticket system    (required)
"""

from __future__ import annotations

import os
import requests
from anthropic import beta_tool

TICKET_BASE_URL = os.environ.get("TICKET_BASE_URL", "").rstrip("/")
TICKET_API_KEY = os.environ.get("TICKET_API_KEY", "")

# Fields Claude actually needs per ticket - keeps tool results small so a
# broad search doesn't blow up the context window.
_SUMMARY_FIELDS = (
    "id", "title", "status", "team", "assignee", "priority", "created", "url",
)


@beta_tool
def search_tickets(query: str, limit: int = 20) -> str:
    """Search tickets using the ticket system's query syntax.

    The query is a boolean expression over fields, combined with AND / OR
    and grouped with parentheses. Examples:

        status:open AND team:platform
        status:open AND team:platform AND created:>2026-01-01
        (status:open OR status:in_progress) AND assignee:jdoe
        priority:>=high AND team:platform AND NOT status:closed
        title:"payment timeout"

    Supported fields include: status, team, assignee, priority, created,
    updated, title, reporter. Comparison operators (>, <, >=, <=) work on
    date and numeric fields (created, updated, priority); other fields use
    exact match with ':'. Quote values containing spaces.

    Args:
        query: The search query string, e.g.
            "status:open AND team:platform AND created:>2026-01-01".
        limit: Max number of tickets to return (default 20, max 100).
    """
    if not TICKET_BASE_URL or not TICKET_API_KEY:
        return (
            "Error: ticket search is not configured - set TICKET_BASE_URL "
            "and TICKET_API_KEY."
        )

    limit = max(1, min(limit, 100))

    try:
        resp = requests.get(
            f"{TICKET_BASE_URL}/api/tickets/search",
            params={"q": query, "limit": limit},
            headers={"Authorization": f"Bearer {TICKET_API_KEY}"},
            timeout=15,
        )
    except requests.exceptions.Timeout:
        return "Error: ticket search timed out. Try a narrower query."
    except requests.exceptions.RequestException as e:
        return f"Error: could not reach ticket system ({e})."

    if resp.status_code == 400:
        # The server rejected the query syntax - surface its message so
        # Claude can fix the query instead of retrying blindly.
        detail = _extract_error_detail(resp)
        return f"Error: invalid query - {detail}"
    if resp.status_code == 401:
        return "Error: ticket search authentication failed (check TICKET_API_KEY)."
    if resp.status_code == 429:
        return "Error: ticket search rate-limited. Wait and retry with a narrower query."
    if resp.status_code >= 500:
        return f"Error: ticket system server error ({resp.status_code}). Try again later."
    if resp.status_code != 200:
        return f"Error: ticket search failed ({resp.status_code}): {resp.text[:500]}"

    try:
        data = resp.json()
    except ValueError:
        return "Error: ticket system returned a non-JSON response."

    tickets = data.get("results", data.get("tickets", []))
    total = data.get("total_count", len(tickets))

    if not tickets:
        return f'No tickets matched query: "{query}"'

    trimmed = [{k: t.get(k) for k in _SUMMARY_FIELDS if k in t} for t in tickets]
    return _format_results(query, total, trimmed)


def _extract_error_detail(resp: requests.Response) -> str:
    try:
        body = resp.json()
        return body.get("message") or body.get("error") or resp.text[:300]
    except ValueError:
        return resp.text[:300]


def _format_results(query: str, total: int, tickets: list[dict]) -> str:
    lines = [f'{total} ticket(s) matched "{query}" (showing {len(tickets)}):']
    for t in tickets:
        lines.append(
            f"- [{t.get('id')}] {t.get('title')} "
            f"(status={t.get('status')}, team={t.get('team')}, "
            f"assignee={t.get('assignee')}, priority={t.get('priority')}, "
            f"created={t.get('created')}) {t.get('url', '')}"
        )
    return "\n".join(lines)
