"""
inventory_agent.py

Lets Claude call your existing get_inventory(sku) / reserve_stock(sku, qty)
functions as tools, using the Anthropic SDK's beta tool runner (it drives
the request -> tool call -> feed-result-back loop for you).

Setup:
    pip install anthropic
    # credentials: ANTHROPIC_API_KEY env var, or `ant auth login`

Swap the import below for wherever your real functions live — nothing
else in this file needs to change as long as the signatures match
get_inventory(sku) -> int and reserve_stock(sku, qty) -> bool.
"""

import anthropic
from anthropic import beta_tool

# TODO: point this at your real module, e.g. `from warehouse.stock import ...`
from inventory_service import get_inventory as _get_inventory, reserve_stock as _reserve_stock

client = anthropic.Anthropic()

MODEL = "claude-opus-5"

SYSTEM_PROMPT = (
    "You help with inventory questions and stock reservations. "
    "Always check inventory with get_inventory before calling reserve_stock. "
    "Never reserve more units than are available. After a successful "
    "reservation, confirm the SKU and quantity back to the user."
)


# Each function below is a thin wrapper around your existing business logic.
# The docstring and type hints are what the SDK uses to generate the tool's
# JSON schema for Claude — keep them accurate. Wrapping (rather than
# decorating your originals) keeps the anthropic SDK dependency out of your
# core inventory module and lets this layer translate exceptions into the
# plain-text tool results Claude expects.


@beta_tool
def get_inventory(sku: str) -> str:
    """Look up how many units of a SKU are currently available to reserve.

    Args:
        sku: The product SKU to look up, e.g. "SKU-100".
    """
    try:
        available = _get_inventory(sku)
    except KeyError:
        return f"Error: unknown SKU '{sku}'."
    return f"{available} unit(s) of '{sku}' available."


@beta_tool
def reserve_stock(sku: str, qty: int) -> str:
    """Reserve a quantity of a SKU against available stock.

    Args:
        sku: The product SKU to reserve, e.g. "SKU-100".
        qty: Number of units to reserve. Must be a positive integer.
    """
    if qty <= 0:
        return f"Error: qty must be a positive integer, got {qty}."
    try:
        reserved = _reserve_stock(sku, qty)
    except KeyError:
        return f"Error: unknown SKU '{sku}'."
    if not reserved:
        return f"Error: insufficient stock to reserve {qty} unit(s) of '{sku}'."
    return f"Reserved {qty} unit(s) of '{sku}'."


def run(user_message: str) -> str:
    """Send one user message through the tool-calling loop and return Claude's final reply."""
    runner = client.beta.messages.tool_runner(
        model=MODEL,
        max_tokens=16000,
        system=SYSTEM_PROMPT,
        tools=[get_inventory, reserve_stock],
        messages=[{"role": "user", "content": user_message}],
    )

    final_text = ""
    for message in runner:
        for block in message.content:
            if block.type == "text":
                final_text = block.text
    return final_text


if __name__ == "__main__":
    print(run("Do we have any SKU-100 in stock? If so, reserve 3 units."))
    print(run("Reserve 10 units of SKU-200."))  # will fail — none available
    print(run("What about a SKU called NOT-REAL?"))

"""
inventory_service.py

Stand-in for your existing inventory functions. This file only exists so
the example in inventory_agent.py is runnable end-to-end — replace it with
your real module and update the import in inventory_agent.py accordingly.

Existing signatures assumed:
    get_inventory(sku: str) -> int
        Returns units currently available to reserve. Raises KeyError for
        an unknown SKU.
    reserve_stock(sku: str, qty: int) -> bool
        Reserves `qty` units of `sku`. Returns True on success, False if
        there isn't enough available stock. Raises KeyError for an
        unknown SKU.
"""

from dataclasses import dataclass
from threading import Lock


@dataclass
class _Stock:
    available: int
    reserved: int = 0


_LOCK = Lock()
_STOCK: dict[str, _Stock] = {
    "SKU-100": _Stock(available=42),
    "SKU-200": _Stock(available=0),
    "SKU-300": _Stock(available=5),
}


def get_inventory(sku: str) -> int:
    with _LOCK:
        return _STOCK[sku].available  # raises KeyError if sku is unknown


def reserve_stock(sku: str, qty: int) -> bool:
    with _LOCK:
        stock = _STOCK[sku]  # raises KeyError if sku is unknown
        if qty <= 0 or stock.available < qty:
            return False
        stock.available -= qty
        stock.reserved += qty
        return True
