### `agent.py`

````py
"""
CRM agent - exposes the 14 CRM REST endpoints in crm_tools.py to Claude as
tool calls, using the Anthropic Python SDK's beta Tool Runner.

Usage:
    export ANTHROPIC_API_KEY=...      # or `ant auth login`
    export CRM_BASE_URL=https://your-crm.example.com/api/v1
    export CRM_API_KEY=...
    python agent.py "Find the contact for Jane Doe and create a follow-up task"

With no argument, starts an interactive REPL.
"""

from __future__ import annotations

import sys

import anthropic

from crm_tools import ALL_TOOLS

MODEL = "claude-opus-5"

SYSTEM_PROMPT = (
    "You are a CRM assistant. Use the provided tools to look up and manage "
    "contacts, companies, deals, and tasks. Prefer listing/searching before "
    "creating or updating a record, and confirm destructive actions (like "
    "delete_contact) by summarizing what will happen before you call the tool."
)


def run_turn(client: anthropic.Anthropic, messages: list) -> str:
    """Run one user turn to completion (including any tool calls) and return
    Claude's final text reply."""
    runner = client.beta.messages.tool_runner(
        model=MODEL,
        max_tokens=16000,
        system=SYSTEM_PROMPT,
        tools=ALL_TOOLS,
        messages=messages,
        thinking={"type": "adaptive"},
        output_config={"effort": "high"},
    )

    final_message = None
    for message in runner:
        final_message = message

    if final_message is None:
        return "(no response)"

    return next(
        (b.text for b in final_message.content if b.type == "text"),
        "(no text response)",
    )


def main() -> None:
    client = anthropic.Anthropic()
    messages: list = []

    if len(sys.argv) > 1:
        user_input = " ".join(sys.argv[1:])
        messages.append({"role": "user", "content": user_input})
        print(run_turn(client, messages))
        return

    print("CRM agent ready. Type a request, or 'quit' to exit.")
    while True:
        try:
            user_input = input("\nyou> ").strip()
        except (EOFError, KeyboardInterrupt):
            break
        if user_input.lower() in {"quit", "exit"}:
            break
        if not user_input:
            continue
        messages.append({"role": "user", "content": user_input})
        reply = run_turn(client, messages)
        print(f"\nagent> {reply}")
        messages.append({"role": "assistant", "content": reply})


if __name__ == "__main__":
    main()

````

### `crm_client.py`

````py
"""
Thin REST client for the CRM API.

Configure via environment variables:
    CRM_BASE_URL - base URL of the CRM API, e.g. https://crm.internal.example.com/api/v1
    CRM_API_KEY  - bearer token sent as `Authorization: Bearer <key>`

This module knows nothing about Claude - it's a plain HTTP wrapper that the
tool functions in crm_tools.py call into.
"""

from __future__ import annotations

import os
from typing import Any

import requests

CRM_BASE_URL = os.environ.get("CRM_BASE_URL", "https://api.example-crm.com/v1")
CRM_API_KEY = os.environ.get("CRM_API_KEY")

DEFAULT_TIMEOUT = 15  # seconds


class CRMAPIError(Exception):
    """Raised when the CRM REST API returns a non-2xx response."""

    def __init__(self, method: str, path: str, status_code: int, body: str):
        self.method = method
        self.path = path
        self.status_code = status_code
        self.body = body
        super().__init__(f"{method} {path} -> HTTP {status_code}: {body}")


def _headers() -> dict:
    headers = {"Content-Type": "application/json"}
    if CRM_API_KEY:
        headers["Authorization"] = f"Bearer {CRM_API_KEY}"
    return headers


def request(
    method: str,
    path: str,
    params: dict | None = None,
    json_body: dict | None = None,
    timeout: int = DEFAULT_TIMEOUT,
) -> Any:
    """Issue one HTTP request against the CRM API and return the parsed JSON body.

    Raises CRMAPIError on any non-2xx response so the calling tool function
    surfaces a clear error back to Claude instead of a stack trace.
    """
    url = f"{CRM_BASE_URL.rstrip('/')}/{path.lstrip('/')}"
    resp = requests.request(
        method,
        url,
        headers=_headers(),
        params=params,
        json=json_body,
        timeout=timeout,
    )
    if not resp.ok:
        raise CRMAPIError(method, path, resp.status_code, resp.text[:2000])
    if resp.status_code == 204 or not resp.content:
        return {}
    return resp.json()

````

"""
Agent tool definitions for the CRM REST API.

ASSUMPTION: the actual 14 endpoints weren't specified, so this file wires up
14 typical CRM endpoints (Contacts / Companies / Deals / Tasks CRUD) as a
placeholder. Each function's body is a one-line `crm.request(...)` call, so
retargeting a function to your real path/method is a small, local edit -
the docstring (which becomes the tool description Claude sees) and the
type hints (which become the input schema) usually need updating too.

Each function is decorated with @beta_tool so the Anthropic Python SDK's
tool runner can call it directly - see agent.py.
"""

from __future__ import annotations

from anthropic import beta_tool

import crm_client as crm

# ---------------------------------------------------------------------------
# Contacts
# ---------------------------------------------------------------------------


@beta_tool
def list_contacts(query: str = "", limit: int = 20) -> dict:
    """List CRM contacts, optionally filtered by a search query.

    Args:
        query: Free-text search over name/email. Empty string returns all contacts.
        limit: Maximum number of contacts to return (1-100).
    """
    return crm.request("GET", "/contacts", params={"q": query, "limit": limit})


@beta_tool
def get_contact(contact_id: str) -> dict:
    """Get a single CRM contact by ID.

    Args:
        contact_id: The contact's unique identifier.
    """
    return crm.request("GET", f"/contacts/{contact_id}")


@beta_tool
def create_contact(name: str, email: str, phone: str = "", company_id: str = "") -> dict:
    """Create a new CRM contact.

    Args:
        name: Full name of the contact.
        email: Email address of the contact.
        phone: Phone number, if known.
        company_id: ID of the company to associate this contact with, if known.
    """
    body = {"name": name, "email": email, "phone": phone, "company_id": company_id}
    return crm.request("POST", "/contacts", json_body=body)


@beta_tool
def update_contact(contact_id: str, name: str = "", email: str = "", phone: str = "") -> dict:
    """Update fields on an existing CRM contact. Only non-empty fields are sent.

    Args:
        contact_id: The contact's unique identifier.
        name: New full name, if changing.
        email: New email address, if changing.
        phone: New phone number, if changing.
    """
    body = {k: v for k, v in {"name": name, "email": email, "phone": phone}.items() if v}
    return crm.request("PATCH", f"/contacts/{contact_id}", json_body=body)


@beta_tool
def delete_contact(contact_id: str) -> dict:
    """Delete a CRM contact by ID.

    Args:
        contact_id: The contact's unique identifier.
    """
    return crm.request("DELETE", f"/contacts/{contact_id}")

    """Create a new CRM company record.

    Args:
        name: Company name.
        domain: Company website domain, if known.
        industry: Industry category, if known.

    body = {"name": name, "domain": domain, "industry": industry}
    return crm.request("POST", "/companies", json_body=body)

def list_deals(stage: str = "", limit: int = 20) -> dict:

    return crm.request("GET", f"/deals/{deal_id}")

def create_deal(name: str, contact_id: str, amount: float, stage: str = "open") -> dict:

        name: Short name/title of the deal.
        contact_id: ID of the primary contact for this deal.
        amount: Deal value in the CRM's base currency.
        stage: Initial pipeline stage, defaults to "open".

    body = {"name": name, "contact_id": contact_id, "amount": amount, "stage": stage}
    return crm.request("POST", "/deals", json_body=body)


@beta_tool
def update_deal_stage(deal_id: str, stage: str) -> dict:

# Tasks

@beta_tool
def list_tasks(assignee: str = "", completed: bool | None = None) -> dict:
    """List CRM tasks, optionally filtered by assignee and completion status.

        related_contact_id: ID of a contact this task relates to, if any.

    body = {"title": title, "due_date": due_date, "related_contact_id": related_contact_id}

## How it's wired

- `crm_client.py` - plain HTTP client (`requests`) for the CRM API. Reads
  `CRM_BASE_URL` and `CRM_API_KEY` from the environment. Raises `CRMAPIError`
  on non-2xx responses, which the tool runner surfaces back to Claude as a
  tool error instead of crashing the agent.
- `crm_tools.py` - one `@beta_tool`-decorated function per CRM endpoint. The
  docstring becomes the tool description Claude sees; the type hints become
  the JSON input schema. `ALL_TOOLS` is the list passed to the runner.
- `agent.py` - creates the Claude client, runs `client.beta.messages.tool_runner(...)`
  with `ALL_TOOLS`, and loops until Claude has no more tool calls to make.
  Model: `claude-opus-5` with adaptive thinking, effort `high`.
