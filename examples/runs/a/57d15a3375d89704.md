"""Summarize a meeting transcript into a short, manager-skimmable paragraph."""

import anthropic

MODEL = "claude-opus-5"

SYSTEM_PROMPT = (
    "You summarize meeting transcripts for a busy manager who was not present. "
    "Write ONE short paragraph (4-6 sentences, no bullet points, no headers) that "
    "covers: what the meeting was about, the key decisions made, any action items "
    "with owners if named, and open questions or blockers. Be concrete and skimmable "
    "in under 20 seconds. Do not include preamble like 'Here is a summary' - just "
    "write the paragraph."
)


def summarize_meeting_transcript(
    transcript: str,
    client: anthropic.Anthropic | None = None,
) -> str:
    """Summarize a meeting transcript into a short paragraph for a manager.

    Args:
        transcript: The raw meeting transcript text (speaker-labeled or plain).
        client: Optional pre-configured Anthropic client (useful for reuse/testing).
            If omitted, a client is created from the environment
            (ANTHROPIC_API_KEY / ANTHROPIC_AUTH_TOKEN / an `ant auth login` profile).

    Returns:
        A short paragraph summarizing the meeting.

    Raises:
        anthropic.APIError: on unrecoverable API failures (auth, bad request, etc.)
            after the SDK's built-in retries for transient errors are exhausted.
    """
    if not transcript or not transcript.strip():
        raise ValueError("transcript must be a non-empty string")

    client = client or anthropic.Anthropic()

    try:
        response = client.messages.create(
            model=MODEL,
            max_tokens=1024,
            system=SYSTEM_PROMPT,
            output_config={"effort": "low"},
            messages=[
                {
                    "role": "user",
                    "content": f"Meeting transcript:\n\n{transcript}",
                }
            ],
        )
    except anthropic.BadRequestError as e:
        raise ValueError(f"Bad request to Claude API: {e.message}") from e
    except anthropic.AuthenticationError as e:
        raise RuntimeError("Invalid or missing Anthropic API credentials") from e
    except anthropic.RateLimitError as e:
        retry_after = e.response.headers.get("retry-after", "unknown")
        raise RuntimeError(f"Rate limited by Claude API; retry after {retry_after}s") from e
    except anthropic.APIConnectionError as e:
        raise RuntimeError("Network error reaching the Claude API") from e
    except anthropic.APIStatusError as e:
        raise RuntimeError(f"Claude API error ({e.status_code}): {e.message}") from e

    if response.stop_reason == "refusal":
        raise RuntimeError("Claude declined to summarize this transcript")

    summary = next((b.text for b in response.content if b.type == "text"), "")
    if not summary:
        raise RuntimeError("Claude API returned no text content")

    return summary.strip()
