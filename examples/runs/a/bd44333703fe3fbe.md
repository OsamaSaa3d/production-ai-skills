#!/usr/bin/env python3
"""
autofix.py - take a failing pytest test and iterate with Claude until it passes.

Usage:
    python autofix.py <pytest-target> [options]

Examples:
    python autofix.py demo/test_mathutils.py --repo demo
    python autofix.py test_mathutils.py::test_average_basic --repo demo
    python autofix.py demo/test_mathutils.py --repo demo --dry-run

How it works:
    1. Run pytest against the target. If it already passes, stop.
    2. On failure, collect the traceback and the source of every file it
       references (skipping stdlib/site-packages), plus the test file.
    3. Send that to Claude and ask for the minimal fix, as whole-file
       replacements in a strict, parseable format.
    4. Write the returned files back to disk (never the test file, unless
       --allow-test-edits is passed) and re-run pytest.
    5. Repeat until the test passes or --max-iters is hit.

Requires the `anthropic` package and credentials (ANTHROPIC_API_KEY, or an
`ant auth login` profile) for anything beyond --dry-run.
"""
from __future__ import annotations

import argparse
import ast
import difflib
import re
import subprocess
import sys
from pathlib import Path

MODEL = "claude-opus-5"
MAX_OUTPUT_CHARS_IN_PROMPT = 8000
MAX_FILE_BYTES = 60_000

FILE_BLOCK_RE = re.compile(
    r"===FILE:\s*(?P<path>[^\n=]+?)\s*===\r?\n(?P<body>.*?)\r?\n===END===",
    re.DOTALL,
)
TRACEBACK_FILE_RE = re.compile(r"([A-Za-z0-9_./\\\-]+\.py):(\d+)")


def run_pytest(repo_root: Path, test_cmd: list[str]) -> tuple[bool, str]:
    proc = subprocess.run(
        test_cmd, cwd=repo_root, capture_output=True, text=True
    )
    output = proc.stdout + "\n" + proc.stderr
    return proc.returncode == 0, output


def is_test_file(path: Path) -> bool:
    name = path.name
    return name.startswith("test_") or name.endswith("_test.py")


def gather_referenced_files(repo_root: Path, output: str) -> list[Path]:
    repo_root = repo_root.resolve()
    found: set[Path] = set()
    for match in TRACEBACK_FILE_RE.finditer(output):
        raw = match.group(1)
        candidate = (repo_root / raw).resolve()
        if not candidate.is_file():
            candidate = Path(raw).resolve()
        if not candidate.is_file():
            continue
        try:
            candidate.relative_to(repo_root)
        except ValueError:
            continue  # outside the repo (stdlib, site-packages, etc.)
        found.add(candidate)
    return sorted(found)


def resolve_module_path(repo_root: Path, module: str) -> Path | None:
    parts = module.split(".")
    base = repo_root.joinpath(*parts)
    for candidate in (base.with_suffix(".py"), base / "__init__.py"):
        if candidate.is_file():
            return candidate.resolve()
    return None


def gather_imported_files(repo_root: Path, test_file: Path) -> list[Path]:
    """Resolve modules the test file imports to in-repo source files.

    Traceback frames alone miss any function that fails via a wrong return
    value rather than a raised exception, so we also look at what the test
    actually imports.
    """
    try:
        tree = ast.parse(test_file.read_text(encoding="utf-8"))
    except (SyntaxError, OSError):
        return []

    modules: set[str] = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for alias in node.names:
                modules.add(alias.name)
        elif isinstance(node, ast.ImportFrom):
            if node.module and node.level == 0:
                modules.add(node.module)
                for alias in node.names:
                    modules.add(f"{node.module}.{alias.name}")

    resolved: set[Path] = set()
    test_file_resolved = test_file.resolve()
    for module in modules:
        path = resolve_module_path(repo_root, module)
        if path and path != test_file_resolved:
            resolved.add(path)
    return sorted(resolved)


def build_prompt(
    repo_root: Path,
    target: str,
    test_output: str,
    files: list[Path],
    allow_test_edits: bool,
) -> str:
    truncated_output = test_output[-MAX_OUTPUT_CHARS_IN_PROMPT:]
    parts = [
        f"Failing pytest target: {target}",
        "",
        "=== TEST OUTPUT (most recent run, tail) ===",
        truncated_output,
        "",
        "=== CURRENT FILE CONTENTS ===",
    ]
    for f in files:
        rel = f.relative_to(repo_root.resolve())
        try:
            content = f.read_text(encoding="utf-8")
        except UnicodeDecodeError:
            continue
        if len(content.encode("utf-8")) > MAX_FILE_BYTES:
            content = content[: MAX_FILE_BYTES] + "\n# ... truncated ..."
        role = "TEST FILE" if is_test_file(f) else "SOURCE FILE"
        parts.append(f"--- {role}: {rel} ---")
        parts.append(content)
        parts.append("")
    if not allow_test_edits:
        parts.append(
            "Note: test files are shown for context only; you may not modify them."
        )
    return "\n".join(parts)


def system_prompt(allow_test_edits: bool) -> str:
    test_rule = (
        "You may edit the test file if it is itself wrong, but prefer fixing "
        "the implementation."
        if allow_test_edits
        else "Do not modify any test file (files named test_*.py or *_test.py). "
        "Fix the implementation instead."
    )
    return (
        "You are an expert software engineer fixing a failing test in a real "
        "codebase.\n\n"
        "Rules:\n"
        f"- {test_rule}\n"
        "- Make the minimal change needed to make the test pass for the right "
        "reason. Do not special-case the test's specific input values.\n"
        "- Preserve unrelated code exactly as-is.\n"
        "- Return ONLY whole-file replacements for every file you change, "
        "using exactly this format, with no other text before, between, or "
        "after the blocks:\n\n"
        "===FILE: relative/path/to/file.py===\n"
        "<complete new file content>\n"
        "===END===\n\n"
        "- Use the same relative path shown in the file's header above.\n"
        "- If you cannot determine a fix, return no FILE blocks and explain "
        "why in a single line prefixed with 'REASON:'."
    )


def call_claude(system: str, user_content: str):
    import anthropic

    client = anthropic.Anthropic()
    try:
        with client.messages.stream(
            model=MODEL,
            max_tokens=16000,
            system=system,
            thinking={"type": "adaptive"},
            output_config={"effort": "high"},
            messages=[{"role": "user", "content": user_content}],
        ) as stream:
            message = stream.get_final_message()
    except anthropic.AuthenticationError:
        print("ERROR: Anthropic authentication failed. Set ANTHROPIC_API_KEY "
              "or run `ant auth login`.", file=sys.stderr)
        raise
    except anthropic.RateLimitError as e:
        retry_after = e.response.headers.get("retry-after", "unknown")
        print(f"ERROR: rate limited, retry after {retry_after}s.", file=sys.stderr)
        raise
    except anthropic.APIStatusError as e:
        print(f"ERROR: API error {e.status_code}: {e.message}", file=sys.stderr)
        raise
    except anthropic.APIConnectionError:
        print("ERROR: network error reaching the Anthropic API.", file=sys.stderr)
        raise
    except TypeError as e:
        if "Could not resolve authentication method" in str(e):
            print(
                "ERROR: no Anthropic credentials found. Set ANTHROPIC_API_KEY "
                "or run `ant auth login`.",
                file=sys.stderr,
            )
        else:
            print(f"ERROR: {e}", file=sys.stderr)
        raise

    text = "".join(b.text for b in message.content if b.type == "text")
    return text


def apply_fix(
    repo_root: Path, response_text: str, allow_test_edits: bool
) -> list[Path]:
    changed: list[Path] = []
    matches = list(FILE_BLOCK_RE.finditer(response_text))
    if not matches:
        reason = next(
            (line for line in response_text.splitlines() if line.startswith("REASON:")),
            None,
        )
        if reason:
            print(f"Claude did not propose a fix: {reason}")
        else:
            print("Claude's response did not contain any FILE blocks:")
            print(response_text[:2000])
        return changed

    for m in matches:
        rel_path = m.group("path").strip()
        new_content = m.group("body")
        target_path = (repo_root / rel_path).resolve()
        try:
            target_path.relative_to(repo_root.resolve())
        except ValueError:
            print(f"Skipping out-of-repo path from model: {rel_path}")
            continue
        if not allow_test_edits and is_test_file(target_path):
            print(f"Skipping edit to protected test file: {rel_path}")
            continue

        old_content = target_path.read_text(encoding="utf-8") if target_path.exists() else ""
        if old_content == new_content:
            continue

        diff = difflib.unified_diff(
            old_content.splitlines(keepends=True),
            new_content.splitlines(keepends=True) if new_content.endswith("\n")
            else (new_content + "\n").splitlines(keepends=True),
            fromfile=f"a/{rel_path}",
            tofile=f"b/{rel_path}",
        )
        diff_text = "".join(diff)
        print(f"\n--- Proposed change to {rel_path} ---")
        print(diff_text if diff_text else "(no textual diff)")

        target_path.parent.mkdir(parents=True, exist_ok=True)
        target_path.write_text(new_content + ("" if new_content.endswith("\n") else "\n"), encoding="utf-8")
        changed.append(target_path)

    return changed


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("target", help="pytest target, e.g. test_x.py or test_x.py::test_y")
    parser.add_argument("--repo", default=".", help="repo root to run pytest from (default: .)")
    parser.add_argument("--max-iters", type=int, default=5, help="max fix attempts (default: 5)")
    parser.add_argument(
        "--allow-test-edits",
        action="store_true",
        help="allow Claude to modify test files too (off by default)",
    )
    parser.add_argument(
        "--dry-run",
        action="store_true",
        help="run pytest and print the prompt that would be sent, without calling the API",
    )
    args = parser.parse_args()

    repo_root = Path(args.repo).resolve()
    if not repo_root.is_dir():
        print(f"ERROR: repo root {repo_root} does not exist", file=sys.stderr)
        return 2

    test_cmd = [sys.executable, "-m", "pytest", args.target, "-q", "--tb=long", "--no-header"]

    for attempt in range(1, args.max_iters + 1):
        print(f"\n=== Attempt {attempt}/{args.max_iters}: running `pytest {args.target}` in {repo_root} ===")
        passed, output = run_pytest(repo_root, test_cmd)
        print(output)

        if passed:
            print(f"PASSED after {attempt - 1} fix iteration(s).")
            return 0

        files = set(gather_referenced_files(repo_root, output))

        target_file = repo_root / args.target.split("::", 1)[0]
        if target_file.is_file():
            files.add(target_file.resolve())
            files.update(gather_imported_files(repo_root, target_file))

        files = sorted(files)
        if not files:
            print("Could not identify any in-repo files from the traceback or imports; giving up.")
            return 1

        prompt = build_prompt(repo_root, args.target, output, files, args.allow_test_edits)
        system = system_prompt(args.allow_test_edits)

        if args.dry_run:
            print("\n--- DRY RUN: system prompt ---")
            print(system)
            print("\n--- DRY RUN: user prompt ---")
            print(prompt)
            print("\n(dry run: not calling the API, not applying any fix)")
            return 0

        try:
            response_text = call_claude(system, prompt)
        except Exception:
            return 1

        changed = apply_fix(repo_root, response_text, args.allow_test_edits)
        if not changed:
            print("No files were changed this iteration; stopping.")
            return 1

    print(f"\nGave up after {args.max_iters} attempts without a passing test.")
    return 1


if __name__ == "__main__":
    sys.exit(main())

# autofix.py

A small tool that takes a failing pytest test and iterates with Claude until
it passes.

## How it works

1. Runs pytest against the target test.
2. On failure, collects:
   - the failure output/traceback
   - every in-repo file mentioned in the traceback
   - every in-repo file the test file imports (this catches bugs where a
     function returns the wrong value without raising, so no traceback frame
     points at it)
3. Sends the test output + file contents to Claude (`claude-opus-5`), asking
   for whole-file replacements in a strict `===FILE: path===...===END===`
   format.
4. Applies the returned files (never a `test_*.py`/`*_test.py` file, unless
   `--allow-test-edits` is passed), printing a unified diff of each change.
5. Re-runs pytest. Repeats up to `--max-iters` times (default 5).

## Setup

```bash
pip install -r requirements.txt
```

Requires Anthropic API credentials: set `ANTHROPIC_API_KEY`, or use
`ant auth login`.

## Usage

```bash
python autofix.py <pytest-target> [--repo PATH] [--max-iters N] [--allow-test-edits] [--dry-run]
```

- `<pytest-target>` - anything pytest accepts: a file (`test_x.py`), a node id
  (`test_x.py::test_y`), or a directory.
- `--repo` - repo root to run pytest from (default: current directory).
- `--dry-run` - run pytest and print exactly the prompt that would be sent to
  Claude, without calling the API or changing any files. Useful for checking
  which files would be gathered before spending a request.

### Demo

`demo/mathutils.py` ships with a deliberate bug (`average()` uses integer
division), and `demo/test_mathutils.py` has a test that fails because of it:

```bash
python autofix.py test_mathutils.py --repo demo
```

Try `--dry-run` first (no API key needed) to see the exact prompt:

```bash
python autofix.py test_mathutils.py --repo demo --dry-run
```

## Notes / limitations

- One test target at a time; it doesn't triage a whole suite of unrelated
  failures.
- File discovery is traceback + import based. A bug reached only through
  deep indirection (e.g. via a plugin system or dynamic import) may not be
  picked up automatically - in that case, `--dry-run` will show an empty or
  incomplete file list.
- No git integration - it edits files in place. Run it inside a git repo (or
  otherwise have a backup) if you want to easily revert an attempt.
- This was validated end-to-end except for the live Claude call itself
  (pytest failure detection, file discovery, prompt construction, and fix
  application were all exercised directly - the latter with a synthetic
  model response). No Anthropic API credentials were available in the
  environment this was built in.
