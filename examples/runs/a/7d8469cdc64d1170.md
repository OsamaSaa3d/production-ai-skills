#!/usr/bin/env node
/**
 * autofix.js — point it at a repo and a test command; it runs the tests,
 * and if they fail, hands the failure output to Claude Code (headless)
 * to patch the source, then re-runs the tests. Repeats until green or
 * out of attempts.
 *
 * Usage:
 *   node autofix.js --cwd <repoDir> --cmd "<test command>" [--max-attempts 5] [--model sonnet]
 *
 * Example:
 *   node autofix.js --cwd ./demo-repo --cmd "npm test"
 */

const { spawnSync } = require('node:child_process');
const path = require('node:path');

function parseArgs(argv) {
  const args = { cwd: process.cwd(), cmd: 'npm test', maxAttempts: 5, model: null };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a === '--cwd') args.cwd = argv[++i];
    else if (a === '--cmd') args.cmd = argv[++i];
    else if (a === '--max-attempts') args.maxAttempts = parseInt(argv[++i], 10);
    else if (a === '--model') args.model = argv[++i];
    else throw new Error(`Unknown argument: ${a}`);
  }
  return args;
}

function runTests(cwd, cmd) {
  const result = spawnSync(cmd, { cwd, shell: true, encoding: 'utf8' });
  const output = `${result.stdout || ''}${result.stderr || ''}`;
  return { passed: result.status === 0, output, status: result.status };
}

function truncate(text, maxChars) {
  if (text.length <= maxChars) return text;
  return `...(truncated)...\n${text.slice(-maxChars)}`;
}

function buildPrompt(cmd, testOutput, attempt) {
  return [
    `A test suite in this repository is failing. The test command is:`,
    '',
    `    ${cmd}`,
    '',
    `Here is the output from running it (attempt ${attempt}):`,
    '',
    '```',
    truncate(testOutput, 6000),
    '```',
    '',
    'Fix the underlying source code so the failing test(s) pass.',
    'Do not weaken, skip, or delete any test to make it pass — fix the implementation.',
    'Only change the test file if it is provably wrong (e.g. a typo in the expected value',
    'that contradicts the stated intent of the test).',
    'Make the smallest change that fixes the root cause.',
    `After editing, run "${cmd}" yourself to confirm it passes before finishing.`,
  ].join('\n');
}

function invokeClaude({ cwd, cmd, prompt, model }) {
  // Prompt is passed via stdin (not argv) and the Bash allow-pattern is
  // pre-quoted: with { shell: true } on Windows, node joins argv elements
  // with plain spaces before handing them to cmd.exe, so any element that
  // itself contains a space (the prompt, or "Bash(<cmd> *)") must be quoted
  // by us or it gets split into multiple tokens.
  const args = [
    '-p',
    '--permission-mode', 'acceptEdits',
    '--allowedTools', 'Read', 'Edit', 'Write', 'Glob', 'Grep', `"Bash(${cmd} *)"`,
    '--output-format', 'text',
  ];
  if (model) args.push('--model', model);

  const result = spawnSync('claude', args, {
    cwd,
    shell: true,
    encoding: 'utf8',
    input: prompt,
    maxBuffer: 1024 * 1024 * 50,
  });
  return {
    ok: result.status === 0,
    stdout: result.stdout || '',
    stderr: result.stderr || '',
  };
}

function main() {
  const args = parseArgs(process.argv.slice(2));
  const cwd = path.resolve(args.cwd);

  console.log(`[autofix] repo: ${cwd}`);
  console.log(`[autofix] test command: ${args.cmd}`);

  let { passed, output } = runTests(cwd, args.cmd);
  if (passed) {
    console.log('[autofix] tests already pass — nothing to do.');
    process.exit(0);
  }

  for (let attempt = 1; attempt <= args.maxAttempts; attempt++) {
    console.log(`\n[autofix] attempt ${attempt}/${args.maxAttempts}: tests failing, asking Claude Code to fix it...`);
    const prompt = buildPrompt(args.cmd, output, attempt);
    const claudeResult = invokeClaude({ cwd, cmd: args.cmd, prompt, model: args.model });

    if (claudeResult.stdout.trim()) console.log(`[claude] ${claudeResult.stdout.trim()}`);
    if (!claudeResult.ok) {
      console.error('[autofix] claude invocation failed:');
      console.error(claudeResult.stderr || claudeResult.stdout);
    } else {
      console.log('[autofix] claude finished this attempt. Re-running tests...');
    }

    const retry = runTests(cwd, args.cmd);
    passed = retry.passed;
    output = retry.output;

    if (passed) {
      console.log(`\n[autofix] tests pass after ${attempt} attempt(s). Done.`);
      process.exit(0);
    }
    console.log(`[autofix] still failing after attempt ${attempt}.`);
  }

  console.error(`\n[autofix] gave up after ${args.maxAttempts} attempts. Tests still failing:\n`);
  console.error(output);
  process.exit(1);
}

main();

````

function capitalize(str) {
  if (!str) return str;
  return str.charAt(0).toUpperCase() + str.slice(1);
}

function slugify(str) {
  return str.trim().toLowerCase().replace(/\s+/g, '-');
}

function truncate(str, maxLen) {
  if (str.length <= maxLen) return str;
  return str.slice(0, maxLen) + '...';
}

module.exports = { capitalize, slugify, truncate };

### `README.md`

````md
# autofix

Points at a repo and a test command, runs it, and if it fails, hands the
failure output to headless Claude Code to patch the source. Repeats until
the tests pass or attempts run out.

## Usage

```
node autofix.js --cwd <repoDir> --cmd "<test command>" [--max-attempts 5] [--model sonnet]
```

- `--cwd` — repo directory to run in (default: current directory)
- `--cmd` — the test command to run, e.g. `"npm test"`, `"pytest tests/"` (default: `"npm test"`)
- `--max-attempts` — how many fix/retest cycles to try before giving up (default: 5)
- `--model` — optional model override passed through to `claude`

Exits `0` once the test command exits `0`. Exits `1` if it's still failing
after `--max-attempts` tries, printing the last failure output.

## How it works

1. Runs the test command. If it already passes, exits immediately.
2. On failure, sends Claude Code (`claude -p`) the test output with
   instructions to fix the implementation — not weaken or delete the test —
   and to verify by re-running the test command itself.
3. Claude Code runs with `--permission-mode acceptEdits` and is restricted to
   `Read`, `Edit`, `Write`, `Glob`, `Grep`, and running the given test command
   via `Bash` — nothing else.
4. Re-runs the test command. If it passes, done; otherwise loops, feeding the
   new failure output back in.

```
