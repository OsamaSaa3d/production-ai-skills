Written to `support_bot.py` — it compiles cleanly. Summary of what it does:

**Tools** (`TOOLS`, `strict: true` so inputs always validate against the schema):
- `lookup_order(order_id)` — returns status, total, currency, items, and `refunded_amount`
- `issue_refund(order_id, amount, reason)` — refunds against the order

**Backend** — a mock in-memory `_ORDERS` dict standing in for your real order/payments system, plus `_approve_refund()`, a stub gate you should replace with real policy logic (amount ceiling, fraud review, human approval) before wiring this to a real payments system — refunds are hard to reverse, so I gave it its own seam rather than baking a check into `_issue_refund`.

**The loop** (`run_turn`):
1. Call `client.messages.create(...)`, always append `response.content` to `messages`
2. If `stop_reason == "tool_use"`: run every `tool_use` block via `execute_tool`, collect all results into **one** `tool_result` user message (required for parallel tool calls), append it, loop back to call Claude again
3. If `stop_reason == "pause_turn"`: just resend — Claude continues where it left off
4. If `stop_reason == "max_tokens"`: raise — silently truncated output shouldn't be presented as a finished answer
5. Otherwise (`end_turn`): return — that turn is done

The `messages` list is returned to the caller after each turn and passed back in on the next `run_turn` call, so state (including tool history) persists across turns — the `__main__` block demonstrates this with two turns: request a refund, then confirm it. Because the API is stateless, everything the model needs to remember must live in that list.

"""
Support bot: looks up an order and issues a refund, using the Claude Messages
API with a manual tool_use / tool_result loop across multiple turns.

Requires: pip install anthropic
Credentials: resolved automatically from ANTHROPIC_API_KEY / ANTHROPIC_AUTH_TOKEN /
an `ant auth login` profile - see anthropic.Anthropic() below.
"""

import json
import anthropic

client = anthropic.Anthropic()
MODEL = "claude-opus-5"

SYSTEM_PROMPT = """You are a support agent for an online store. You can look up
orders and issue refunds.

Rules:
- Always call lookup_order before issue_refund - never guess an order's total
  or refund history.
- Never refund more than the order's remaining refundable balance
  (total - refunded_amount, both returned by lookup_order).
- If the customer doesn't give a reason for the refund, ask for one before
  calling issue_refund.
- If lookup_order or issue_refund returns an error, explain it to the
  customer in plain language and suggest a next step - don't retry blindly.
- Be concise and confirm what happened after a refund succeeds (amount,
  order id, refund id)."""

TOOLS = [
    {
        "name": "lookup_order",
        "description": (
            "Look up an order by its order ID. Returns status, items, total, "
            "currency, and how much has already been refunded."
        ),
        "strict": True,
        "input_schema": {
            "type": "object",
            "properties": {
                "order_id": {
                    "type": "string",
                    "description": "The order ID, e.g. ORD-1001.",
                }
            },
            "required": ["order_id"],
            "additionalProperties": False,
        },
    },
    {
        "name": "issue_refund",
        "description": (
            "Issue a refund against an order. Only call this after "
            "lookup_order confirms the order exists and the amount does not "
            "exceed the remaining refundable balance."
        ),
        "strict": True,
        "input_schema": {
            "type": "object",
            "properties": {
                "order_id": {"type": "string", "description": "The order ID to refund."},
                "amount": {
                    "type": "number",
                    "description": "Refund amount in the order's currency.",
                },
                "reason": {
                    "type": "string",
                    "description": "Short reason for the refund, e.g. 'damaged in transit'.",
                },
            },
            "required": ["order_id", "amount", "reason"],
            "additionalProperties": False,
        },
    },
]

# --- Mock order backend -----------------------------------------------
# Swap this for real calls to your order/payments system.

_ORDERS = {
    "ORD-1001": {
        "order_id": "ORD-1001",
        "customer_email": "alice@example.com",
        "status": "delivered",
        "currency": "USD",
        "total": 49.99,
        "refunded_amount": 0.0,
        "items": [{"sku": "SKU-1", "name": "Wireless Mouse", "qty": 1, "price": 49.99}],
    },
    "ORD-1002": {
        "order_id": "ORD-1002",
        "customer_email": "bob@example.com",
        "status": "delivered",
        "currency": "USD",
        "total": 129.50,
        "refunded_amount": 129.50,
        "items": [{"sku": "SKU-2", "name": "Desk Lamp", "qty": 1, "price": 129.50}],
    },
}


def _lookup_order(order_id: str) -> dict:
    order = _ORDERS.get(order_id)
    if order is None:
        return {"error": f"No order found with id {order_id!r}."}
    return order


def _approve_refund(order: dict, amount: float, reason: str) -> bool:
    """Refunds are hard to reverse - gate them here.

    Replace this stub with your real policy check (amount ceiling, fraud
    review, human-in-the-loop queue, etc.) before wiring this up to a real
    payments system.
    """
    return True


def _issue_refund(order_id: str, amount: float, reason: str) -> dict:
    order = _ORDERS.get(order_id)
    if order is None:
        return {"error": f"No order found with id {order_id!r}."}
    if amount <= 0:
        return {"error": "Refund amount must be greater than zero."}
    remaining = round(order["total"] - order["refunded_amount"], 2)
    if amount > remaining:
        return {
            "error": (
                f"Refund amount {amount} exceeds the remaining refundable "
                f"balance of {remaining} {order['currency']} for {order_id}."
            )
        }
    if not _approve_refund(order, amount, reason):
        return {"error": "Refund declined by approval policy."}

    order["refunded_amount"] = round(order["refunded_amount"] + amount, 2)
    return {
        "refund_id": f"RFND-{order_id}-{int(order['refunded_amount'] * 100)}",
        "order_id": order_id,
        "amount": amount,
        "currency": order["currency"],
        "reason": reason,
        "status": "succeeded",
    }


def execute_tool(name: str, tool_input: dict) -> tuple[str, bool]:
    """Dispatch a tool_use block. Returns (content_str, is_error)."""
    try:
        if name == "lookup_order":
            result = _lookup_order(tool_input["order_id"])
        elif name == "issue_refund":
            result = _issue_refund(
                tool_input["order_id"], tool_input["amount"], tool_input["reason"]
            )
        else:
            return f"Unknown tool: {name}", True
    except Exception as exc:  # tool bugs shouldn't crash the loop
        return f"Tool execution failed: {exc}", True

    is_error = "error" in result
    return json.dumps(result), is_error


# --- The agentic loop ---------------------------------------------------


def run_turn(messages: list) -> list:
    """Drive one user turn to completion, looping over tool calls until
    Claude produces a final text response. Mutates and returns `messages`
    so the caller can keep it around for the next turn."""
    while True:
        try:
            response = client.messages.create(
                model=MODEL,
                max_tokens=2048,
                system=SYSTEM_PROMPT,
                tools=TOOLS,
                messages=messages,
            )
        except anthropic.RateLimitError as e:
            raise RuntimeError(f"Rate limited: {e}") from e
        except anthropic.APIStatusError as e:
            raise RuntimeError(f"API error ({e.status_code}): {e.message}") from e
        except anthropic.APIConnectionError as e:
            raise RuntimeError(f"Network error talking to Claude: {e}") from e

        messages.append({"role": "assistant", "content": response.content})

        if response.stop_reason == "tool_use":
            tool_use_blocks = [b for b in response.content if b.type == "tool_use"]
            tool_results = []
            for block in tool_use_blocks:
                content, is_error = execute_tool(block.name, block.input)
                tool_results.append(
                    {
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": content,
                        "is_error": is_error,
                    }
                )
            # All results for this turn go back in a single user message.
            messages.append({"role": "user", "content": tool_results})
            continue  # let Claude see the results and decide what's next

        if response.stop_reason == "pause_turn":
            # Server-side tool hit an internal limit mid-turn; just resend
            # to let Claude continue where it left off.
            continue

        if response.stop_reason == "max_tokens":
            raise RuntimeError("Response was cut off at max_tokens; increase it or retry.")

        # end_turn (or another terminal stop_reason) - the turn is done.
        return messages


def final_text(messages: list) -> str:
    """Extract the text of the most recent assistant message."""
    last = messages[-1]
    if last["role"] != "assistant":
        return ""
    return "".join(b.text for b in last["content"] if b.type == "text")


# --- Example multi-turn conversation ------------------------------------

if __name__ == "__main__":
    messages: list = []

    for user_text in [
        "Hi, I'd like a refund for order ORD-1001, the mouse arrived broken.",
        "Yes please go ahead and refund the full amount.",
    ]:
        print(f"\nUser: {user_text}")
        messages.append({"role": "user", "content": user_text})
        messages = run_turn(messages)
        print(f"Bot: {final_text(messages)}")
