"""Record filtering for the agent.

filter_records(field, value, mode) selects records whose `field` equals
`value`, then either keeps ("include") or drops ("exclude") those matches.
"""


class RecordAgent:
    def __init__(self, records):
        self.records = records

    def filter_records(self, field, value, mode="include"):
        """Filter self.records by field/value, honoring `mode`.

        mode="include": keep records where record[field] == value.
        mode="exclude": keep records where record[field] != value.
        """
        if mode == "include":
            return [r for r in self.records if r.get(field) == value]
        elif mode == "exclude":
            return [r for r in self.records if r.get(field) != value]
        else:
            raise ValueError(f"Unknown mode: {mode!r} (expected 'include' or 'exclude')")


if __name__ == "__main__":

    agent = RecordAgent(records)

    included = agent.filter_records("status", "active", mode="include")
    excluded = agent.filter_records("status", "active", mode="exclude")
