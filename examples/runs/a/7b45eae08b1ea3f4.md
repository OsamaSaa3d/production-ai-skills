````md
# CRM MCP Server

Exposes your CRM's 14-endpoint REST API to an AI agent via the [Model Context
Protocol](https://modelcontextprotocol.io) (MCP). Any MCP-compatible agent
(Claude Code, Claude Desktop, etc.) can then call `list_contacts`,
`create_deal`, `update_company`, and so on as tools.

## ⚠️ Built without your actual API spec

No OpenAPI spec or endpoint docs were available when this was generated, so
`src/endpoints.ts` assumes a conventional REST CRM shape:

| Resource  | Endpoints |
|-----------|-----------|
| Contacts  | list, get, create, update, delete |
| Companies | list, get, create, update, delete |
| Deals     | list, get, create, update |

That's the 14. **If your real API differs** (different paths, verbs, field
names, nesting, activities/tasks instead of a 5th deals endpoint, etc.),
`src/endpoints.ts` is the *only* file you should need to edit — every tool's
name, description, HTTP method, path, and parameters are declared there as
plain data. `src/client.ts` and `src/index.ts` are generic drivers that read
that config; they shouldn't need to change for typical adjustments.

## Setup

```bash
npm install
cp .env.example .env   # then fill in CRM_API_BASE_URL and CRM_API_KEY
npm run build
```

## Run standalone

```bash
npm start
```

The server speaks MCP over stdio and logs a one-line startup confirmation to
stderr (tool count + base URL) — stdout is reserved for the MCP protocol
itself.

## Wire it into an agent

For Claude Code / Claude Desktop, add to your MCP config:

```json
{
  "mcpServers": {
    "crm": {
      "command": "node",
      "args": ["C:/tmp/skills-eval/arm-a/t10-r2/dist/index.js"],
      "env": {
        "CRM_API_BASE_URL": "https://api.yourcrm.com/v1",
        "CRM_API_KEY": "your-key-here"
      }
    }
  }
}
```

## Auth

Defaults to `Authorization: Bearer <CRM_API_KEY>`. Set
`CRM_AUTH_HEADER=ApiKey` in `.env` to send the raw key with no `Bearer`
prefix instead. If your API uses a custom header name (e.g. `X-API-Key`)
rather than `Authorization`, adjust the header line in `src/client.ts`.

## Error handling

Non-2xx responses from the CRM API are surfaced back to the agent as tool
errors (`isError: true`) with the HTTP status and response body included, so
the agent can see, e.g., a 404 or validation error and react instead of
treating a failure as a normal result.
````

````ts
import { EndpointDef } from "./endpoints.js";

export interface CrmClientConfig {
  baseUrl: string;
  apiKey?: string;
  authHeader: "Bearer" | "ApiKey";
  timeoutMs: number;
}

export function loadConfigFromEnv(): CrmClientConfig {
  const baseUrl = process.env.CRM_API_BASE_URL;
  if (!baseUrl) {
    throw new Error(
      "CRM_API_BASE_URL is not set. Point it at your CRM API, e.g. https://api.yourcrm.com/v1"
    );
  }
  return {
    baseUrl: baseUrl.replace(/\/+$/, ""),
    apiKey: process.env.CRM_API_KEY,
    authHeader: (process.env.CRM_AUTH_HEADER as "Bearer" | "ApiKey") ?? "Bearer",
    timeoutMs: Number(process.env.CRM_API_TIMEOUT_MS ?? 15000),
  };
}

/** Raised for non-2xx responses; message includes status + body so the agent can react sensibly. */
export class CrmApiError extends Error {
  constructor(public status: number, public body: string) {
    super(`CRM API error ${status}: ${body}`);
  }
}

export class CrmClient {
  constructor(private config: CrmClientConfig) {}

  async call(endpoint: EndpointDef, args: Record<string, unknown>): Promise<unknown> {
    let path = endpoint.path;
    const query = new URLSearchParams();
    const body: Record<string, unknown> = {};

    for (const [key, def] of Object.entries(endpoint.params)) {
      const value = args[key];
      if (value === undefined || value === null) continue;
      if (def.location === "path") {
        path = path.replace(`{${key}}`, encodeURIComponent(String(value)));
      } else if (def.location === "query") {
        query.set(key, String(value));
      } else {
        body[key] = value;
      }
    }

    const url = new URL(this.config.baseUrl + path);
    for (const [k, v] of query) url.searchParams.set(k, v);

    const headers: Record<string, string> = { Accept: "application/json" };
    if (this.config.apiKey) {
      headers.Authorization =
        this.config.authHeader === "Bearer"
          ? `Bearer ${this.config.apiKey}`
          : this.config.apiKey;
    }

    const hasBody = endpoint.method === "POST" || endpoint.method === "PATCH";
    if (hasBody) headers["Content-Type"] = "application/json";

    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), this.config.timeoutMs);
    try {
      const res = await fetch(url, {
        method: endpoint.method,
        headers,
        body: hasBody ? JSON.stringify(body) : undefined,
        signal: controller.signal,
      });

      const text = await res.text();
      if (!res.ok) {
        throw new CrmApiError(res.status, text);
      }
      return text ? JSON.parse(text) : { ok: true };
    } finally {
      clearTimeout(timeout);
    }
  }
}
````

### `src/index.ts`

````ts
#!/usr/bin/env node
import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { z, ZodRawShape } from "zod";
import { ENDPOINTS, ParamDef } from "./endpoints.js";
import { CrmClient, CrmApiError, loadConfigFromEnv } from "./client.js";

function zodForParam(def: ParamDef) {
  let schema: z.ZodTypeAny =
    def.type === "number" ? z.number() : def.type === "boolean" ? z.boolean() : z.string();
  schema = schema.describe(def.description);
  return def.required ? schema : schema.optional();
}

function buildInputShape(params: Record<string, ParamDef>): ZodRawShape {
  const shape: ZodRawShape = {};
  for (const [key, def] of Object.entries(params)) {
    shape[key] = zodForParam(def);
  }
  return shape;
}

async function main() {
  const config = loadConfigFromEnv();
  const client = new CrmClient(config);

  const server = new McpServer({
    name: "crm-mcp-server",
    version: "1.0.0",
  });

  for (const endpoint of ENDPOINTS) {
    server.registerTool(
      endpoint.name,
      {
        title: endpoint.name,
        description: `${endpoint.description} (${endpoint.method} ${endpoint.path})`,
        inputSchema: buildInputShape(endpoint.params),
      },
      async (args: Record<string, unknown>) => {
        try {
          const result = await client.call(endpoint, args);
          return {
            content: [{ type: "text", text: JSON.stringify(result, null, 2) }],
          };
        } catch (err) {
          const message = err instanceof CrmApiError ? err.message : String(err);
          return {
            content: [{ type: "text", text: message }],
            isError: true,
          };
        }
      }
    );
  }

  const transport = new StdioServerTransport();
  await server.connect(transport);
  console.error(
    `crm-mcp-server: ${ENDPOINTS.length} tools registered, talking to ${config.baseUrl}`
  );
}

main().catch((err) => {
  console.error("Fatal error starting crm-mcp-server:", err);
  process.exit(1);
});
````
