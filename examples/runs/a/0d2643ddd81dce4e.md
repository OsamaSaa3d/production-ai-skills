````py
"""
Orders database helpers.

Demo uses SQLite so this runs with zero setup. If your real pipeline talks to
Postgres/MySQL/etc., replace `get_readonly_connection` and the DDL below with
your actual connection logic — the tool in tools.py only needs a connection
that can run a SELECT and return rows, so nothing else changes.
"""

import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).parent / "orders.db"

_SCHEMA = """
CREATE TABLE IF NOT EXISTS orders (
    order_id      INTEGER PRIMARY KEY,
    customer_name TEXT NOT NULL,
    product       TEXT NOT NULL,
    quantity      INTEGER NOT NULL,
    unit_price    REAL NOT NULL,
    status        TEXT NOT NULL,   -- pending | shipped | delivered | cancelled
    order_date    TEXT NOT NULL    -- ISO 8601 date
);
"""

_SEED = [
    (1, "Alice Chen", "Wireless Mouse", 2, 19.99, "delivered", "2026-08-01"),
    (2, "Bob Martinez", "Mechanical Keyboard", 1, 89.50, "shipped", "2026-08-03"),
    (3, "Alice Chen", "USB-C Hub", 1, 34.00, "pending", "2026-08-20"),
    (4, "Priya Nair", "Monitor Stand", 3, 25.00, "delivered", "2026-08-10"),
    (5, "Bob Martinez", "Webcam", 1, 59.99, "cancelled", "2026-08-12"),
    (6, "Diego Alvarez", "Wireless Mouse", 5, 19.99, "shipped", "2026-09-01"),
    (7, "Priya Nair", "Laptop Sleeve", 2, 15.00, "pending", "2026-09-05"),
]


def ensure_seeded() -> None:
    """Create the orders table and seed it if the DB doesn't exist yet."""
    is_new = not DB_PATH.exists()
    conn = sqlite3.connect(DB_PATH)
    try:
        conn.execute(_SCHEMA)
        if is_new:
            conn.executemany(
                "INSERT INTO orders VALUES (?, ?, ?, ?, ?, ?, ?)", _SEED
            )
            conn.commit()
    finally:
        conn.close()


def get_readonly_connection() -> sqlite3.Connection:
    """
    Open the orders DB in SQLite's read-only URI mode. This is a hard backstop:
    even if the query text somehow slipped past the guard in tools.py, SQLite
    itself will refuse to execute a write against this connection.
    """
    ensure_seeded()
    uri = f"file:{DB_PATH.as_posix()}?mode=ro"
    conn = sqlite3.connect(uri, uri=True)
    conn.row_factory = sqlite3.Row
    return conn
````

````py
"""
LangGraph wiring for the orders-query step.

This file is deliberately split into two parts:

1. `add_orders_query_step(builder, model, from_node, to_node=END)` — call this
   from YOUR existing StateGraph builder to splice the step into your real
   pipeline. It adds two nodes ("query_orders_llm" and "query_orders_tools")
   between whatever node you specify and wherever you want control to go
   next.

2. `build_demo_graph()` / `main()` — a minimal standalone graph so this step
   can be run and checkpointed on its own to verify it works, using
   MessagesState + SqliteSaver for durable checkpointing between steps, the
   same pattern the rest of the pipeline uses.
"""

from __future__ import annotations

from typing import Optional

from langchain_anthropic import ChatAnthropic
from langgraph.graph import END, START, MessagesState, StateGraph
from langgraph.prebuilt import ToolNode, tools_condition

from orders_pipeline.tools import ORDERS_TOOLS

ORDERS_SYSTEM_PROMPT = (
    "You can answer questions about the orders table using the query_orders "
    "tool. Only issue SELECT queries. Prefer narrow WHERE clauses over "
    "pulling the whole table. When you have enough information, answer the "
    "user in plain language — don't just dump raw rows."
)


def add_orders_query_step(
    builder: StateGraph,
    model: ChatAnthropic,
    from_node: str,
    to_node: str = END,
    *,
    llm_node_name: str = "query_orders_llm",
    tool_node_name: str = "query_orders_tools",
) -> None:
    """Splice a model + orders-SQL-tool loop into an existing graph builder.

    Wire it in as: `add_orders_query_step(builder, model, from_node="plan")`
    then edit your existing edge from "plan" to point at "query_orders_llm"
    instead (or add a conditional edge as needed).
    """
    model_with_tools = model.bind_tools(ORDERS_TOOLS)

    def call_model(state: MessagesState) -> dict:
        messages = state["messages"]
        if not any(getattr(m, "type", None) == "system" for m in messages):
            from langchain_core.messages import SystemMessage

            messages = [SystemMessage(content=ORDERS_SYSTEM_PROMPT)] + messages
        response = model_with_tools.invoke(messages)
        return {"messages": [response]}

    builder.add_node(llm_node_name, call_model)
    builder.add_node(tool_node_name, ToolNode(ORDERS_TOOLS))

    builder.add_edge(from_node, llm_node_name)
    builder.add_conditional_edges(
        llm_node_name,
        tools_condition,
        {"tools": tool_node_name, END: to_node},
    )
    builder.add_edge(tool_node_name, llm_node_name)


def build_demo_graph(checkpointer) -> StateGraph:
    """A minimal graph containing only the orders-query step, for standalone testing."""
    model = ChatAnthropic(model="claude-sonnet-5", temperature=0)

    builder = StateGraph(MessagesState)
    builder.add_node("noop_start", lambda state: state)
    builder.add_edge(START, "noop_start")
    add_orders_query_step(builder, model, from_node="noop_start", to_node=END)

    return builder.compile(checkpointer=checkpointer)


def main(question: Optional[str] = None) -> None:
    from langchain_core.messages import HumanMessage
    from langgraph.checkpoint.sqlite import SqliteSaver

    question = question or "How many orders are still pending, and for whom?"

    with SqliteSaver.from_conn_string("checkpoints.sqlite") as checkpointer:
        graph = build_demo_graph(checkpointer)
        config = {"configurable": {"thread_id": "demo-thread-1"}}

        result = graph.invoke({"messages": [HumanMessage(content=question)]}, config)
        print(result["messages"][-1].content)


if __name__ == "__main__":
    main()
````

### `orders_pipeline/tools.py`

````py
"""
The `query_orders` tool: lets the model read from the orders table.

Guardrails (defense in depth — each one alone would probably be enough, but
LLM-generated SQL is adversarial input, so we stack them):
  1. Only a single statement is allowed (reject embedded `;`).
  2. Only SELECT statements are allowed (reject INSERT/UPDATE/DELETE/DROP/
     ALTER/ATTACH/PRAGMA/etc. via an allowlist check on the first keyword).
  3. Only the `orders` table may be referenced.
  4. A LIMIT is enforced server-side so one query can't dump an unbounded
     table.
  5. The underlying connection is opened read-only at the SQLite level
     (see db.get_readonly_connection), so even a guard bypass can't write.
"""

import re
import sqlite3

from langchain_core.tools import tool

from orders_pipeline.db import get_readonly_connection

MAX_ROWS = 200

_DISALLOWED_KEYWORDS = re.compile(
    r"\b(insert|update|delete|drop|alter|create|replace|attach|detach|"
    r"pragma|vacuum|reindex|trigger)\b",
    re.IGNORECASE,
)
_ALLOWED_TABLE = "orders"


class UnsafeQueryError(ValueError):
    pass


def _validate(sql: str) -> str:
    stripped = sql.strip().rstrip(";")

    if ";" in stripped:
        raise UnsafeQueryError("Only a single SQL statement is allowed.")

    if not re.match(r"^\s*select\b", stripped, re.IGNORECASE):
        raise UnsafeQueryError("Only SELECT statements are allowed.")

    if _DISALLOWED_KEYWORDS.search(stripped):
        raise UnsafeQueryError("Query contains a disallowed keyword.")

    if not re.search(rf"\b{_ALLOWED_TABLE}\b", stripped, re.IGNORECASE):
        raise UnsafeQueryError(f"Query must reference the '{_ALLOWED_TABLE}' table.")

    # Cheap table-scope check: flag obvious references to other tables via
    # JOIN, which we don't support for this tool.
    if re.search(r"\bjoin\b", stripped, re.IGNORECASE):
        raise UnsafeQueryError("Joins are not supported; query the 'orders' table only.")

    if not re.search(r"\blimit\b", stripped, re.IGNORECASE):
        stripped = f"{stripped} LIMIT {MAX_ROWS}"

    return stripped


@tool
def query_orders(sql: str) -> str:
    """Run a read-only SQL SELECT query against the `orders` table and return the results.

    The `orders` table has columns: order_id (int), customer_name (text),
    product (text), quantity (int), unit_price (real), status (text: pending
    | shipped | delivered | cancelled), order_date (text, ISO 8601 date).

    Only SELECT statements against `orders` are permitted — no writes, no
    other tables, no joins. Results are capped at 200 rows.

    Args:
        sql: A single SELECT statement, e.g.
             "SELECT * FROM orders WHERE status = 'pending'"
    """
    try:
        safe_sql = _validate(sql)
    except UnsafeQueryError as exc:
        return f"Query rejected: {exc}"

    conn = get_readonly_connection()
    try:
        cursor = conn.execute(safe_sql)
        rows = [dict(row) for row in cursor.fetchall()]
    except sqlite3.Error as exc:
        return f"Query failed: {exc}"
    finally:
        conn.close()

    if not rows:
        return "No rows returned."
    return str(rows)


ORDERS_TOOLS = [query_orders]
````

### `README.md`

````md
# Orders-query step for the LangGraph pipeline

This directory was empty when this step was added, so it's set up as a
standalone, runnable example rather than a patch to existing code. Lift the
pieces below into your real pipeline directly.

## What's here

- `orders_pipeline/db.py` — SQLite orders table + read-only connection helper.
  **Swap this for your real DB connection** (Postgres/MySQL/etc.); nothing
  else needs to change as long as the connection can run a SELECT.
- `orders_pipeline/tools.py` — the `query_orders` tool the model calls. It
  only allows single-statement `SELECT`s against the `orders` table (no
  writes, no other tables, no joins, capped at 200 rows), and the DB
  connection itself is opened read-only as a backstop.
- `orders_pipeline/graph.py` — `add_orders_query_step(builder, model,
  from_node=...)` splices a model node + tool node into any existing
  `StateGraph`. `build_demo_graph` / `main()` show it running standalone with
  `SqliteSaver` for durable checkpointing, matching the rest of the pipeline.

## Wiring it into your existing graph

```python
from orders_pipeline.graph import add_orders_query_step

builder = StateGraph(YourState)
# ... your existing nodes ...
add_orders_query_step(builder, model, from_node="your_upstream_node", to_node="your_downstream_node")
# then compile with your existing checkpointer, e.g.:
graph = builder.compile(checkpointer=your_checkpointer)
```

This adds two nodes, `query_orders_llm` and `query_orders_tools`, forming a
model↔tool loop (via `tools_condition`) between the node you specify and
wherever you want control to continue. Because it's added to your existing
`StateGraph` before `.compile()`, it inherits whatever checkpointer you
already compile with — no separate persistence setup needed.

## Running the standalone demo

```bash
pip install -r requirements.txt
export ANTHROPIC_API_KEY=...   # your key
python -m orders_pipeline.graph
```

First run creates `orders_pipeline/orders.db` (seeded with sample rows) and
`checkpoints.sqlite` (LangGraph's checkpoint store) in the working directory.

## Notes / things to revisit for production

- The table/column allowlist in `tools.py` is intentionally strict (single
  `orders` table, no joins). If you need joins against other tables, extend
  the validator explicitly rather than loosening it — don't just delete the
  join check.
- Consider also enforcing row-level access control (e.g. a `customer_id`
  filter tied to the calling user) if this tool is ever exposed to
  multi-tenant end users rather than internal pipeline use.
- `db.get_readonly_connection` seeds the demo DB on first call
  (`ensure_seeded`) — drop that once you point it at your real orders table.
````
