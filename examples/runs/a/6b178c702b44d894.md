### `issue_store.py`

````py
"""Local JSON-backed issue store standing in for a real issue tracker
(GitHub/Jira/Linear). Swap `search`/`add`/`add_duplicate_report` for API
calls to a real tracker without touching the agent logic in triage_agent.py.
"""
import json
import re
from pathlib import Path

STOPWORDS = {
    "the", "a", "an", "is", "are", "was", "were", "be", "been", "to", "of",
    "in", "on", "for", "and", "or", "it", "this", "that", "with", "when",
    "at", "by", "from", "as", "our", "we", "users", "user", "report",
    "reports", "issue", "bug",
}


def _tokenize(text: str) -> set[str]:
    words = re.findall(r"[a-z0-9]+", text.lower())
    return {w for w in words if w not in STOPWORDS and len(w) > 2}


class IssueStore:
    def __init__(self, path: str = "data/issues.json"):
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        if self.path.exists():
            self._issues = json.loads(self.path.read_text(encoding="utf-8"))
        else:
            self._issues = []

    def _save(self):
        self.path.write_text(
            json.dumps(self._issues, indent=2, ensure_ascii=False), encoding="utf-8"
        )

    def search(self, query: str, top_k: int = 5) -> list[dict]:
        """Keyword-overlap search. Deliberately simple/dependency-free;
        the LLM does the actual semantic duplicate judgment on the
        candidates this returns."""
        query_tokens = _tokenize(query)
        if not query_tokens:
            return []
        scored = []
        for issue in self._issues:
            issue_tokens = _tokenize(issue["title"] + " " + issue["description"])
            if not issue_tokens:
                continue
            overlap = query_tokens & issue_tokens
            if not overlap:
                continue
            score = len(overlap) / len(query_tokens | issue_tokens)
            scored.append((score, issue))
        scored.sort(key=lambda x: x[0], reverse=True)
        return [
            {
                "id": i["id"],
                "title": i["title"],
                "description": i["description"],
                "severity": i["severity"],
                "team": i["team"],
                "status": i["status"],
                "match_score": round(s, 3),
            }
            for s, i in scored[:top_k]
        ]

    def next_id(self) -> str:
        nums = [
            int(m.group(1))
            for i in self._issues
            if (m := re.match(r"BUG-(\d+)", i["id"]))
        ]
        return f"BUG-{(max(nums) + 1) if nums else 1001}"

    def add(self, title: str, description: str, severity: str, team: str,
            reporter: str, created_at: str) -> dict:
        issue = {
            "id": self.next_id(),
            "title": title,
            "description": description,
            "severity": severity,
            "team": team,
            "status": "open",
            "reporter": reporter,
            "created_at": created_at,
        }
        self._issues.append(issue)
        self._save()
        return issue

    def add_duplicate_report(self, issue_id: str, reporter: str, created_at: str) -> dict | None:
        for issue in self._issues:
            if issue["id"] == issue_id:
                issue.setdefault("duplicate_reports", []).append(
                    {"reporter": reporter, "created_at": created_at}
                )
                self._save()
                return issue
        return None

    def get(self, issue_id: str) -> dict | None:
        for issue in self._issues:
            if issue["id"] == issue_id:
                return issue
        return None

````

````md
# Bug Report Triage Agent

Reads an incoming bug report, checks it against existing issues for
duplicates, assigns a severity label (Critical/High/Medium/Low), and routes
it to the right team — using Claude with tool use.

## How it works

- `issue_store.py` — a local JSON file (`data/issues.json`) standing in for
  a real issue tracker (GitHub/Jira/Linear). It exposes keyword search plus
  add/lookup. Swap this module's internals for real API calls to switch
  trackers without touching the agent logic.
- `teams_config.json` — the list of teams, their descriptions, and routing
  keywords. Edit this to match your org.
- `triage_agent.py` — the agent loop. Claude is given a `search_issues` tool
  (to look for duplicates) and a `submit_triage` tool (its structured final
  answer: title, severity, team, duplicate verdict, confidence, rationale).
  The severity rubric and team list are embedded in the system prompt.

On each run:
1. The report is sent to Claude, which searches the issue store for likely
   duplicates (trying a couple of query phrasings).
2. Claude assigns severity per the rubric and picks the best-fit team.
3. If it's a duplicate, the existing issue gets a linked report instead of a
   new issue. Otherwise a new issue is filed in `data/issues.json` with the
   assigned severity/team.

## Setup

```
pip install -r requirements.txt
cp .env.example .env   # fill in ANTHROPIC_API_KEY, then export it
```

```

```

python triage_agent.py --file report.txt --reporter bob@example.com

## Notes / assumptions

- No issue tracker or team list was specified, so this ships with a local
  JSON store (seeded with 6 sample issues in `data/issues.json`) and a
  6-team routing config (`teams_config.json`) as reasonable defaults —
  both are meant to be replaced with your real tracker/teams.
- Duplicate detection uses simple keyword-overlap search to surface
  *candidates*, then Claude makes the actual semantic same-root-cause
  judgment — this avoids needing an embeddings/vector DB dependency while
  still catching near-duplicate phrasing.
- Severity/routing rubrics live in `triage_agent.py`'s system prompt and
  `teams_config.json` — tune the wording there for your org's conventions.
````

### `triage_agent.py`

````py
"""Bug report triage agent.

Reads an incoming bug report, checks it against existing issues for
duplicates, assigns a severity label, and routes it to the right team —
using Claude with tool use to search the issue store and to emit a
structured decision.

Usage:
    python triage_agent.py --file report.txt --reporter someone@example.com
    python triage_agent.py --text "Checkout page 500s for large carts" --reporter someone@example.com
    echo "Checkout page 500s for large carts" | python triage_agent.py --reporter someone@example.com

Requires ANTHROPIC_API_KEY in the environment (see .env.example).
"""
from __future__ import annotations

import argparse
import json
import sys
from datetime import datetime, timezone

from anthropic import Anthropic

from issue_store import IssueStore

MODEL = "claude-sonnet-5"
MAX_TOOL_ROUNDS = 6

SEVERITY_RUBRIC = """
- Critical: Production down, data loss/corruption, security breach, or a core
  workflow (checkout, login, auth) completely broken for all/most users. No
  workaround. Page the on-call team.
- High: A major feature is broken or badly degraded for a large subset of
  users, or a security issue with limited exposure. Workaround may exist but
  is painful. Fix within 1-2 days.
- Medium: A feature is impaired or broken for a subset of users/browsers/
  platforms, has a reasonable workaround, or is a non-critical data accuracy
  issue. Fix within the current or next sprint.
- Low: Cosmetic issues, minor UX annoyances, edge cases affecting very few
  users, or requests better filed as enhancements. No urgency.
""".strip()

TOOLS = [
    {
        "name": "search_issues",
        "description": (
            "Search the existing issue tracker for issues that might be "
            "duplicates of the incoming report. Uses keyword matching, so "
            "try a couple of different phrasings/queries (e.g. once with "
            "the report's core symptom, once with the likely component "
            "name) before concluding there's no duplicate."
        ),
        "input_schema": {
            "type": "object",
            "properties": {
                "query": {"type": "string", "description": "Search text, e.g. a symptom or component name."}
            },
            "required": ["query"],
        },
    },
    {
        "name": "submit_triage",
        "description": (
            "Submit the final triage decision for this bug report. Call "
            "this exactly once, after you've searched for duplicates."
        ),
        "input_schema": {
            "type": "object",
            "properties": {
                "title": {
                    "type": "string",
                    "description": "Short, clear issue title (write one even if the report didn't include one).",
                },
                "is_duplicate": {"type": "boolean"},
                "duplicate_of": {
                    "type": ["string", "null"],
                    "description": "Existing issue ID if is_duplicate is true, else null.",
                },
                "severity": {
                    "type": "string",
                    "enum": ["Critical", "High", "Medium", "Low"],
                },
                "team": {
                    "type": "string",
                    "description": "Team name from the provided team list, or the default team if none fit.",
                },
                "confidence": {
                    "type": "number",
                    "description": "0-1 confidence in the duplicate determination and routing.",
                },
                "rationale": {
                    "type": "string",
                    "description": "2-4 sentences explaining the severity, duplicate, and routing calls.",
                },
            },
            "required": ["title", "is_duplicate", "duplicate_of", "severity", "team", "confidence", "rationale"],
        },
    },
]


def build_system_prompt(teams_config: dict) -> str:
    teams_desc = "\n".join(
        f"- {t['name']}: {t['description']} (keywords: {', '.join(t['keywords'][:6])}, ...)"
        for t in teams_config["teams"]
    )
    return f"""You are a bug triage agent for an engineering organization. Given an
incoming bug report, you must:

1. Search the existing issue tracker (search_issues tool) for likely
   duplicates. Try at least one search using the report's main symptom.
   Only mark something a duplicate if it describes the same underlying
   problem, not just a similar area of the product.
2. Assign a severity label using this rubric:
{SEVERITY_RUBRIC}
3. Route it to exactly one team from this list (pick the closest match; use
   "{teams_config['default_team']}" only if truly none fit):
{teams_desc}
4. Call submit_triage exactly once with your final decision. Do not call it
   more than once, and do not skip searching for duplicates first.

Be decisive. If the report is vague, make the best judgment call from the
rubric rather than asking clarifying questions — this is an offline triage
pass, not a conversation with the reporter."""


def run_agent(client: Anthropic, store: IssueStore, teams_config: dict, report_text: str) -> dict:
    system_prompt = build_system_prompt(teams_config)
    messages = [
        {
            "role": "user",
            "content": f"Incoming bug report:\n\n{report_text}",
        }
    ]

    for _ in range(MAX_TOOL_ROUNDS):
        response = client.messages.create(
            model=MODEL,
            max_tokens=1500,
            system=system_prompt,
            tools=TOOLS,
            messages=messages,
        )

        tool_uses = [b for b in response.content if b.type == "tool_use"]
        if not tool_uses:
            break

        messages.append({"role": "assistant", "content": response.content})
        tool_results = []
        submitted = None

        for block in tool_uses:
            if block.name == "search_issues":
                results = store.search(block.input["query"])
                tool_results.append(
                    {
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": json.dumps(results) if results else "No matches found.",
                    }
                )
            elif block.name == "submit_triage":
                submitted = block.input
                tool_results.append(
                    {
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": "Triage decision recorded.",
                    }
                )

        if submitted is not None:
            return submitted

        messages.append({"role": "user", "content": tool_results})

    raise RuntimeError("Agent did not submit a triage decision within the tool-call budget.")


def main():
    parser = argparse.ArgumentParser(description="Triage an incoming bug report.")
    src = parser.add_mutually_exclusive_group()
    src.add_argument("--file", help="Path to a text file containing the bug report.")
    src.add_argument("--text", help="Bug report text given directly on the command line.")
    parser.add_argument("--reporter", default="unknown@example.com", help="Reporter email/handle.")
    parser.add_argument("--store", default="data/issues.json", help="Path to the JSON issue store.")
    parser.add_argument("--teams-config", default="teams_config.json", help="Path to the teams config JSON.")
    args = parser.parse_args()

    if args.file:
        report_text = open(args.file, encoding="utf-8").read().strip()
    elif args.text:
        report_text = args.text.strip()
    else:
        report_text = sys.stdin.read().strip()

    if not report_text:
        parser.error("No bug report text provided (use --file, --text, or pipe via stdin).")

    teams_config = json.loads(open(args.teams_config, encoding="utf-8").read())
    store = IssueStore(args.store)
    client = Anthropic()

    decision = run_agent(client, store, teams_config, report_text)
    now = datetime.now(timezone.utc).isoformat()

    print("=" * 60)
    print("TRIAGE DECISION")
    print("=" * 60)
    print(f"Title:      {decision['title']}")
    print(f"Severity:   {decision['severity']}")
    print(f"Team:       {decision['team']}")
    print(f"Duplicate:  {decision['is_duplicate']}"
          + (f" (of {decision['duplicate_of']})" if decision.get("duplicate_of") else ""))
    print(f"Confidence: {decision['confidence']}")
    print(f"Rationale:  {decision['rationale']}")
    print("=" * 60)

    if decision["is_duplicate"] and decision.get("duplicate_of"):
        updated = store.add_duplicate_report(decision["duplicate_of"], args.reporter, now)
        if updated:
            print(f"Linked as an additional report on {decision['duplicate_of']} (no new issue created).")
        else:
            print(
                f"WARNING: model referenced duplicate_of={decision['duplicate_of']!r} "
                "which doesn't exist in the store; filing as a new issue instead."
            )
            issue = store.add(decision["title"], report_text, decision["severity"], decision["team"], args.reporter, now)
            print(f"Filed as {issue['id']}.")
    else:
        issue = store.add(decision["title"], report_text, decision["severity"], decision["team"], args.reporter, now)
        print(f"Filed as {issue['id']}, routed to {issue['team']}.")


if __name__ == "__main__":
    main()
````
